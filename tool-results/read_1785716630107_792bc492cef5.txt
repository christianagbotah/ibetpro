    78→    78→  });

   157→   157→      }

   159→   159→      addToast("error", "Failed to connect broker");

   161→   161→      setConnecting(false);
   162→   162→    }

   180→   180→        addToast("error", result.error || "Sync failed");
   181→   181→      }
   182→   182→    } catch {
   183→   183→      addToast("error", "Sync error occurred");
   184→   184→    }
   185→   185→  }, [addToast, refetch]);

   191→   191→    setStep("region");

   195→   195→  };

   230→   230→  };

   272→   272→              <DialogContent className="max-w-lg max-h-[90vh] overflow-y-auto sm:max-h-[90vh] max-[640px]:h-screen max-[640px]:max-h-screen max-[640px]:w-screen max-[640px]:max-w-screen max-[640px]:rounded-none max-[640px]:p-4">
   273→   273→                <DialogHeader>
   274→   274→                  <DialogTitle>Connect Your Broker Account</DialogTitle>

   276→   276→
   277→   277→                {/* Step indicator */}
   278→   278→                <div className="flex items-center gap-2 mb-2">

   280→   280→                    <div className={`h-5 w-5 rounded-full flex items-center justify-center text-[10px] font-bold ${step === "region" ? "bg-primary text-primary-foreground" : "bg-secondary text-muted-foreground"}`}>

   282→   282→                    </div>

   295→   295→                      3

   297→   297→                    Connect

   299→   299→                </div>

   301→   301→                {/* ====== STEP 1: REGION SELECTION ====== */}

   306→   306→                    </p>
   307→   307→
   308→   308→                    <div className="relative">
   309→   309→                      <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   310→   310→                      <Input
   311→   311→                        value={regionSearch}
   312→   312→                        onChange={(e) => setRegionSearch(e.target.value)}

   314→   314→                        className="bg-secondary border-border pl-9"
   315→   315→                      />
