   127→   127→// ==================== RETRY LOGIC ====================
   128→   128→
   129→   129→interface RetryConfig {
   130→   130→  maxRetries: number;
   131→   131→  baseDelayMs: number;
   132→   132→  maxDelayMs: number;
   133→   133→  retryableErrors: string[];
   134→   134→}
   135→   135→
   136→   136→const DEFAULT_RETRY_CONFIG: RetryConfig = {
   137→   137→  maxRetries: 3,
   138→   138→  baseDelayMs: 1000,
   139→   139→  maxDelayMs: 10000,
   140→   140→  retryableErrors: [
   141→   141→    "ETIMEDOUT",
   142→   142→    "ECONNRESET",
   143→   143→    "ECONNREFUSED",
   144→   144→    "rate_limit",
   145→   145→    "server_error",
   146→   146→    "session_expired",
   147→   147→    "temporarily_unavailable",
   148→   148→  ],
   149→   149→};
   150→   150→
   151→   151→async function withRetry<T>(
   152→   152→  fn: () => Promise<T>,
   153→   153→  retryConfig: Partial<RetryConfig> = {}
   154→   154→): Promise<T> {
   155→   155→  const cfg = { ...DEFAULT_RETRY_CONFIG, ...retryConfig };
   156→   156→  let lastError: Error | null = null;
   157→   157→
   158→   158→  for (let attempt = 0; attempt <= cfg.maxRetries; attempt++) {
   159→   159→    try {
   160→   160→      return await fn();
   161→   161→    } catch (error) {
   162→   162→      lastError = error instanceof Error ? error : new Error(String(error));
   163→   163→
   164→   164→      // Check if error is retryable
   165→   165→      const isRetryable = cfg.retryableErrors.some(
   166→   166→        (err) => lastError!.message.toLowerCase().includes(err.toLowerCase())
   167→   167→      );
   168→   168→
   169→   169→      if (!isRetryable || attempt === cfg.maxRetries) {
   170→   170→        throw lastError;
   171→   171→      }
   172→   172→
   173→   173→      // Exponential backoff with jitter
   174→   174→      const delay = Math.min(
   175→   175→        cfg.baseDelayMs * Math.pow(2, attempt) + Math.random() * 1000,
   176→   176→        cfg.maxDelayMs
   177→   177→      );
   178→   178→
   179→   179→      await new Promise((resolve) => setTimeout(resolve, delay));
   180→   180→    }
   181→   181→  }
   182→   182→
   183→   183→  throw lastError;
   184→   184→}

   186→   186→// ==================== IDEMPOTENCY ====================
   187→   187→
   188→   188→// In-memory idempotency store (use Redis in production)
   189→   189→const idempotencyStore = new Map<string, { result: unknown; timestamp: number }>();
   190→   190→const IDEMPOTENCY_TTL = 3600000; // 1 hour
   191→   191→
   192→   192→function checkIdempotency<T>(key: string): T | null {
   193→   193→  const entry = idempotencyStore.get(key);
   194→   194→  if (!entry) return null;
   195→   195→
   196→   196→  if (Date.now() - entry.timestamp > IDEMPOTENCY_TTL) {
   197→   197→    idempotencyStore.delete(key);
   198→   198→    return null;
   199→   199→  }
   200→   200→
   201→   201→  return entry.result as T;
   202→   202→}
   203→   203→
   204→   204→function setIdempotencyResult(key: string, result: unknown): void {
   205→   205→  idempotencyStore.set(key, { result, timestamp: Date.now() });
   206→   206→
   207→   207→  // Clean up old entries periodically
   208→   208→  if (idempotencyStore.size > 10000) {
   209→   209→    const now = Date.now();
   210→   210→    for (const [k, v] of idempotencyStore) {
   211→   211→      if (now - v.timestamp > IDEMPOTENCY_TTL) {
   212→   212→        idempotencyStore.delete(k);
   213→   213→      }
   214→   214→    }
   215→   215→  }
   216→   216→}

   230→   230→   */

   232→   232→    if (this._demoMode !== undefined) return this._demoMode;
   233→   233→    return config.broker.demoMode;
   234→   234→  }

   236→   236→  abstract authenticate(credentials: BrokerCredentials): Promise<BrokerAuthResult>;
   237→   237→  abstract refreshSession(refreshToken: string): Promise<BrokerAuthResult>;

   240→   240→  abstract placeBet(accessToken: string, bet: BrokerBetRequest): Promise<BrokerBetResult>;
   241→   241→  abstract getBetStatus(accessToken: string, brokerBetId: string): Promise<BrokerBetStatus>;

   243→   243→  abstract transferToWallet(accessToken: string, amount: number, walletAddress: string, reference: string): Promise<BrokerTransferResult>;
   244→   244→  abstract getAvailableMarkets(accessToken: string, matchId: string): Promise<BrokerMarket[]>;
   245→   245→  abstract getLiveEvents(accessToken: string): Promise<BrokerLiveEvent[]>;
   246→   246→
   247→   247→  validateSession(sessionToken: string, sessionExpiry: Date): BrokerSession {
   248→   248→    return validateBrokerSession(sessionToken, sessionExpiry);
   249→   249→  }
   250→   250→
   251→   251→  // Common helper: make authenticated API call with retry
   252→   252→  protected async apiCall<T>(
   253→   253→    accessToken: string,
   254→   254→    method: string,
   255→   255→    path: string,
   256→   256→    body?: unknown,
   257→   257→    options?: { idempotencyKey?: string }
   258→   258→  ): Promise<T> {
   259→   259→    const url = `${this.platform.baseUrl}${path}`;
   260→   260→
   261→   261→    // Check idempotency
   262→   262→    if (options?.idempotencyKey) {
   263→   263→      const cached = checkIdempotency<T>(options.idempotencyKey);
   264→   264→      if (cached) return cached;
   265→   265→    }
   266→   266→
   267→   267→    const result = await withRetry(async () => {
   268→   268→      const response = await fetch(url, {
   269→   269→        method,
   270→   270→        headers: {
   271→   271→          "Authorization": `Bearer ${accessToken}`,
   272→   272→          "Content-Type": "application/json",
   273→   273→          "X-Idempotency-Key": options?.idempotencyKey || "",
   274→   274→          "X-Platform": this.platformId,
   275→   275→          "X-App-Version": "1.0.0",
   276→   276→        },
   277→   277→        body: body ? JSON.stringify(body) : undefined,
   278→   278→      });
   279→   279→
   280→   280→      if (!response.ok) {
   281→   281→        const errorData = await response.json().catch(() => ({}));
   282→   282→        const errorCode = errorData.code || errorData.error || `HTTP_${response.status}`;
   283→   283→        const errorMessage = errorData.message || errorData.error || `API error: ${response.status}`;
   284→   284→
   285→   285→        // Map common error codes
   286→   286→        if (response.status === 401) {
   287→   287→          throw new Error(`session_expired: ${errorMessage}`);
   288→   288→        }
   289→   289→        if (response.status === 429) {
   290→   290→          throw new Error(`rate_limit: ${errorMessage}`);
   291→   291→        }
   292→   292→        if (response.status >= 500) {
   293→   293→          throw new Error(`server_error: ${errorMessage}`);
   294→   294→        }
   295→   295→
   296→   296→        throw new Error(`${errorCode}: ${errorMessage}`);
   297→   297→      }
   298→   298→
   299→   299→      return response.json() as Promise<T>;
   300→   300→    });
   301→   301→
   302→   302→    // Store idempotency result
   303→   303→    if (options?.idempotencyKey) {
   304→   304→      setIdempotencyResult(options.idempotencyKey, result);
   305→   305→    }
   306→   306→
   307→   307→    return result;
   308→   308→  }
   309→   309→}

   313→   313→class OAuthBrokerAdapter extends BaseBrokerAdapter {
   314→   314→  platformId: string;

   322→   322→
   323→   323→  async authenticate(credentials: BrokerCredentials): Promise<BrokerAuthResult> {
   324→   324→    if (!credentials.token) {
   325→   325→      return { success: false, error: "OAuth authorization code required" };
   326→   326→    }
   327→   327→
   328→   328→    // Demo/Sandbox mode: simulate successful authentication without real API calls
   329→   329→    if (this.isDemoMode()) {
   330→   330→      return {
   331→   331→        success: true,
   332→   332→        accessToken: `demo_oauth_${this.platformId}_${Date.now()}`,
   333→   333→        refreshToken: `demo_refresh_${this.platformId}_${Date.now()}`,
   334→   334→        sessionExpiry: new Date(Date.now() + 3600000),
   335→   335→        brokerUserId: `demo_oauth_${credentials.token.substring(0, 8)}_${Date.now()}`,
   336→   336→      };
   337→   337→    }
   338→   338→
   339→   339→    try {
   340→   340→      // In production, exchange authorization code for access token
   341→   341→      // POST /oauth/token with client_id, client_secret, code, redirect_uri
   342→   342→      const result = await withRetry(async () => {
   343→   343→        const response = await fetch(`${this.platform.baseUrl}/oauth/token`, {
   344→   344→          method: "POST",
   345→   345→          headers: { "Content-Type": "application/json" },
   346→   346→          body: JSON.stringify({
   347→   347→            grant_type: "authorization_code",
   348→   348→            code: credentials.token,
   349→   349→            client_id: process.env[`${this.platformId.toUpperCase()}_CLIENT_ID`],
   350→   350→            client_secret: process.env[`${this.platformId.toUpperCase()}_CLIENT_SECRET`],
   351→   351→            redirect_uri: `${config.platform.url}/api/broker/callback/${this.platformId}`,
   352→   352→          }),
   353→   353→        });
   354→   354→
   355→   355→        if (!response.ok) {
   356→   356→          throw new Error(`OAuth token exchange failed: ${response.status}`);
   357→   357→        }
   358→   358→
   359→   359→        return response.json();
   360→   360→      });
   361→   361→
   362→   362→      return {
   363→   363→        success: true,
   364→   364→        accessToken: result.access_token,
   365→   365→        refreshToken: result.refresh_token,
   366→   366→        sessionExpiry: new Date(Date.now() + (result.expires_in || 3600) * 1000),
   367→   367→        brokerUserId: result.user_id,
   368→   368→      };
   369→   369→    } catch (error) {
   370→   370→      return {
   371→   371→        success: false,
   372→   372→        error: error instanceof Error ? error.message : "OAuth authentication failed",
   373→   373→      };
   374→   374→    }
   375→   375→  }

   395→   395→        return response.json();

   402→   402→        sessionExpiry: new Date(Date.now() + (result.expires_in || 3600) * 1000),

   404→   404→    } catch (error) {

   934→   934→// ==================== ADAPTER FACTORY ====================
   935→   935→
   936→   936→const adapterCache = new Map<string, BrokerAdapter>();
   937→   937→
   938→   938→/**
   939→   939→ * Get or create a broker adapter for the given platform
   940→   940→ * @param platformId - The platform ID
   941→   941→ * @param demoMode - If true, the adapter will use demo/sandbox mode (no real API calls)
   942→   942→ */
   943→   943→export function getBrokerAdapter(platformId: string, demoMode?: boolean): BrokerAdapter | null {
   944→   944→  // Check cache (but demo mode changes per-user, so we need a keyed cache)
   945→   945→  const cacheKey = demoMode !== undefined ? `${platformId}_${demoMode ? "demo" : "real"}` : platformId;
   946→   946→  if (adapterCache.has(cacheKey)) {
   947→   947→    return adapterCache.get(cacheKey)!;
   948→   948→  }
   949→   949→
   950→   950→  // Manual adapter
   951→   951→  if (platformId === "manual") {
   952→   952→    const adapter = new ManualBrokerAdapter();
   953→   953→    adapterCache.set(cacheKey, adapter);
   954→   954→    return adapter;
   955→   955→  }
   956→   956→
   957→   957→  // Get platform config
   958→   958→  const platform = getBrokerPlatform(platformId);
   959→   959→  if (!platform) {
   960→   960→    console.error(`Unknown broker platform: ${platformId}`);
   961→   961→    return null;
   962→   962→  }
   963→   963→
   964→   964→  // Create appropriate adapter based on auth type
   965→   965→  let adapter: BrokerAdapter;
   966→   966→
   967→   967→  switch (platform.authType) {
   968→   968→    case "oauth":
   969→   969→      adapter = new OAuthBrokerAdapter(platform);
   970→   970→      break;
   971→   971→    case "api_key":
   972→   972→      adapter = new ApiKeyBrokerAdapter(platform);
   973→   973→      break;
   974→   974→    case "web_session":
   975→   975→      adapter = new WebSessionBrokerAdapter(platform);
