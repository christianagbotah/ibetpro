   302→          </div>
   303→          <div className="grid grid-cols-2 gap-3">

   307→                {profile.currency === "NGN" ? "₦" : profile.currency === "GBP" ? "£" : profile.currency === "EUR" ? "€" : "$"}

   397→                  setSettings({ ...settings, minOddsThreshold: parseFloat(e.target.value) || 0 })
