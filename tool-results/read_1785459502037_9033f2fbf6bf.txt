     1→"use client";
     2→
     3→import { useState, useCallback, useMemo } from "react";
     4→import { useFetch } from "@/lib/hooks";
     5→import { useToast } from "@/components/ui/toast";
     6→import { Card, CardContent, CardHeader, CardTitle } from "@/components/ui/card";
     7→import { Button } from "@/components/ui/button";
     8→import { Badge } from "@/components/ui/badge";
     9→import { Input } from "@/components/ui/input";
    10→import { Label } from "@/components/ui/label";
    11→import { Separator } from "@/components/ui/separator";
    12→import {
    13→  Dialog,
    14→  DialogContent,
    15→  DialogHeader,
    16→  DialogTitle,
    17→} from "@/components/ui/dialog";
    18→import {
    19→  Link2, RefreshCw, Wallet, Shield, CheckCircle2,
    20→  AlertCircle, Globe, Key, User, Lock, ChevronRight,
    21→  Search, MapPin, Smartphone, Star, ArrowLeft,
    22→} from "lucide-react";
    23→import {
    24→  REGIONS,
    25→  getPlatformsForRegion,
    26→  getCurrencyForRegion,
    27→  getContinents,
    28→  searchRegions,
    29→  type RegionInfo,
    30→  type BrokerPlatformInfo,
    31→} from "@/lib/regions";
    32→
    33→interface ConnectedAccount {
    34→  id: string;
    35→  platform: string;
    36→  platformName: string;
    37→  accountName: string;
    38→  isConnected: boolean;
    39→  sessionValid: boolean;
    40→  needsRefresh: boolean;
    41→  sessionExpiry: string | null;
    42→  brokerType: string;
    43→  brokerRegion: string | null;
    44→  balance: number;
    45→  currency: string;
    46→  allocatedAmount: number;
    47→  allocationLock: boolean;
    48→  totalBrokerBets: number;
    49→  totalBrokerProfit: number;
    50→  lastBetPlacedAt: string | null;
    51→  features: {
    52→    liveBetting: boolean;
    53→    cashout: boolean;
    54→    partialCashout: boolean;
    55→    accumulators: boolean;
    56→    maxAccumulatorLegs: number;
    57→    minStake: number;
    58→    maxStake: number;
    59→  } | null;
    60→}
    61→
    62→type ConnectStep = "region" | "platform" | "credentials";
    63→
    64→export function BrokerConnect() {
    65→  const { toast } = useToast();
    66→  const [connectDialogOpen, setConnectDialogOpen] = useState(false);
    67→  const [step, setStep] = useState<ConnectStep>("region");
    68→  const [selectedRegion, setSelectedRegion] = useState<string>("");
    69→  const [selectedPlatform, setSelectedPlatform] = useState<string | null>(null);
    70→  const [connecting, setConnecting] = useState(false);
    71→  const [credentials, setCredentials] = useState({
    72→    username: "",
    73→    password: "",
    74→    apiKey: "",
    75→    token: "",
    76→  });
    77→  const [accountName, setAccountName] = useState("");
    78→  const [regionSearch, setRegionSearch] = useState("");
    79→  const [selectedContinent, setSelectedContinent] = useState<string>("all");
    80→
    81→  const { data, loading, refresh } = useFetch<{
    82→    accounts: ConnectedAccount[];
    83→    availablePlatforms: BrokerPlatformInfo[];
    84→  }>("/api/broker/connect", { accounts: [], availablePlatforms: [] });
    85→
    86→  const accounts = data?.accounts || [];
    87→
    88→  // Get filtered regions based on search and continent
    89→  const filteredRegions = useMemo(() => {
    90→    let regions = regionSearch ? searchRegions(regionSearch) : REGIONS;
    91→    if (selectedContinent !== "all") {
    92→      regions = regions.filter((r) => r.continent === selectedContinent);
    93→    }
    94→    return regions;
    95→  }, [regionSearch, selectedContinent]);
    96→
    97→  // Get platforms for selected region
    98→  const platformsForRegion = useMemo(() => {
    99→    if (!selectedRegion) return [];
   100→    return getPlatformsForRegion(selectedRegion);
   101→  }, [selectedRegion]);
   102→
   103→  const selectedRegionInfo = useMemo(() => {
   104→    return REGIONS.find((r) => r.code === selectedRegion);
   105→  }, [selectedRegion]);
   106→
   107→  const selectedPlatformData = platformsForRegion.find((p) => p.id === selectedPlatform);
   108→
   109→  const handleConnect = useCallback(async () => {
   110→    if (!selectedPlatform) return;
   111→    setConnecting(true);
   112→
   113→    try {
   114→      const res = await fetch("/api/broker/connect", {
   115→        method: "POST",
   116→        headers: { "Content-Type": "application/json" },
   117→        body: JSON.stringify({
   118→          platformId: selectedPlatform,
   119→          credentials,
   120→          region: selectedRegion,
   121→          accountName: accountName || undefined,
   122→        }),
   123→      });
   124→
   125→      const result = await res.json();
   126→
   127→      if (result.success) {
   128→        toast({
   129→          title: "Broker Connected!",
   130→          description: `${result.account.accountName} connected successfully`,
   131→        });
   132→        setConnectDialogOpen(false);
   133→        resetForm();
   134→        refresh();
   135→      } else {
   136→        toast({
   137→          title: "Connection Failed",
   138→          description: result.error || "Failed to connect broker",
   139→          variant: "destructive",
   140→        });
   141→      }
   142→    } catch {
   143→      toast({
   144→        title: "Error",
   145→        description: "Failed to connect broker",
   146→        variant: "destructive",
   147→      });
   148→    } finally {
   149→      setConnecting(false);
   150→    }
   151→  }, [selectedPlatform, credentials, accountName, selectedRegion, toast, refresh]);
   152→
   153→  const handleSync = useCallback(async (accountId: string) => {
   154→    try {
   155→      const res = await fetch("/api/broker/sync", {
   156→        method: "POST",
   157→        headers: { "Content-Type": "application/json" },
   158→        body: JSON.stringify({ bettingAccountId: accountId }),
   159→      });
   160→
   161→      const result = await res.json();
   162→
   163→      if (result.success) {
   164→        const currency = result.balance?.currency || "USD";
   165→        toast({
   166→          title: "Account Synced",
   167→          description: `Balance: ${currency} ${result.balance.total.toFixed(2)}`,
   168→        });
   169→        refresh();
   170→      } else {
   171→        toast({
   172→          title: "Sync Failed",
   173→          description: result.error,
   174→          variant: "destructive",
   175→        });
   176→      }
   177→    } catch {
   178→      toast({ title: "Sync Error", variant: "destructive" });
   179→    }
   180→  }, [toast, refresh]);
   181→
   182→  const resetForm = () => {
   183→    setCredentials({ username: "", password: "", apiKey: "", token: "" });
   184→    setAccountName("");
   185→    setSelectedPlatform(null);
   186→    setStep("region");
   187→    setRegionSearch("");
   188→    setSelectedContinent("all");
   189→  };
   190→
   191→  const handleDialogClose = (open: boolean) => {
   192→    setConnectDialogOpen(open);
   193→    if (!open) resetForm();
   194→  };
   195→
   196→  const formatBalance = (amount: number, currency: string) => {
   197→    const region = REGIONS.find((r) => r.currencyCode === currency);
   198→    const symbol = region?.currencySymbol || "$";
   199→    return `${symbol}${amount.toLocaleString(undefined, { minimumFractionDigits: 2, maximumFractionDigits: 2 })}`;
   200→  };
   201→
   202→  const continents = getContinents();
   203→
   204→  return (
   205→    <div className="space-y-6">
   206→      <Card className="bg-card border-border">
   207→        <CardHeader className="pb-3">
   208→          <div className="flex items-center justify-between">
   209→            <CardTitle className="flex items-center gap-2 text-base">
   210→              <Link2 className="h-4 w-4 text-primary" />
   211→              Connected Brokers
   212→            </CardTitle>
   213→            <Dialog open={connectDialogOpen} onOpenChange={handleDialogClose}>
   214→              <Button
   215→                size="sm"
   216→                className="bg-primary text-primary-foreground"
   217→                onClick={() => setConnectDialogOpen(true)}
   218→              >
   219→                <Link2 className="h-3.5 w-3.5 mr-1.5" />
   220→                Connect Broker
   221→              </Button>
   222→
   223→              <DialogContent className="max-w-lg max-h-[90vh] overflow-y-auto">
   224→                <DialogHeader>
   225→                  <DialogTitle>Connect Your Broker Account</DialogTitle>
   226→                </DialogHeader>
   227→
   228→                {/* Step indicator */}
   229→                <div className="flex items-center gap-2 mb-2">
   230→                  <div className={`flex items-center gap-1.5 text-xs ${step === "region" ? "text-primary font-medium" : "text-muted-foreground"}`}>
   231→                    <div className={`h-5 w-5 rounded-full flex items-center justify-center text-[10px] font-bold ${step === "region" ? "bg-primary text-primary-foreground" : "bg-secondary text-muted-foreground"}`}>
   232→                      {step !== "region" ? "✓" : "1"}
   233→                    </div>
   234→                    Region
   235→                  </div>
   236→                  <div className="h-px flex-1 bg-border" />
   237→                  <div className={`flex items-center gap-1.5 text-xs ${step === "platform" ? "text-primary font-medium" : "text-muted-foreground"}`}>
   238→                    <div className={`h-5 w-5 rounded-full flex items-center justify-center text-[10px] font-bold ${step === "platform" ? "bg-primary text-primary-foreground" : step === "credentials" ? "bg-primary text-primary-foreground" : "bg-secondary text-muted-foreground"}`}>
   239→                      {step === "credentials" ? "✓" : "2"}
   240→                    </div>
   241→                    Platform
   242→                  </div>
   243→                  <div className="h-px flex-1 bg-border" />
   244→                  <div className={`flex items-center gap-1.5 text-xs ${step === "credentials" ? "text-primary font-medium" : "text-muted-foreground"}`}>
   245→                    <div className={`h-5 w-5 rounded-full flex items-center justify-center text-[10px] font-bold ${step === "credentials" ? "bg-primary text-primary-foreground" : "bg-secondary text-muted-foreground"}`}>
   246→                      3
   247→                    </div>
   248→                    Connect
   249→                  </div>
   250→                </div>
   251→
   252→                {/* ====== STEP 1: REGION SELECTION ====== */}
   253→                {step === "region" && (
   254→                  <div className="space-y-3">
   255→                    <p className="text-sm text-muted-foreground">
   256→                      Select your country or region. This helps us show betting platforms available in your area and set the right currency.
   257→                    </p>
   258→
   259→                    <div className="relative">
   260→                      <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   261→                      <Input
   262→                        value={regionSearch}
   263→                        onChange={(e) => setRegionSearch(e.target.value)}
   264→                        placeholder="Search country or currency..."
   265→                        className="bg-secondary border-border pl-9"
   266→                      />
   267→                    </div>
   268→
   269→                    <div className="flex flex-wrap gap-1.5">
   270→                      <button
   271→                        onClick={() => setSelectedContinent("all")}
   272→                        className={`rounded-full px-3 py-1 text-xs font-medium transition-all ${
   273→                          selectedContinent === "all"
   274→                            ? "bg-primary/10 text-primary border border-primary/30"
   275→                            : "bg-secondary/50 text-muted-foreground border border-border"
   276→                        }`}
   277→                      >
   278→                        All
   279→                      </button>
   280→                      {continents.map((c) => (
   281→                        <button
   282→                          key={c}
   283→                          onClick={() => setSelectedContinent(c)}
   284→                          className={`rounded-full px-3 py-1 text-xs font-medium transition-all ${
   285→                            selectedContinent === c
   286→                              ? "bg-primary/10 text-primary border border-primary/30"
   287→                              : "bg-secondary/50 text-muted-foreground border border-border"
   288→                          }`}
   289→                        >
   290→                          {c}
   291→                        </button>
   292→                      ))}
   293→                    </div>
   294→
   295→                    <div className="grid grid-cols-2 sm:grid-cols-3 gap-2 max-h-64 overflow-y-auto">
   296→                      {filteredRegions.map((region) => {
   297→                        const platformCount = getPlatformsForRegion(region.code).length;
   298→                        return (
   299→                          <button
   300→                            key={region.code}
   301→                            onClick={() => {
   302→                              setSelectedRegion(region.code);
   303→                              setStep("platform");
   304→                            }}
   305→                            className={`flex items-center gap-2 rounded-lg border p-2.5 transition-all text-left hover:bg-primary/5 ${
   306→                              selectedRegion === region.code
   307→                                ? "border-primary/30 bg-primary/5"
   308→                                : "border-border"
   309→                            }`}
   310→                          >
   311→                            <span className="text-lg">{region.flag}</span>
   312→                            <div className="min-w-0 flex-1">
   313→                              <p className="text-xs font-medium text-foreground truncate">{region.name}</p>
   314→                              <p className="text-[10px] text-muted-foreground">
   315→                                {region.currencySymbol} {region.currencyCode} · {platformCount} platforms
   316→                              </p>
   317→                            </div>
   318→                          </button>
   319→                        );
   320→                      })}
   321→                    </div>
   322→
   323→                    {filteredRegions.length === 0 && (
   324→                      <div className="text-center py-4">
   325→                        <p className="text-sm text-muted-foreground">No regions found matching your search</p>
   326→                      </div>
   327→                    )}
   328→                  </div>
   329→                )}
   330→
   331→                {/* ====== STEP 2: PLATFORM SELECTION ====== */}
   332→                {step === "platform" && selectedRegionInfo && (
   333→                  <div className="space-y-3">
   334→                    <button
   335→                      onClick={() => {
   336→                        setStep("region");
   337→                        setSelectedPlatform(null);
   338→                      }}
   339→                      className="flex items-center gap-1 text-xs text-muted-foreground hover:text-primary transition-colors"
   340→                    >
   341→                      <ArrowLeft className="h-3 w-3" /> Change region
   342→                    </button>
   343→
   344→                    <div className="flex items-center gap-2 p-2.5 rounded-lg bg-primary/5 border border-primary/10">
   345→                      <span className="text-lg">{selectedRegionInfo.flag}</span>
   346→                      <div>
   347→                        <p className="text-sm font-medium text-foreground">{selectedRegionInfo.name}</p>
   348→                        <p className="text-xs text-muted-foreground">
   349→                          Currency: {selectedRegionInfo.currencySymbol} {selectedRegionInfo.currencyCode} ({selectedRegionInfo.currencyName})
   350→                        </p>
   351→                      </div>
   352→                    </div>
   353→
   354→                    <p className="text-sm text-muted-foreground">
   355→                      {platformsForRegion.length} platforms available in {selectedRegionInfo.name}. Select your betting platform.
   356→                    </p>
   357→
   358→                    <div className="space-y-2 max-h-64 overflow-y-auto">
   359→                      {platformsForRegion.map((platform) => {
   360→                        const isPopular = platform.popularIn.includes(selectedRegion);
   361→                        return (
   362→                          <button
   363→                            key={platform.id}
   364→                            onClick={() => {
   365→                              setSelectedPlatform(platform.id);
   366→                              setStep("credentials");
   367→                            }}
   368→                            className="flex items-center justify-between rounded-lg border border-border p-3 hover:bg-primary/5 transition-colors text-left w-full"
   369→                          >
   370→                            <div className="flex items-center gap-3">
   371→                              <div
   372→                                className="flex h-10 w-10 items-center justify-center rounded-lg text-xs font-bold text-white shrink-0"
   373→                                style={{ backgroundColor: platform.color || "#10b981" }}
   374→                              >
   375→                                {platform.logo}
   376→                              </div>
   377→                              <div>
   378→                                <div className="flex items-center gap-1.5">
   379→                                  <p className="text-sm font-medium text-foreground">{platform.name}</p>
   380→                                  {isPopular && (
   381→                                    <Badge className="bg-amber-400/10 text-amber-400 border-amber-400/30 text-[9px] px-1 py-0">
   382→                                      <Star className="h-2.5 w-2.5 mr-0.5" /> Popular
   383→                                    </Badge>
   384→                                  )}
   385→                                </div>
   386→                                <p className="text-xs text-muted-foreground">
   387→                                  {platform.authType.replace(/_/g, " ")} · {platform.mobileApp ? "📱 App" : "Web only"}
   388→                                </p>
   389→                              </div>
   390→                            </div>
   391→                            <ChevronRight className="h-4 w-4 text-muted-foreground shrink-0" />
   392→                          </button>
   393→                        );
   394→                      })}
   395→                    </div>
   396→                  </div>
   397→                )}
   398→
   399→                {/* ====== STEP 3: CREDENTIALS ====== */}
   400→                {step === "credentials" && selectedPlatformData && selectedRegionInfo && (
   401→                  <div className="space-y-4">
   402→                    <button
   403→                      onClick={() => {
   404→                        setStep("platform");
   405→                        setSelectedPlatform(null);
   406→                      }}
   407→                      className="flex items-center gap-1 text-xs text-muted-foreground hover:text-primary transition-colors"
   408→                    >
   409→                      <ArrowLeft className="h-3 w-3" /> Change platform
   410→                    </button>
   411→
   412→                    <div className="flex items-center gap-3 p-3 rounded-lg bg-primary/5 border border-primary/10">
   413→                      <div
   414→                        className="flex h-10 w-10 items-center justify-center rounded-lg text-xs font-bold text-white shrink-0"
   415→                        style={{ backgroundColor: selectedPlatformData.color || "#10b981" }}
   416→                      >
   417→                        {selectedPlatformData.logo}
   418→                      </div>
   419→                      <div className="flex-1 min-w-0">
   420→                        <p className="text-sm font-medium text-foreground">{selectedPlatformData.name}</p>
   421→                        <p className="text-xs text-muted-foreground">
   422→                          {selectedPlatformData.authType.replace(/_/g, " ")} authentication
   423→                        </p>
   424→                      </div>
   425→                      <div className="text-right shrink-0">
   426→                        <p className="text-xs text-muted-foreground">{selectedRegionInfo.flag} {selectedRegionInfo.name}</p>
   427→                        <p className="text-xs font-medium text-primary">{selectedRegionInfo.currencySymbol} {selectedRegionInfo.currencyCode}</p>
   428→                      </div>
   429→                    </div>
   430→
   431→                    <div className="space-y-3">
   432→                      <div>
   433→                        <Label className="text-xs text-muted-foreground">Account Name</Label>
   434→                        <Input
   435→                          value={accountName}
   436→                          onChange={(e) => setAccountName(e.target.value)}
   437→                          placeholder={`My ${selectedPlatformData?.name} Account`}
   438→                          className="bg-secondary border-border mt-1"
   439→                        />
   440→                      </div>
   441→
   442→                      <Separator />
   443→
   444→                      {selectedPlatformData?.authType === "web_session" && (
   445→                        <>
   446→                          <div>
   447→                            <Label className="text-xs text-muted-foreground flex items-center gap-1">
   448→                              <User className="h-3 w-3" /> Username / Email
   449→                            </Label>
   450→                            <Input
   451→                              value={credentials.username}
   452→                              onChange={(e) => setCredentials({ ...credentials, username: e.target.value })}
   453→                              placeholder="Your broker username"
   454→                              className="bg-secondary border-border mt-1"
   455→                            />
   456→                          </div>
   457→                          <div>
   458→                            <Label className="text-xs text-muted-foreground flex items-center gap-1">
   459→                              <Lock className="h-3 w-3" /> Password
   460→                            </Label>
   461→                            <Input
   462→                              type="password"
   463→                              value={credentials.password}
   464→                              onChange={(e) => setCredentials({ ...credentials, password: e.target.value })}
   465→                              placeholder="Your broker password"
   466→                              className="bg-secondary border-border mt-1"
   467→                            />
   468→                          </div>
   469→                        </>
   470→                      )}
   471→
   472→                      {selectedPlatformData?.authType === "api_key" && (
   473→                        <div>
   474→                          <Label className="text-xs text-muted-foreground flex items-center gap-1">
   475→                            <Key className="h-3 w-3" /> API Key
   476→                          </Label>
   477→                          <Input
   478→                            value={credentials.apiKey}
   479→                            onChange={(e) => setCredentials({ ...credentials, apiKey: e.target.value })}
   480→                            placeholder="Your broker API key"
   481→                            className="bg-secondary border-border mt-1"
   482→                          />
   483→                        </div>
   484→                      )}
   485→
   486→                      {selectedPlatformData?.authType === "oauth" && (
   487→                        <div>
   488→                          <Label className="text-xs text-muted-foreground flex items-center gap-1">
   489→                            <Shield className="h-3 w-3" /> OAuth Token
   490→                          </Label>
   491→                          <Input
   492→                            value={credentials.token}
   493→                            onChange={(e) => setCredentials({ ...credentials, token: e.target.value })}
   494→                            placeholder="OAuth access token"
   495→                            className="bg-secondary border-border mt-1"
   496→                          />
   497→                        </div>
   498→                      )}
   499→
   500→                      {selectedPlatformData?.authType === "manual" && (
   501→                        <div className="rounded-lg bg-amber-400/5 border border-amber-400/10 p-3">
   502→                          <p className="text-xs text-muted-foreground">
   503→                            Manual connection: You provide your account details and manage bets manually.
   504→                            The app will track your allocation and commission.
   505→                          </p>
   506→                        </div>
   507→                      )}
   508→                    </div>
   509→
   510→                    {/* Platform features */}
   511→                    <div className="rounded-lg bg-secondary/30 p-3 space-y-2">
   512→                      <p className="text-xs font-medium text-muted-foreground">Platform Features</p>
   513→                      <div className="flex flex-wrap gap-1.5">
   514→                        {selectedPlatformData.features.liveBetting && (
   515→                          <Badge variant="secondary" className="text-[10px]">Live Betting</Badge>
   516→                        )}
   517→                        {selectedPlatformData.features.cashout && (
   518→                          <Badge variant="secondary" className="text-[10px]">Cashout</Badge>
   519→                        )}
   520→                        {selectedPlatformData.features.partialCashout && (
   521→                          <Badge variant="secondary" className="text-[10px]">Partial Cashout</Badge>
   522→                        )}
   523→                        {selectedPlatformData.features.accumulators && (
   524→                          <Badge variant="secondary" className="text-[10px]">Accumulators (up to {selectedPlatformData.features.maxAccumulatorLegs} legs)</Badge>
   525→                        )}
   526→                        {selectedPlatformData.mobileApp && (
   527→                          <Badge variant="secondary" className="text-[10px]">
   528→                            <Smartphone className="h-2.5 w-2.5 mr-0.5" /> Mobile App
   529→                          </Badge>
   530→                        )}
   531→                      </div>
   532→                      <p className="text-[10px] text-muted-foreground">
   533→                        Min stake: {selectedRegionInfo.currencySymbol}{selectedPlatformData.features.minStake.toLocaleString()} ·
   534→                        Max stake: {selectedRegionInfo.currencySymbol}{selectedPlatformData.features.maxStake.toLocaleString()}
   535→                      </p>
   536→                    </div>
   537→
   538→                    <div className="rounded-lg bg-primary/5 border border-primary/10 p-3">
   539→                      <p className="text-xs text-muted-foreground">
   540→                        <Shield className="h-3 w-3 inline mr-1" />
   541→                        Your credentials are encrypted and stored securely. The app only uses your
   542→                        allocation to place bets and never withdraws more than you authorize.
   543→                      </p>
   544→                    </div>
