   113→export default function BettingPage() {
   114→  const { addToast } = useToast();
   115→  const { user } = useAuth();
   116→  const { data: bets, loading, refetch: refetchBets } = useFetch<Bet[]>("/api/bets", []);
   117→  const { data: accounts } = useFetch<Array<{ id: string; platform: string }>>("/api/accounts", []);
   118→  const [settings, setSettings] = useState<UserSettings>(defaultSettings);
   119→  const [betFilter, setBetFilter] = useState<string>("all");
   120→  const [botRunning, setBotRunning] = useState(false);
   121→  const [botLoading, setBotLoading] = useState(false); // for start/stop API calls
   122→  const [settleLoading, setSettleLoading] = useState(false);
   123→  const [botSession, setBotSession] = useState<{
   124→    totalScans: number;
   125→    totalBetsPlaced: number;
   126→    totalStakeUsed: number;
   127→    totalProfit: number;
   128→    lastScanAt: string | null;
   129→    lastBetAt: string | null;
   130→    startedAt: string | null;
   131→  } | null>(null);
   132→  const [engineRunning, setEngineRunning] = useState(false);
   133→  const pollIntervalRef = useRef<NodeJS.Timeout | null>(null);
   134→
   135→  // Load user settings and bot status
   136→  useEffect(() => {
   137→    async function loadSettings() {
   138→      try {
   139→        const res = await fetch("/api/settings");
   140→        if (res.ok) {
   141→          const data = await res.json();
   142→          if (data) {
   143→            setSettings({
   144→              ...defaultSettings,
   145→              ...data,
   146→            });
   147→          }
   148→        }
   149→      } catch {
   150→        // Use defaults
   151→      }
   152→    }
   153→    async function loadBotStatus() {
   154→      try {
   155→        const res = await fetch("/api/bot/control");
   156→        if (res.ok) {
   157→          const data = await res.json();
   158→          if (data.botStatus === "running") {
   159→            setBotRunning(true);
   160→          }
   161→          if (data.engineRunning) {
   162→            setEngineRunning(true);
   163→          }
   164→          if (data.session) {
   165→            setBotSession(data.session);
   166→          }
   167→        }
   168→      } catch {
   169→        // Ignore
   170→      }
   171→    }
   172→    // Initialize the bot engine (recovers running bots on server restart)
   173→    fetch("/api/bot/init").catch(() => {});
   174→    loadSettings();
   175→    loadBotStatus();
   176→  }, []);
   177→
   178→  // Poll for bot status updates when running (read-only, doesn't trigger scans)
   179→  useEffect(() => {
   180→    if (botRunning) {
   181→      // Poll for status updates every 15 seconds (the bot engine runs scans independently)
   182→      pollIntervalRef.current = setInterval(async () => {
   183→        try {
   184→          const res = await fetch("/api/bot/control");
   185→          if (res.ok) {
   186→            const data = await res.json();
   187→            if (data.botStatus === "stopped") {
   188→              // Bot was auto-stopped (risk limit, no allocation, etc.)
   189→              setBotRunning(false);
   190→              setEngineRunning(false);
   191→              if (data.session?.stopReason && data.session.stopReason !== "user_stopped") {
   192→                const reasonMap: Record<string, string> = {
   193→                  stop_loss: "Stop-loss limit reached",
   194→                  profit_target: "Profit target reached",
   195→                  schedule: "Outside betting schedule",
   196→                  no_allocation: "No allocation remaining",
   197→                  error: "An error occurred",
   198→                  too_many_errors: "Too many errors occurred",
   199→                };
   200→                addToast("warning", `Bot stopped: ${reasonMap[data.session.stopReason] || data.session.stopReason}`);
   201→              }
   202→            }
   203→            if (data.engineRunning !== undefined) {
   204→              setEngineRunning(data.engineRunning);
   205→            }
   206→            // Update session stats from engine
   207→            if (data.engineStats) {
   208→              setBotSession((prev) => ({
   209→                totalScans: data.engineStats.totalScans ?? prev?.totalScans ?? 0,
   210→                totalBetsPlaced: data.engineStats.totalBetsPlaced ?? prev?.totalBetsPlaced ?? 0,
   211→                totalStakeUsed: data.engineStats.totalStakeUsed ?? prev?.totalStakeUsed ?? 0,
   212→                totalProfit: data.engineStats.totalProfit ?? prev?.totalProfit ?? 0,
   213→                lastScanAt: data.engineStats.lastScanAt ?? prev?.lastScanAt ?? null,
   214→                lastBetAt: data.engineStats.lastBetAt ?? prev?.lastBetAt ?? null,
   215→                startedAt: data.engineStats.startedAt ?? prev?.startedAt ?? null,
   216→              }));
   217→            } else if (data.session) {
   218→              setBotSession(data.session);
   219→            }
   220→            // Check for new bets placed
   221→            const prevBets = botSession?.totalBetsPlaced ?? 0;
   222→            const newBetsPlaced = (data.engineStats?.totalBetsPlaced ?? data.session?.totalBetsPlaced ?? 0) - prevBets;
   223→            if (newBetsPlaced > 0) {
   224→              addToast("success", `Bot placed ${newBetsPlaced} bet(s)!`);
   225→              refetchBets();
   226→            }
   227→          }
   228→        } catch {
   229→          // Ignore poll errors
   230→        }
   231→      }, 15000); // 15-second status poll (engine scans independently)
   232→    } else {
   233→      // Stop polling
   234→      if (pollIntervalRef.current) {
   235→        clearInterval(pollIntervalRef.current);
   236→        pollIntervalRef.current = null;
   237→      }
   238→    }
   239→    return () => {
   240→      if (pollIntervalRef.current) {
   241→        clearInterval(pollIntervalRef.current);
   242→        pollIntervalRef.current = null;
   243→      }
   244→    };
   245→  }, [botRunning, addToast, refetchBets]);
   246→
   247→  // Start the bot
   248→  const handleStartBot = async () => {
   249→    if (!user?.id) {
   250→      addToast("error", "Please log in to run the bot");
   251→      return;
   252→    }
   253→    setBotLoading(true);
   254→    try {
   255→      const res = await fetch("/api/bot/control", {
   256→        method: "POST",
   257→        headers: { "Content-Type": "application/json" },
   258→        body: JSON.stringify({ action: "start" }),
   259→      });
   260→      const data = await res.json();
   261→      if (res.ok && data.success) {
   262→        setBotRunning(true);
   263→        if (data.firstScan?.betsPlaced > 0) {
   264→          addToast("success", data.message);
   265→        } else {
   266→          addToast("success", "AI Bot started! It will scan for matches and place bets automatically.");
   267→        }
   268→        refetchBets();
   269→      } else {
   270→        addToast("error", data.error || "Failed to start bot");
   271→      }
   272→    } catch {
   273→      addToast("error", "Failed to start bot");
   274→    } finally {
   275→      setBotLoading(false);
   276→    }
   277→  };
   278→
   279→  // Stop the bot
   280→  const handleStopBot = async () => {
   281→    setBotLoading(true);
   282→    try {
   283→      const res = await fetch("/api/bot/control", {
   284→        method: "POST",
   285→        headers: { "Content-Type": "application/json" },
   286→        body: JSON.stringify({ action: "stop" }),
   287→      });
   288→      const data = await res.json();
   289→      if (res.ok && data.success) {
   290→        setBotRunning(false);
   291→        addToast("info", data.message || "Bot stopped.");
   292→      } else {
   293→        addToast("error", data.error || "Failed to stop bot");
   294→      }
   295→    } catch {
   296→      addToast("error", "Failed to stop bot");
   297→    } finally {
   298→      setBotLoading(false);
   299→    }
   300→  };
   301→
   302→  const handleCashout = async (betId: string, type: "full" | "partial" = "full") => {
   303→    try {
   304→      const res = await fetch("/api/cashout", {
   305→        method: "POST",
   306→        headers: { "Content-Type": "application/json" },
   307→        body: JSON.stringify({ betId, cashoutType: type }),
   308→      });
   309→      if (res.ok) {
   310→        const result = await res.json();
   311→        refetchBets();
   312→        addToast("success", `${type === "partial" ? "Partial" : "Full"} cashout processed: $${result.amount?.toFixed(2)}`);
   313→      } else {
   314→        const data = await res.json();
   315→        addToast("error", data.error || "Cashout failed");
   316→      }
   317→    } catch (error) {
   318→      console.error("Cashout failed:", error);
   319→      addToast("error", "Cashout failed");
   320→    }
   321→  };
   322→
   323→  const handleSettle = async () => {
   324→    if (!user?.id) return;
   325→    setSettleLoading(true);
   326→    try {
   327→      const res = await fetch("/api/settle", {
   328→        method: "POST",
   329→        headers: { "Content-Type": "application/json" },
   330→        body: JSON.stringify({ userId: user.id }),
   331→      });
   332→      if (res.ok) {
   333→        const result = await res.json();
   334→        if (result.settled > 0) {
   335→          addToast("success", `Settled ${result.settled} bet(s). Profit: $${result.totalProfit?.toFixed(2)}`);
   336→        } else {
   337→          addToast("info", "No bets to settle");
   338→        }
   339→        refetchBets();
   340→      }
   341→    } catch {
   342→      addToast("error", "Settlement failed");
   343→    } finally {
   344→      setSettleLoading(false);
   345→    }
   346→  };
   347→
   348→  const handleSaveSettings = async (newSettings: UserSettings) => {
   349→    setSettings(newSettings);
   350→    try {
   351→      await fetch("/api/settings", {
   352→        method: "PUT",
   353→        headers: { "Content-Type": "application/json" },
   354→        body: JSON.stringify(newSettings),
   355→      });
   356→      addToast("success", "Settings saved successfully!");
   357→    } catch {
   358→      addToast("error", "Failed to save settings");
   359→    }
   360→  };
   361→
   362→  // Merge betting account info
   363→  const enrichedBets = bets.map((bet) => {
   364→    const account = accounts.find((a) => a.id === bet.bettingAccountId);
   365→    return {
   366→      ...bet,
   367→      bettingAccount: account ? { platform: account.platform } : undefined,
   368→    };
   369→  });
   370→
   371→  const filteredBets =
   372→    betFilter === "all"
   373→      ? enrichedBets
   374→      : betFilter === "auto"
   375→      ? enrichedBets.filter((b) => b.isAutoPlaced)
   376→      : betFilter === "manual"
   377→      ? enrichedBets.filter((b) => !b.isAutoPlaced)
   378→      : betFilter === "accumulator"
   379→      ? enrichedBets.filter((b) => b.accumulatorId)
   380→      : enrichedBets.filter((b) => b.status === betFilter);
   381→
   382→  const autoBets = enrichedBets.filter((b) => b.isAutoPlaced);
   383→  const accumulatorBets = enrichedBets.filter((b) => b.accumulatorId);
   384→  const totalStake = enrichedBets.reduce((sum, b) => sum + b.stake, 0);
   385→  const totalProfit = enrichedBets.reduce((sum, b) => sum + (b.profit || 0), 0);
   386→
   387→  // AI recommended bets
   388→  const aiRecommendedBets = enrichedBets
   389→    .filter((b) => b.status === "pending" && b.aiConfidence && b.aiConfidence > 0.5)
   390→    .sort((a, b) => (b.aiConfidence || 0) - (a.aiConfidence || 0));
   391→
   392→  // Daily bet limit progress
   393→  const todayBets = enrichedBets.filter((b) => {
   394→    const today = new Date();
   395→    const placedAt = new Date(b.placedAt);
   396→    return placedAt.toDateString() === today.toDateString();
   397→  });
   398→  const dailyStake = todayBets.reduce((sum, b) => sum + b.stake, 0);
   399→  const dailyLimitProgress = Math.min((dailyStake / settings.dailyBetLimit) * 100, 100);
   400→
   401→  if (loading) {
   402→    return (
   403→      <div className="flex items-center justify-center h-64">
   404→        <div className="flex items-center gap-2 text-muted-foreground">
   405→          <div className="h-5 w-5 border-2 border-primary border-t-transparent rounded-full animate-spin" />
   406→          Loading...
   407→        </div>
   408→      </div>
   409→    );
   410→  }
   411→
   412→  return (
   413→    <div className="space-y-6">
   414→      {/* Auto-Betting Disabled Banner */}
   415→      {!settings.autoBettingEnabled && !botRunning && (
   416→        <div className="rounded-xl border-2 border-amber-400/30 bg-amber-400/5 p-4">
   417→          <div className="flex items-start justify-between gap-4">
   418→            <div className="flex items-start gap-3">
   419→              <div className="flex h-8 w-8 shrink-0 items-center justify-center rounded-full bg-amber-400/10">
   420→                <Zap className="h-4 w-4 text-amber-400" />
   421→              </div>
   422→              <div>
   423→                <p className="text-sm font-medium text-amber-400">AI Auto-Betting is Disabled</p>
   424→                <p className="text-xs text-muted-foreground mt-1">
   425→                  Toggle the switch to enable the AI bot, then click "Start Bot" to begin scanning for matches automatically.
   426→                </p>
   427→              </div>
   428→            </div>
   429→            <div className="flex items-center gap-2 shrink-0">
   430→              <span className="text-xs text-muted-foreground">Enable</span>
   431→              <Switch
   432→                checked={settings.autoBettingEnabled}
   433→                onCheckedChange={async (checked) => {
   434→                  const updated = { ...settings, autoBettingEnabled: checked as boolean };
   435→                  setSettings(updated);
   436→                  try {
   437→                    await fetch("/api/settings", {
   438→                      method: "PUT",
   439→                      headers: { "Content-Type": "application/json" },
   440→                      body: JSON.stringify({ autoBettingEnabled: checked }),
   441→                    });
   442→                    addToast("success", checked ? "Auto-betting enabled! You can now start the bot." : "Auto-betting disabled");
   443→                  } catch {
   444→                    addToast("error", "Failed to update setting");
   445→                    setSettings(settings); // revert
   446→                  }
   447→                }}
   448→              />
   449→            </div>
   450→          </div>
   451→        </div>
   452→      )}
   453→
   454→      {/* Bot Running Banner */}
   455→      {botRunning && (
   456→        <div className="rounded-xl border-2 border-red-500/30 bg-red-500/5 p-4">
   457→          <div className="flex items-center justify-between gap-4">
   458→            <div className="flex items-center gap-3">
   459→              <div className="flex h-8 w-8 shrink-0 items-center justify-center rounded-full bg-red-500/10">
   460→                <Radio className="h-4 w-4 text-red-500 animate-pulse" />
   461→              </div>
   462→              <div>
   463→                <p className="text-sm font-medium text-red-400">AI Bot is Running</p>
   464→                <p className="text-xs text-muted-foreground mt-0.5">
   465→                  {engineRunning
   466→                    ? "Running in background — scans happen automatically even when you close this page."
   467→                    : "Session active but engine not detected — try restarting the bot."}{" "}
   468→                  {botSession?.totalScans ? `${botSession.totalScans} scan(s) completed` : "First scan in progress"}...
   469→                  {botSession?.totalBetsPlaced ? ` ${botSession.totalBetsPlaced} bet(s) placed.` : ""}
   470→                </p>
   471→              </div>
   472→            </div>
   473→            <div className="flex items-center gap-3 shrink-0">
   474→              {botSession?.lastScanAt && (
   475→                <span className="text-[10px] text-muted-foreground">
   476→                  Last scan: {new Date(botSession.lastScanAt).toLocaleTimeString()}
   477→                </span>
   478→              )}
   479→              <Badge className={`text-xs animate-pulse ${
   480→                engineRunning
   481→                  ? "bg-red-500/10 text-red-400 border-red-500/20"
   482→                  : "bg-amber-400/10 text-amber-400 border-amber-400/20"
   483→              }`}>
   484→                <Activity className="h-3 w-3 mr-1" />
   485→                {engineRunning ? "Live" : "Reconnect"}
   486→              </Badge>
   487→            </div>
   488→          </div>
   489→        </div>
   490→      )}
   491→
   492→      <div className="flex items-center justify-between">
   493→        <div>
   494→          <h1 className="text-2xl font-bold text-foreground">Automated Betting</h1>
   495→          <p className="text-sm text-muted-foreground mt-1">
   496→            {botRunning ? "AI Bot is scanning for the best opportunities" : "Configure auto-betting and manage your bets"}
   497→          </p>
   498→        </div>
   499→        <div className="flex items-center gap-2">
   500→          {/* Bot Status Indicator */}
   501→          <Badge className={`px-3 py-1.5 ${
   502→            botRunning
   503→              ? "bg-red-500/10 text-red-400 border-red-500/20"
   504→              : settings.autoBettingEnabled
   505→              ? "bg-primary/10 text-primary border-primary/20"
   506→              : "bg-secondary text-muted-foreground border-border"
   507→          }`}>
   508→            <span className="relative flex h-2 w-2 mr-2">
   509→              {botRunning ? (
   510→                <>
   511→                  <span className="animate-live-pulse absolute inline-flex h-full w-full rounded-full bg-red-500 opacity-75" />
   512→                  <span className="relative inline-flex rounded-full h-2 w-2 bg-red-500" />
   513→                </>
   514→              ) : settings.autoBettingEnabled ? (
   515→                <>
   516→                  <span className="animate-live-pulse absolute inline-flex h-full w-full rounded-full bg-primary opacity-75" />
   517→                  <span className="relative inline-flex rounded-full h-2 w-2 bg-primary" />
   518→                </>
   519→              ) : (
   520→                <span className="relative inline-flex rounded-full h-2 w-2 bg-muted-foreground" />
   521→              )}
   522→            </span>
   523→            {botRunning ? (engineRunning ? "Running" : "Session Active") : settings.autoBettingEnabled ? "Ready" : "Inactive"}
   524→          </Badge>
   525→
   526→          {/* Start/Stop Bot Button */}
   527→          {botRunning ? (
   528→            <Button
   529→              size="sm"
   530→              className="bg-red-600 hover:bg-red-700 text-white"
   531→              onClick={handleStopBot}
   532→              disabled={botLoading}
   533→            >
   534→              {botLoading ? (
   535→                <div className="h-4 w-4 border-2 border-white border-t-transparent rounded-full animate-spin" />
   536→              ) : (
   537→                <Square className="h-4 w-4" />
   538→              )}
   539→              {botLoading ? "Stopping..." : "Stop Bot"}
   540→            </Button>
   541→          ) : (
   542→            <Button
   543→              size="sm"
   544→              className="bg-primary text-primary-foreground hover:bg-primary/80"
   545→              onClick={handleStartBot}
   546→              disabled={botLoading || !settings.autoBettingEnabled}
   547→            >
   548→              {botLoading ? (
   549→                <div className="h-4 w-4 border-2 border-primary-foreground border-t-transparent rounded-full animate-spin" />
   550→              ) : (
   551→                <Play className="h-4 w-4" />
   552→              )}
   553→              {botLoading ? "Starting..." : "Start Bot"}
   554→            </Button>
   555→          )}
   556→
   557→          {/* Settle Button */}
