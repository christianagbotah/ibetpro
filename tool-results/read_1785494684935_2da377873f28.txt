     1→"use client";
     2→
     3→import { useState, useCallback, useMemo } from "react";
     4→import { useFetch } from "@/lib/hooks";
     5→import { useToast } from "@/components/ui/toast";
     6→import { Card, CardContent, CardHeader, CardTitle } from "@/components/ui/card";
     7→import { Button } from "@/components/ui/button";
     8→import { Badge } from "@/components/ui/badge";
     9→import { Input } from "@/components/ui/input";
    10→import { Label } from "@/components/ui/label";
    11→import { Separator } from "@/components/ui/separator";
    12→import {
    13→  Dialog,
    14→  DialogContent,
    15→  DialogHeader,
    16→  DialogTitle,
    17→} from "@/components/ui/dialog";
    18→import {
    19→  Link2, RefreshCw, Wallet, Shield, CheckCircle2,
    20→  AlertCircle, Key, User, Lock, ChevronRight,
    21→  Search, MapPin, Smartphone, Star, ArrowLeft,
    22→  FlaskConical, Globe, Loader2,
    23→} from "lucide-react";
    24→import { BrokerLogo } from "@/components/broker/broker-logo";
    25→import {
    26→  REGIONS,
    27→  getPlatformsForRegion,
    28→  getCurrencyForRegion,
    29→  getContinents,
    30→  searchRegions,
    31→  type RegionInfo,
    32→  type BrokerPlatformInfo,
    33→} from "@/lib/regions";
    34→
    35→interface ConnectedAccount {
    36→  id: string;
    37→  platform: string;
    38→  platformName: string;
    39→  accountName: string;
    40→  isConnected: boolean;
    41→  sessionValid: boolean;
    42→  needsRefresh: boolean;
    43→  sessionExpiry: string | null;
    44→  brokerType: string;
    45→  brokerRegion: string | null;
    46→  balance: number;
    47→  currency: string;
    48→  allocatedAmount: number;
    49→  allocationLock: boolean;
    50→  totalBrokerBets: number;
    51→  totalBrokerProfit: number;
    52→  lastBetPlacedAt: string | null;
    53→  features: {
    54→    liveBetting: boolean;
    55→    cashout: boolean;
    56→    partialCashout: boolean;
    57→    accumulators: boolean;
    58→    maxAccumulatorLegs: number;
    59→    minStake: number;
    60→    maxStake: number;
    61→  } | null;
    62→}
    63→
    64→type ConnectStep = "region" | "platform" | "credentials";
    65→
    66→export function BrokerConnect() {
    67→  const { addToast } = useToast();
    68→  const [connectDialogOpen, setConnectDialogOpen] = useState(false);
    69→  const [step, setStep] = useState<ConnectStep>("region");
    70→  const [selectedRegion, setSelectedRegion] = useState<string>("");
    71→  const [selectedPlatform, setSelectedPlatform] = useState<string | null>(null);
    72→  const [connecting, setConnecting] = useState(false);
    73→  const [credentials, setCredentials] = useState({
    74→    username: "",
    75→    password: "",
    76→    apiKey: "",
    77→    token: "",
    78→  });
    79→  const [accountName, setAccountName] = useState("");
    80→  const [regionSearch, setRegionSearch] = useState("");
    81→  const [brokerSearch, setBrokerSearch] = useState("");
    82→  const [selectedContinent, setSelectedContinent] = useState<string>("all");
    83→
    84→  const { data, loading, refetch } = useFetch<{
    85→    accounts: ConnectedAccount[];
    86→    availablePlatforms: BrokerPlatformInfo[];
    87→    brokerMode: "demo" | "real";
    88→  }>("/api/broker/connect", { accounts: [], availablePlatforms: [], brokerMode: "demo" });
    89→
    90→  const accounts = data?.accounts || [];
    91→  const brokerMode = data?.brokerMode || "demo";
    92→
    93→  // Get filtered regions based on search and continent
    94→  const filteredRegions = useMemo(() => {
    95→    let regions = regionSearch ? searchRegions(regionSearch) : REGIONS;
    96→    if (selectedContinent !== "all") {
    97→      regions = regions.filter((r) => r.continent === selectedContinent);
    98→    }
    99→    // Deduplicate by region code to prevent React key warnings
   100→    const seen = new Set<string>();
   101→    return regions.filter((r) => {
   102→      if (seen.has(r.code)) return false;
   103→      seen.add(r.code);
   104→      return true;
   105→    });
   106→  }, [regionSearch, selectedContinent]);
   107→
   108→  // Get platforms for selected region
   109→  const platformsForRegion = useMemo(() => {
   110→    if (!selectedRegion) return [];
   111→    return getPlatformsForRegion(selectedRegion);
   112→  }, [selectedRegion]);
   113→
   114→  // Filter platforms by search query
   115→  const filteredPlatforms = useMemo(() => {
   116→    if (!brokerSearch.trim()) return platformsForRegion;
   117→    const q = brokerSearch.toLowerCase().trim();
   118→    return platformsForRegion.filter(
   119→      (p) =>
   120→        p.name.toLowerCase().includes(q) ||
   121→        p.id.toLowerCase().includes(q) ||
   122→        p.authType.toLowerCase().includes(q)
   123→    );
   124→  }, [platformsForRegion, brokerSearch]);
   125→
   126→  const selectedRegionInfo = useMemo(() => {
   127→    return REGIONS.find((r) => r.code === selectedRegion);
   128→  }, [selectedRegion]);
   129→
   130→  const selectedPlatformData = platformsForRegion.find((p) => p.id === selectedPlatform);
   131→
   132→  const handleConnect = useCallback(async () => {
   133→    if (!selectedPlatform) return;
   134→    setConnecting(true);
   135→
   136→    try {
   137→      const res = await fetch("/api/broker/connect", {
   138→        method: "POST",
   139→        headers: { "Content-Type": "application/json" },
   140→        body: JSON.stringify({
   141→          platformId: selectedPlatform,
   142→          credentials,
   143→          region: selectedRegion,
   144→          accountName: accountName || undefined,
   145→        }),
   146→      });
   147→
   148→      const result = await res.json();
   149→
   150→      if (result.success) {
   151→        addToast("success", `${result.account.accountName} connected successfully!`);
   152→        setConnectDialogOpen(false);
   153→        resetForm();
   154→        refetch();
   155→      } else {
   156→        addToast("error", result.error || "Failed to connect broker");
   157→      }
   158→    } catch {
   159→      addToast("error", "Failed to connect broker");
   160→    } finally {
   161→      setConnecting(false);
   162→    }
   163→  }, [selectedPlatform, credentials, accountName, selectedRegion, addToast, refetch]);
   164→
   165→  const handleSync = useCallback(async (accountId: string) => {
   166→    try {
   167→      const res = await fetch("/api/broker/sync", {
   168→        method: "POST",
   169→        headers: { "Content-Type": "application/json" },
   170→        body: JSON.stringify({ bettingAccountId: accountId }),
   171→      });
   172→
   173→      const result = await res.json();
   174→
   175→      if (result.success) {
   176→        const currency = result.balance?.currency || "USD";
   177→        addToast("success", `Account synced! Balance: ${currency} ${result.balance.total.toFixed(2)}`);
   178→        refetch();
   179→      } else {
   180→        addToast("error", result.error || "Sync failed");
   181→      }
   182→    } catch {
   183→      addToast("error", "Sync error occurred");
   184→    }
   185→  }, [addToast, refetch]);
   186→
   187→  const resetForm = () => {
   188→    setCredentials({ username: "", password: "", apiKey: "", token: "" });
   189→    setAccountName("");
   190→    setSelectedPlatform(null);
   191→    setStep("region");
   192→    setRegionSearch("");
   193→    setBrokerSearch("");
   194→    setSelectedContinent("all");
   195→  };
   196→
   197→  const handleDialogClose = (open: boolean) => {
   198→    setConnectDialogOpen(open);
   199→    if (!open) resetForm();
   200→  };
   201→
   202→  const [switchingMode, setSwitchingMode] = useState(false);
   203→
   204→  // Switch broker mode
   205→  const handleBrokerModeSwitch = useCallback(async (mode: "demo" | "real") => {
   206→    setSwitchingMode(true);
   207→    try {
   208→      const res = await fetch("/api/user/broker-mode", {
   209→        method: "PATCH",
   210→        headers: { "Content-Type": "application/json" },
   211→        body: JSON.stringify({ brokerMode: mode }),
   212→      });
   213→      if (res.ok) {
   214→        addToast("success", `Switched to ${mode === "demo" ? "Demo (Sandbox)" : "Live (Real)"} mode`);
   215→        refetch();
   216→      } else {
   217→        addToast("error", "Failed to switch broker mode");
   218→      }
   219→    } catch {
   220→      addToast("error", "Failed to switch broker mode");
   221→    } finally {
   222→      setSwitchingMode(false);
   223→    }
   224→  }, [addToast, refetch]);
   225→
   226→  const formatBalance = (amount: number, currency: string) => {
   227→    const region = REGIONS.find((r) => r.currencyCode === currency);
   228→    const symbol = region?.currencySymbol || "$";
   229→    return `${symbol}${amount.toLocaleString(undefined, { minimumFractionDigits: 2, maximumFractionDigits: 2 })}`;
   230→  };
   231→
   232→  const continents = getContinents();
   233→
   234→  return (
   235→    <div className="space-y-6">
   236→      <Card className="bg-card border-border">
   237→        <CardHeader className="pb-3">
   238→          <div className="flex items-center justify-between">
   239→            <CardTitle className="flex items-center gap-2 text-base">
   240→              <Link2 className="h-4 w-4 text-primary" />
   241→              Connected Brokers
   242→              <button
   243→                onClick={() => handleBrokerModeSwitch(brokerMode === "demo" ? "real" : "demo")}
   244→                disabled={switchingMode}
   245→                className={`inline-flex items-center gap-1 rounded-full px-2 py-0.5 text-[10px] font-medium border transition-all cursor-pointer ${
   246→                  brokerMode === "demo"
   247→                    ? "bg-blue-400/10 text-blue-400 border-blue-400/30 hover:bg-blue-400/20"
   248→                    : "bg-emerald-400/10 text-emerald-400 border-emerald-400/30 hover:bg-emerald-400/20"
   249→                } ${switchingMode ? "opacity-50" : ""}`}
   250→                title={`Click to switch to ${brokerMode === "demo" ? "Live" : "Demo"} mode`}
   251→              >
   252→                {switchingMode ? (
   253→                  <Loader2 className="h-3 w-3 animate-spin" />
   254→                ) : brokerMode === "demo" ? (
   255→                  <FlaskConical className="h-3 w-3" />
   256→                ) : (
   257→                  <Globe className="h-3 w-3" />
   258→                )}
   259→                {brokerMode === "demo" ? "Sandbox" : "Live"}
   260→              </button>
   261→            </CardTitle>
   262→            <Dialog open={connectDialogOpen} onOpenChange={handleDialogClose}>
   263→              <Button
   264→                size="sm"
   265→                className="bg-primary text-primary-foreground"
   266→                onClick={() => setConnectDialogOpen(true)}
   267→              >
   268→                <Link2 className="h-3.5 w-3.5 mr-1.5" />
   269→                Connect Broker
   270→              </Button>
   271→
   272→              <DialogContent className="max-w-lg max-h-[90vh] overflow-y-auto sm:max-h-[90vh] max-[640px]:h-screen max-[640px]:max-h-screen max-[640px]:w-screen max-[640px]:max-w-screen max-[640px]:rounded-none max-[640px]:p-4">
   273→                <DialogHeader>
   274→                  <DialogTitle>Connect Your Broker Account</DialogTitle>
   275→                </DialogHeader>
   276→
   277→                {/* Step indicator */}
   278→                <div className="flex items-center gap-2 mb-2">
   279→                  <div className={`flex items-center gap-1.5 text-xs ${step === "region" ? "text-primary font-medium" : "text-muted-foreground"}`}>
   280→                    <div className={`h-5 w-5 rounded-full flex items-center justify-center text-[10px] font-bold ${step === "region" ? "bg-primary text-primary-foreground" : "bg-secondary text-muted-foreground"}`}>
   281→                      {step !== "region" ? "✓" : "1"}
   282→                    </div>
   283→                    Region
   284→                  </div>
   285→                  <div className="h-px flex-1 bg-border" />
   286→                  <div className={`flex items-center gap-1.5 text-xs ${step === "platform" ? "text-primary font-medium" : "text-muted-foreground"}`}>
   287→                    <div className={`h-5 w-5 rounded-full flex items-center justify-center text-[10px] font-bold ${step === "platform" ? "bg-primary text-primary-foreground" : step === "credentials" ? "bg-primary text-primary-foreground" : "bg-secondary text-muted-foreground"}`}>
   288→                      {step === "credentials" ? "✓" : "2"}
   289→                    </div>
   290→                    Platform
   291→                  </div>
   292→                  <div className="h-px flex-1 bg-border" />
   293→                  <div className={`flex items-center gap-1.5 text-xs ${step === "credentials" ? "text-primary font-medium" : "text-muted-foreground"}`}>
   294→                    <div className={`h-5 w-5 rounded-full flex items-center justify-center text-[10px] font-bold ${step === "credentials" ? "bg-primary text-primary-foreground" : "bg-secondary text-muted-foreground"}`}>
   295→                      3
   296→                    </div>
   297→                    Connect
   298→                  </div>
   299→                </div>
   300→
   301→                {/* ====== STEP 1: REGION SELECTION ====== */}
   302→                {step === "region" && (
   303→                  <div className="space-y-3">
   304→                    <p className="text-sm text-muted-foreground">
   305→                      Select your country or region. This helps us show betting platforms available in your area and set the right currency.
   306→                    </p>
   307→
   308→                    <div className="relative">
   309→                      <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   310→                      <Input
   311→                        value={regionSearch}
   312→                        onChange={(e) => setRegionSearch(e.target.value)}
   313→                        placeholder="Search country or currency..."
   314→                        className="bg-secondary border-border pl-9"
   315→                      />
   316→                    </div>
   317→
   318→                    <div className="flex flex-wrap gap-1.5">
   319→                      <button
   320→                        onClick={() => setSelectedContinent("all")}
   321→                        className={`rounded-full px-3 py-1 text-xs font-medium transition-all ${
   322→                          selectedContinent === "all"
   323→                            ? "bg-primary/10 text-primary border border-primary/30"
   324→                            : "bg-secondary/50 text-muted-foreground border border-border"
   325→                        }`}
   326→                      >
   327→                        All
   328→                      </button>
   329→                      {continents.map((c) => (
   330→                        <button
   331→                          key={c}
   332→                          onClick={() => setSelectedContinent(c)}
   333→                          className={`rounded-full px-3 py-1 text-xs font-medium transition-all ${
   334→                            selectedContinent === c
   335→                              ? "bg-primary/10 text-primary border border-primary/30"
   336→                              : "bg-secondary/50 text-muted-foreground border border-border"
   337→                          }`}
   338→                        >
   339→                          {c}
   340→                        </button>
   341→                      ))}
   342→                    </div>
   343→
   344→                    <div className="grid grid-cols-2 sm:grid-cols-3 gap-2 max-h-64 overflow-y-auto -webkit-overflow-scrolling-touch">
   345→                      {filteredRegions.map((region) => {
   346→                        const platformCount = getPlatformsForRegion(region.code).length;
   347→                        return (
   348→                          <button
   349→                            key={region.code}
   350→                            onClick={() => {
   351→                              setSelectedRegion(region.code);
   352→                              setStep("platform");
   353→                            }}
   354→                            className={`flex items-center gap-2 rounded-lg border p-2.5 transition-all text-left hover:bg-primary/5 active:scale-[0.98] ${
   355→                              selectedRegion === region.code
   356→                                ? "border-primary/30 bg-primary/5"
   357→                                : "border-border"
   358→                            }`}
   359→                          >
   360→                            <span className="text-lg">{region.flag}</span>
   361→                            <div className="min-w-0 flex-1">
   362→                              <p className="text-xs font-medium text-foreground truncate">{region.name}</p>
   363→                              <p className="text-[10px] text-muted-foreground">
   364→                                {region.currencySymbol} {region.currencyCode} · {platformCount} platforms
   365→                              </p>
   366→                            </div>
   367→                          </button>
   368→                        );
   369→                      })}
   370→                    </div>
   371→
   372→                    {filteredRegions.length === 0 && (
   373→                      <div className="text-center py-4">
   374→                        <p className="text-sm text-muted-foreground">No regions found matching your search</p>
   375→                      </div>
   376→                    )}
   377→                  </div>
   378→                )}
   379→
   380→                {/* ====== STEP 2: PLATFORM SELECTION ====== */}
   381→                {step === "platform" && selectedRegionInfo && (
   382→                  <div className="space-y-3">
   383→                    <button
   384→                      onClick={() => {
   385→                        setStep("region");
   386→                        setSelectedPlatform(null);
   387→                      }}
   388→                      className="flex items-center gap-1 text-xs text-muted-foreground hover:text-primary transition-colors"
   389→                    >
   390→                      <ArrowLeft className="h-3 w-3" /> Change region
   391→                    </button>
   392→
   393→                    <div className="flex items-center gap-2 p-2.5 rounded-lg bg-primary/5 border border-primary/10">
   394→                      <span className="text-lg">{selectedRegionInfo.flag}</span>
   395→                      <div>
   396→                        <p className="text-sm font-medium text-foreground">{selectedRegionInfo.name}</p>
   397→                        <p className="text-xs text-muted-foreground">
   398→                          Currency: {selectedRegionInfo.currencySymbol} {selectedRegionInfo.currencyCode} ({selectedRegionInfo.currencyName})
   399→                        </p>
   400→                      </div>
   401→                    </div>
   402→
   403→                    <p className="text-sm text-muted-foreground">
   404→                      {platformsForRegion.length} platforms available in {selectedRegionInfo.name}. Select your betting platform.
   405→                    </p>
   406→
   407→                    <div className="relative">
   408→                      <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   409→                      <Input
   410→                        type="text"
   411→                        placeholder="Search brokers..."
   412→                        value={brokerSearch}
   413→                        onChange={(e) => setBrokerSearch(e.target.value)}
   414→                        className="pl-9 h-9 text-sm"
   415→                      />
   416→                    </div>
   417→
   418→                    <div className="space-y-2 max-h-64 overflow-y-auto -webkit-overflow-scrolling-touch">
   419→                      {filteredPlatforms.length === 0 ? (
   420→                        <div className="text-center py-4 text-sm text-muted-foreground">
   421→                          No brokers found matching &quot;{brokerSearch}&quot;
   422→                        </div>
   423→                      ) : (
   424→                        filteredPlatforms.map((platform) => {
   425→                        const isPopular = platform.popularIn.includes(selectedRegion);
   426→                        return (
   427→                          <button
   428→                            key={platform.id}
   429→                            onClick={() => {
   430→                              setSelectedPlatform(platform.id);
   431→                              setStep("credentials");
   432→                            }}
   433→                            className="flex items-center justify-between rounded-lg border border-border p-3 hover:bg-primary/5 active:scale-[0.98] transition-colors text-left w-full"
   434→                          >
   435→                            <div className="flex items-center gap-3">
   436→                              <BrokerLogo platform={platform} size={40} className="shrink-0" />
   437→                              <div>
   438→                                <div className="flex items-center gap-1.5">
   439→                                  <p className="text-sm font-medium text-foreground">{platform.name}</p>
   440→                                  {isPopular && (
   441→                                    <Badge className="bg-amber-400/10 text-amber-400 border-amber-400/30 text-[9px] px-1 py-0">
   442→                                      <Star className="h-2.5 w-2.5 mr-0.5" /> Popular
   443→                                    </Badge>
   444→                                  )}
   445→                                </div>
   446→                                <p className="text-xs text-muted-foreground">
   447→                                  {platform.authType.replace(/_/g, " ")} · {platform.mobileApp ? "📱 App" : "Web only"}
   448→                                </p>
   449→                              </div>
   450→                            </div>
   451→                            <ChevronRight className="h-4 w-4 text-muted-foreground shrink-0" />
   452→                          </button>
   453→                        );
   454→                      })
   455→                      )}
   456→                    </div>
   457→                  </div>
   458→                )}
   459→
   460→                {/* ====== STEP 3: CREDENTIALS ====== */}
   461→                {step === "credentials" && selectedPlatformData && selectedRegionInfo && (
   462→                  <div className="space-y-4">
   463→                    <button
   464→                      onClick={() => {
   465→                        setStep("platform");
   466→                        setSelectedPlatform(null);
   467→                      }}
   468→                      className="flex items-center gap-1 text-xs text-muted-foreground hover:text-primary transition-colors"
   469→                    >
   470→                      <ArrowLeft className="h-3 w-3" /> Change platform
   471→                    </button>
   472→
   473→                    <div className="flex items-center gap-3 p-3 rounded-lg bg-primary/5 border border-primary/10">
   474→                      <BrokerLogo platform={selectedPlatformData} size={40} className="shrink-0" />
   475→                      <div className="flex-1 min-w-0">
   476→                        <p className="text-sm font-medium text-foreground">{selectedPlatformData.name}</p>
   477→                        <p className="text-xs text-muted-foreground">
   478→                          {selectedPlatformData.authType.replace(/_/g, " ")} authentication
   479→                        </p>
   480→                      </div>
   481→                      <div className="text-right shrink-0">
   482→                        <p className="text-xs text-muted-foreground">{selectedRegionInfo.flag} {selectedRegionInfo.name}</p>
   483→                        <p className="text-xs font-medium text-primary">{selectedRegionInfo.currencySymbol} {selectedRegionInfo.currencyCode}</p>
   484→                      </div>
   485→                    </div>
   486→
   487→                    {/* Sandbox/Demo mode banner */}
   488→                    {brokerMode === "demo" && (
   489→                      <div className="flex items-center gap-2 p-2.5 rounded-lg bg-blue-400/5 border border-blue-400/20 text-xs text-blue-400">
   490→                        <FlaskConical className="h-4 w-4 shrink-0" />
   491→                        <div>
   492→                          <span className="font-medium">Sandbox Mode</span> — Your credentials are simulated. No real API calls will be made. Switch to Live mode in Settings for real broker connections.
   493→                        </div>
   494→                      </div>
   495→                    )}
   496→
   497→                    <div className="space-y-3">
   498→                      <div>
   499→                        <Label className="text-xs text-muted-foreground">Account Name</Label>
   500→                        <Input
   501→                          value={accountName}
   502→                          onChange={(e) => setAccountName(e.target.value)}
   503→                          placeholder={`My ${selectedPlatformData?.name} Account`}
   504→                          className="bg-secondary border-border mt-1"
   505→                        />
   506→                      </div>
   507→
   508→                      {brokerMode === "demo" ? (
   509→                        <div className="rounded-lg bg-blue-400/5 border border-blue-400/10 p-3">
   510→                          <p className="text-xs text-blue-400">
   511→                            <Shield className="h-3 w-3 inline mr-1" />
   512→                            Sandbox Mode: Any credentials will work for testing. Switch to Real mode in the Mode tab to use live broker APIs.
   513→                          </p>
   514→                        </div>
   515→                      ) : (
   516→                        <div className="rounded-lg bg-amber-400/5 border border-amber-400/10 p-3">
   517→                          <p className="text-xs text-amber-400">
   518→                            <Shield className="h-3 w-3 inline mr-1" />
   519→                            Live Mode: Your real broker credentials will be used to connect. Real API calls will be made.
   520→                          </p>
   521→                        </div>
   522→                      )}
   523→
   524→                      <Separator />
   525→
