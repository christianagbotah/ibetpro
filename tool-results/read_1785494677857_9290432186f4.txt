   127→// ==================== RETRY LOGIC ====================
   128→
   129→interface RetryConfig {
   130→  maxRetries: number;
   131→  baseDelayMs: number;
   132→  maxDelayMs: number;
   133→  retryableErrors: string[];
   134→}
   135→
   136→const DEFAULT_RETRY_CONFIG: RetryConfig = {
   137→  maxRetries: 3,
   138→  baseDelayMs: 1000,
   139→  maxDelayMs: 10000,
   140→  retryableErrors: [
   141→    "ETIMEDOUT",
   142→    "ECONNRESET",
   143→    "ECONNREFUSED",
   144→    "rate_limit",
   145→    "server_error",
   146→    "session_expired",
   147→    "temporarily_unavailable",
   148→  ],
   149→};
   150→
   151→async function withRetry<T>(
   152→  fn: () => Promise<T>,
   153→  retryConfig: Partial<RetryConfig> = {}
   154→): Promise<T> {
   155→  const cfg = { ...DEFAULT_RETRY_CONFIG, ...retryConfig };
   156→  let lastError: Error | null = null;
   157→
   158→  for (let attempt = 0; attempt <= cfg.maxRetries; attempt++) {
   159→    try {
   160→      return await fn();
   161→    } catch (error) {
   162→      lastError = error instanceof Error ? error : new Error(String(error));
   163→
   164→      // Check if error is retryable
   165→      const isRetryable = cfg.retryableErrors.some(
   166→        (err) => lastError!.message.toLowerCase().includes(err.toLowerCase())
   167→      );
   168→
   169→      if (!isRetryable || attempt === cfg.maxRetries) {
   170→        throw lastError;
   171→      }
   172→
   173→      // Exponential backoff with jitter
   174→      const delay = Math.min(
   175→        cfg.baseDelayMs * Math.pow(2, attempt) + Math.random() * 1000,
   176→        cfg.maxDelayMs
   177→      );
   178→
   179→      await new Promise((resolve) => setTimeout(resolve, delay));
   180→    }
   181→  }
   182→
   183→  throw lastError;
   184→}

   186→// ==================== IDEMPOTENCY ====================
   187→
   188→// In-memory idempotency store (use Redis in production)
   189→const idempotencyStore = new Map<string, { result: unknown; timestamp: number }>();
   190→const IDEMPOTENCY_TTL = 3600000; // 1 hour
   191→
   192→function checkIdempotency<T>(key: string): T | null {
   193→  const entry = idempotencyStore.get(key);
   194→  if (!entry) return null;
   195→
   196→  if (Date.now() - entry.timestamp > IDEMPOTENCY_TTL) {
   197→    idempotencyStore.delete(key);
   198→    return null;
   199→  }
   200→
   201→  return entry.result as T;
   202→}
   203→
   204→function setIdempotencyResult(key: string, result: unknown): void {
   205→  idempotencyStore.set(key, { result, timestamp: Date.now() });
   206→
   207→  // Clean up old entries periodically
   208→  if (idempotencyStore.size > 10000) {
   209→    const now = Date.now();
   210→    for (const [k, v] of idempotencyStore) {
   211→      if (now - v.timestamp > IDEMPOTENCY_TTL) {
   212→        idempotencyStore.delete(k);
   213→      }
   214→    }
   215→  }
   216→}

   218→// ==================== BASE ADAPTER ====================
   219→
   220→abstract class BaseBrokerAdapter implements BrokerAdapter {
   221→  abstract platformId: string;
   222→  abstract platform: BrokerPlatform;
   223→
   224→  // Per-instance demo mode override (set by getBrokerAdapter)
   225→  _demoMode?: boolean;
   226→
   227→  /**
   228→   * Check if this adapter should operate in demo mode.
   229→   * Per-instance override takes priority over global config.
   230→   */
   231→  protected isDemoMode(): boolean {
   232→    if (this._demoMode !== undefined) return this._demoMode;
   233→    return config.broker.demoMode;
   234→  }
   235→
   236→  abstract authenticate(credentials: BrokerCredentials): Promise<BrokerAuthResult>;
   237→  abstract refreshSession(refreshToken: string): Promise<BrokerAuthResult>;
   238→  abstract getBalance(accessToken: string, regionCode?: string): Promise<BrokerBalance>;
   239→  abstract getProfile(accessToken: string): Promise<BrokerUserProfile>;
   240→  abstract placeBet(accessToken: string, bet: BrokerBetRequest): Promise<BrokerBetResult>;
   241→  abstract getBetStatus(accessToken: string, brokerBetId: string): Promise<BrokerBetStatus>;
   242→  abstract cashout(accessToken: string, brokerBetId: string, type: "full" | "partial", partialPercent?: number): Promise<BrokerCashoutResult>;
   243→  abstract transferToWallet(accessToken: string, amount: number, walletAddress: string, reference: string): Promise<BrokerTransferResult>;
   244→  abstract getAvailableMarkets(accessToken: string, matchId: string): Promise<BrokerMarket[]>;
   245→  abstract getLiveEvents(accessToken: string): Promise<BrokerLiveEvent[]>;
   246→
   247→  validateSession(sessionToken: string, sessionExpiry: Date): BrokerSession {
   248→    return validateBrokerSession(sessionToken, sessionExpiry);
   249→  }
   250→
   251→  // Common helper: make authenticated API call with retry
   252→  protected async apiCall<T>(
   253→    accessToken: string,
   254→    method: string,
   255→    path: string,
   256→    body?: unknown,
   257→    options?: { idempotencyKey?: string }
   258→  ): Promise<T> {
   259→    const url = `${this.platform.baseUrl}${path}`;
   260→
   261→    // Check idempotency
   262→    if (options?.idempotencyKey) {
   263→      const cached = checkIdempotency<T>(options.idempotencyKey);
   264→      if (cached) return cached;
   265→    }
   266→
   267→    const result = await withRetry(async () => {
   268→      const response = await fetch(url, {
   269→        method,
   270→        headers: {
   271→          "Authorization": `Bearer ${accessToken}`,
   272→          "Content-Type": "application/json",
   273→          "X-Idempotency-Key": options?.idempotencyKey || "",
   274→          "X-Platform": this.platformId,
   275→          "X-App-Version": "1.0.0",
   276→        },
   277→        body: body ? JSON.stringify(body) : undefined,
   278→      });
   279→
   280→      if (!response.ok) {
   281→        const errorData = await response.json().catch(() => ({}));
   282→        const errorCode = errorData.code || errorData.error || `HTTP_${response.status}`;
   283→        const errorMessage = errorData.message || errorData.error || `API error: ${response.status}`;
   284→
   285→        // Map common error codes
   286→        if (response.status === 401) {
   287→          throw new Error(`session_expired: ${errorMessage}`);
   288→        }
   289→        if (response.status === 429) {
   290→          throw new Error(`rate_limit: ${errorMessage}`);
   291→        }
   292→        if (response.status >= 500) {
   293→          throw new Error(`server_error: ${errorMessage}`);
   294→        }
   295→
   296→        throw new Error(`${errorCode}: ${errorMessage}`);
   297→      }
   298→
   299→      return response.json() as Promise<T>;
   300→    });
   301→
   302→    // Store idempotency result
   303→    if (options?.idempotencyKey) {
   304→      setIdempotencyResult(options.idempotencyKey, result);
   305→    }
   306→
   307→    return result;
   308→  }
   309→}

   313→class OAuthBrokerAdapter extends BaseBrokerAdapter {
   314→  platformId: string;

   318→    super();

   323→  async authenticate(credentials: BrokerCredentials): Promise<BrokerAuthResult> {
   324→    if (!credentials.token) {
   325→      return { success: false, error: "OAuth authorization code required" };
   326→    }
   327→
   328→    // Demo/Sandbox mode: simulate successful authentication without real API calls
   329→    if (this.isDemoMode()) {
   330→      return {
   331→        success: true,
   332→        accessToken: `demo_oauth_${this.platformId}_${Date.now()}`,
   333→        refreshToken: `demo_refresh_${this.platformId}_${Date.now()}`,
   334→        sessionExpiry: new Date(Date.now() + 3600000),
   335→        brokerUserId: `demo_oauth_${credentials.token.substring(0, 8)}_${Date.now()}`,
   336→      };
   337→    }
   338→
   339→    try {
   340→      // In production, exchange authorization code for access token
   341→      // POST /oauth/token with client_id, client_secret, code, redirect_uri
   342→      const result = await withRetry(async () => {
   343→        const response = await fetch(`${this.platform.baseUrl}/oauth/token`, {
   344→          method: "POST",
   345→          headers: { "Content-Type": "application/json" },
   346→          body: JSON.stringify({
   347→            grant_type: "authorization_code",
   348→            code: credentials.token,
   349→            client_id: process.env[`${this.platformId.toUpperCase()}_CLIENT_ID`],
   350→            client_secret: process.env[`${this.platformId.toUpperCase()}_CLIENT_SECRET`],
   351→            redirect_uri: `${config.platform.url}/api/broker/callback/${this.platformId}`,
   352→          }),
   353→        });
   354→
   355→        if (!response.ok) {
   356→          throw new Error(`OAuth token exchange failed: ${response.status}`);
   357→        }
   358→
   359→        return response.json();
   360→      });
   361→
   362→      return {
   363→        success: true,
   364→        accessToken: result.access_token,
   365→        refreshToken: result.refresh_token,
   366→        sessionExpiry: new Date(Date.now() + (result.expires_in || 3600) * 1000),
   367→        brokerUserId: result.user_id,
   368→      };
   369→    } catch (error) {
   370→      return {
   371→        success: false,
   372→        error: error instanceof Error ? error.message : "OAuth authentication failed",
   373→      };
   374→    }
   375→  }
   376→
   377→  async refreshSession(refreshToken: string): Promise<BrokerAuthResult> {
   378→    try {
   379→      const result = await withRetry(async () => {
   380→        const response = await fetch(`${this.platform.baseUrl}/oauth/token`, {
   381→          method: "POST",
   382→          headers: { "Content-Type": "application/json" },
   383→          body: JSON.stringify({
   384→            grant_type: "refresh_token",
   385→            refresh_token: refreshToken,
   386→            client_id: process.env[`${this.platformId.toUpperCase()}_CLIENT_ID`],
   387→            client_secret: process.env[`${this.platformId.toUpperCase()}_CLIENT_SECRET`],
   388→          }),
   389→        });
   390→
   391→        if (!response.ok) {
   392→          throw new Error(`Token refresh failed: ${response.status}`);
   393→        }
   394→
   395→        return response.json();
   396→      });
   397→
   398→      return {
   399→        success: true,
   400→        accessToken: result.access_token,
   401→        refreshToken: result.refresh_token || refreshToken,
   402→        sessionExpiry: new Date(Date.now() + (result.expires_in || 3600) * 1000),
   403→      };
   404→    } catch (error) {
   405→      return {
   406→        success: false,
   407→        error: error instanceof Error ? error.message : "Session refresh failed",
   408→      };
   409→    }

   934→// ==================== ADAPTER FACTORY ====================
   935→
   936→const adapterCache = new Map<string, BrokerAdapter>();
   937→
   938→/**
   939→ * Get or create a broker adapter for the given platform
   940→ * @param platformId - The platform ID
   941→ * @param demoMode - If true, the adapter will use demo/sandbox mode (no real API calls)
   942→ */
   943→export function getBrokerAdapter(platformId: string, demoMode?: boolean): BrokerAdapter | null {
   944→  // Check cache (but demo mode changes per-user, so we need a keyed cache)
   945→  const cacheKey = demoMode !== undefined ? `${platformId}_${demoMode ? "demo" : "real"}` : platformId;
   946→  if (adapterCache.has(cacheKey)) {
   947→    return adapterCache.get(cacheKey)!;
   948→  }
   949→
   950→  // Manual adapter
   951→  if (platformId === "manual") {
   952→    const adapter = new ManualBrokerAdapter();
   953→    adapterCache.set(cacheKey, adapter);
   954→    return adapter;
   955→  }
   956→
   957→  // Get platform config
   958→  const platform = getBrokerPlatform(platformId);
   959→  if (!platform) {
   960→    console.error(`Unknown broker platform: ${platformId}`);
   961→    return null;
   962→  }
   963→
   964→  // Create appropriate adapter based on auth type
   965→  let adapter: BrokerAdapter;
   966→
   967→  switch (platform.authType) {
   968→    case "oauth":
   969→      adapter = new OAuthBrokerAdapter(platform);
   970→      break;
   971→    case "api_key":
   972→      adapter = new ApiKeyBrokerAdapter(platform);
   973→      break;
   974→    case "web_session":
   975→      adapter = new WebSessionBrokerAdapter(platform);
   976→      break;
   977→    case "manual":
   978→      adapter = new ManualBrokerAdapter();
   979→      break;
   980→    default:
   981→      console.error(`Unsupported auth type: ${platform.authType}`);
   982→      return null;
   983→  }
   984→
   985→  // Override demo mode if specified
   986→  if (demoMode !== undefined) {
   987→    (adapter as any)._demoMode = demoMode;
   988→  }
   989→
   990→  adapterCache.set(cacheKey, adapter);
   991→  return adapter;
   992→}

   994→/**
   995→ * Get all available broker adapters
   996→ * Uses lazy import to avoid circular dependency at module load time
   997→ */
   998→export function getAvailableAdapters(): BrokerAdapter[] {
   999→  const adapters: BrokerAdapter[] = [new ManualBrokerAdapter()];
  1000→
  1001→  // Lazy import to avoid circular dependency at module load time
  1002→  // eslint-disable-next-line @typescript-eslint/no-require-imports
  1003→  const { BROKER_PLATFORMS } = require("./regions") as { BROKER_PLATFORMS: Array<{ id: string }> };
  1004→
  1005→  const seen = new Set<string>();
  1006→  for (const platform of BROKER_PLATFORMS) {
  1007→    if (seen.has(platform.id)) continue;
  1008→    seen.add(platform.id);
  1009→    const adapter = getBrokerAdapter(platform.id);
  1010→    if (adapter) {
  1011→      adapters.push(adapter);
  1012→    }
  1013→  }
  1014→
  1015→  return adapters;
  1016→}

  1020→/**
  1021→ * Place a bet on a broker with full safety checks
  1022→ */
  1023→export async function safePlaceBet(
  1024→  platformId: string,
  1025→  accessToken: string,
  1026→  bet: BrokerBetRequest,
  1027→  maxStake: number
  1028→): Promise<BrokerBetResult> {
  1029→  // Validate stake limits
  1030→  if (bet.stake <= 0) {
  1031→    return { success: false, error: "Stake must be positive" };
  1032→  }
  1033→  if (bet.stake > maxStake) {
  1034→    return { success: false, error: `Stake exceeds maximum allowed (${maxStake})` };
  1035→  }
  1036→
  1037→  // Validate odds
  1038→  if (bet.odds < 1.01) {
  1039→    return { success: false, error: "Invalid odds" };
  1040→  }
  1041→
  1042→  const adapter = getBrokerAdapter(platformId);
  1043→  if (!adapter) {
  1044→    return { success: false, error: `No adapter for platform: ${platformId}` };
  1045→  }
  1046→
  1047→  // Generate idempotency key
  1048→  const idempotencyKey = `${platformId}_${bet.matchId}_${bet.selection}_${bet.stake}_${Date.now()}`;
  1049→
  1050→  return adapter.placeBet(accessToken, {
  1051→    ...bet,
  1052→    idempotencyKey,
  1053→  });
  1054→}
