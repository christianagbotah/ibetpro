  1170→export function getPlatformsForRegion(regionCode: string): BrokerPlatformInfo[] {
  1171→  return BROKER_PLATFORMS.filter(
  1172→    (p) => (p.regions && p.regions.includes(regionCode)) || (p.regions && p.regions.includes("global"))
  1173→  );
  1174→}
  1175→
  1176→export function getPopularPlatformsForRegion(regionCode: string): BrokerPlatformInfo[] {
  1177→  const platforms = getPlatformsForRegion(regionCode);
  1178→  // Sort by popularity in the region first, then by name
  1179→  return platforms.sort((a, b) => {
  1180→    const aPopular = a.popularIn?.includes(regionCode) ? 1 : 0;
  1181→    const bPopular = b.popularIn?.includes(regionCode) ? 1 : 0;
  1182→    if (bPopular !== aPopular) return bPopular - aPopular;
  1183→    return a.name.localeCompare(b.name);
  1184→  });
  1185→}

  1187→export function getCurrencyForRegion(regionCode: string): { code: string; symbol: string; name: string } {
  1188→  const region = getRegionInfo(regionCode);
  1189→  if (region) {
  1190→    return { code: region.currencyCode, symbol: region.currencySymbol, name: region.currencyName };
  1191→  }
  1192→  return { code: "USD", symbol: "$", name: "Dollar" };
  1193→}

  1195→export function formatCurrency(amount: number, currencyCode: string): string {
  1196→  const region = REGIONS.find((r) => r.currencyCode === currencyCode);
  1197→  const symbol = region?.currencySymbol || "$";
  1198→  return `${symbol}${amount.toLocaleString(undefined, { minimumFractionDigits: 2, maximumFractionDigits: 2 })}`;
  1199→}

  1205→export function searchRegions(query: string): RegionInfo[] {

  1208→    (r) =>
  1209→      r.name.toLowerCase().includes(q) ||

  1212→      r.continent.toLowerCase().includes(q)
  1213→  );
  1214→}
