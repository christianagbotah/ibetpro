   127→// ==================== RETRY LOGIC ====================
   128→
   129→interface RetryConfig {
   130→  maxRetries: number;
   131→  baseDelayMs: number;
   132→  maxDelayMs: number;
   133→  retryableErrors: string[];
   134→}
   135→
   136→const DEFAULT_RETRY_CONFIG: RetryConfig = {
   137→  maxRetries: 3,
   138→  baseDelayMs: 1000,
   139→  maxDelayMs: 10000,
   140→  retryableErrors: [
   141→    "ETIMEDOUT",
   142→    "ECONNRESET",
   143→    "ECONNREFUSED",
   144→    "rate_limit",
   145→    "server_error",
   146→    "session_expired",
   147→    "temporarily_unavailable",
   148→  ],
   149→};
   150→
   151→async function withRetry<T>(
   152→  fn: () => Promise<T>,
   153→  retryConfig: Partial<RetryConfig> = {}
   154→): Promise<T> {
   155→  const cfg = { ...DEFAULT_RETRY_CONFIG, ...retryConfig };
   156→  let lastError: Error | null = null;
   157→
   158→  for (let attempt = 0; attempt <= cfg.maxRetries; attempt++) {
   159→    try {
   160→      return await fn();
   161→    } catch (error) {
   162→      lastError = error instanceof Error ? error : new Error(String(error));
   163→
   164→      // Check if error is retryable
   165→      const isRetryable = cfg.retryableErrors.some(
   166→        (err) => lastError!.message.toLowerCase().includes(err.toLowerCase())
   167→      );
   168→
   169→      if (!isRetryable || attempt === cfg.maxRetries) {
   170→        throw lastError;
   171→      }
   172→
   173→      // Exponential backoff with jitter
   174→      const delay = Math.min(
   175→        cfg.baseDelayMs * Math.pow(2, attempt) + Math.random() * 1000,
   176→        cfg.maxDelayMs
   177→      );
   178→
   179→      await new Promise((resolve) => setTimeout(resolve, delay));
   180→    }
   181→  }
   182→
   183→  throw lastError;
   184→}

   186→// ==================== IDEMPOTENCY ====================
   187→
   188→// In-memory idempotency store (use Redis in production)
   189→const idempotencyStore = new Map<string, { result: unknown; timestamp: number }>();
   190→const IDEMPOTENCY_TTL = 3600000; // 1 hour
   191→
   192→function checkIdempotency<T>(key: string): T | null {
   193→  const entry = idempotencyStore.get(key);
   194→  if (!entry) return null;
   195→
   196→  if (Date.now() - entry.timestamp > IDEMPOTENCY_TTL) {
   197→    idempotencyStore.delete(key);
   198→    return null;
   199→  }
   200→
   201→  return entry.result as T;
   202→}
   203→
   204→function setIdempotencyResult(key: string, result: unknown): void {
   205→  idempotencyStore.set(key, { result, timestamp: Date.now() });
   206→
   207→  // Clean up old entries periodically
   208→  if (idempotencyStore.size > 10000) {
   209→    const now = Date.now();
   210→    for (const [k, v] of idempotencyStore) {
   211→      if (now - v.timestamp > IDEMPOTENCY_TTL) {
   212→        idempotencyStore.delete(k);
   213→      }
   214→    }
   215→  }
   216→}

   218→// ==================== BASE ADAPTER ====================
   219→
   220→abstract class BaseBrokerAdapter implements BrokerAdapter {
   221→  abstract platformId: string;
   222→  abstract platform: BrokerPlatform;
   223→
   224→  // Per-instance demo mode override (set by getBrokerAdapter)
   225→  _demoMode?: boolean;
   226→
   227→  /**
   228→   * Check if this adapter should operate in demo mode.
   229→   * Per-instance override takes priority over global config.
   230→   */
   231→  protected isDemoMode(): boolean {
   232→    if (this._demoMode !== undefined) return this._demoMode;
   233→    return config.broker.demoMode;
   234→  }
   235→
   236→  abstract authenticate(credentials: BrokerCredentials): Promise<BrokerAuthResult>;
   237→  abstract refreshSession(refreshToken: string): Promise<BrokerAuthResult>;
   238→  abstract getBalance(accessToken: string, regionCode?: string): Promise<BrokerBalance>;
   239→  abstract getProfile(accessToken: string): Promise<BrokerUserProfile>;
   240→  abstract placeBet(accessToken: string, bet: BrokerBetRequest): Promise<BrokerBetResult>;
   241→  abstract getBetStatus(accessToken: string, brokerBetId: string): Promise<BrokerBetStatus>;
   242→  abstract cashout(accessToken: string, brokerBetId: string, type: "full" | "partial", partialPercent?: number): Promise<BrokerCashoutResult>;
   243→  abstract transferToWallet(accessToken: string, amount: number, walletAddress: string, reference: string): Promise<BrokerTransferResult>;
   244→  abstract getAvailableMarkets(accessToken: string, matchId: string): Promise<BrokerMarket[]>;
   245→  abstract getLiveEvents(accessToken: string): Promise<BrokerLiveEvent[]>;
   246→
   247→  validateSession(sessionToken: string, sessionExpiry: Date): BrokerSession {
   248→    return validateBrokerSession(sessionToken, sessionExpiry);
   249→  }
   250→
   251→  // Common helper: make authenticated API call with retry
   252→  protected async apiCall<T>(
   253→    accessToken: string,
   254→    method: string,
   255→    path: string,
   256→    body?: unknown,
   257→    options?: { idempotencyKey?: string }
   258→  ): Promise<T> {
   259→    const url = `${this.platform.baseUrl}${path}`;
   260→
   261→    // Check idempotency
   262→    if (options?.idempotencyKey) {
   263→      const cached = checkIdempotency<T>(options.idempotencyKey);
   264→      if (cached) return cached;
   265→    }
   266→
   267→    const result = await withRetry(async () => {
   268→      const response = await fetch(url, {
   269→        method,
   270→        headers: {
   271→          "Authorization": `Bearer ${accessToken}`,
   272→          "Content-Type": "application/json",
   273→          "X-Idempotency-Key": options?.idempotencyKey || "",
   274→          "X-Platform": this.platformId,
   275→          "X-App-Version": "1.0.0",
   276→        },
   277→        body: body ? JSON.stringify(body) : undefined,
   278→      });
   279→
   280→      if (!response.ok) {
   281→        const errorData = await response.json().catch(() => ({}));
   282→        const errorCode = errorData.code || errorData.error || `HTTP_${response.status}`;
   283→        const errorMessage = errorData.message || errorData.error || `API error: ${response.status}`;
   284→
   285→        // Map common error codes
   286→        if (response.status === 401) {
   287→          throw new Error(`session_expired: ${errorMessage}`);
   288→        }
   289→        if (response.status === 429) {
   290→          throw new Error(`rate_limit: ${errorMessage}`);
   291→        }
   292→        if (response.status >= 500) {
   293→          throw new Error(`server_error: ${errorMessage}`);
   294→        }
   295→
   296→        throw new Error(`${errorCode}: ${errorMessage}`);
   297→      }
   298→
   299→      return response.json() as Promise<T>;
   300→    });
   301→
   302→    // Store idempotency result
   303→    if (options?.idempotencyKey) {
   304→      setIdempotencyResult(options.idempotencyKey, result);
   305→    }
   306→
   307→    return result;
   308→  }
   309→}

   314→  platformId: string;

   323→  async authenticate(credentials: BrokerCredentials): Promise<BrokerAuthResult> {
   324→    // Demo/Sandbox mode: simulate successful authentication without real API calls
   325→    // MUST be checked BEFORE credential validation — in demo mode, any credentials (even empty) work
   326→    if (this.isDemoMode()) {
   327→      const demoToken = credentials.token || `demo_${Date.now()}`;
   328→      return {
   329→        success: true,
   330→        accessToken: `demo_oauth_${this.platformId}_${Date.now()}`,
   331→        refreshToken: `demo_refresh_${this.platformId}_${Date.now()}`,
   332→        sessionExpiry: new Date(Date.now() + 3600000),
   333→        brokerUserId: `demo_oauth_${demoToken.substring(0, 8)}_${Date.now()}`,
   334→      };
   335→    }
   336→
   337→    if (!credentials.token) {
   338→      return { success: false, error: "OAuth authorization code required" };
   339→    }
   340→
   341→    try {
   342→      // In production, exchange authorization code for access token
   343→      // POST /oauth/token with client_id, client_secret, code, redirect_uri
   344→      const result = await withRetry(async () => {
   345→        const response = await fetch(`${this.platform.baseUrl}/oauth/token`, {
   346→          method: "POST",
   347→          headers: { "Content-Type": "application/json" },
   348→          body: JSON.stringify({
   349→            grant_type: "authorization_code",
   350→            code: credentials.token,
   351→            client_id: process.env[`${this.platformId.toUpperCase()}_CLIENT_ID`],
   352→            client_secret: process.env[`${this.platformId.toUpperCase()}_CLIENT_SECRET`],
   353→            redirect_uri: `${config.platform.url}/api/broker/callback/${this.platformId}`,
   354→          }),
   355→        });
   356→
   357→        if (!response.ok) {
   358→          throw new Error(`OAuth token exchange failed: ${response.status}`);
   359→        }
   360→
   361→        return response.json();
   362→      });
   363→
   364→      return {
   365→        success: true,
   366→        accessToken: result.access_token,
   367→        refreshToken: result.refresh_token,
   368→        sessionExpiry: new Date(Date.now() + (result.expires_in || 3600) * 1000),
   369→        brokerUserId: result.user_id,
   370→      };
   371→    } catch (error) {
   372→      return {
   373→        success: false,
   374→        error: error instanceof Error ? error.message : "OAuth authentication failed",
   375→      };
   376→    }
   377→  }

   393→          method: "POST",
   394→          headers: { "Content-Type": "application/json" },
   395→          body: JSON.stringify({
   396→            grant_type: "refresh_token",
   397→            refresh_token: refreshToken,
   398→            client_id: process.env[`${this.platformId.toUpperCase()}_CLIENT_ID`],

   404→          throw new Error(`Token refresh failed: ${response.status}`);

   407→        return response.json();
   408→      });
   409→
   410→      return {
   411→        success: true,
   412→        accessToken: result.access_token,
   413→        refreshToken: result.refresh_token || refreshToken,
   414→        sessionExpiry: new Date(Date.now() + (result.expires_in || 3600) * 1000),

   416→    } catch (error) {
   417→      return {
   418→        success: false,
   419→        error: error instanceof Error ? error.message : "Session refresh failed",

   963→// ==================== ADAPTER FACTORY ====================
   964→
   965→const adapterCache = new Map<string, BrokerAdapter>();
   966→
   967→/**
   968→ * Get or create a broker adapter for the given platform
   969→ * @param platformId - The platform ID
   970→ * @param demoMode - If true, the adapter will use demo/sandbox mode (no real API calls)
   971→ */
   972→export function getBrokerAdapter(platformId: string, demoMode?: boolean): BrokerAdapter | null {
   973→  // Check cache (but demo mode changes per-user, so we need a keyed cache)
   974→  const cacheKey = demoMode !== undefined ? `${platformId}_${demoMode ? "demo" : "real"}` : platformId;
   975→  if (adapterCache.has(cacheKey)) {
   976→    return adapterCache.get(cacheKey)!;
   977→  }
   978→
   979→  // Manual adapter
   980→  if (platformId === "manual") {
   981→    const adapter = new ManualBrokerAdapter();
   982→    adapterCache.set(cacheKey, adapter);
   983→    return adapter;
   984→  }
   985→
   986→  // Get platform config
   987→  const platform = getBrokerPlatform(platformId);
   988→  if (!platform) {
   989→    console.error(`Unknown broker platform: ${platformId}`);
   990→    return null;
   991→  }
   992→
   993→  // Create appropriate adapter based on auth type
   994→  let adapter: BrokerAdapter;
   995→
   996→  switch (platform.authType) {
   997→    case "oauth":
   998→      adapter = new OAuthBrokerAdapter(platform);
   999→      break;
  1000→    case "api_key":
  1001→      adapter = new ApiKeyBrokerAdapter(platform);
  1002→      break;
  1003→    case "web_session":
  1004→      adapter = new WebSessionBrokerAdapter(platform);
  1005→      break;
  1006→    case "manual":
  1007→      adapter = new ManualBrokerAdapter();
  1008→      break;
  1009→    default:
  1010→      console.error(`Unsupported auth type: ${platform.authType}`);
  1011→      return null;
  1012→  }
  1013→
  1014→  // Override demo mode if specified
  1015→  if (demoMode !== undefined) {
  1016→    (adapter as any)._demoMode = demoMode;
  1017→  }
  1018→
  1019→  adapterCache.set(cacheKey, adapter);
  1020→  return adapter;
  1021→}

  1023→/**
  1024→ * Get all available broker adapters
  1025→ * Uses lazy import to avoid circular dependency at module load time
  1026→ */
  1027→export function getAvailableAdapters(): BrokerAdapter[] {
  1028→  const adapters: BrokerAdapter[] = [new ManualBrokerAdapter()];
  1029→
  1030→  // Lazy import to avoid circular dependency at module load time
  1031→  // eslint-disable-next-line @typescript-eslint/no-require-imports
  1032→  const { BROKER_PLATFORMS } = require("./regions") as { BROKER_PLATFORMS: Array<{ id: string }> };
  1033→
  1034→  const seen = new Set<string>();
  1035→  for (const platform of BROKER_PLATFORMS) {
  1036→    if (seen.has(platform.id)) continue;
  1037→    seen.add(platform.id);
  1038→    const adapter = getBrokerAdapter(platform.id);
  1039→    if (adapter) {
  1040→      adapters.push(adapter);
  1041→    }
  1042→  }
  1043→
  1044→  return adapters;
  1045→}

  1052→export async function safePlaceBet(
  1053→  platformId: string,
  1054→  accessToken: string,
  1055→  bet: BrokerBetRequest,
  1056→  maxStake: number
  1057→): Promise<BrokerBetResult> {
  1058→  // Validate stake limits
  1059→  if (bet.stake <= 0) {
  1060→    return { success: false, error: "Stake must be positive" };
  1061→  }
  1062→  if (bet.stake > maxStake) {
  1063→    return { success: false, error: `Stake exceeds maximum allowed (${maxStake})` };
  1064→  }
  1065→
  1066→  // Validate odds
  1067→  if (bet.odds < 1.01) {
  1068→    return { success: false, error: "Invalid odds" };
  1069→  }
  1070→
  1071→  const adapter = getBrokerAdapter(platformId);
  1072→  if (!adapter) {
  1073→    return { success: false, error: `No adapter for platform: ${platformId}` };
  1074→  }
  1075→
  1076→  // Generate idempotency key
  1077→  const idempotencyKey = `${platformId}_${bet.matchId}_${bet.selection}_${bet.stake}_${Date.now()}`;
  1078→
  1079→  return adapter.placeBet(accessToken, {
  1080→    ...bet,
  1081→    idempotencyKey,
  1082→  });
  1083→}
