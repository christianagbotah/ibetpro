    56→function expandSportKeys(sports: string[]): string[] {
    57→  const expanded = new Set<string>();
    58→  for (const sport of sports) {
    59→    const lower = sport.toLowerCase().trim();
    60→    if (SPORT_CATEGORY_MAP[lower]) {
    61→      for (const key of SPORT_CATEGORY_MAP[lower]) {
    62→        expanded.add(key);
    63→      }
    64→    } else {
    65→      // Keep the original key as-is (might be a specific key like "soccer_epl")
    66→      expanded.add(lower);
    67→    }
    68→  }
    69→  return Array.from(expanded);
    70→}

    98→// ==================== SINGLETON ENGINE ====================
    99→
   100→/**
   101→ * BotEngine is a singleton that manages all running bot instances.
   102→ * Each user has their own timer that runs scan cycles on a configurable interval.
   103→ *
   104→ * The engine is designed to:
   105→ * - Run independently of the frontend (no polling required)
   106→ * - Auto-recover from errors (continues running after failed scans)
   107→ * - Auto-stop on risk limits, schedule violations, or no allocation
   108→ * - Persist state in the database (BotSession) for recovery after restarts
   109→ * - Support graceful shutdown
   110→ */
   111→class BotEngine {
   112→  private static instance: BotEngine | null = null;
   113→  private timers: Map<string, NodeJS.Timeout> = new Map();
   114→  private runningUsers: Map<string, BotEngineStats> = new Map();
   115→  private isShuttingDown = false;
   116→
   117→  private constructor() {
   118→    // Register graceful shutdown handlers
   119→    process.on("SIGTERM", () => this.shutdown());
   120→    process.on("SIGINT", () => this.shutdown());
   121→  }
   122→
   123→  static getInstance(): BotEngine {
   124→    if (!BotEngine.instance) {
   125→      BotEngine.instance = new BotEngine();
   126→    }
   127→    return BotEngine.instance;
   128→  }

   130→  // ==================== LIFECYCLE ====================
   131→
   132→  /**
   133→   * Start the bot for a specific user. Creates a timer that runs scan cycles
   134→   * on the configured interval (default 30 seconds).
   135→   */
   136→  async start(userId: string, scanIntervalSec: number = 30): Promise<{ success: boolean; message: string }> {
   137→    // Don't start if already running
   138→    if (this.timers.has(userId)) {
   139→      return { success: false, message: "Bot is already running for this user" };
   140→    }
   141→
   142→    // Validate prerequisites
   143→    const validation = await this.validatePrerequisites(userId);
   144→    if (!validation.valid) {
   145→      return { success: false, message: validation.error! };
   146→    }
   147→
   148→    // Initialize stats tracking
   149→    const stats: BotEngineStats = {
   150→      userId,
   151→      status: "running",
   152→      scanIntervalSec,
   153→      totalScans: 0,
   154→      totalBetsPlaced: 0,
   155→      totalStakeUsed: 0,
   156→      totalProfit: 0,
   157→      lastScanAt: null,
   158→      lastBetAt: null,
   159→      startedAt: new Date(),
   160→      errorCount: 0,
   161→      lastError: null,
   162→    };
   163→    this.runningUsers.set(userId, stats);
   164→
   165→    // Run the first scan immediately
   166→    try {
   167→      const result = await this.runScanCycle(userId);
   168→      stats.totalScans++;
   169→      stats.totalBetsPlaced += result.betsPlaced;
   170→      stats.totalStakeUsed += result.totalStake;
   171→      stats.lastScanAt = new Date();
   172→      if (result.betsPlaced > 0 || result.tipsGenerated > 0) {
   173→        stats.lastBetAt = new Date();
   174→      }
   175→      if (result.error) {
   176→        stats.errorCount++;
   177→        stats.lastError = result.error;
   178→      }
   179→    } catch (error) {
   180→      console.error(`[BotEngine] First scan failed for user ${userId}:`, error);
   181→      stats.errorCount++;
   182→      stats.lastError = error instanceof Error ? error.message : "Unknown error";
   183→    }
   184→
   185→    // Set up the recurring timer
   186→    const timer = setInterval(async () => {
   187→      if (this.isShuttingDown) return;
   188→
   189→      try {
   190→        const currentStats = this.runningUsers.get(userId);
   191→        if (!currentStats || currentStats.status !== "running") {
   192→          // Bot was stopped externally, clean up
   193→          this.stop(userId, "auto_stopped");
   194→          return;
   195→        }
   196→
   197→        // Check if the session is still running in the DB
   198→        const session = await prisma.botSession.findUnique({ where: { userId } });
   199→        if (!session || session.status !== "running") {
   200→          this.stop(userId, "session_expired");
   201→          return;
   202→        }
   203→
   204→        const result = await this.runScanCycle(userId);
   205→
   206→        currentStats.totalScans++;
   207→        currentStats.totalBetsPlaced += result.betsPlaced;
   208→        currentStats.totalStakeUsed += result.totalStake;
   209→        currentStats.lastScanAt = new Date();
   210→        currentStats.lastError = null; // Clear error on success
   211→
   212→        if (result.betsPlaced > 0) {
   213→          currentStats.lastBetAt = new Date();
   214→        }
   215→
   216→        if (result.error) {
   217→          currentStats.errorCount++;
   218→          currentStats.lastError = result.error;
   219→        }
   220→
   221→        // Update session stats in DB
   222→        await prisma.botSession.update({
   223→          where: { userId },
   224→          data: {
   225→            totalScans: { increment: 1 },
   226→            totalBetsPlaced: { increment: result.betsPlaced },
   227→            totalStakeUsed: { increment: result.totalStake },
   228→            lastScanAt: new Date(),
   229→            lastBetAt: result.betsPlaced > 0 ? new Date() : undefined,
   230→          },
   231→        });
   232→
   233→        // Log scan activity periodically (every 10 scans)
   234→        if (currentStats.totalScans % 10 === 0) {
   235→          await prisma.botLog.create({
   236→            data: {
   237→              userId,
   238→              action: "bot_scan",
   239→              reasoning: `Auto-scan #${currentStats.totalScans}: ${result.matches} matches found, ${result.betsPlaced} bets placed, ${result.skipped} skipped`,
   240→              details: JSON.stringify({
   241→                scanNumber: currentStats.totalScans,
   242→                betsPlaced: result.betsPlaced,
   243→                totalStake: result.totalStake,
   244→                matches: result.matches,
   245→                skipped: result.skipped,
   246→              }),
   247→            },
   248→          });
   249→        }
   250→
   251→        // Auto-settle finished matches every 3rd scan
   252→        if (currentStats.totalScans % 3 === 0) {
   253→          try {
   254→            await this.autoSettle(userId);
   255→          } catch (settleError) {
   256→            console.error(`[BotEngine] Auto-settle error for user ${userId}:`, settleError);
   257→          }
   258→        }
   259→
   260→        // Auto-cashout evaluation every 5th scan
   261→        if (currentStats.totalScans % 5 === 0) {
   262→          try {
   263→            await this.autoCashoutEvaluation(userId);
   264→          } catch (cashoutError) {
   265→            console.error(`[BotEngine] Auto-cashout error for user ${userId}:`, cashoutError);
   266→          }
   267→        }
   268→      } catch (error) {
   269→        console.error(`[BotEngine] Scan cycle error for user ${userId}:`, error);
   270→        const currentStats = this.runningUsers.get(userId);
   271→        if (currentStats) {
   272→          currentStats.errorCount++;
   273→          currentStats.lastError = error instanceof Error ? error.message : "Unknown error";
   274→
   275→          // Auto-stop after too many consecutive errors
   276→          if (currentStats.errorCount >= 10) {
   277→            console.error(`[BotEngine] Too many errors for user ${userId}, stopping bot`);
   278→            await this.stop(userId, "too_many_errors");
   279→          }
   280→        }
   281→      }
   282→    }, scanIntervalSec * 1000);
   283→
   284→    // Prevent Node.js from keeping the process alive just for this timer
   285→    if (timer.unref) {
   286→      timer.unref();
   287→    }
   288→
   289→    this.timers.set(userId, timer);
   290→
   291→    console.log(`[BotEngine] Bot started for user ${userId} with interval ${scanIntervalSec}s`);
   292→
   293→    // Log bot start
   294→    await prisma.botLog.create({
   295→      data: {
   296→        userId,
   297→        action: "bot_started",
   298→        reasoning: `Bot engine started in background mode. Scanning every ${scanIntervalSec}s.`,
   299→        details: JSON.stringify({
   300→          mode: "background",
   301→          scanIntervalSec,
   302→          broker: validation.connectedAccount?.platform,
   303→          allocation: validation.connectedAccount?.allocatedAmount,
   304→        }),
   305→      },
   306→    });
   307→
   308→    return {
   309→      success: true,
   310→      message: stats.tipsGenerated > 0
   311→        ? `Advisor started! Generated ${stats.tipsGenerated} tip(s) in first scan. Running in background.`
   312→        : stats.betsPlaced > 0
   313→        ? `Bot started! Placed ${stats.betsPlaced} bet(s) in first scan. Running in background.`
   314→        : "Bot started! Scanning for matches in background. Will generate tips automatically.",
   315→    };
   316→  }

   318→  /**
   319→   * Stop the bot for a specific user. Clears the timer and updates the session.
   320→   */
   321→  async stop(userId: string, reason: string = "user_stopped"): Promise<void> {
   322→    const timer = this.timers.get(userId);
   323→    if (timer) {
   324→      clearInterval(timer);
   325→      this.timers.delete(userId);
   326→    }
   327→
   328→    const stats = this.runningUsers.get(userId);
   329→    if (stats) {
   330→      stats.status = "stopped";
   331→      this.runningUsers.delete(userId);
   332→    }
   333→
   334→    // Update the database session
   335→    try {
   336→      await prisma.botSession.upsert({
   337→        where: { userId },
   338→        update: {
   339→          status: "stopped",
   340→          stoppedAt: new Date(),
   341→          stopReason: reason,
   342→        },
   343→        create: {
   344→          userId,
   345→          status: "stopped",
   346→          stoppedAt: new Date(),
   347→          stopReason: reason,
   348→        },
   349→      });
   350→
   351→      // Log bot stop
   352→      await prisma.botLog.create({
   353→        data: {
   354→          userId,
   355→          action: "bot_stopped",
   356→          reasoning: `Bot engine stopped. Reason: ${reason}`,
   357→          details: JSON.stringify({
   358→            reason,
   359→            totalScans: stats?.totalScans || 0,
   360→            totalBetsPlaced: stats?.totalBetsPlaced || 0,
   361→            totalStakeUsed: stats?.totalStakeUsed || 0,
   362→            totalProfit: stats?.totalProfit || 0,
   363→            runDuration: stats?.startedAt
   364→              ? Math.round((Date.now() - stats.startedAt.getTime()) / 60000)
   365→              : 0,
   366→            errorCount: stats?.errorCount || 0,
   367→          }),
   368→        },
   369→      });
   370→    } catch (error) {
   371→      console.error(`[BotEngine] Error stopping bot for user ${userId}:`, error);
   372→    }
   373→
   374→    console.log(`[BotEngine] Bot stopped for user ${userId}. Reason: ${reason}`);
   375→  }

   377→  /**
   378→   * Gracefully shut down all running bots.
   379→   */
   380→  async shutdown(): Promise<void> {
   381→    if (this.isShuttingDown) return;
   382→    this.isShuttingDown = true;
   383→
   384→    console.log(`[BotEngine] Shutting down ${this.timers.size} running bots...`);
   385→
   386→    for (const [userId] of this.timers) {
   387→      await this.stop(userId, "server_shutdown");
   388→    }
   389→
   390→    this.timers.clear();
   391→    this.runningUsers.clear();
   392→  }

   458→  // ==================== SCAN CYCLE ====================
   459→
   460→  /**
   461→   * Run a single scan cycle - analyze matches and generate tips or place bets.
   462→   * In advisor mode: AI finds value bets → creates Tip → sends Telegram alert.
   463→   * In auto mode: AI finds value bets → places bet via broker → creates Tip.
   464→   */
   465→  private async runScanCycle(userId: string): Promise<ScanResult> {
   466→    const result: ScanResult = { betsPlaced: 0, tipsGenerated: 0, totalStake: 0, matches: 0, skipped: 0 };
   467→
   468→    try {
   469→      // Refresh match data from external APIs (throttled to avoid rate limits)
   470→      try {
   471→        await syncMatchData(false);
   472→      } catch (syncErr) {
   473→        console.warn("[BotEngine] Match sync failed, continuing with existing data:", syncErr);
   474→      }
   475→
   476→      // Get user settings
   477→      const settings = await prisma.userSettings.findUnique({ where: { userId } });
   478→      if (!settings) {
   479→        result.error = "Settings not found";
   480→        return result;
   481→      }
   482→
   483→      const isAdvisorMode = settings.botMode === "advisor";
   484→
   485→      if (!settings.autoBettingEnabled && !isAdvisorMode) {
   486→        result.error = "Auto-betting is disabled";
   487→        await this.stop(userId, "auto_betting_disabled");
   488→        return result;
   489→      }
   490→
   491→      // Check schedule (using user's timezone setting)
   492→      if (!isWithinBetSchedule(settings.betScheduleStart, settings.betScheduleEnd, settings.timezone)) {
   493→        return result;
   494→      }
   495→
   496→      // Check risk limits
   497→      const user = await prisma.user.findUnique({ where: { id: userId } });
   498→      const riskCheck = checkRiskLimits(user?.dailyPnl || 0, user?.weeklyPnl || 0, {
   499→        stopLossDaily: settings.stopLossDaily,
   500→        stopLossWeekly: settings.stopLossWeekly,
   501→        profitTargetDaily: settings.profitTargetDaily,
   502→        profitTargetWeekly: settings.profitTargetWeekly,
   503→      });
   504→
   505→      if (!riskCheck.canBet) {
   506→        const reason = (user?.dailyPnl ?? 0) <= -settings.stopLossDaily ? "stop_loss" : "profit_target";
   507→        await this.stop(userId, reason);
   508→        await prisma.botLog.create({
   509→          data: {
   510→            userId,
   511→            action: reason === "stop_loss" ? "stop_loss_hit" : "profit_target_hit",
   512→            reasoning: riskCheck.reason,
   513→          },
   514→        });
   515→        return result;
   516→      }
   517→
   518→      // ---- Advisor mode: no broker required ----
   519→      // ---- Auto mode: need connected broker ----
   520→      let bettingAccount: Awaited<ReturnType<typeof prisma.bettingAccount.findFirst>> = null;
   521→      let activeAllocation: Awaited<ReturnType<typeof prisma.allocation.findFirst>> = null;
   522→      let remainingDailyLimit = settings.dailyBetLimit;
   523→      let existingBetMatchIds: string[] = [];
   524→
   525→      if (!isAdvisorMode) {
   526→        bettingAccount = await prisma.bettingAccount.findFirst({
   527→          where: { userId, isConnected: true },
   528→          orderBy: { allocatedAmount: "desc" },
   529→        });
   530→
   531→        if (!bettingAccount || bettingAccount.allocatedAmount <= 0) {
   532→          await this.stop(userId, "no_allocation");
   533→          await prisma.botLog.create({
   534→            data: {
   535→              userId,
   536→              action: "bot_scan",
   537→              reasoning: "No connected broker or allocation available. Stopping bot.",
   538→            },
   539→          });
   540→          return result;
   541→        }
   542→
   543→        // Check daily limit
   544→        const todayStart = new Date();
   545→        todayStart.setHours(0, 0, 0, 0);
   546→        const todayBets = await prisma.bet.findMany({
   547→          where: {
   548→            userId,
   549→            placedAt: { gte: todayStart },
   550→            status: { in: ["pending", "won", "lost", "cashed_out", "partial_cashout"] },
   551→          },
   552→        });
   553→        const dailyStake = todayBets.reduce((sum, b) => sum + b.stake, 0);
   554→        existingBetMatchIds = todayBets.map((b) => b.matchId);
   555→
   556→        if (dailyStake >= settings.dailyBetLimit) {
   557→          return result;
   558→        }
   559→
   560→        remainingDailyLimit = Math.min(
   561→          settings.dailyBetLimit - dailyStake,
   562→          bettingAccount.allocatedAmount
   563→        );
   564→
   565→        if (remainingDailyLimit < 5) {
   566→          return result;
   567→        }
   568→
   569→        activeAllocation = await prisma.allocation.findFirst({
   570→          where: { userId, bettingAccountId: bettingAccount.id, status: "active" },
   571→        });
   572→      } else {
   573→        // In advisor mode, check for existing tips to avoid duplicates

   576→        const todayTips = await prisma.tip.findMany({
   577→          where: { userId, createdAt: { gte: todayStart } },

   763→

   767→          // ============ AUTO MODE ============
   768→          else {
   769→            const stake = Math.min(
   770→              autoBetCheck.suggestedStake || settings.maxBetAmount * 0.5,

   780→            const brokerResult = await placeBetOnBroker(
   781→              bettingAccount!.platform,
   782→              bettingAccount!.accessToken || "",
   783→              {
   784→                matchId: match.id,

   787→                stake,

   796→                bettingAccountId: bettingAccount!.id,

   832→            });
   833→

   845→            });
   846→
   847→            // Update match AI data
   848→            await prisma.match.update({
   849→              where: { id: match.id },
   850→              data: {
   851→                aiHomeWinProb: prediction.homeWinProb,
   852→                aiDrawProb: prediction.drawProb,
   853→                aiAwayWinProb: prediction.awayWinProb,
   854→                aiConfidence: prediction.confidence,
   855→                aiRecommended: prediction.recommended,
   856→                aiAnalysis: prediction.analysis,

   865→              data: {
   866→                userId,
   867→                action: "bet_placed",
   868→                matchId: match.id,
   869→                betId: bet.id,
   870→                details: JSON.stringify({
   871→                  stake,
   872→                  odds: recOdds,
   873→                  selection,
   874→                  potentialWin,
   875→                  broker: bettingAccount!.platform,
   876→                  brokerBetId: brokerResult.brokerBetId,
   877→                }),
   878→                reasoning: autoBetCheck.reason,
   879→                confidence: prediction.confidence,
   880→                profitImpact: -stake,
   881→              },
   882→            });
   883→
   884→            // Also create a Tip for the auto-placed bet
   885→            try {
   886→              const tip = await prisma.tip.create({
   887→                data: {
   888→                  userId,
   889→                  matchId: match.id,
   890→                  sport: match.sport,
   891→                  league: match.league,
   892→                  homeTeam: match.homeTeam,
   893→                  awayTeam: match.awayTeam,
   894→                  selection,
   895→                  odds: recOdds,
   896→                  aiConfidence: prediction.confidence,
   897→                  valueEdge: prediction.valueEdge,
   898→                  kellyStake: prediction.kellyStake,
   899→                  riskLevel: settings.riskLevel,
   900→                  aiReasoning: prediction.analysis,
   901→                  tracked: true, // auto-placed = tracked
   902→                  commencesAt: match.commenceTime,
   903→                },
   904→              });
   905→
   906→              // Send Telegram + in-app notification
   907→              const tipAlert: TipAlert = {
   908→                sport: match.sport,
   909→                league: match.league,
   910→                homeTeam: match.homeTeam,
   911→                awayTeam: match.awayTeam,

   913→                odds: recOdds,
   914→                confidence: prediction.confidence,
   915→                valueEdge: prediction.valueEdge,
   916→                kellyStake: prediction.kellyStake,
   917→                riskLevel: settings.riskLevel,
   918→                matchTime: new Date(match.commenceTime).toLocaleString(),
   919→                matchId: match.id,
   920→              };
   921→              await sendTipAlert(userId, tipAlert);
   922→            } catch (tipError) {
   923→              console.error("[BotEngine] Failed to create tip/notification for auto-bet:", tipError);
   924→            }
   925→
   926→            result.betsPlaced++;
   927→            result.totalStake += stake;
   928→          }
   929→        } catch (matchError) {
   930→          console.error(`[BotEngine] Error processing match ${match.id}:`, matchError);
   931→          result.skipped++;
   932→        }
   933→      }
   934→
   935→      // Log scan summary
   936→      if (result.betsPlaced === 0 && result.tipsGenerated === 0 && result.skipped > 0) {
   937→        await prisma.botLog.create({
   938→          data: {
   939→            userId,
   940→            action: "bot_scan",
   941→            reasoning: `Scan completed: ${result.matches} matches found, ${result.skipped} skipped (no qualifying ${isAdvisorMode ? "tips" : "bets"})`,
   942→          },
   943→        });
   944→      }
   945→
   946→    } catch (error) {
   947→      console.error(`[BotEngine] Scan cycle error for user ${userId}:`, error);
   948→      result.error = error instanceof Error ? error.message : "Unknown error";
   949→    }
   950→
   951→    return result;
   952→  }

   956→  private async validatePrerequisites(userId: string): Promise<{
   957→    valid: boolean;
   958→    error?: string;
   959→    connectedAccount?: { platform: string; allocatedAmount: number };
   960→  }> {
   961→    const settings = await prisma.userSettings.findUnique({ where: { userId } });
   962→    if (!settings) {
   963→      return { valid: false, error: "Settings not found. Please configure your settings first." };
   964→    }
   965→
   966→    const isAdvisorMode = settings.botMode === "advisor";
   967→
   968→    if (!settings.autoBettingEnabled && !isAdvisorMode) {
   969→      return { valid: false, error: "Auto-betting is disabled. Enable it first." };
   970→    }
   971→
   972→    // In advisor mode, no broker is required — just need settings
   973→    if (isAdvisorMode) {
   974→      return { valid: true };
   975→    }
   976→
   977→    // Auto mode requires a connected broker
   978→    const connectedAccount = await prisma.bettingAccount.findFirst({
   979→      where: { userId, isConnected: true },
   980→    });
   981→
   982→    if (!connectedAccount) {
   983→      return { valid: false, error: "No connected broker account. Connect a broker and set allocation first." };
   984→    }
   985→
   986→    if (connectedAccount.allocatedAmount <= 0) {
   987→      return { valid: false, error: "No allocation set. Allocate funds from your broker account first." };
   988→    }
   989→
   990→    return {
   991→      valid: true,
   992→      connectedAccount: {
   993→        platform: connectedAccount.platform,
   994→        allocatedAmount: connectedAccount.allocatedAmount,
   995→      },
   996→    };
   997→  }

   999→  // ==================== AUTO SETTLEMENT ====================
  1000→
  1001→  /**
  1002→   * Automatically settle bets for finished matches.
  1003→   * Called periodically by the engine (every 3rd scan cycle).
  1004→   */
  1005→  private async autoSettle(userId: string): Promise<number> {
  1006→    const user = await prisma.user.findUnique({
  1007→      where: { id: userId },
  1008→      include: { settings: true },
  1009→    });
  1010→
  1011→    if (!user) return 0;
  1012→
  1013→    const commissionRate = user.settings?.commissionRate || 0.10;
  1014→
  1015→    // Find pending bets for finished matches
  1016→    const unsettledBets = await prisma.bet.findMany({
  1017→      where: {
  1018→        userId,
  1019→        status: "pending",
  1020→        match: { status: "finished" },
  1021→      },
  1022→      include: { match: true, bettingAccount: true },
  1023→    });
  1024→
  1025→    if (unsettledBets.length === 0) return 0;
  1026→
  1027→    let settledCount = 0;
  1028→
  1029→    for (const bet of unsettledBets) {
  1030→      if (!bet.match) continue;
  1031→
  1032→      const match = bet.match;
  1033→      const homeScore = match.homeScore ?? 0;
  1034→      const awayScore = match.awayScore ?? 0;
  1035→
  1036→      let betWon = false;
  1037→      let resultReason = "";
  1038→
  1039→      if (bet.betType === "single" || bet.betType === "accumulator_leg") {
  1040→        if (bet.selection === match.homeTeam) {
  1041→          betWon = homeScore > awayScore;
  1042→          resultReason = betWon
  1043→            ? `${match.homeTeam} won ${homeScore}-${awayScore}`
  1044→            : `${match.homeTeam} did not win (${homeScore}-${awayScore})`;
  1045→        } else if (bet.selection === match.awayTeam) {
  1046→          betWon = awayScore > homeScore;
  1047→          resultReason = betWon
  1048→            ? `${match.awayTeam} won ${awayScore}-${homeScore}`
  1049→            : `${match.awayTeam} did not win (${awayScore}-${homeScore})`;
  1050→        } else if (bet.selection === "Draw") {
  1051→          betWon = homeScore === awayScore;
  1052→          resultReason = betWon
  1053→            ? `Match drawn ${homeScore}-${awayScore}`
  1054→            : `Match not drawn (${homeScore}-${awayScore})`;
  1055→        } else if (bet.selection === "Over 2.5") {
  1056→          const totalGoals = homeScore + awayScore;
  1057→          betWon = totalGoals > 2.5;
  1058→          resultReason = betWon
  1059→            ? `Total goals ${totalGoals} > 2.5`
  1060→            : `Total goals ${totalGoals} <= 2.5`;
  1061→        } else if (bet.selection === "Under 2.5") {
  1062→          const totalGoals = homeScore + awayScore;
  1063→          betWon = totalGoals < 2.5;
  1064→          resultReason = betWon
  1065→            ? `Total goals ${totalGoals} < 2.5`
  1066→            : `Total goals ${totalGoals} >= 2.5`;
  1067→        }
  1068→      }
  1069→
  1070→      const now = new Date();
  1071→
  1072→      if (betWon) {
  1073→        const effectiveStake = bet.partialCashoutAmount ? bet.stake - bet.partialCashoutAmount : bet.stake;
  1074→        const grossProfit = bet.potentialWin - effectiveStake;
  1075→        const commission = grossProfit * commissionRate;
  1076→        const netProfit = grossProfit - commission;
  1077→
  1078→        await prisma.bet.update({
  1079→          where: { id: bet.id },
  1080→          data: {
  1081→            status: "won",
  1082→            profit: Math.round(netProfit * 100) / 100,
  1083→            commission: Math.round(commission * 100) / 100,
  1084→            settledAt: now,
  1085→            settlementReason: "match_finished",
  1086→          },
  1087→        });
  1088→
  1089→        await prisma.user.update({
  1090→          where: { id: userId },
  1091→          data: {
  1092→            balance: { increment: bet.potentialWin },
  1093→            totalProfit: { increment: netProfit },
  1094→            commissionPaid: { increment: commission },
  1095→            dailyPnl: { increment: netProfit },
  1096→            weeklyPnl: { increment: netProfit },
  1097→          },
  1098→        });
  1099→
  1100→        await prisma.transaction.create({
  1101→          data: {
  1102→            userId,
  1103→            type: "bet_won",
  1104→            amount: bet.potentialWin,
  1105→            currency: "USD",
  1106→            status: "completed",
  1107→            description: `Auto-settled: ${match.homeTeam} vs ${match.awayTeam} - ${bet.selection} @ ${bet.odds} (profit: $${netProfit.toFixed(2)})`,
  1108→            betId: bet.id,
  1109→          },
  1110→        });
  1111→
  1112→        if (commission > 0 && bet.bettingAccountId) {
  1113→          await prisma.commissionLedger.create({
  1114→            data: {
  1115→              userId,
  1116→              bettingAccountId: bet.bettingAccountId,
  1117→              betId: bet.id,
  1118→              accumulatorId: bet.accumulatorId,
  1119→              grossProfit,
  1120→              commissionRate,
  1121→              commissionAmount: commission,
  1122→              netProfit,
  1123→              status: "pending",
  1124→            },
  1125→          });
  1126→        }
  1127→
  1128→        await prisma.botLog.create({
  1129→          data: {
  1130→            userId,
  1131→            action: "bet_settled",
  1132→            betId: bet.id,
  1133→            matchId: match.id,
  1134→            accumulatorId: bet.accumulatorId,
  1135→            details: JSON.stringify({ result: "won", profit: netProfit, commission, autoSettled: true }),
  1136→            reasoning: resultReason,
  1137→            profitImpact: netProfit,
  1138→          },
  1139→        });
  1140→      } else {
  1141→        const effectiveStake = bet.partialCashoutAmount ? bet.stake - bet.partialCashoutAmount : bet.stake;
  1142→
  1143→        await prisma.bet.update({
  1144→          where: { id: bet.id },
  1145→          data: {
  1146→            status: "lost",
  1147→            profit: -effectiveStake,
  1148→            settledAt: now,
  1149→            settlementReason: "match_finished",
  1150→          },
  1151→        });
  1152→
  1153→        await prisma.user.update({
  1154→          where: { id: userId },
  1155→          data: {
  1156→            totalLoss: { increment: effectiveStake },
  1157→            dailyPnl: { decrement: effectiveStake },
  1158→            weeklyPnl: { decrement: effectiveStake },
  1159→          },
  1160→        });
  1161→
  1162→        await prisma.botLog.create({
  1163→          data: {
  1164→            userId,
  1165→            action: "bet_settled",
  1166→            betId: bet.id,
  1167→            matchId: match.id,

  1191→   * Uses the AI engine's cashout recommendation system.
  1192→   */
  1193→  private async autoCashoutEvaluation(userId: string): Promise<number> {
  1194→    const settings = await prisma.userSettings.findUnique({ where: { userId } });
  1195→    if (!settings?.autoCashoutEnabled) return 0;
  1196→

  1198→    const liveBets = await prisma.bet.findMany({
  1199→      where: {
  1200→        userId,

  1204→      },
  1205→      include: { match: true },
  1206→    });
  1207→
  1208→    if (liveBets.length === 0) return 0;
  1209→

  1211→
  1212→    for (const bet of liveBets) {
  1213→      if (!bet.match) continue;
  1214→
  1215→      // Simple cashout evaluation based on match state
  1216→      const match = bet.match;
  1217→      const homeScore = match.homeScore ?? 0;
  1218→      const awayScore = match.awayScore ?? 0;
  1219→      const minute = match.minute ?? 0;
  1220→
  1221→      // Determine if the bet is winning
  1222→      let isWinning = false;
  1223→      if (bet.selection === match.homeTeam) {
  1224→        isWinning = homeScore > awayScore;
  1225→      } else if (bet.selection === match.awayTeam) {
  1226→        isWinning = awayScore > homeScore;
  1227→      } else if (bet.selection === "Draw") {
  1228→        isWinning = homeScore === awayScore;
  1229→      } else if (bet.selection === "Over 2.5") {
  1230→        isWinning = (homeScore + awayScore) > 2.5;

  1232→        isWinning = (homeScore + awayScore) < 2.5;
  1233→      }
  1234→
