   127→   127→// ==================== RETRY LOGIC ====================
   128→   128→
   129→   129→interface RetryConfig {
   130→   130→  maxRetries: number;
   131→   131→  baseDelayMs: number;
   132→   132→  maxDelayMs: number;
   133→   133→  retryableErrors: string[];
   134→   134→}
   135→   135→
   136→   136→const DEFAULT_RETRY_CONFIG: RetryConfig = {
   137→   137→  maxRetries: 3,
   138→   138→  baseDelayMs: 1000,
   139→   139→  maxDelayMs: 10000,
   140→   140→  retryableErrors: [
   141→   141→    "ETIMEDOUT",
   142→   142→    "ECONNRESET",
   143→   143→    "ECONNREFUSED",
   144→   144→    "rate_limit",
   145→   145→    "server_error",
   146→   146→    "session_expired",
   147→   147→    "temporarily_unavailable",
   148→   148→  ],
   149→   149→};
   150→   150→
   151→   151→async function withRetry<T>(
   152→   152→  fn: () => Promise<T>,
   153→   153→  retryConfig: Partial<RetryConfig> = {}
   154→   154→): Promise<T> {
   155→   155→  const cfg = { ...DEFAULT_RETRY_CONFIG, ...retryConfig };
   156→   156→  let lastError: Error | null = null;
   157→   157→
   158→   158→  for (let attempt = 0; attempt <= cfg.maxRetries; attempt++) {
   159→   159→    try {
   160→   160→      return await fn();
   161→   161→    } catch (error) {
   162→   162→      lastError = error instanceof Error ? error : new Error(String(error));
   163→   163→
   164→   164→      // Check if error is retryable
   165→   165→      const isRetryable = cfg.retryableErrors.some(
   166→   166→        (err) => lastError!.message.toLowerCase().includes(err.toLowerCase())
   167→   167→      );
   168→   168→
   169→   169→      if (!isRetryable || attempt === cfg.maxRetries) {
   170→   170→        throw lastError;
   171→   171→      }
   172→   172→
   173→   173→      // Exponential backoff with jitter
   174→   174→      const delay = Math.min(
   175→   175→        cfg.baseDelayMs * Math.pow(2, attempt) + Math.random() * 1000,
   176→   176→        cfg.maxDelayMs
   177→   177→      );
   178→   178→
   179→   179→      await new Promise((resolve) => setTimeout(resolve, delay));
   180→   180→    }
   181→   181→  }
   182→   182→
   183→   183→  throw lastError;
   184→   184→}

   186→   186→// ==================== IDEMPOTENCY ====================
   187→   187→
   188→   188→// In-memory idempotency store (use Redis in production)
   189→   189→const idempotencyStore = new Map<string, { result: unknown; timestamp: number }>();
   190→   190→const IDEMPOTENCY_TTL = 3600000; // 1 hour
   191→   191→
   192→   192→function checkIdempotency<T>(key: string): T | null {
   193→   193→  const entry = idempotencyStore.get(key);
   194→   194→  if (!entry) return null;
   195→   195→
   196→   196→  if (Date.now() - entry.timestamp > IDEMPOTENCY_TTL) {
   197→   197→    idempotencyStore.delete(key);
   198→   198→    return null;
   199→   199→  }
   200→   200→
   201→   201→  return entry.result as T;
   202→   202→}
   203→   203→
   204→   204→function setIdempotencyResult(key: string, result: unknown): void {
   205→   205→  idempotencyStore.set(key, { result, timestamp: Date.now() });
   206→   206→
   207→   207→  // Clean up old entries periodically
   208→   208→  if (idempotencyStore.size > 10000) {
   209→   209→    const now = Date.now();
   210→   210→    for (const [k, v] of idempotencyStore) {
   211→   211→      if (now - v.timestamp > IDEMPOTENCY_TTL) {
   212→   212→        idempotencyStore.delete(k);
   213→   213→      }
   214→   214→    }
   215→   215→  }
   216→   216→}

   249→   249→  }
   250→   250→
   251→   251→  // Common helper: make authenticated API call with retry
   252→   252→  protected async apiCall<T>(
   253→   253→    accessToken: string,
   254→   254→    method: string,
   255→   255→    path: string,
   256→   256→    body?: unknown,
   257→   257→    options?: { idempotencyKey?: string }
   258→   258→  ): Promise<T> {
   259→   259→    const url = `${this.platform.baseUrl}${path}`;
   260→   260→
   261→   261→    // Check idempotency
   262→   262→    if (options?.idempotencyKey) {
   263→   263→      const cached = checkIdempotency<T>(options.idempotencyKey);
   264→   264→      if (cached) return cached;
   265→   265→    }
   266→   266→
   267→   267→    const result = await withRetry(async () => {
   268→   268→      const response = await fetch(url, {
   269→   269→        method,
   270→   270→        headers: {
   271→   271→          "Authorization": `Bearer ${accessToken}`,
   272→   272→          "Content-Type": "application/json",
   273→   273→          "X-Idempotency-Key": options?.idempotencyKey || "",
   274→   274→          "X-Platform": this.platformId,
   275→   275→          "X-App-Version": "1.0.0",
   276→   276→        },
   277→   277→        body: body ? JSON.stringify(body) : undefined,
   278→   278→      });
   279→   279→
   280→   280→      if (!response.ok) {
   281→   281→        const errorData = await response.json().catch(() => ({}));
   282→   282→        const errorCode = errorData.code || errorData.error || `HTTP_${response.status}`;
   283→   283→        const errorMessage = errorData.message || errorData.error || `API error: ${response.status}`;
   284→   284→
   285→   285→        // Map common error codes
   286→   286→        if (response.status === 401) {
   287→   287→          throw new Error(`session_expired: ${errorMessage}`);
   288→   288→        }
   289→   289→        if (response.status === 429) {
   290→   290→          throw new Error(`rate_limit: ${errorMessage}`);
   291→   291→        }
   292→   292→        if (response.status >= 500) {
   293→   293→          throw new Error(`server_error: ${errorMessage}`);
   294→   294→        }
   295→   295→
   296→   296→        throw new Error(`${errorCode}: ${errorMessage}`);
   297→   297→      }
   298→   298→
   299→   299→      return response.json() as Promise<T>;
   300→   300→    });
   301→   301→
   302→   302→    // Store idempotency result
   303→   303→    if (options?.idempotencyKey) {
   304→   304→      setIdempotencyResult(options.idempotencyKey, result);
   305→   305→    }
   306→   306→
   307→   307→    return result;
   308→   308→  }
   309→   309→}

   323→   323→  async authenticate(credentials: BrokerCredentials): Promise<BrokerAuthResult> {
   324→   324→    // Demo/Sandbox mode: simulate successful authentication without real API calls
   325→   325→    // MUST be checked BEFORE credential validation — in demo mode, any credentials (even empty) work
   326→   326→    if (this.isDemoMode()) {
   327→   327→      const demoToken = credentials.token || `demo_${Date.now()}`;
   328→   328→      return {
   329→   329→        success: true,
   330→   330→        accessToken: `demo_oauth_${this.platformId}_${Date.now()}`,
   331→   331→        refreshToken: `demo_refresh_${this.platformId}_${Date.now()}`,
   332→   332→        sessionExpiry: new Date(Date.now() + 3600000),
   333→   333→        brokerUserId: `demo_oauth_${demoToken.substring(0, 8)}_${Date.now()}`,
   334→   334→      };
   335→   335→    }
   336→   336→
   337→   337→    if (!credentials.token) {
   338→   338→      return { success: false, error: "OAuth authorization code required" };
   339→   339→    }
   340→   340→
   341→   341→    try {
   342→   342→      // In production, exchange authorization code for access token
   343→   343→      // POST /oauth/token with client_id, client_secret, code, redirect_uri
   344→   344→      const result = await withRetry(async () => {
   345→   345→        const response = await fetch(`${this.platform.baseUrl}/oauth/token`, {
   346→   346→          method: "POST",
   347→   347→          headers: { "Content-Type": "application/json" },
   348→   348→          body: JSON.stringify({
   349→   349→            grant_type: "authorization_code",
   350→   350→            code: credentials.token,
   351→   351→            client_id: process.env[`${this.platformId.toUpperCase()}_CLIENT_ID`],
   352→   352→            client_secret: process.env[`${this.platformId.toUpperCase()}_CLIENT_SECRET`],
   353→   353→            redirect_uri: `${config.platform.url}/api/broker/callback/${this.platformId}`,
   354→   354→          }),
   355→   355→        });
   356→   356→
   357→   357→        if (!response.ok) {
   358→   358→          throw new Error(`OAuth token exchange failed: ${response.status}`);
   359→   359→        }
   360→   360→
   361→   361→        return response.json();
   362→   362→      });
   363→   363→
   364→   364→      return {
   365→   365→        success: true,
   366→   366→        accessToken: result.access_token,
   367→   367→        refreshToken: result.refresh_token,
   368→   368→        sessionExpiry: new Date(Date.now() + (result.expires_in || 3600) * 1000),
   369→   369→        brokerUserId: result.user_id,
   370→   370→      };
   371→   371→    } catch (error) {
   372→   372→      return {
   373→   373→        success: false,
   374→   374→        error: error instanceof Error ? error.message : "OAuth authentication failed",
   375→   375→      };
   376→   376→    }
   377→   377→  }

   970→   970→ * @param demoMode - If true, the adapter will use demo/sandbox mode (no real API calls)
   971→   971→ */
   972→   972→export function getBrokerAdapter(platformId: string, demoMode?: boolean): BrokerAdapter | null {
   973→   973→  // Check cache (but demo mode changes per-user, so we need a keyed cache)
   974→   974→  const cacheKey = demoMode !== undefined ? `${platformId}_${demoMode ? "demo" : "real"}` : platformId;
   975→   975→  if (adapterCache.has(cacheKey)) {
   976→   976→    return adapterCache.get(cacheKey)!;
   977→   977→  }
   978→   978→
   979→   979→  // Manual adapter
   980→   980→  if (platformId === "manual") {
   981→   981→    const adapter = new ManualBrokerAdapter();
   982→   982→    adapterCache.set(cacheKey, adapter);
   983→   983→    return adapter;
   984→   984→  }
   985→   985→
   986→   986→  // Get platform config
   987→   987→  const platform = getBrokerPlatform(platformId);
   988→   988→  if (!platform) {
   989→   989→    console.error(`Unknown broker platform: ${platformId}`);
   990→   990→    return null;
   991→   991→  }
   992→   992→
   993→   993→  // Create appropriate adapter based on auth type
   994→   994→  let adapter: BrokerAdapter;
   995→   995→
   996→   996→  switch (platform.authType) {
   997→   997→    case "oauth":
   998→   998→      adapter = new OAuthBrokerAdapter(platform);
   999→   999→      break;
  1000→  1000→    case "api_key":
  1001→  1001→      adapter = new ApiKeyBrokerAdapter(platform);
  1002→  1002→      break;
  1003→  1003→    case "web_session":
  1004→  1004→      adapter = new WebSessionBrokerAdapter(platform);
  1005→  1005→      break;
  1006→  1006→    case "manual":
  1007→  1007→      adapter = new ManualBrokerAdapter();
  1008→  1008→      break;
  1009→  1009→    default:
  1010→  1010→      console.error(`Unsupported auth type: ${platform.authType}`);
  1011→  1011→      return null;
  1012→  1012→  }
  1013→  1013→
  1014→  1014→  // Override demo mode if specified
  1015→  1015→  if (demoMode !== undefined) {
  1016→  1016→    (adapter as any)._demoMode = demoMode;
  1017→  1017→  }
  1018→  1018→
  1019→  1019→  adapterCache.set(cacheKey, adapter);
  1020→  1020→  return adapter;
  1021→  1021→}

  1027→  1027→export function getAvailableAdapters(): BrokerAdapter[] {
  1028→  1028→  const adapters: BrokerAdapter[] = [new ManualBrokerAdapter()];
  1029→  1029→

  1032→  1032→  const { BROKER_PLATFORMS } = require("./regions") as { BROKER_PLATFORMS: Array<{ id: string }> };

  1041→  1041→    }
  1042→  1042→  }

  1053→  1053→  platformId: string,
  1054→  1054→  accessToken: string,
  1055→  1055→  bet: BrokerBetRequest,
  1056→  1056→  maxStake: number

  1059→  1059→  if (bet.stake <= 0) {

  1061→  1061→  }
  1062→  1062→  if (bet.stake > maxStake) {

  1064→  1064→  }

  1066→  1066→  // Validate odds
  1067→  1067→  if (bet.odds < 1.01) {
  1068→  1068→    return { success: false, error: "Invalid odds" };
  1069→  1069→  }
  1070→  1070→
  1071→  1071→  const adapter = getBrokerAdapter(platformId);
  1072→  1072→  if (!adapter) {

  1074→  1074→  }
  1075→  1075→
  1076→  1076→  // Generate idempotency key
  1077→  1077→  const idempotencyKey = `${platformId}_${bet.matchId}_${bet.selection}_${bet.stake}_${Date.now()}`;
  1078→  1078→
  1079→  1079→  return adapter.placeBet(accessToken, {
  1080→  1080→    ...bet,
  1081→  1081→    idempotencyKey,
  1082→  1082→  });
  1083→  1083→}
