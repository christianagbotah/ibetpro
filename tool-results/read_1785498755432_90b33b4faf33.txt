   112→export default function BettingPage() {
   113→  const { addToast } = useToast();
   114→  const { user } = useAuth();
   115→  const { data: bets, loading, refetch: refetchBets } = useFetch<Bet[]>("/api/bets", []);
   116→  const { data: accounts } = useFetch<Array<{ id: string; platform: string }>>("/api/accounts", []);
   117→  const [settings, setSettings] = useState<UserSettings>(defaultSettings);
   118→  const [betFilter, setBetFilter] = useState<string>("all");
   119→  const [botRunning, setBotRunning] = useState(false);
   120→  const [settleLoading, setSettleLoading] = useState(false);
   121→
   122→  // Load user settings
   123→  useEffect(() => {
   124→    async function loadSettings() {
   125→      try {
   126→        const res = await fetch("/api/settings");
   127→        if (res.ok) {
   128→          const data = await res.json();
   129→          if (data) {
   130→            setSettings({
   131→              ...defaultSettings,
   132→              ...data,
   133→            });
   134→          }
   135→        }
   136→      } catch {
   137→        // Use defaults
   138→      }
   139→    }
   140→    loadSettings();
   141→  }, []);
   142→
   143→  const handleCashout = async (betId: string, type: "full" | "partial" = "full") => {
   144→    try {
   145→      const res = await fetch("/api/cashout", {
   146→        method: "POST",
   147→        headers: { "Content-Type": "application/json" },
   148→        body: JSON.stringify({ betId, cashoutType: type }),
   149→      });
   150→      if (res.ok) {
   151→        const result = await res.json();
   152→        refetchBets();
   153→        addToast("success", `${type === "partial" ? "Partial" : "Full"} cashout processed: $${result.amount?.toFixed(2)}`);
   154→      } else {
   155→        const data = await res.json();
   156→        addToast("error", data.error || "Cashout failed");
   157→      }
   158→    } catch (error) {
   159→      console.error("Cashout failed:", error);
   160→      addToast("error", "Cashout failed");
   161→    }
   162→  };
   163→
   164→  const handleRunBot = async () => {
   165→    if (!user?.id) {
   166→      addToast("error", "Please log in to run the bot");
   167→      return;
   168→    }
   169→
   170→    setBotRunning(true);
   171→    try {
   172→      const res = await fetch("/api/auto-bet", {
   173→        method: "POST",
   174→        headers: { "Content-Type": "application/json" },
   175→        body: JSON.stringify({ userId: user.id }),
   176→      });
   177→      if (res.ok) {
   178→        const result = await res.json();
   179→        if (result.betsPlaced > 0) {
   180→          addToast("success", `Bot placed ${result.betsPlaced} bet(s)`);
   181→        } else {
   182→          addToast("info", result.error || result.message || "No suitable matches found for betting");
   183→        }
   184→        refetchBets();
   185→      } else {
   186→        const data = await res.json();
   187→        addToast("error", data.error || "Bot failed");
   188→      }
   189→    } catch {
   190→      addToast("error", "Bot execution failed");
   191→    } finally {
   192→      setBotRunning(false);
   193→    }
   194→  };
   195→
   196→  const handleSettle = async () => {
   197→    if (!user?.id) return;
   198→    setSettleLoading(true);
   199→    try {
   200→      const res = await fetch("/api/settle", {
   201→        method: "POST",
   202→        headers: { "Content-Type": "application/json" },
   203→        body: JSON.stringify({ userId: user.id }),
   204→      });
   205→      if (res.ok) {
   206→        const result = await res.json();
   207→        if (result.settled > 0) {
   208→          addToast("success", `Settled ${result.settled} bet(s). Profit: $${result.totalProfit?.toFixed(2)}`);
   209→        } else {
   210→          addToast("info", "No bets to settle");
   211→        }
   212→        refetchBets();
   213→      }
   214→    } catch {
   215→      addToast("error", "Settlement failed");
   216→    } finally {
   217→      setSettleLoading(false);
   218→    }
   219→  };
   220→
   221→  const handleSaveSettings = async (newSettings: UserSettings) => {
   222→    setSettings(newSettings);
   223→    try {
   224→      await fetch("/api/settings", {
   225→        method: "PUT",
   226→        headers: { "Content-Type": "application/json" },
   227→        body: JSON.stringify(newSettings),
   228→      });
   229→      addToast("success", "Settings saved successfully!");
   230→    } catch {
   231→      addToast("error", "Failed to save settings");
   232→    }
   233→  };
   234→
   235→  // Merge betting account info
   236→  const enrichedBets = bets.map((bet) => {

   238→    return {
   239→      ...bet,
   240→      bettingAccount: account ? { platform: account.platform } : undefined,
   241→    };
   242→  });
   243→
   244→  const filteredBets =
   245→    betFilter === "all"
   246→      ? enrichedBets
   247→      : betFilter === "auto"
   248→      ? enrichedBets.filter((b) => b.isAutoPlaced)
   249→      : betFilter === "manual"
   250→      ? enrichedBets.filter((b) => !b.isAutoPlaced)
   251→      : betFilter === "accumulator"
   252→      ? enrichedBets.filter((b) => b.accumulatorId)
   253→      : enrichedBets.filter((b) => b.status === betFilter);
   254→
   255→  const autoBets = enrichedBets.filter((b) => b.isAutoPlaced);
   256→  const accumulatorBets = enrichedBets.filter((b) => b.accumulatorId);
   257→  const totalStake = enrichedBets.reduce((sum, b) => sum + b.stake, 0);
   258→  const totalProfit = enrichedBets.reduce((sum, b) => sum + (b.profit || 0), 0);
   259→
   260→  // AI recommended bets
   261→  const aiRecommendedBets = enrichedBets
   262→    .filter((b) => b.status === "pending" && b.aiConfidence && b.aiConfidence > 0.5)
   263→    .sort((a, b) => (b.aiConfidence || 0) - (a.aiConfidence || 0));
   264→
   265→  // Daily bet limit progress
   266→  const todayBets = enrichedBets.filter((b) => {
   267→    const today = new Date();
   268→    const placedAt = new Date(b.placedAt);
   269→    return placedAt.toDateString() === today.toDateString();
   270→  });
   271→  const dailyStake = todayBets.reduce((sum, b) => sum + b.stake, 0);
   272→  const dailyLimitProgress = Math.min((dailyStake / settings.dailyBetLimit) * 100, 100);
   273→
   274→  if (loading) {
   275→    return (
   276→      <div className="flex items-center justify-center h-64">
   277→        <div className="flex items-center gap-2 text-muted-foreground">
   278→          <div className="h-5 w-5 border-2 border-primary border-t-transparent rounded-full animate-spin" />
   279→          Loading...
   280→        </div>
   281→      </div>
   282→    );
   283→  }
   284→
   285→  return (
   286→    <div className="space-y-6">
   287→      {/* Auto-Betting Disabled Banner */}
   288→      {!settings.autoBettingEnabled && (
   289→        <div className="rounded-xl border-2 border-amber-400/30 bg-amber-400/5 p-4">
   290→          <div className="flex items-start justify-between gap-4">
   291→            <div className="flex items-start gap-3">
   292→              <div className="flex h-8 w-8 shrink-0 items-center justify-center rounded-full bg-amber-400/10">
   293→                <Zap className="h-4 w-4 text-amber-400" />
   294→              </div>
   295→              <div>
   296→                <p className="text-sm font-medium text-amber-400">AI Auto-Betting is Disabled</p>
   297→                <p className="text-xs text-muted-foreground mt-1">
   298→                  Toggle the switch to enable the AI bot, then click "Run Bot" to start placing bets automatically.
   299→                  You can also configure advanced settings in the Auto-Betting Configuration panel below.
   300→                </p>
   301→              </div>
   302→            </div>
   303→            <div className="flex items-center gap-2 shrink-0">
   304→              <span className="text-xs text-muted-foreground">Enable</span>
   305→              <Switch
   306→                checked={settings.autoBettingEnabled}
   307→                onCheckedChange={async (checked) => {
   308→                  const updated = { ...settings, autoBettingEnabled: checked as boolean };
   309→                  setSettings(updated);
   310→                  try {
   311→                    await fetch("/api/settings", {
   312→                      method: "PUT",
   313→                      headers: { "Content-Type": "application/json" },
   314→                      body: JSON.stringify({ autoBettingEnabled: checked }),
   315→                    });
   316→                    addToast("success", checked ? "Auto-betting enabled! You can now run the bot." : "Auto-betting disabled");
   317→                  } catch {
   318→                    addToast("error", "Failed to update setting");
   319→                    setSettings(settings); // revert
   320→                  }
   321→                }}
   322→              />
   323→            </div>
   324→          </div>
   325→        </div>
   326→      )}
   327→
   328→      <div className="flex items-center justify-between">
   329→        <div>
   330→          <h1 className="text-2xl font-bold text-foreground">Automated Betting</h1>
   331→          <p className="text-sm text-muted-foreground mt-1">
   332→            Configure auto-betting and manage your bets
   333→          </p>
   334→        </div>
   335→        <div className="flex items-center gap-2">
   336→          {/* Bot Status Indicator */}
   337→          <Badge className={`px-3 py-1.5 ${
   338→            settings.autoBettingEnabled
   339→              ? "bg-primary/10 text-primary border-primary/20"
   340→              : "bg-secondary text-muted-foreground border-border"
   341→          }`}>
   342→            <span className="relative flex h-2 w-2 mr-2">
   343→              {settings.autoBettingEnabled && (
   344→                <>
   345→                  <span className="animate-live-pulse absolute inline-flex h-full w-full rounded-full bg-primary opacity-75" />
   346→                  <span className="relative inline-flex rounded-full h-2 w-2 bg-primary" />
   347→                </>
   348→              )}
   349→              {!settings.autoBettingEnabled && (
   350→                <span className="relative inline-flex rounded-full h-2 w-2 bg-muted-foreground" />
   351→              )}
   352→            </span>
   353→            Bot {settings.autoBettingEnabled ? "Active" : "Inactive"}
   354→          </Badge>
   355→
   356→          {/* Run Bot Button */}
   357→          <Button
   358→            size="sm"
   359→            className="bg-primary text-primary-foreground hover:bg-primary/80"
   360→            onClick={handleRunBot}
   361→            disabled={botRunning || !settings.autoBettingEnabled}
   362→          >
   363→            {botRunning ? (
   364→              <div className="h-4 w-4 border-2 border-primary-foreground border-t-transparent rounded-full animate-spin" />
   365→            ) : (
   366→              <Play className="h-4 w-4" />
   367→            )}
   368→            {botRunning ? "Running..." : "Run Bot"}
   369→          </Button>
   370→
   371→          {/* Settle Button */}
   372→          <Button
   373→            size="sm"
   374→            variant="outline"
   375→            onClick={handleSettle}
   376→            disabled={settleLoading}
   377→            className="border-emerald-400/30 text-emerald-400 hover:bg-emerald-400/10"
   378→          >
   379→            {settleLoading ? (
   380→              <div className="h-4 w-4 border-2 border-emerald-400 border-t-transparent rounded-full animate-spin" />
   381→            ) : (
   382→              <RefreshCw className="h-4 w-4" />
   383→            )}
   384→            Settle
   385→          </Button>
   386→        </div>
   387→      </div>
   388→
   389→      <div className="grid grid-cols-2 sm:grid-cols-5 gap-4">
   390→        <Card className="bg-card border-border">
   391→          <CardContent className="p-3">
   392→            <p className="text-xs text-muted-foreground">Total Bets</p>
   393→            <p className="text-lg font-bold text-foreground">{enrichedBets.length}</p>
   394→          </CardContent>
   395→        </Card>
   396→        <Card className="bg-card border-border">
   397→          <CardContent className="p-3">
   398→            <p className="text-xs text-muted-foreground">AI-Placed</p>
   399→            <p className="text-lg font-bold text-primary">{autoBets.length}</p>
   400→          </CardContent>
   401→        </Card>
   402→        <Card className="bg-card border-border">
   403→          <CardContent className="p-3">
   404→            <p className="text-xs text-muted-foreground">Accumulators</p>
   405→            <p className="text-lg font-bold text-purple-400">{accumulatorBets.length}</p>
   406→          </CardContent>
   407→        </Card>
   408→        <Card className="bg-card border-border">
   409→          <CardContent className="p-3">
   410→            <p className="text-xs text-muted-foreground">Total Stake</p>
   411→            <p className="text-lg font-bold text-foreground">${totalStake.toFixed(2)}</p>
   412→          </CardContent>
   413→        </Card>
   414→        <Card className="bg-card border-border">
   415→          <CardContent className="p-3">
   416→            <p className="text-xs text-muted-foreground">Net Profit</p>
   417→            <p className={`text-lg font-bold ${totalProfit >= 0 ? "text-emerald-400" : "text-red-400"}`}>
   418→              {totalProfit >= 0 ? "+" : ""}${totalProfit.toFixed(2)}
   419→            </p>
   420→          </CardContent>
   421→        </Card>
   422→      </div>
   423→
   424→      {/* Daily Bet Limit Progress Bar */}
   425→      <Card className="bg-card border-border">
   426→        <CardContent className="p-4">
   427→          <div className="flex items-center justify-between mb-2">
   428→            <div className="flex items-center gap-2">
   429→              <Shield className="h-4 w-4 text-amber-400" />
   430→              <span className="text-sm font-medium text-foreground">Daily Bet Limit</span>
   431→            </div>
   432→            <span className="text-sm text-muted-foreground">
   433→              ${dailyStake.toFixed(0)} / ${settings.dailyBetLimit.toFixed(0)}
   434→            </span>
   435→          </div>
   436→          <div className="h-2.5 rounded-full bg-secondary overflow-hidden">
   437→            <div
   438→              className={`h-full rounded-full transition-all ${
   439→                dailyLimitProgress >= 90 ? "bg-red-500" : dailyLimitProgress >= 70 ? "bg-amber-400" : "bg-primary"
   440→              }`}
   441→              style={{ width: `${dailyLimitProgress}%` }}

   443→          </div>

   451→          </div>

   454→
   455→      {/* Risk Limits Display */}
   456→      <div className="grid grid-cols-2 sm:grid-cols-4 gap-3">

   463→        </div>

   465→          <div className="flex items-center gap-1.5 mb-1">

   486→
   487→      {/* AI Recommended Bets */}
   488→      {aiRecommendedBets.length > 0 && (

   533→                  </div>
   534→                  {/* Cashout buttons */}
   535→                  <div className="flex gap-2">
   536→                    {bet.cashoutAmount && (
   537→                      <button
   538→                        onClick={() => handleCashout(bet.id, "partial")}
   539→                        className="flex-1 text-[10px] bg-primary/10 text-primary border border-primary/20 rounded px-2 py-1 hover:bg-primary/20 transition-colors"
   540→                      >
   541→                        <DollarSign className="h-3 w-3 inline" />
   542→                        Partial ${((bet.cashoutAmount || 0) * settings.partialCashoutPercent).toFixed(2)}
   543→                      </button>
   544→                    )}
   545→                    {bet.cashoutAmount && (
   546→                      <button
   547→                        onClick={() => handleCashout(bet.id, "full")}
   548→                        className="flex-1 text-[10px] bg-amber-400/10 text-amber-400 border border-amber-400/20 rounded px-2 py-1 hover:bg-amber-400/20 transition-colors"
   549→                      >
   550→                        <DollarSign className="h-3 w-3 inline" />
   551→                        Full ${bet.cashoutAmount.toFixed(2)}
   552→                      </button>
   553→                    )}
   554→                  </div>
   555→                </div>
   556→              ))}
   557→            </div>
   558→          </CardContent>
   559→        </Card>
   560→      )}
   561→
   562→      <div className="grid grid-cols-1 lg:grid-cols-3 gap-6">
   563→        <div className="lg:col-span-2 space-y-4">
   564→          <div className="flex items-center justify-between">
   565→            <h2 className="text-lg font-semibold text-foreground">Your Bets</h2>
   566→            <div className="flex items-center gap-2">
   567→              {["all", "pending", "won", "lost", "auto", "accumulator"].map((filter) => (
   568→                <Button
   569→                  key={filter}
   570→                  size="xs"
   571→                  variant={betFilter === filter ? "default" : "ghost"}
   572→                  onClick={() => setBetFilter(filter)}
   573→                  className={betFilter === filter ? "bg-primary text-primary-foreground" : ""}
   574→                >
   575→                  {filter.charAt(0).toUpperCase() + filter.slice(1)}
   576→                </Button>
   577→              ))}
   578→            </div>
   579→          </div>
   580→
   581→          <div className="space-y-3 max-h-[calc(100vh-400px)] overflow-y-auto">
   582→            {filteredBets.length === 0 ? (
   583→              <Card className="bg-card border-border">
   584→                <CardContent className="p-8 text-center">
   585→                  <Zap className="h-12 w-12 text-muted-foreground mx-auto mb-3" />
   586→                  <p className="text-sm text-muted-foreground">
   587→                    No bets found for this filter.
   588→                  </p>
   589→                </CardContent>
   590→              </Card>
   591→            ) : (
   592→              filteredBets.map((bet) => (
   593→                <EnhancedBetCard
   594→                  key={bet.id}
   595→                  bet={bet}
   596→                  onCashout={handleCashout}
   597→                  settings={settings}
   598→                />
   599→              ))
   600→            )}
   601→          </div>
   602→        </div>
   603→
   604→        <div className="space-y-4">
   605→          <AutoBetConfig settings={settings} onSave={handleSaveSettings} />
   606→          {user?.id && <BotActivityFeed userId={user.id} compact />}
   607→        </div>
   608→      </div>
   609→    </div>
   610→  );
   611→}
