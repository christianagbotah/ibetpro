   104→   104→              ...prev,
   105→   105→              [bet.id]: result.cashoutRecommendation,
   106→   106→            }));
   107→   107→          }
   108→   108→        } catch {
   109→   109→          // Ignore cashout errors
   110→   110→        }
   111→   111→      }
   112→   112→    }
   113→   113→
   114→   114→    if (liveBets.length > 0) {
   115→   115→      fetchCashouts();
   116→   116→    }
   117→   117→
   118→   118→    return () => {
   119→   119→      cancelled = true;
   120→   120→    };
   121→   121→  }, [bets, cashedOutBets]);

   127→   127→        method: "POST",

   134→   134→      } else {
   135→   135→        const data = await res.json();
   136→   136→        addToast("error", data.error || "Cashout failed");
   137→   137→      }
   138→   138→    } catch {
   139→   139→      addToast("error", "Cashout failed");
   140→   140→    }

   143→   143→

   148→   148→        method: "POST",

   159→   159→            ].slice(0, 20));
   160→   160→          });

   162→   162→        } else {

   164→   164→        }
   165→   165→        refetchMatches();

   256→   256→      <div className="grid grid-cols-2 sm:grid-cols-3 lg:grid-cols-5 gap-3">

   270→   270→            <div className="flex h-9 w-9 items-center justify-center rounded-lg bg-primary/10">
