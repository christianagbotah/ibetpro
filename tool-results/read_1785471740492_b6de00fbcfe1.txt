     1→"use client";
     2→
     3→import { useState, useCallback, useMemo } from "react";
     4→import { useFetch } from "@/lib/hooks";
     5→import { useToast } from "@/components/ui/toast";
     6→import { Card, CardContent, CardHeader, CardTitle } from "@/components/ui/card";
     7→import { Button } from "@/components/ui/button";
     8→import { Badge } from "@/components/ui/badge";
     9→import { Input } from "@/components/ui/input";
    10→import { Label } from "@/components/ui/label";
    11→import { Separator } from "@/components/ui/separator";
    12→import {
    13→  Dialog,
    14→  DialogContent,
    15→  DialogHeader,
    16→  DialogTitle,
    17→} from "@/components/ui/dialog";
    18→import {
    19→  Link2, RefreshCw, Wallet, Shield, CheckCircle2,
    20→  AlertCircle, Key, User, Lock, ChevronRight,
    21→  Search, MapPin, Smartphone, Star, ArrowLeft,
    22→} from "lucide-react";
    23→import { getPlatformLogoPath, getBrokerLogoUrl } from "@/lib/broker-logos";
    24→import {
    25→  REGIONS,
    26→  getPlatformsForRegion,
    27→  getCurrencyForRegion,
    28→  getContinents,
    29→  searchRegions,
    30→  type RegionInfo,
    31→  type BrokerPlatformInfo,
    32→} from "@/lib/regions";
    33→
    34→interface ConnectedAccount {
    35→  id: string;
    36→  platform: string;
    37→  platformName: string;
    38→  accountName: string;
    39→  isConnected: boolean;
    40→  sessionValid: boolean;
    41→  needsRefresh: boolean;
    42→  sessionExpiry: string | null;
    43→  brokerType: string;
    44→  brokerRegion: string | null;
    45→  balance: number;
    46→  currency: string;
    47→  allocatedAmount: number;
    48→  allocationLock: boolean;
    49→  totalBrokerBets: number;
    50→  totalBrokerProfit: number;
    51→  lastBetPlacedAt: string | null;
    52→  features: {
    53→    liveBetting: boolean;
    54→    cashout: boolean;
    55→    partialCashout: boolean;
    56→    accumulators: boolean;
    57→    maxAccumulatorLegs: number;
    58→    minStake: number;
    59→    maxStake: number;
    60→  } | null;
    61→}
    62→
    63→type ConnectStep = "region" | "platform" | "credentials";
    64→
    65→export function BrokerConnect() {
    66→  const { addToast } = useToast();
    67→  const [connectDialogOpen, setConnectDialogOpen] = useState(false);
    68→  const [step, setStep] = useState<ConnectStep>("region");
    69→  const [selectedRegion, setSelectedRegion] = useState<string>("");
    70→  const [selectedPlatform, setSelectedPlatform] = useState<string | null>(null);
    71→  const [connecting, setConnecting] = useState(false);
    72→  const [credentials, setCredentials] = useState({
    73→    username: "",
    74→    password: "",
    75→    apiKey: "",
    76→    token: "",
    77→  });
    78→  const [accountName, setAccountName] = useState("");
    79→  const [regionSearch, setRegionSearch] = useState("");
    80→  const [brokerSearch, setBrokerSearch] = useState("");
    81→  const [selectedContinent, setSelectedContinent] = useState<string>("all");
    82→
    83→  const { data, loading, refresh } = useFetch<{
    84→    accounts: ConnectedAccount[];
    85→    availablePlatforms: BrokerPlatformInfo[];
    86→  }>("/api/broker/connect", { accounts: [], availablePlatforms: [] });
    87→
    88→  const accounts = data?.accounts || [];
    89→
    90→  // Get filtered regions based on search and continent
    91→  const filteredRegions = useMemo(() => {
    92→    let regions = regionSearch ? searchRegions(regionSearch) : REGIONS;
    93→    if (selectedContinent !== "all") {
    94→      regions = regions.filter((r) => r.continent === selectedContinent);
    95→    }
    96→    // Deduplicate by region code to prevent React key warnings
    97→    const seen = new Set<string>();
    98→    return regions.filter((r) => {
    99→      if (seen.has(r.code)) return false;
   100→      seen.add(r.code);
   101→      return true;
   102→    });
   103→  }, [regionSearch, selectedContinent]);
   104→
   105→  // Get platforms for selected region
   106→  const platformsForRegion = useMemo(() => {
   107→    if (!selectedRegion) return [];
   108→    return getPlatformsForRegion(selectedRegion);
   109→  }, [selectedRegion]);
   110→
   111→  // Filter platforms by search query
   112→  const filteredPlatforms = useMemo(() => {
   113→    if (!brokerSearch.trim()) return platformsForRegion;
   114→    const q = brokerSearch.toLowerCase().trim();
   115→    return platformsForRegion.filter(
   116→      (p) =>
   117→        p.name.toLowerCase().includes(q) ||
   118→        p.id.toLowerCase().includes(q) ||
   119→        p.authType.toLowerCase().includes(q)
   120→    );
   121→  }, [platformsForRegion, brokerSearch]);
   122→
   123→  const selectedRegionInfo = useMemo(() => {
   124→    return REGIONS.find((r) => r.code === selectedRegion);
   125→  }, [selectedRegion]);
   126→
   127→  const selectedPlatformData = platformsForRegion.find((p) => p.id === selectedPlatform);
   128→
   129→  const handleConnect = useCallback(async () => {
   130→    if (!selectedPlatform) return;
   131→    setConnecting(true);
   132→
   133→    try {
   134→      const res = await fetch("/api/broker/connect", {
   135→        method: "POST",
   136→        headers: { "Content-Type": "application/json" },
   137→        body: JSON.stringify({
   138→          platformId: selectedPlatform,
   139→          credentials,
   140→          region: selectedRegion,
   141→          accountName: accountName || undefined,
   142→        }),
   143→      });
   144→
   145→      const result = await res.json();
   146→
   147→      if (result.success) {
   148→        addToast("success", `${result.account.accountName} connected successfully!`);
   149→        setConnectDialogOpen(false);
   150→        resetForm();
   151→        refresh();
   152→      } else {
   153→        addToast("error", result.error || "Failed to connect broker");
   154→      }
   155→    } catch {
   156→      addToast("error", "Failed to connect broker");
   157→    } finally {
   158→      setConnecting(false);
   159→    }
   160→  }, [selectedPlatform, credentials, accountName, selectedRegion, addToast, refresh]);
   161→
   162→  const handleSync = useCallback(async (accountId: string) => {
   163→    try {
   164→      const res = await fetch("/api/broker/sync", {
   165→        method: "POST",
   166→        headers: { "Content-Type": "application/json" },
   167→        body: JSON.stringify({ bettingAccountId: accountId }),
   168→      });
   169→
   170→      const result = await res.json();
   171→
   172→      if (result.success) {
   173→        const currency = result.balance?.currency || "USD";
   174→        addToast("success", `Account synced! Balance: ${currency} ${result.balance.total.toFixed(2)}`);
   175→        refresh();
   176→      } else {
   177→        addToast("error", result.error || "Sync failed");
   178→      }
   179→    } catch {
   180→      addToast("error", "Sync error occurred");
   181→    }
   182→  }, [addToast, refresh]);
   183→
   184→  const resetForm = () => {
   185→    setCredentials({ username: "", password: "", apiKey: "", token: "" });
   186→    setAccountName("");
   187→    setSelectedPlatform(null);
   188→    setStep("region");
   189→    setRegionSearch("");
   190→    setBrokerSearch("");
   191→    setSelectedContinent("all");
   192→  };
   193→
   194→  const handleDialogClose = (open: boolean) => {
   195→    setConnectDialogOpen(open);
   196→    if (!open) resetForm();
   197→  };
   198→
   199→  const formatBalance = (amount: number, currency: string) => {
   200→    const region = REGIONS.find((r) => r.currencyCode === currency);
   201→    const symbol = region?.currencySymbol || "$";
   202→    return `${symbol}${amount.toLocaleString(undefined, { minimumFractionDigits: 2, maximumFractionDigits: 2 })}`;
   203→  };
   204→
   205→  const continents = getContinents();
   206→
   207→  return (
   208→    <div className="space-y-6">
   209→      <Card className="bg-card border-border">
   210→        <CardHeader className="pb-3">
   211→          <div className="flex items-center justify-between">
   212→            <CardTitle className="flex items-center gap-2 text-base">
   213→              <Link2 className="h-4 w-4 text-primary" />
   214→              Connected Brokers
   215→            </CardTitle>
   216→            <Dialog open={connectDialogOpen} onOpenChange={handleDialogClose}>
   217→              <Button
   218→                size="sm"
   219→                className="bg-primary text-primary-foreground"
   220→                onClick={() => setConnectDialogOpen(true)}
   221→              >
   222→                <Link2 className="h-3.5 w-3.5 mr-1.5" />
   223→                Connect Broker
   224→              </Button>
   225→
   226→              <DialogContent className="max-w-lg max-h-[90vh] overflow-y-auto sm:max-h-[90vh] max-[640px]:h-screen max-[640px]:max-h-screen max-[640px]:w-screen max-[640px]:max-w-screen max-[640px]:rounded-none max-[640px]:p-4">
   227→                <DialogHeader>
   228→                  <DialogTitle>Connect Your Broker Account</DialogTitle>
   229→                </DialogHeader>
   230→
   231→                {/* Step indicator */}
   232→                <div className="flex items-center gap-2 mb-2">
   233→                  <div className={`flex items-center gap-1.5 text-xs ${step === "region" ? "text-primary font-medium" : "text-muted-foreground"}`}>
   234→                    <div className={`h-5 w-5 rounded-full flex items-center justify-center text-[10px] font-bold ${step === "region" ? "bg-primary text-primary-foreground" : "bg-secondary text-muted-foreground"}`}>
   235→                      {step !== "region" ? "✓" : "1"}
   236→                    </div>
   237→                    Region
   238→                  </div>
   239→                  <div className="h-px flex-1 bg-border" />
   240→                  <div className={`flex items-center gap-1.5 text-xs ${step === "platform" ? "text-primary font-medium" : "text-muted-foreground"}`}>
   241→                    <div className={`h-5 w-5 rounded-full flex items-center justify-center text-[10px] font-bold ${step === "platform" ? "bg-primary text-primary-foreground" : step === "credentials" ? "bg-primary text-primary-foreground" : "bg-secondary text-muted-foreground"}`}>
   242→                      {step === "credentials" ? "✓" : "2"}
   243→                    </div>
   244→                    Platform
   245→                  </div>
   246→                  <div className="h-px flex-1 bg-border" />
   247→                  <div className={`flex items-center gap-1.5 text-xs ${step === "credentials" ? "text-primary font-medium" : "text-muted-foreground"}`}>
   248→                    <div className={`h-5 w-5 rounded-full flex items-center justify-center text-[10px] font-bold ${step === "credentials" ? "bg-primary text-primary-foreground" : "bg-secondary text-muted-foreground"}`}>
   249→                      3
   250→                    </div>
   251→                    Connect
   252→                  </div>
   253→                </div>
   254→
   255→                {/* ====== STEP 1: REGION SELECTION ====== */}
   256→                {step === "region" && (
   257→                  <div className="space-y-3">
   258→                    <p className="text-sm text-muted-foreground">
   259→                      Select your country or region. This helps us show betting platforms available in your area and set the right currency.
   260→                    </p>
   261→
   262→                    <div className="relative">
   263→                      <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   264→                      <Input
   265→                        value={regionSearch}
   266→                        onChange={(e) => setRegionSearch(e.target.value)}
   267→                        placeholder="Search country or currency..."
   268→                        className="bg-secondary border-border pl-9"
   269→                      />
   270→                    </div>
   271→
   272→                    <div className="flex flex-wrap gap-1.5">
   273→                      <button
   274→                        onClick={() => setSelectedContinent("all")}
   275→                        className={`rounded-full px-3 py-1 text-xs font-medium transition-all ${
   276→                          selectedContinent === "all"
   277→                            ? "bg-primary/10 text-primary border border-primary/30"
   278→                            : "bg-secondary/50 text-muted-foreground border border-border"
   279→                        }`}
   280→                      >
   281→                        All
   282→                      </button>
   283→                      {continents.map((c) => (
   284→                        <button
   285→                          key={c}
   286→                          onClick={() => setSelectedContinent(c)}
   287→                          className={`rounded-full px-3 py-1 text-xs font-medium transition-all ${
   288→                            selectedContinent === c
   289→                              ? "bg-primary/10 text-primary border border-primary/30"
   290→                              : "bg-secondary/50 text-muted-foreground border border-border"
   291→                          }`}
   292→                        >
   293→                          {c}
   294→                        </button>
   295→                      ))}
   296→                    </div>
   297→
   298→                    <div className="grid grid-cols-2 sm:grid-cols-3 gap-2 max-h-64 overflow-y-auto -webkit-overflow-scrolling-touch">
   299→                      {filteredRegions.map((region) => {
   300→                        const platformCount = getPlatformsForRegion(region.code).length;
   301→                        return (
   302→                          <button
   303→                            key={region.code}
   304→                            onClick={() => {
   305→                              setSelectedRegion(region.code);
   306→                              setStep("platform");
   307→                            }}
   308→                            className={`flex items-center gap-2 rounded-lg border p-2.5 transition-all text-left hover:bg-primary/5 active:scale-[0.98] ${
   309→                              selectedRegion === region.code
   310→                                ? "border-primary/30 bg-primary/5"
   311→                                : "border-border"
   312→                            }`}
   313→                          >
   314→                            <span className="text-lg">{region.flag}</span>
   315→                            <div className="min-w-0 flex-1">
   316→                              <p className="text-xs font-medium text-foreground truncate">{region.name}</p>
   317→                              <p className="text-[10px] text-muted-foreground">
   318→                                {region.currencySymbol} {region.currencyCode} · {platformCount} platforms
   319→                              </p>
   320→                            </div>
   321→                          </button>
   322→                        );
   323→                      })}
   324→                    </div>
   325→
   326→                    {filteredRegions.length === 0 && (
   327→                      <div className="text-center py-4">
   328→                        <p className="text-sm text-muted-foreground">No regions found matching your search</p>
   329→                      </div>
   330→                    )}
   331→                  </div>
   332→                )}
   333→
   334→                {/* ====== STEP 2: PLATFORM SELECTION ====== */}
   335→                {step === "platform" && selectedRegionInfo && (
   336→                  <div className="space-y-3">
   337→                    <button
   338→                      onClick={() => {
   339→                        setStep("region");
   340→                        setSelectedPlatform(null);
   341→                      }}
   342→                      className="flex items-center gap-1 text-xs text-muted-foreground hover:text-primary transition-colors"
   343→                    >
   344→                      <ArrowLeft className="h-3 w-3" /> Change region
   345→                    </button>
   346→
   347→                    <div className="flex items-center gap-2 p-2.5 rounded-lg bg-primary/5 border border-primary/10">
   348→                      <span className="text-lg">{selectedRegionInfo.flag}</span>
   349→                      <div>
   350→                        <p className="text-sm font-medium text-foreground">{selectedRegionInfo.name}</p>
   351→                        <p className="text-xs text-muted-foreground">
   352→                          Currency: {selectedRegionInfo.currencySymbol} {selectedRegionInfo.currencyCode} ({selectedRegionInfo.currencyName})
   353→                        </p>
   354→                      </div>
   355→                    </div>
   356→
   357→                    <p className="text-sm text-muted-foreground">
   358→                      {platformsForRegion.length} platforms available in {selectedRegionInfo.name}. Select your betting platform.
   359→                    </p>
   360→
   361→                    <div className="relative">
   362→                      <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   363→                      <Input
   364→                        type="text"
   365→                        placeholder="Search brokers..."
   366→                        value={brokerSearch}
   367→                        onChange={(e) => setBrokerSearch(e.target.value)}
   368→                        className="pl-9 h-9 text-sm"
   369→                      />
   370→                    </div>
   371→
   372→                    <div className="space-y-2 max-h-64 overflow-y-auto -webkit-overflow-scrolling-touch">
   373→                      {filteredPlatforms.length === 0 ? (
   374→                        <div className="text-center py-4 text-sm text-muted-foreground">
   375→                          No brokers found matching &quot;{brokerSearch}&quot;
   376→                        </div>
   377→                      ) : (
   378→                        filteredPlatforms.map((platform) => {
   379→                        const isPopular = platform.popularIn.includes(selectedRegion);
   380→                        return (
   381→                          <button
   382→                            key={platform.id}
   383→                            onClick={() => {
   384→                              setSelectedPlatform(platform.id);
   385→                              setStep("credentials");
   386→                            }}
   387→                            className="flex items-center justify-between rounded-lg border border-border p-3 hover:bg-primary/5 active:scale-[0.98] transition-colors text-left w-full"
   388→                          >
   389→                            <div className="flex items-center gap-3">
   390→                              <div
   391→                                className="flex h-10 w-10 items-center justify-center rounded-lg shrink-0 overflow-hidden bg-secondary/50 border border-border"
   392→                              >
   393→                                <img
   394→                                  src={getPlatformLogoPath(platform)}
   395→                                  alt={platform.name}
   396→                                  className="h-10 w-10 object-contain rounded-lg"
   397→                                  onError={(e) => {
   398→                                    const target = e.currentTarget;
   399→                                    if (!target.dataset.retried) {
   400→                                      target.dataset.retried = "true";
   401→                                      // Try SVG fallback
   402→                                      target.src = getBrokerLogoUrl(platform.id, platform.name, platform.color || "#10b981");
   403→                                    } else if (!target.dataset.retried2) {
   404→                                      target.dataset.retried2 = "true";
   405→                                      // Last resort: hide image and show color abbreviation
   406→                                      target.style.display = "none";
   407→                                      const fallback = document.createElement("div");
   408→                                      fallback.style.cssText = `background:${platform.color || "#10b981"};width:100%;height:100%;display:flex;align-items:center;justify-content:center;border-radius:0.5rem;color:#fff;font-size:0.65rem;font-weight:700;`;
   409→                                      fallback.textContent = platform.name.substring(0, 2).toUpperCase();
   410→                                      target.parentElement?.appendChild(fallback);
   411→                                    }
   412→                                  }}
   413→                                />
   414→                              </div>
   415→                              <div>
   416→                                <div className="flex items-center gap-1.5">
   417→                                  <p className="text-sm font-medium text-foreground">{platform.name}</p>
   418→                                  {isPopular && (
   419→                                    <Badge className="bg-amber-400/10 text-amber-400 border-amber-400/30 text-[9px] px-1 py-0">
   420→                                      <Star className="h-2.5 w-2.5 mr-0.5" /> Popular
   421→                                    </Badge>
   422→                                  )}
   423→                                </div>
   424→                                <p className="text-xs text-muted-foreground">
   425→                                  {platform.authType.replace(/_/g, " ")} · {platform.mobileApp ? "📱 App" : "Web only"}
   426→                                </p>
   427→                              </div>
   428→                            </div>
   429→                            <ChevronRight className="h-4 w-4 text-muted-foreground shrink-0" />
   430→                          </button>
   431→                        );
   432→                      })
   433→                      )}
   434→                    </div>
   435→                  </div>
   436→                )}
   437→
   438→                {/* ====== STEP 3: CREDENTIALS ====== */}
   439→                {step === "credentials" && selectedPlatformData && selectedRegionInfo && (
   440→                  <div className="space-y-4">
   441→                    <button
   442→                      onClick={() => {
   443→                        setStep("platform");
   444→                        setSelectedPlatform(null);
   445→                      }}
   446→                      className="flex items-center gap-1 text-xs text-muted-foreground hover:text-primary transition-colors"
   447→                    >
   448→                      <ArrowLeft className="h-3 w-3" /> Change platform
   449→                    </button>
   450→
   451→                    <div className="flex items-center gap-3 p-3 rounded-lg bg-primary/5 border border-primary/10">
   452→                      <div
   453→                        className="flex h-10 w-10 items-center justify-center rounded-lg shrink-0 overflow-hidden bg-secondary/50 border border-border"
   454→                      >
   455→                        <img
   456→                          src={getPlatformLogoPath(selectedPlatformData)}
   457→                          alt={selectedPlatformData.name}
   458→                          className="h-10 w-10 object-contain rounded-lg"
   459→                          onError={(e) => {
   460→                            const target = e.currentTarget;
   461→                            if (!target.dataset.retried) {
   462→                              target.dataset.retried = "true";
   463→                              target.src = getBrokerLogoUrl(selectedPlatformData.id, selectedPlatformData.name, selectedPlatformData.color || "#10b981");
   464→                            } else if (!target.dataset.retried2) {
   465→                              target.dataset.retried2 = "true";
   466→                              target.style.display = "none";
   467→                              const fallback = document.createElement("div");
   468→                              fallback.style.cssText = `background:${selectedPlatformData.color || "#10b981"};width:100%;height:100%;display:flex;align-items:center;justify-content:center;border-radius:0.5rem;color:#fff;font-size:0.65rem;font-weight:700;`;
   469→                              fallback.textContent = selectedPlatformData.name.substring(0, 2).toUpperCase();
   470→                              target.parentElement?.appendChild(fallback);
   471→                            }
   472→                          }}
   473→                        />
   474→                      </div>
   475→                      <div className="flex-1 min-w-0">
   476→                        <p className="text-sm font-medium text-foreground">{selectedPlatformData.name}</p>
   477→                        <p className="text-xs text-muted-foreground">
   478→                          {selectedPlatformData.authType.replace(/_/g, " ")} authentication
   479→                        </p>
   480→                      </div>
   481→                      <div className="text-right shrink-0">
   482→                        <p className="text-xs text-muted-foreground">{selectedRegionInfo.flag} {selectedRegionInfo.name}</p>
   483→                        <p className="text-xs font-medium text-primary">{selectedRegionInfo.currencySymbol} {selectedRegionInfo.currencyCode}</p>
   484→                      </div>
   485→                    </div>
   486→
   487→                    <div className="space-y-3">
   488→                      <div>
   489→                        <Label className="text-xs text-muted-foreground">Account Name</Label>
   490→                        <Input
   491→                          value={accountName}
   492→                          onChange={(e) => setAccountName(e.target.value)}
   493→                          placeholder={`My ${selectedPlatformData?.name} Account`}
   494→                          className="bg-secondary border-border mt-1"
   495→                        />
   496→                      </div>
   497→
   498→                      <Separator />
   499→
   500→                      {selectedPlatformData?.authType === "web_session" && (
   501→                        <>
   502→                          <div>
   503→                            <Label className="text-xs text-muted-foreground flex items-center gap-1">
   504→                              <User className="h-3 w-3" /> Username / Email
   505→                            </Label>
   506→                            <Input
   507→                              value={credentials.username}
   508→                              onChange={(e) => setCredentials({ ...credentials, username: e.target.value })}
   509→                              placeholder="Your broker username"
   510→                              className="bg-secondary border-border mt-1"
   511→                            />
   512→                          </div>
   513→                          <div>
   514→                            <Label className="text-xs text-muted-foreground flex items-center gap-1">
   515→                              <Lock className="h-3 w-3" /> Password
   516→                            </Label>
   517→                            <Input
   518→                              type="password"
   519→                              value={credentials.password}
   520→                              onChange={(e) => setCredentials({ ...credentials, password: e.target.value })}
   521→                              placeholder="Your broker password"
   522→                              className="bg-secondary border-border mt-1"
   523→                            />
