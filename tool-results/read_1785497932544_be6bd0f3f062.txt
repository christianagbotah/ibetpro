   110→
   111→export default function BettingPage() {
   112→  const { addToast } = useToast();
   113→  const { user } = useAuth();
   114→  const { data: bets, loading, refetch: refetchBets } = useFetch<Bet[]>("/api/bets", []);
   115→  const { data: accounts } = useFetch<Array<{ id: string; platform: string }>>("/api/accounts", []);
   116→  const [settings, setSettings] = useState<UserSettings>(defaultSettings);
   117→  const [betFilter, setBetFilter] = useState<string>("all");
   118→  const [botRunning, setBotRunning] = useState(false);
   119→  const [settleLoading, setSettleLoading] = useState(false);
   120→
   121→  // Load user settings
   122→  useEffect(() => {
   123→    async function loadSettings() {
   124→      try {
   125→        const res = await fetch("/api/settings");
   126→        if (res.ok) {
   127→          const data = await res.json();
   128→          if (data) {
   129→            setSettings({
   130→              ...defaultSettings,
   131→              ...data,
   132→            });
   133→          }
   134→        }
   135→      } catch {
   136→        // Use defaults
   137→      }
   138→    }
   139→    loadSettings();
   140→  }, []);
   141→
   142→  const handleCashout = async (betId: string, type: "full" | "partial" = "full") => {
   143→    try {
   144→      const res = await fetch("/api/cashout", {
   145→        method: "POST",
   146→        headers: { "Content-Type": "application/json" },
   147→        body: JSON.stringify({ betId, cashoutType: type }),
   148→      });
   149→      if (res.ok) {
   150→        const result = await res.json();
   151→        refetchBets();
   152→        addToast("success", `${type === "partial" ? "Partial" : "Full"} cashout processed: $${result.amount?.toFixed(2)}`);
   153→      } else {
   154→        const data = await res.json();
   155→        addToast("error", data.error || "Cashout failed");
   156→      }
   157→    } catch (error) {
   158→      console.error("Cashout failed:", error);
   159→      addToast("error", "Cashout failed");
   160→    }
   161→  };
   162→
   163→  const handleRunBot = async () => {
   164→    if (!user?.id) {
   165→      addToast("error", "Please log in to run the bot");
   166→      return;
   167→    }
   168→
   169→    setBotRunning(true);
   170→    try {
   171→      const res = await fetch("/api/auto-bet", {
   172→        method: "POST",
   173→        headers: { "Content-Type": "application/json" },
   174→        body: JSON.stringify({ userId: user.id }),
   175→      });
   176→      if (res.ok) {
   177→        const result = await res.json();
   178→        if (result.betsPlaced > 0) {
   179→          addToast("success", `Bot placed ${result.betsPlaced} bet(s)`);
   180→        } else {
   181→          addToast("info", result.error || result.message || "No suitable matches found for betting");
   182→        }
   183→        refetchBets();
   184→      } else {
   185→        const data = await res.json();
   186→        addToast("error", data.error || "Bot failed");
   187→      }
   188→    } catch {
   189→      addToast("error", "Bot execution failed");
   190→    } finally {
   191→      setBotRunning(false);
   192→    }
   193→  };
   194→
   195→  const handleSettle = async () => {
   196→    if (!user?.id) return;
   197→    setSettleLoading(true);
   198→    try {
   199→      const res = await fetch("/api/settle", {
   200→        method: "POST",
   201→        headers: { "Content-Type": "application/json" },
   202→        body: JSON.stringify({ userId: user.id }),
   203→      });
   204→      if (res.ok) {
   205→        const result = await res.json();
   206→        if (result.settled > 0) {
   207→          addToast("success", `Settled ${result.settled} bet(s). Profit: $${result.totalProfit?.toFixed(2)}`);
   208→        } else {
   209→          addToast("info", "No bets to settle");
   210→        }
   211→        refetchBets();
   212→      }
   213→    } catch {
   214→      addToast("error", "Settlement failed");
   215→    } finally {
   216→      setSettleLoading(false);
   217→    }
   218→  };
   219→
   220→  const handleSaveSettings = async (newSettings: UserSettings) => {
   221→    setSettings(newSettings);
   222→    try {
   223→      await fetch("/api/settings", {
   224→        method: "PUT",
   225→        headers: { "Content-Type": "application/json" },
   226→        body: JSON.stringify(newSettings),
   227→      });
   228→      addToast("success", "Settings saved successfully!");
   229→    } catch {
   230→      addToast("error", "Failed to save settings");
   231→    }
   232→  };
   233→
   234→  // Merge betting account info
   235→  const enrichedBets = bets.map((bet) => {
   236→    const account = accounts.find((a) => a.id === bet.bettingAccountId);
   237→    return {
   238→      ...bet,
   239→      bettingAccount: account ? { platform: account.platform } : undefined,
   240→    };
   241→  });
   242→
   243→  const filteredBets =
   244→    betFilter === "all"
   245→      ? enrichedBets
   246→      : betFilter === "auto"
   247→      ? enrichedBets.filter((b) => b.isAutoPlaced)
   248→      : betFilter === "manual"
   249→      ? enrichedBets.filter((b) => !b.isAutoPlaced)
   250→      : betFilter === "accumulator"
   251→      ? enrichedBets.filter((b) => b.accumulatorId)
   252→      : enrichedBets.filter((b) => b.status === betFilter);
   253→
   254→  const autoBets = enrichedBets.filter((b) => b.isAutoPlaced);
   255→  const accumulatorBets = enrichedBets.filter((b) => b.accumulatorId);
   256→  const totalStake = enrichedBets.reduce((sum, b) => sum + b.stake, 0);
   257→  const totalProfit = enrichedBets.reduce((sum, b) => sum + (b.profit || 0), 0);
   258→
   259→  // AI recommended bets
   260→  const aiRecommendedBets = enrichedBets
   261→    .filter((b) => b.status === "pending" && b.aiConfidence && b.aiConfidence > 0.5)
   262→    .sort((a, b) => (b.aiConfidence || 0) - (a.aiConfidence || 0));
   263→
   264→  // Daily bet limit progress
   265→  const todayBets = enrichedBets.filter((b) => {
   266→    const today = new Date();
   267→    const placedAt = new Date(b.placedAt);
   268→    return placedAt.toDateString() === today.toDateString();
   269→  });
   270→  const dailyStake = todayBets.reduce((sum, b) => sum + b.stake, 0);
   271→  const dailyLimitProgress = Math.min((dailyStake / settings.dailyBetLimit) * 100, 100);
   272→
   273→  if (loading) {
   274→    return (
   275→      <div className="flex items-center justify-center h-64">
   276→        <div className="flex items-center gap-2 text-muted-foreground">
   277→          <div className="h-5 w-5 border-2 border-primary border-t-transparent rounded-full animate-spin" />
   278→          Loading...
   279→        </div>
   280→      </div>
   281→    );
   282→  }
   283→
   284→  return (
   285→    <div className="space-y-6">
   286→      <div className="flex items-center justify-between">
   287→        <div>
   288→          <h1 className="text-2xl font-bold text-foreground">Automated Betting</h1>
   289→          <p className="text-sm text-muted-foreground mt-1">
   290→            Configure auto-betting and manage your bets
   291→          </p>
   292→        </div>
   293→        <div className="flex items-center gap-2">
   294→          {/* Bot Status Indicator */}
   295→          <Badge className={`px-3 py-1.5 ${
   296→            settings.autoBettingEnabled
   297→              ? "bg-primary/10 text-primary border-primary/20"
   298→              : "bg-secondary text-muted-foreground border-border"
   299→          }`}>
   300→            <span className="relative flex h-2 w-2 mr-2">
   301→              {settings.autoBettingEnabled && (
   302→                <>
   303→                  <span className="animate-live-pulse absolute inline-flex h-full w-full rounded-full bg-primary opacity-75" />
   304→                  <span className="relative inline-flex rounded-full h-2 w-2 bg-primary" />
   305→                </>
   306→              )}
   307→              {!settings.autoBettingEnabled && (
   308→                <span className="relative inline-flex rounded-full h-2 w-2 bg-muted-foreground" />
   309→              )}
   310→            </span>
   311→            Bot {settings.autoBettingEnabled ? "Active" : "Inactive"}
   312→          </Badge>
   313→
   314→          {/* Run Bot Button */}
   315→          <Button
   316→            size="sm"
   317→            className="bg-primary text-primary-foreground hover:bg-primary/80"
   318→            onClick={handleRunBot}
   319→            disabled={botRunning || !settings.autoBettingEnabled}
   320→          >
   321→            {botRunning ? (
   322→              <div className="h-4 w-4 border-2 border-primary-foreground border-t-transparent rounded-full animate-spin" />
   323→            ) : (
   324→              <Play className="h-4 w-4" />
   325→            )}
   326→            {botRunning ? "Running..." : "Run Bot"}
   327→          </Button>
   328→
   329→          {/* Settle Button */}
   330→          <Button
   331→            size="sm"
   332→            variant="outline"
   333→            onClick={handleSettle}
   334→            disabled={settleLoading}
   335→            className="border-emerald-400/30 text-emerald-400 hover:bg-emerald-400/10"
   336→          >
   337→            {settleLoading ? (
   338→              <div className="h-4 w-4 border-2 border-emerald-400 border-t-transparent rounded-full animate-spin" />
   339→            ) : (
   340→              <RefreshCw className="h-4 w-4" />
   341→            )}
   342→            Settle
   343→          </Button>
   344→        </div>
   345→      </div>
   346→
   347→      <div className="grid grid-cols-2 sm:grid-cols-5 gap-4">
   348→        <Card className="bg-card border-border">
   349→          <CardContent className="p-3">
   350→            <p className="text-xs text-muted-foreground">Total Bets</p>
   351→            <p className="text-lg font-bold text-foreground">{enrichedBets.length}</p>
   352→          </CardContent>
   353→        </Card>
   354→        <Card className="bg-card border-border">
   355→          <CardContent className="p-3">
   356→            <p className="text-xs text-muted-foreground">AI-Placed</p>
   357→            <p className="text-lg font-bold text-primary">{autoBets.length}</p>
   358→          </CardContent>
   359→        </Card>
   360→        <Card className="bg-card border-border">
   361→          <CardContent className="p-3">
   362→            <p className="text-xs text-muted-foreground">Accumulators</p>
   363→            <p className="text-lg font-bold text-purple-400">{accumulatorBets.length}</p>
   364→          </CardContent>
   365→        </Card>
   366→        <Card className="bg-card border-border">
   367→          <CardContent className="p-3">
   368→            <p className="text-xs text-muted-foreground">Total Stake</p>
   369→            <p className="text-lg font-bold text-foreground">${totalStake.toFixed(2)}</p>
   370→          </CardContent>
   371→        </Card>
   372→        <Card className="bg-card border-border">
   373→          <CardContent className="p-3">
   374→            <p className="text-xs text-muted-foreground">Net Profit</p>
   375→            <p className={`text-lg font-bold ${totalProfit >= 0 ? "text-emerald-400" : "text-red-400"}`}>
   376→              {totalProfit >= 0 ? "+" : ""}${totalProfit.toFixed(2)}
   377→            </p>
   378→          </CardContent>
   379→        </Card>
   380→      </div>
   381→
   382→      {/* Daily Bet Limit Progress Bar */}
   383→      <Card className="bg-card border-border">
   384→        <CardContent className="p-4">
   385→          <div className="flex items-center justify-between mb-2">
   386→            <div className="flex items-center gap-2">
   387→              <Shield className="h-4 w-4 text-amber-400" />
   388→              <span className="text-sm font-medium text-foreground">Daily Bet Limit</span>
   389→            </div>
   390→            <span className="text-sm text-muted-foreground">
   391→              ${dailyStake.toFixed(0)} / ${settings.dailyBetLimit.toFixed(0)}
   392→            </span>
   393→          </div>
   394→          <div className="h-2.5 rounded-full bg-secondary overflow-hidden">
   395→            <div
   396→              className={`h-full rounded-full transition-all ${
   397→                dailyLimitProgress >= 90 ? "bg-red-500" : dailyLimitProgress >= 70 ? "bg-amber-400" : "bg-primary"
   398→              }`}
   399→              style={{ width: `${dailyLimitProgress}%` }}
   400→            />
   401→          </div>
   402→          <div className="flex items-center justify-between mt-1">
   403→            <span className="text-[10px] text-muted-foreground">
   404→              {dailyLimitProgress >= 90 ? "Limit almost reached!" : `${Math.round(dailyLimitProgress)}% of daily limit used`}
   405→            </span>
   406→            <span className="text-[10px] text-muted-foreground">
   407→              ${Math.max(0, settings.dailyBetLimit - dailyStake).toFixed(0)} remaining
   408→            </span>
   409→          </div>
   410→        </CardContent>
   411→      </Card>
   412→
   413→      {/* Risk Limits Display */}
   414→      <div className="grid grid-cols-2 sm:grid-cols-4 gap-3">
   415→        <div className="rounded-lg bg-red-400/5 border border-red-400/10 p-3">
   416→          <div className="flex items-center gap-1.5 mb-1">
   417→            <Shield className="h-3 w-3 text-red-400" />
   418→            <span className="text-xs text-muted-foreground">Daily Stop-Loss</span>
   419→          </div>
   420→          <p className="text-sm font-bold text-red-400">-${settings.stopLossDaily}</p>
   421→        </div>
   422→        <div className="rounded-lg bg-red-400/5 border border-red-400/10 p-3">
   423→          <div className="flex items-center gap-1.5 mb-1">
   424→            <Shield className="h-3 w-3 text-red-400" />
   425→            <span className="text-xs text-muted-foreground">Weekly Stop-Loss</span>
   426→          </div>
   427→          <p className="text-sm font-bold text-red-400">-${settings.stopLossWeekly}</p>
   428→        </div>
   429→        <div className="rounded-lg bg-emerald-400/5 border border-emerald-400/10 p-3">
   430→          <div className="flex items-center gap-1.5 mb-1">
   431→            <Target className="h-3 w-3 text-emerald-400" />
   432→            <span className="text-xs text-muted-foreground">Daily Target</span>
   433→          </div>
   434→          <p className="text-sm font-bold text-emerald-400">+${settings.profitTargetDaily}</p>
   435→        </div>
   436→        <div className="rounded-lg bg-emerald-400/5 border border-emerald-400/10 p-3">
   437→          <div className="flex items-center gap-1.5 mb-1">
   438→            <Target className="h-3 w-3 text-emerald-400" />
   439→            <span className="text-xs text-muted-foreground">Weekly Target</span>
   440→          </div>
   441→          <p className="text-sm font-bold text-emerald-400">+${settings.profitTargetWeekly}</p>
   442→        </div>
   443→      </div>
   444→
   445→      {/* AI Recommended Bets */}
   446→      {aiRecommendedBets.length > 0 && (
   447→        <Card className="bg-card border-border">
   448→          <CardHeader className="pb-3">
   449→            <CardTitle className="flex items-center gap-2 text-base">
   450→              <Brain className="h-4 w-4 text-primary" />
   451→              AI Recommended Bets
   452→            </CardTitle>
   453→          </CardHeader>
   454→          <CardContent>
   455→            <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-3">
   456→              {aiRecommendedBets.slice(0, 6).map((bet) => (
   457→                <div
   458→                  key={bet.id}
   459→                  className="rounded-lg bg-primary/5 border border-primary/10 p-3"
   460→                >
   461→                  <div className="flex items-center justify-between mb-2">
   462→                    <div className="flex items-center gap-1.5">
   463→                      <Brain className="h-3 w-3 text-primary" />
   464→                      <span className="text-xs font-medium text-foreground">
   465→                        {bet.match?.homeTeam} vs {bet.match?.awayTeam}
   466→                      </span>
   467→                    </div>
   468→                    <div className="flex items-center gap-1">
   469→                      <div className="w-12 h-1.5 rounded-full bg-secondary overflow-hidden">
   470→                        <div
   471→                          className={`h-full rounded-full ${
   472→                            (bet.aiConfidence || 0) >= 0.7 ? "bg-primary" : (bet.aiConfidence || 0) >= 0.5 ? "bg-amber-400" : "bg-red-400"
   473→                          }`}
   474→                          style={{ width: `${(bet.aiConfidence || 0) * 100}%` }}
   475→                        />
   476→                      </div>
   477→                      <span className="text-[10px] font-bold text-primary">
   478→                        {Math.round((bet.aiConfidence || 0) * 100)}%
   479→                      </span>
   480→                    </div>
   481→                  </div>
   482→                  <div className="flex items-center justify-between text-xs mb-2">
   483→                    <span className="text-muted-foreground">
   484→                      {bet.selection} @ {bet.odds}
   485→                    </span>
   486→                    <span className="text-foreground">${bet.stake.toFixed(0)} stake</span>
   487→                  </div>
   488→                  <div className="flex items-center justify-between text-[10px] text-muted-foreground mb-2">
   489→                    <span>Commission: {(settings.commissionRate * 100).toFixed(0)}%</span>
   490→                    <span>Projected: ${(bet.potentialWin * (1 - settings.commissionRate)).toFixed(2)}</span>
   491→                  </div>
   492→                  {/* Cashout buttons */}
   493→                  <div className="flex gap-2">
   494→                    {bet.cashoutAmount && (
   495→                      <button
   496→                        onClick={() => handleCashout(bet.id, "partial")}
   497→                        className="flex-1 text-[10px] bg-primary/10 text-primary border border-primary/20 rounded px-2 py-1 hover:bg-primary/20 transition-colors"
   498→                      >
   499→                        <DollarSign className="h-3 w-3 inline" />
   500→                        Partial ${((bet.cashoutAmount || 0) * settings.partialCashoutPercent).toFixed(2)}
   501→                      </button>
   502→                    )}
   503→                    {bet.cashoutAmount && (
   504→                      <button
   505→                        onClick={() => handleCashout(bet.id, "full")}
   506→                        className="flex-1 text-[10px] bg-amber-400/10 text-amber-400 border border-amber-400/20 rounded px-2 py-1 hover:bg-amber-400/20 transition-colors"
   507→                      >
   508→                        <DollarSign className="h-3 w-3 inline" />
   509→                        Full ${bet.cashoutAmount.toFixed(2)}
   510→                      </button>
   511→                    )}
   512→                  </div>
   513→                </div>
   514→              ))}
   515→            </div>
   516→          </CardContent>
   517→        </Card>
   518→      )}
   519→
   520→      <div className="grid grid-cols-1 lg:grid-cols-3 gap-6">
   521→        <div className="lg:col-span-2 space-y-4">
   522→          <div className="flex items-center justify-between">
   523→            <h2 className="text-lg font-semibold text-foreground">Your Bets</h2>
   524→            <div className="flex items-center gap-2">
   525→              {["all", "pending", "won", "lost", "auto", "accumulator"].map((filter) => (
   526→                <Button
   527→                  key={filter}
   528→                  size="xs"
   529→                  variant={betFilter === filter ? "default" : "ghost"}
   530→                  onClick={() => setBetFilter(filter)}
   531→                  className={betFilter === filter ? "bg-primary text-primary-foreground" : ""}
   532→                >
   533→                  {filter.charAt(0).toUpperCase() + filter.slice(1)}
   534→                </Button>
   535→              ))}
   536→            </div>
   537→          </div>
   538→
   539→          <div className="space-y-3 max-h-[calc(100vh-400px)] overflow-y-auto">
   540→            {filteredBets.length === 0 ? (
   541→              <Card className="bg-card border-border">
   542→                <CardContent className="p-8 text-center">
   543→                  <Zap className="h-12 w-12 text-muted-foreground mx-auto mb-3" />
   544→                  <p className="text-sm text-muted-foreground">
   545→                    No bets found for this filter.
   546→                  </p>
   547→                </CardContent>
   548→              </Card>
   549→            ) : (
   550→              filteredBets.map((bet) => (
   551→                <EnhancedBetCard
   552→                  key={bet.id}
   553→                  bet={bet}
   554→                  onCashout={handleCashout}
   555→                  settings={settings}
   556→                />
   557→              ))
   558→            )}
   559→          </div>
   560→        </div>
   561→
   562→        <div className="space-y-4">
   563→          <AutoBetConfig settings={settings} onSave={handleSaveSettings} />
   564→          {user?.id && <BotActivityFeed userId={user.id} compact />}
   565→        </div>
   566→      </div>
   567→    </div>
   568→  );
   569→}
