    13→    13→// ==================== TYPES ====================
    14→    14→
    15→    15→interface ScanResult {
    16→    16→  betsPlaced: number;
    17→    17→  tipsGenerated: number;
    18→    18→  totalStake: number;
    19→    19→  matches: number;

    21→    21→  error?: string;

    23→    23→
    24→    24→interface BotEngineStats {
    25→    25→  userId: string;
    26→    26→  status: "running" | "stopped" | "paused";
    27→    27→  scanIntervalSec: number;
    28→    28→  totalScans: number;
    29→    29→  totalBetsPlaced: number;
    30→    30→  totalStakeUsed: number;
    31→    31→  totalProfit: number;
    32→    32→  lastScanAt: Date | null;
    33→    33→  lastBetAt: Date | null;
    34→    34→  startedAt: Date | null;
    35→    35→  errorCount: number;
    36→    36→  lastError: string | null;
    37→    37→}
    38→    38→
    39→    39→// ==================== SINGLETON ENGINE ====================
    40→    40→
    41→    41→/**
    42→    42→ * BotEngine is a singleton that manages all running bot instances.
    43→    43→ * Each user has their own timer that runs scan cycles on a configurable interval.
    44→    44→ *
    45→    45→ * The engine is designed to:
    46→    46→ * - Run independently of the frontend (no polling required)
    47→    47→ * - Auto-recover from errors (continues running after failed scans)
    48→    48→ * - Auto-stop on risk limits, schedule violations, or no allocation
    49→    49→ * - Persist state in the database (BotSession) for recovery after restarts
    50→    50→ * - Support graceful shutdown
    51→    51→ */
    52→    52→class BotEngine {
    53→    53→  private static instance: BotEngine | null = null;
    54→    54→  private timers: Map<string, NodeJS.Timeout> = new Map();
    55→    55→  private runningUsers: Map<string, BotEngineStats> = new Map();
    56→    56→  private isShuttingDown = false;
    57→    57→
    58→    58→  private constructor() {
    59→    59→    // Register graceful shutdown handlers
    60→    60→    process.on("SIGTERM", () => this.shutdown());
    61→    61→    process.on("SIGINT", () => this.shutdown());
    62→    62→  }
    63→    63→
    64→    64→  static getInstance(): BotEngine {
    65→    65→    if (!BotEngine.instance) {
    66→    66→      BotEngine.instance = new BotEngine();
    67→    67→    }
    68→    68→    return BotEngine.instance;
    69→    69→  }
    70→    70→
    71→    71→  // ==================== LIFECYCLE ====================
    72→    72→
    73→    73→  /**
    74→    74→   * Start the bot for a specific user. Creates a timer that runs scan cycles
    75→    75→   * on the configured interval (default 30 seconds).
    76→    76→   */
    77→    77→  async start(userId: string, scanIntervalSec: number = 30): Promise<{ success: boolean; message: string }> {
    78→    78→    // Don't start if already running
    79→    79→    if (this.timers.has(userId)) {
    80→    80→      return { success: false, message: "Bot is already running for this user" };
    81→    81→    }
    82→    82→
    83→    83→    // Validate prerequisites
    84→    84→    const validation = await this.validatePrerequisites(userId);
    85→    85→    if (!validation.valid) {
    86→    86→      return { success: false, message: validation.error! };
    87→    87→    }
    88→    88→
    89→    89→    // Initialize stats tracking
    90→    90→    const stats: BotEngineStats = {
    91→    91→      userId,
    92→    92→      status: "running",
    93→    93→      scanIntervalSec,
    94→    94→      totalScans: 0,
    95→    95→      totalBetsPlaced: 0,
    96→    96→      totalStakeUsed: 0,
    97→    97→      totalProfit: 0,
    98→    98→      lastScanAt: null,
    99→    99→      lastBetAt: null,
   100→   100→      startedAt: new Date(),
   101→   101→      errorCount: 0,
   102→   102→      lastError: null,
   103→   103→    };
   104→   104→    this.runningUsers.set(userId, stats);
   105→   105→
   106→   106→    // Run the first scan immediately
   107→   107→    try {
   108→   108→      const result = await this.runScanCycle(userId);
   109→   109→      stats.totalScans++;
   110→   110→      stats.totalBetsPlaced += result.betsPlaced;
   111→   111→      stats.totalStakeUsed += result.totalStake;
   112→   112→      stats.lastScanAt = new Date();
   113→   113→      if (result.betsPlaced > 0 || result.tipsGenerated > 0) {
   114→   114→        stats.lastBetAt = new Date();
   115→   115→      }
   116→   116→      if (result.error) {
   117→   117→        stats.errorCount++;
   118→   118→        stats.lastError = result.error;
   119→   119→      }
   120→   120→    } catch (error) {
   121→   121→      console.error(`[BotEngine] First scan failed for user ${userId}:`, error);
   122→   122→      stats.errorCount++;
   123→   123→      stats.lastError = error instanceof Error ? error.message : "Unknown error";
   124→   124→    }
   125→   125→
   126→   126→    // Set up the recurring timer
   127→   127→    const timer = setInterval(async () => {
   128→   128→      if (this.isShuttingDown) return;
   129→   129→
   130→   130→      try {
   131→   131→        const currentStats = this.runningUsers.get(userId);
   132→   132→        if (!currentStats || currentStats.status !== "running") {
   133→   133→          // Bot was stopped externally, clean up
   134→   134→          this.stop(userId, "auto_stopped");
   135→   135→          return;
   136→   136→        }
   137→   137→
   138→   138→        // Check if the session is still running in the DB
   139→   139→        const session = await prisma.botSession.findUnique({ where: { userId } });
   140→   140→        if (!session || session.status !== "running") {
   141→   141→          this.stop(userId, "session_expired");
   142→   142→          return;
   143→   143→        }
   144→   144→
   145→   145→        const result = await this.runScanCycle(userId);
   146→   146→
   147→   147→        currentStats.totalScans++;
   148→   148→        currentStats.totalBetsPlaced += result.betsPlaced;
   149→   149→        currentStats.totalStakeUsed += result.totalStake;
   150→   150→        currentStats.lastScanAt = new Date();
   151→   151→        currentStats.lastError = null; // Clear error on success
   152→   152→
   153→   153→        if (result.betsPlaced > 0) {
   154→   154→          currentStats.lastBetAt = new Date();
   155→   155→        }
   156→   156→
   157→   157→        if (result.error) {
   158→   158→          currentStats.errorCount++;
   159→   159→          currentStats.lastError = result.error;
   160→   160→        }
   161→   161→
   162→   162→        // Update session stats in DB
   163→   163→        await prisma.botSession.update({
   164→   164→          where: { userId },
   165→   165→          data: {
   166→   166→            totalScans: { increment: 1 },
   167→   167→            totalBetsPlaced: { increment: result.betsPlaced },
   168→   168→            totalStakeUsed: { increment: result.totalStake },
   169→   169→            lastScanAt: new Date(),
   170→   170→            lastBetAt: result.betsPlaced > 0 ? new Date() : undefined,
   171→   171→          },
   172→   172→        });
   173→   173→
   174→   174→        // Log scan activity periodically (every 10 scans)
   175→   175→        if (currentStats.totalScans % 10 === 0) {
   176→   176→          await prisma.botLog.create({
   177→   177→            data: {
   178→   178→              userId,
   179→   179→              action: "bot_scan",
   180→   180→              reasoning: `Auto-scan #${currentStats.totalScans}: ${result.matches} matches found, ${result.betsPlaced} bets placed, ${result.skipped} skipped`,
   181→   181→              details: JSON.stringify({
   182→   182→                scanNumber: currentStats.totalScans,
   183→   183→                betsPlaced: result.betsPlaced,
   184→   184→                totalStake: result.totalStake,
   185→   185→                matches: result.matches,
   186→   186→                skipped: result.skipped,
   187→   187→              }),
   188→   188→            },
   189→   189→          });
   190→   190→        }
   191→   191→
   192→   192→        // Auto-settle finished matches every 3rd scan
   193→   193→        if (currentStats.totalScans % 3 === 0) {
   194→   194→          try {
   195→   195→            await this.autoSettle(userId);
   196→   196→          } catch (settleError) {
   197→   197→            console.error(`[BotEngine] Auto-settle error for user ${userId}:`, settleError);
   198→   198→          }
   199→   199→        }
   200→   200→
   201→   201→        // Auto-cashout evaluation every 5th scan
   202→   202→        if (currentStats.totalScans % 5 === 0) {
   203→   203→          try {
   204→   204→            await this.autoCashoutEvaluation(userId);
   205→   205→          } catch (cashoutError) {
   206→   206→            console.error(`[BotEngine] Auto-cashout error for user ${userId}:`, cashoutError);
   207→   207→          }
   208→   208→        }
   209→   209→      } catch (error) {
   210→   210→        console.error(`[BotEngine] Scan cycle error for user ${userId}:`, error);
   211→   211→        const currentStats = this.runningUsers.get(userId);
   212→   212→        if (currentStats) {
   213→   213→          currentStats.errorCount++;
   214→   214→          currentStats.lastError = error instanceof Error ? error.message : "Unknown error";
   215→   215→
   216→   216→          // Auto-stop after too many consecutive errors
   217→   217→          if (currentStats.errorCount >= 10) {
   218→   218→            console.error(`[BotEngine] Too many errors for user ${userId}, stopping bot`);
   219→   219→            await this.stop(userId, "too_many_errors");
   220→   220→          }
   221→   221→        }
   222→   222→      }
   223→   223→    }, scanIntervalSec * 1000);
   224→   224→
   225→   225→    // Prevent Node.js from keeping the process alive just for this timer
   226→   226→    if (timer.unref) {
   227→   227→      timer.unref();
   228→   228→    }
   229→   229→
   230→   230→    this.timers.set(userId, timer);
   231→   231→
   232→   232→    console.log(`[BotEngine] Bot started for user ${userId} with interval ${scanIntervalSec}s`);
   233→   233→
   234→   234→    // Log bot start
   235→   235→    await prisma.botLog.create({
   236→   236→      data: {
   237→   237→        userId,
   238→   238→        action: "bot_started",
   239→   239→        reasoning: `Bot engine started in background mode. Scanning every ${scanIntervalSec}s.`,
   240→   240→        details: JSON.stringify({
   241→   241→          mode: "background",
   242→   242→          scanIntervalSec,
   243→   243→          broker: validation.connectedAccount?.platform,
   244→   244→          allocation: validation.connectedAccount?.allocatedAmount,
   245→   245→        }),
   246→   246→      },
   247→   247→    });
   248→   248→
   249→   249→    return {
   250→   250→      success: true,
   251→   251→      message: stats.tipsGenerated > 0
   252→   252→        ? `Advisor started! Generated ${stats.tipsGenerated} tip(s) in first scan. Running in background.`
   253→   253→        : stats.betsPlaced > 0
   254→   254→        ? `Bot started! Placed ${stats.betsPlaced} bet(s) in first scan. Running in background.`
   255→   255→        : "Bot started! Scanning for matches in background. Will generate tips automatically.",
   256→   256→    };
   257→   257→  }

   259→   259→  /**
   260→   260→   * Stop the bot for a specific user. Clears the timer and updates the session.
   261→   261→   */
   262→   262→  async stop(userId: string, reason: string = "user_stopped"): Promise<void> {
   263→   263→    const timer = this.timers.get(userId);
   264→   264→    if (timer) {
   265→   265→      clearInterval(timer);
   266→   266→      this.timers.delete(userId);
   267→   267→    }
   268→   268→
   269→   269→    const stats = this.runningUsers.get(userId);
   270→   270→    if (stats) {
   271→   271→      stats.status = "stopped";
   272→   272→      this.runningUsers.delete(userId);
   273→   273→    }
   274→   274→
   275→   275→    // Update the database session
   276→   276→    try {
   277→   277→      await prisma.botSession.upsert({
   278→   278→        where: { userId },
   279→   279→        update: {
   280→   280→          status: "stopped",
   281→   281→          stoppedAt: new Date(),
   282→   282→          stopReason: reason,
   283→   283→        },
   284→   284→        create: {
   285→   285→          userId,
   286→   286→          status: "stopped",
   287→   287→          stoppedAt: new Date(),
   288→   288→          stopReason: reason,
   289→   289→        },
   290→   290→      });
   291→   291→
   292→   292→      // Log bot stop
   293→   293→      await prisma.botLog.create({
   294→   294→        data: {
   295→   295→          userId,
   296→   296→          action: "bot_stopped",
   297→   297→          reasoning: `Bot engine stopped. Reason: ${reason}`,
   298→   298→          details: JSON.stringify({
   299→   299→            reason,
   300→   300→            totalScans: stats?.totalScans || 0,
   301→   301→            totalBetsPlaced: stats?.totalBetsPlaced || 0,
   302→   302→            totalStakeUsed: stats?.totalStakeUsed || 0,
   303→   303→            totalProfit: stats?.totalProfit || 0,
   304→   304→            runDuration: stats?.startedAt
   305→   305→              ? Math.round((Date.now() - stats.startedAt.getTime()) / 60000)
   306→   306→              : 0,
   307→   307→            errorCount: stats?.errorCount || 0,
   308→   308→          }),
   309→   309→        },
   310→   310→      });
   311→   311→    } catch (error) {
   312→   312→      console.error(`[BotEngine] Error stopping bot for user ${userId}:`, error);
   313→   313→    }
   314→   314→
   315→   315→    console.log(`[BotEngine] Bot stopped for user ${userId}. Reason: ${reason}`);
   316→   316→  }
   317→   317→
   318→   318→  /**
   319→   319→   * Gracefully shut down all running bots.
   320→   320→   */
   321→   321→  async shutdown(): Promise<void> {
   322→   322→    if (this.isShuttingDown) return;
   323→   323→    this.isShuttingDown = true;
   324→   324→
   325→   325→    console.log(`[BotEngine] Shutting down ${this.timers.size} running bots...`);
   326→   326→
   327→   327→    for (const [userId] of this.timers) {
   328→   328→      await this.stop(userId, "server_shutdown");
   329→   329→    }
   330→   330→
   331→   331→    this.timers.clear();
   332→   332→    this.runningUsers.clear();
   333→   333→  }
   334→   334→
   335→   335→  /**
   336→   336→   * Recover bots that were running before a server restart.
   337→   337→   * Checks the database for sessions with status "running" and restarts them.
   338→   338→   */
   339→   339→  async recoverRunningBots(): Promise<number> {
   340→   340→    try {
   341→   341→      const runningSessions = await prisma.botSession.findMany({
   342→   342→        where: { status: "running" },
   343→   343→      });
   344→   344→
   345→   345→      console.log(`[BotEngine] Found ${runningSessions.length} running sessions to recover`);
   346→   346→
   347→   347→      let recovered = 0;
   348→   348→      for (const session of runningSessions) {
   349→   349→        try {
   350→   350→          const result = await this.start(session.userId, session.scanIntervalSec || 30);
   351→   351→          if (result.success) {
   352→   352→            recovered++;
   353→   353→            console.log(`[BotEngine] Recovered bot for user ${session.userId}`);
   354→   354→          } else {
   355→   355→            console.warn(`[BotEngine] Could not recover bot for user ${session.userId}: ${result.message}`);
   356→   356→          }
   357→   357→        } catch (error) {
   358→   358→          console.error(`[BotEngine] Error recovering bot for user ${session.userId}:`, error);
   359→   359→        }
   360→   360→      }
   361→   361→
   362→   362→      return recovered;
   363→   363→    } catch (error) {
   364→   364→      console.error("[BotEngine] Error recovering bots:", error);
   365→   365→      return 0;
   366→   366→    }
   367→   367→  }

   399→   399→  // ==================== SCAN CYCLE ====================
   400→   400→
   401→   401→  /**
   402→   402→   * Run a single scan cycle - analyze matches and generate tips or place bets.
   403→   403→   * In advisor mode: AI finds value bets → creates Tip → sends Telegram alert.
   404→   404→   * In auto mode: AI finds value bets → places bet via broker → creates Tip.
   405→   405→   */
   406→   406→  private async runScanCycle(userId: string): Promise<ScanResult> {
   407→   407→    const result: ScanResult = { betsPlaced: 0, tipsGenerated: 0, totalStake: 0, matches: 0, skipped: 0 };
   408→   408→
   409→   409→    try {
   410→   410→      // Get user settings
   411→   411→      const settings = await prisma.userSettings.findUnique({ where: { userId } });
   412→   412→      if (!settings) {
   413→   413→        result.error = "Settings not found";
   414→   414→        return result;
   415→   415→      }
   416→   416→
   417→   417→      const isAdvisorMode = settings.botMode === "advisor";
   418→   418→
   419→   419→      if (!settings.autoBettingEnabled && !isAdvisorMode) {
   420→   420→        result.error = "Auto-betting is disabled";
   421→   421→        await this.stop(userId, "auto_betting_disabled");
   422→   422→        return result;
   423→   423→      }
   424→   424→
   425→   425→      // Check schedule
   426→   426→      if (!isWithinBetSchedule(settings.betScheduleStart, settings.betScheduleEnd)) {
   427→   427→        return result;
   428→   428→      }
   429→   429→
   430→   430→      // Check risk limits
   431→   431→      const user = await prisma.user.findUnique({ where: { id: userId } });
   432→   432→      const riskCheck = checkRiskLimits(user?.dailyPnl || 0, user?.weeklyPnl || 0, {
   433→   433→        stopLossDaily: settings.stopLossDaily,
   434→   434→        stopLossWeekly: settings.stopLossWeekly,
   435→   435→        profitTargetDaily: settings.profitTargetDaily,
   436→   436→        profitTargetWeekly: settings.profitTargetWeekly,
   437→   437→      });
   438→   438→
   439→   439→      if (!riskCheck.canBet) {
   440→   440→        const reason = (user?.dailyPnl ?? 0) <= -settings.stopLossDaily ? "stop_loss" : "profit_target";
   441→   441→        await this.stop(userId, reason);
   442→   442→        await prisma.botLog.create({
   443→   443→          data: {
   444→   444→            userId,
   445→   445→            action: reason === "stop_loss" ? "stop_loss_hit" : "profit_target_hit",
   446→   446→            reasoning: riskCheck.reason,
   447→   447→          },
   448→   448→        });
   449→   449→        return result;
   450→   450→      }
   451→   451→
   452→   452→      // ---- Advisor mode: no broker required ----
   453→   453→      // ---- Auto mode: need connected broker ----
   454→   454→      let bettingAccount: Awaited<ReturnType<typeof prisma.bettingAccount.findFirst>> = null;
   455→   455→      let activeAllocation: Awaited<ReturnType<typeof prisma.allocation.findFirst>> = null;
   456→   456→      let remainingDailyLimit = settings.dailyBetLimit;
   457→   457→      let existingBetMatchIds: string[] = [];
   458→   458→
   459→   459→      if (!isAdvisorMode) {
   460→   460→        bettingAccount = await prisma.bettingAccount.findFirst({
   461→   461→          where: { userId, isConnected: true },
   462→   462→          orderBy: { allocatedAmount: "desc" },
   463→   463→        });
   464→   464→
   465→   465→        if (!bettingAccount || bettingAccount.allocatedAmount <= 0) {
   466→   466→          await this.stop(userId, "no_allocation");
   467→   467→          await prisma.botLog.create({
   468→   468→            data: {
   469→   469→              userId,
   470→   470→              action: "bot_scan",
   471→   471→              reasoning: "No connected broker or allocation available. Stopping bot.",
   472→   472→            },
   473→   473→          });
   474→   474→          return result;
   475→   475→        }
   476→   476→
   477→   477→        // Check daily limit
   478→   478→        const todayStart = new Date();
   479→   479→        todayStart.setHours(0, 0, 0, 0);
   480→   480→        const todayBets = await prisma.bet.findMany({
   481→   481→          where: {
   482→   482→            userId,
   483→   483→            placedAt: { gte: todayStart },
   484→   484→            status: { in: ["pending", "won", "lost", "cashed_out", "partial_cashout"] },
   485→   485→          },
   486→   486→        });
   487→   487→        const dailyStake = todayBets.reduce((sum, b) => sum + b.stake, 0);
   488→   488→        existingBetMatchIds = todayBets.map((b) => b.matchId);
   489→   489→
   490→   490→        if (dailyStake >= settings.dailyBetLimit) {
   491→   491→          return result;
   492→   492→        }
   493→   493→
   494→   494→        remainingDailyLimit = Math.min(
   495→   495→          settings.dailyBetLimit - dailyStake,
   496→   496→          bettingAccount.allocatedAmount
   497→   497→        );
   498→   498→
   499→   499→        if (remainingDailyLimit < 5) {
   500→   500→          return result;
   501→   501→        }
   502→   502→
   503→   503→        activeAllocation = await prisma.allocation.findFirst({

   882→   882→  private async validatePrerequisites(userId: string): Promise<{
   883→   883→    valid: boolean;
   884→   884→    error?: string;
   885→   885→    connectedAccount?: { platform: string; allocatedAmount: number };
   886→   886→  }> {
   887→   887→    const settings = await prisma.userSettings.findUnique({ where: { userId } });
   888→   888→    if (!settings) {
   889→   889→      return { valid: false, error: "Settings not found. Please configure your settings first." };
   890→   890→    }
   891→   891→
   892→   892→    const isAdvisorMode = settings.botMode === "advisor";
   893→   893→
   894→   894→    if (!settings.autoBettingEnabled && !isAdvisorMode) {
   895→   895→      return { valid: false, error: "Auto-betting is disabled. Enable it first." };
   896→   896→    }
   897→   897→
   898→   898→    // In advisor mode, no broker is required — just need settings
   899→   899→    if (isAdvisorMode) {
   900→   900→      return { valid: true };
   901→   901→    }
   902→   902→
   903→   903→    // Auto mode requires a connected broker
   904→   904→    const connectedAccount = await prisma.bettingAccount.findFirst({
   905→   905→      where: { userId, isConnected: true },
   906→   906→    });
   907→   907→
   908→   908→    if (!connectedAccount) {
   909→   909→      return { valid: false, error: "No connected broker account. Connect a broker and set allocation first." };
   910→   910→    }
   911→   911→
   912→   912→    if (connectedAccount.allocatedAmount <= 0) {
   913→   913→      return { valid: false, error: "No allocation set. Allocate funds from your broker account first." };
   914→   914→    }
   915→   915→
   916→   916→    return {
   917→   917→      valid: true,
   918→   918→      connectedAccount: {
   919→   919→        platform: connectedAccount.platform,
   920→   920→        allocatedAmount: connectedAccount.allocatedAmount,
   921→   921→      },
   922→   922→    };
   923→   923→  }
   924→   924→
   925→   925→  // ==================== AUTO SETTLEMENT ====================
   926→   926→
   927→   927→  /**
   928→   928→   * Automatically settle bets for finished matches.
   929→   929→   * Called periodically by the engine (every 3rd scan cycle).
   930→   930→   */
   931→   931→  private async autoSettle(userId: string): Promise<number> {
   932→   932→    const user = await prisma.user.findUnique({
   933→   933→      where: { id: userId },
   934→   934→      include: { settings: true },
   935→   935→    });
   936→   936→
   937→   937→    if (!user) return 0;
   938→   938→
   939→   939→    const commissionRate = user.settings?.commissionRate || 0.10;
   940→   940→
   941→   941→    // Find pending bets for finished matches
   942→   942→    const unsettledBets = await prisma.bet.findMany({
   943→   943→      where: {
   944→   944→        userId,
   945→   945→        status: "pending",
   946→   946→        match: { status: "finished" },
   947→   947→      },
   948→   948→      include: { match: true, bettingAccount: true },
   949→   949→    });
   950→   950→
   951→   951→    if (unsettledBets.length === 0) return 0;
   952→   952→
   953→   953→    let settledCount = 0;
   954→   954→
   955→   955→    for (const bet of unsettledBets) {
   956→   956→      if (!bet.match) continue;
   957→   957→
   958→   958→      const match = bet.match;
   959→   959→      const homeScore = match.homeScore ?? 0;
   960→   960→      const awayScore = match.awayScore ?? 0;
   961→   961→
   962→   962→      let betWon = false;
   963→   963→      let resultReason = "";
   964→   964→
   965→   965→      if (bet.betType === "single" || bet.betType === "accumulator_leg") {
   966→   966→        if (bet.selection === match.homeTeam) {
   967→   967→          betWon = homeScore > awayScore;
   968→   968→          resultReason = betWon
   969→   969→            ? `${match.homeTeam} won ${homeScore}-${awayScore}`
   970→   970→            : `${match.homeTeam} did not win (${homeScore}-${awayScore})`;
   971→   971→        } else if (bet.selection === match.awayTeam) {
   972→   972→          betWon = awayScore > homeScore;
   973→   973→          resultReason = betWon
   974→   974→            ? `${match.awayTeam} won ${awayScore}-${homeScore}`
   975→   975→            : `${match.awayTeam} did not win (${awayScore}-${homeScore})`;
   976→   976→        } else if (bet.selection === "Draw") {
   977→   977→          betWon = homeScore === awayScore;
   978→   978→          resultReason = betWon
   979→   979→            ? `Match drawn ${homeScore}-${awayScore}`
   980→   980→            : `Match not drawn (${homeScore}-${awayScore})`;
   981→   981→        } else if (bet.selection === "Over 2.5") {
   982→   982→          const totalGoals = homeScore + awayScore;
   983→   983→          betWon = totalGoals > 2.5;
   984→   984→          resultReason = betWon
   985→   985→            ? `Total goals ${totalGoals} > 2.5`
   986→   986→            : `Total goals ${totalGoals} <= 2.5`;
   987→   987→        } else if (bet.selection === "Under 2.5") {
   988→   988→          const totalGoals = homeScore + awayScore;
   989→   989→          betWon = totalGoals < 2.5;
   990→   990→          resultReason = betWon
   991→   991→            ? `Total goals ${totalGoals} < 2.5`
   992→   992→            : `Total goals ${totalGoals} >= 2.5`;
   993→   993→        }
   994→   994→      }
   995→   995→
   996→   996→      const now = new Date();
   997→   997→
