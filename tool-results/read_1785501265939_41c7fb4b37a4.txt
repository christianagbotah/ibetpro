     1→"use client";
     2→
     3→import { useState, useEffect, useCallback } from "react";
     4→import { usePolling, useFetch } from "@/lib/hooks";
     5→import { useToast } from "@/components/ui/toast";
     6→import { Card, CardContent, CardHeader, CardTitle } from "@/components/ui/card";
     7→import { Badge } from "@/components/ui/badge";
     8→import { Button } from "@/components/ui/button";
     9→import { Radio, Eye, DollarSign, Clock, Zap, Volume2, VolumeX, Play, TrendingUp, RefreshCw, Layers, Brain } from "lucide-react";
    10→import { useAuth } from "@/components/auth/auth-provider";
    11→import Link from "next/link";
    12→

    30→  aiRecommended: string | null;
    31→  aiAnalysis: string | null;

    33→

    35→  id: string;
    36→  betType: string;
    37→  selection: string;
    38→  odds: number;
    39→  stake: number;
    40→  potentialWin: number;
    41→  status: string;
    42→  isAutoPlaced: boolean;
    43→  aiConfidence: number | null;
    44→  aiReasoning: string | null;
    45→  accumulatorId: string | null;
    46→  partialCashoutAmount: number | null;
    47→  match?: {
    48→    id: string;
    49→    homeTeam: string;
    50→    awayTeam: string;
    51→    sport: string;
    52→    league: string;
    53→    status: string;
    54→    homeScore: number | null;
    55→    awayScore: number | null;
    56→    minute: number | null;
    57→  };
    58→}
    59→
    60→interface CashoutRec {
    61→  shouldCashout: boolean;
    62→  cashoutAmount: number;
    63→  partialCashoutAmount: number;
    64→  reasoning: string;
    65→  urgency: "low" | "medium" | "high";
    66→  waitOrCashout: "wait" | "cashout_partial" | "cashout_full" | "wait_for_settlement";
    67→  settlementProbability: number;
    68→}
    69→
    70→interface MatchEvent {
    71→  matchId: string;
    72→  text: string;

    75→
    76→export default function MonitorPage() {
    77→  const { addToast } = useToast();
    78→  const { user } = useAuth();
    79→  const { data: allMatches, loading: matchesLoading, refetch: refetchMatches } = usePolling<LiveMatch[]>("/api/matches", 15000, []);
    80→  const { data: bets, loading: betsLoading, refetch: refetchBets } = usePolling<ActiveBet[]>("/api/bets?status=pending", 15000, []);
    81→  const [cashoutRecs, setCashoutRecs] = useState<Record<string, CashoutRec>>({});
    82→  const [cashedOutBets, setCashedOutBets] = useState<Set<string>>(new Set());
    83→  const [soundEnabled, setSoundEnabled] = useState(false);
    84→  const [simulating, setSimulating] = useState<string | null>(null);
    85→  const [matchEvents, setMatchEvents] = useState<MatchEvent[]>([]);
    86→  const [autoSettleRunning, setAutoSettleRunning] = useState(false);
    87→
    88→  const loading = matchesLoading || betsLoading;
    89→
    90→  // Fetch cashout recommendations for live bets
    91→  useEffect(() => {
    92→    const liveBets = bets.filter((b) => b.match?.status === "live" && !cashedOutBets.has(b.id));
    93→    let cancelled = false;
    94→
    95→    async function fetchCashouts() {
    96→      for (const bet of liveBets) {
    97→        if (cancelled) break;
    98→        try {
    99→          const res = await fetch(`/api/cashout?betId=${bet.id}`);
   100→          if (res.ok && !cancelled) {
   101→            const result = await res.json();
   102→            setCashoutRecs((prev) => ({
   103→              ...prev,
   104→              [bet.id]: result.cashoutRecommendation,
   105→            }));
   106→          }
   107→        } catch {
   108→          // Ignore cashout errors
   109→        }
   110→      }
   111→    }
   112→
   113→    if (liveBets.length > 0) {
   114→      fetchCashouts();
   115→    }
   116→
   117→    return () => {
   118→      cancelled = true;
   119→    };
   120→  }, [bets, cashedOutBets]);
   121→
   122→  const handleCashout = useCallback(async (betId: string, type: "full" | "partial" = "full") => {
   123→    setCashedOutBets((prev) => new Set(prev).add(betId));
   124→    try {
   125→      const res = await fetch("/api/cashout", {
   126→        method: "POST",
   127→        headers: { "Content-Type": "application/json" },
   128→        body: JSON.stringify({ betId, cashoutType: type }),
   129→      });
   130→      if (res.ok) {
   131→        const result = await res.json();
   132→        addToast("success", `${type === "partial" ? "Partial" : "Full"} cashout: $${result.amount?.toFixed(2)}`);
   133→      } else {
   134→        const data = await res.json();
   135→        addToast("error", data.error || "Cashout failed");
   136→      }
   137→    } catch {
   138→      addToast("error", "Cashout failed");
   139→    }
   140→    refetchBets();
   141→  }, [addToast, refetchBets]);
   142→
   143→  const handleSimulate = useCallback(async (matchId: string) => {
   144→    setSimulating(matchId);
   145→    try {
   146→      const res = await fetch("/api/matches/simulate", {
   147→        method: "POST",
   148→        headers: { "Content-Type": "application/json" },
   149→        body: JSON.stringify({ matchId }),
   150→      });
   151→      if (res.ok) {
   152→        const result = await res.json();
   153→        if (result.events && result.events.length > 0) {
   154→          result.events.forEach((event: string) => {
   155→            setMatchEvents((prev) => [
   156→              { matchId, text: event, timestamp: Date.now() },
   157→              ...prev,
   158→            ].slice(0, 20));
   159→          });
   160→          addToast("info", `Match progressed: ${result.events.join(", ")}`);
   161→        } else {
   162→          addToast("info", "Match progressed, no goals scored");
   163→        }
   164→        refetchMatches();
   165→        refetchBets();
   166→      } else {
   167→        addToast("error", "Failed to simulate match progress");
   168→      }
   169→    } catch {
   170→      addToast("error", "Failed to simulate match progress");
   171→    } finally {
   172→      setSimulating(null);
   173→    }
   174→  }, [addToast, refetchMatches, refetchBets]);
   175→
   176→  const handleAutoSettle = useCallback(async () => {
   177→    if (!user?.id) return;
   178→    setAutoSettleRunning(true);
   179→    try {
   180→      const res = await fetch("/api/settle", {
   181→        method: "POST",
   182→        headers: { "Content-Type": "application/json" },
   183→        body: JSON.stringify({ userId: user.id }),
   184→      });
   185→      if (res.ok) {
   186→        const result = await res.json();
   187→        if (result.settled > 0) {
   188→          addToast("success", `Settled ${result.settled} bet(s). Profit: $${result.totalProfit?.toFixed(2)}`);
   189→        } else {
   190→          addToast("info", "No bets to settle");
   191→        }
   192→        refetchBets();
   193→      }
   194→    } catch {
   195→      addToast("error", "Settlement failed");
   196→    } finally {
   197→      setAutoSettleRunning(false);
   198→    }
   199→  }, [user?.id, addToast, refetchBets]);
   200→
   201→  const liveMatches = allMatches.filter((m) => m.status === "live" || m.status === "upcoming");
   202→  const liveMatchesList = liveMatches.filter((m) => m.status === "live");
   203→  const upcomingMatchesList = liveMatches.filter((m) => m.status === "upcoming");
   204→  const activeBets = bets.filter((b) => !cashedOutBets.has(b.id));
   205→  const accumulatorBets = activeBets.filter((b) => b.accumulatorId);
   206→  const singleBets = activeBets.filter((b) => !b.accumulatorId);
   207→
   208→  if (loading) {
   209→    return (
   210→      <div className="flex items-center justify-center h-64">
   211→        <div className="flex items-center gap-2 text-muted-foreground">
   212→          <div className="h-5 w-5 border-2 border-primary border-t-transparent rounded-full animate-spin" />
   213→          Loading monitor...
   214→        </div>
   215→      </div>
   216→    );
   217→  }
   218→
   219→  return (
   220→    <div className="space-y-6">
   221→      <div className="flex items-center justify-between">
   222→        <div>
   223→          <h1 className="text-2xl font-bold text-foreground">Live Monitor</h1>
   224→          <p className="text-sm text-muted-foreground mt-1">
   225→            Track live matches and manage active bets in real-time
   226→          </p>
   227→        </div>
   228→        <div className="flex items-center gap-2">
   229→          <Button
   230→            variant="outline"
   231→            size="sm"
   232→            onClick={() => setSoundEnabled(!soundEnabled)}
   233→            className={`${soundEnabled ? "border-primary/30 text-primary" : "border-border text-muted-foreground"}`}
   234→          >
   235→            {soundEnabled ? <Volume2 className="h-4 w-4 mr-1.5" /> : <VolumeX className="h-4 w-4 mr-1.5" />}
   236→            {soundEnabled ? "Sound On" : "Sound Off"}
   237→          </Button>
   238→          <Button
   239→            size="sm"
   240→            variant="outline"
   241→            onClick={handleAutoSettle}
   242→            disabled={autoSettleRunning}
   243→            className="border-emerald-400/30 text-emerald-400 hover:bg-emerald-400/10"
   244→          >
   245→            {autoSettleRunning ? (
   246→              <div className="h-4 w-4 border-2 border-emerald-400 border-t-transparent rounded-full animate-spin" />
   247→            ) : (
   248→              <RefreshCw className="h-4 w-4 mr-1.5" />
   249→            )}
   250→            Settle Bets
   251→          </Button>
   252→        </div>
   253→      </div>
   254→
   255→      <div className="grid grid-cols-2 sm:grid-cols-5 gap-4">
   256→        <Card className="bg-card border-border">
   257→          <CardContent className="p-3 flex items-center gap-3">
   258→            <div className="flex h-9 w-9 items-center justify-center rounded-lg bg-red-500/10">
   259→              <Radio className="h-4 w-4 text-red-500" />
   260→            </div>
   261→            <div>
   262→              <p className="text-xs text-muted-foreground">Live Matches</p>
   263→              <p className="text-lg font-bold text-foreground">{liveMatchesList.length}</p>
   264→            </div>
   265→          </CardContent>
   266→        </Card>
   267→        <Card className="bg-card border-border">
   268→          <CardContent className="p-3 flex items-center gap-3">
   269→            <div className="flex h-9 w-9 items-center justify-center rounded-lg bg-primary/10">
   270→              <Zap className="h-4 w-4 text-primary" />
   271→            </div>
   272→            <div>
   273→              <p className="text-xs text-muted-foreground">Active Bets</p>
   274→              <p className="text-lg font-bold text-foreground">{activeBets.length}</p>
   275→            </div>
   276→          </CardContent>
   277→        </Card>
   278→        <Card className="bg-card border-border">
   279→          <CardContent className="p-3 flex items-center gap-3">
   280→            <div className="flex h-9 w-9 items-center justify-center rounded-lg bg-purple-400/10">
   281→              <Layers className="h-4 w-4 text-purple-400" />
   282→            </div>
   283→            <div>
   284→              <p className="text-xs text-muted-foreground">Accumulators</p>
   285→              <p className="text-lg font-bold text-foreground">{accumulatorBets.length}</p>
   286→            </div>
   287→          </CardContent>
   288→        </Card>
   289→        <Card className="bg-card border-border">
   290→          <CardContent className="p-3 flex items-center gap-3">
   291→            <div className="flex h-9 w-9 items-center justify-center rounded-lg bg-amber-400/10">
   292→              <DollarSign className="h-4 w-4 text-amber-400" />
   293→            </div>
   294→            <div>
   295→              <p className="text-xs text-muted-foreground">Cashout Offers</p>
   296→              <p className="text-lg font-bold text-foreground">
   297→                {Object.keys(cashoutRecs).length}
   298→              </p>
   299→            </div>
   300→          </CardContent>
   301→        </Card>
   302→        <Card className="bg-card border-border">
   303→          <CardContent className="p-3 flex items-center gap-3">
   304→            <div className="flex h-9 w-9 items-center justify-center rounded-lg bg-emerald-400/10">
   305→              <Eye className="h-4 w-4 text-emerald-400" />
   306→            </div>
   307→            <div>
   308→              <p className="text-xs text-muted-foreground">Upcoming</p>
   309→              <p className="text-lg font-bold text-foreground">{upcomingMatchesList.length}</p>
   310→            </div>
   311→          </CardContent>
   312→        </Card>
   313→      </div>
   314→
   315→      <div className="grid grid-cols-1 lg:grid-cols-2 gap-6">
   316→        {/* Live Matches with Simulate */}
   317→        <Card className="bg-card border-border">
   318→          <CardHeader className="pb-3">
   319→            <CardTitle className="flex items-center gap-2 text-base">
   320→              <Radio className="h-4 w-4 text-red-500" />
   321→              Live Matches
   322→            </CardTitle>
   323→          </CardHeader>
   324→          <CardContent className="space-y-3 max-h-[calc(100vh-360px)] overflow-y-auto">
   325→            {liveMatchesList.length === 0 ? (
   326→              <p className="text-sm text-muted-foreground text-center py-4">
   327→                No live matches at the moment
   328→              </p>
   329→            ) : (
   330→              liveMatchesList.map((match) => (
   331→                <div
   332→                  key={match.id}
   333→                  className="rounded-lg border border-red-500/20 bg-red-500/5 p-4"
   334→                >
   335→                  <div className="flex items-center justify-between mb-2">
   336→                    <div className="flex items-center gap-2">
   337→                      <span className="relative flex h-2.5 w-2.5">
   338→                        <span className="animate-live-pulse absolute inline-flex h-full w-full rounded-full bg-red-500 opacity-75" />
   339→                        <span className="relative inline-flex rounded-full h-2.5 w-2.5 bg-red-500" />
   340→                      </span>
   341→                      <Badge className="text-[10px] bg-red-500/20 text-red-400 border-red-500/30">
   342→                        LIVE
   343→                      </Badge>
   344→                      <span className="text-xs text-muted-foreground">{match.minute}&apos;</span>
   345→                    </div>
   346→                    <div className="flex items-center gap-2">
   347→                      <span className="text-xs text-muted-foreground">{match.league}</span>
   348→                      <Button
   349→                        size="xs"
   350→                        variant="outline"
   351→                        className="border-primary/30 text-primary hover:bg-primary/10"
   352→                        onClick={() => handleSimulate(match.id)}
   353→                        disabled={simulating === match.id}
   354→                      >
   355→                        {simulating === match.id ? (
   356→                          <div className="h-3 w-3 border-2 border-primary border-t-transparent rounded-full animate-spin" />
   357→                        ) : (
   358→                          <Play className="h-3 w-3" />
   359→                        )}
   360→                        Simulate
   361→                      </Button>
   362→                    </div>
   363→                  </div>
   364→
   365→                  <Link href={`/matches/${match.id}`} className="block">
   366→                    <div className="flex items-center justify-center gap-6 my-3">
   367→                      <div className="text-center">
   368→                        <p className="text-sm font-medium text-foreground">{match.homeTeam}</p>
   369→                        <p className="text-3xl font-bold text-foreground">{match.homeScore}</p>
   370→                      </div>
   371→                      <div className="text-lg text-muted-foreground">-</div>
   372→                      <div className="text-center">
   373→                        <p className="text-sm font-medium text-foreground">{match.awayTeam}</p>
   374→                        <p className="text-3xl font-bold text-foreground">{match.awayScore}</p>
   375→                      </div>
   376→                    </div>
   377→                  </Link>
   378→
   379→                  <div className="mt-2">
   380→                    <div className="h-1.5 rounded-full bg-secondary overflow-hidden">
   381→                      <div
   382→                        className="h-full rounded-full bg-red-500 transition-all"
   383→                        style={{ width: `${((match.minute || 0) / 90) * 100}%` }}
   384→                      />
   385→                    </div>
   386→                    <div className="flex justify-between text-[10px] text-muted-foreground mt-1">
   387→                      <span>KO</span>
   388→                      <span>HT</span>
   389→                      <span>FT</span>
   390→                    </div>
   391→                  </div>
   392→
   393→                  {match.aiRecommended && (
   394→                    <div className="mt-3 rounded bg-primary/5 border border-primary/10 p-2">
   395→                      <div className="flex items-center gap-1">
   396→                        <Brain className="h-3 w-3 text-primary" />
   397→                        <span className="text-[10px] text-muted-foreground">AI recommends:</span>
   398→                        <span className="text-xs font-medium text-primary">
   399→                          {match.aiRecommended === "home" ? match.homeTeam : match.aiRecommended === "away" ? match.awayTeam : match.aiRecommended}
   400→                        </span>
   401→                      </div>
   402→                    </div>
   403→                  )}
   404→                </div>
   405→              ))
   406→            )}
   407→
   408→            {upcomingMatchesList.length > 0 && (
   409→              <>
   410→                <div className="text-xs font-medium text-muted-foreground uppercase tracking-wider mt-4">
   411→                  Upcoming
   412→                </div>
   413→                {upcomingMatchesList.slice(0, 3).map((match) => (
   414→                  <Link key={match.id} href={`/matches/${match.id}`} className="block">
   415→                    <div className="rounded-lg bg-secondary/50 p-3 hover:bg-secondary transition-colors">
   416→                      <div className="flex items-center justify-between">
   417→                        <div>
   418→                          <p className="text-sm font-medium text-foreground">
   419→                            {match.homeTeam} vs {match.awayTeam}
   420→                          </p>
   421→                          <p className="text-xs text-muted-foreground">
   422→                            {match.league} &middot; {new Date(match.commenceTime).toLocaleTimeString([], { hour: "2-digit", minute: "2-digit" })}
   423→                          </p>
   424→                        </div>
   425→                        <div className="flex items-center gap-1">
   426→                          <Clock className="h-3 w-3 text-muted-foreground" />
   427→                          <span className="text-xs text-muted-foreground">Upcoming</span>
   428→                        </div>
   429→                      </div>
   430→                    </div>
   431→                  </Link>
   432→                ))}
   433→              </>
   434→            )}
   435→          </CardContent>
   436→        </Card>
   437→
   438→        {/* Active Bets & Cashout */}
   439→        <Card className="bg-card border-border">
   440→          <CardHeader className="pb-3">
   441→            <CardTitle className="flex items-center gap-2 text-base">
   442→              <Zap className="h-4 w-4 text-amber-400" />
   443→              Active Bets & Cashout
   444→            </CardTitle>
   445→          </CardHeader>
   446→          <CardContent className="space-y-3 max-h-[calc(100vh-360px)] overflow-y-auto">
   447→            {activeBets.length === 0 ? (
   448→              <p className="text-sm text-muted-foreground text-center py-4">
   449→                No active bets
   450→              </p>
   451→            ) : (
   452→              activeBets.map((bet) => {
   453→                const cashoutRec = cashoutRecs[bet.id];
   454→                const isSelectedOnLiveMatch = bet.match?.status === "live";
   455→                const isWinning =
   456→                  isSelectedOnLiveMatch &&
   457→                  ((bet.selection === bet.match?.homeTeam && (bet.match?.homeScore ?? 0) > (bet.match?.awayScore ?? 0)) ||
   458→                    (bet.selection === bet.match?.awayTeam && (bet.match?.awayScore ?? 0) > (bet.match?.homeScore ?? 0)));
   459→                const isAcca = !!bet.accumulatorId;
   460→
   461→                return (
   462→                  <div
   463→                    key={bet.id}
   464→                    className={`rounded-lg border p-4 ${
   465→                      isSelectedOnLiveMatch
   466→                        ? isWinning
   467→                          ? "border-emerald-400/20 bg-emerald-400/5"
   468→                          : "border-red-400/20 bg-red-400/5"
   469→                        : isAcca
   470→                        ? "border-purple-400/20 bg-purple-400/5"
   471→                        : "border-border bg-secondary/50"
   472→                    }`}
   473→                  >
   474→                    <div className="flex items-center justify-between mb-2">
   475→                      <div className="flex items-center gap-2">
   476→                        {bet.isAutoPlaced && (
   477→                          <Badge className="text-[10px] bg-primary/10 text-primary border-primary/20">
   478→                            AI
   479→                          </Badge>
   480→                        )}
   481→                        {isAcca && (
   482→                          <Badge className="text-[10px] bg-purple-400/10 text-purple-400 border-purple-400/20">
   483→                            <Layers className="h-3 w-3 mr-0.5" />
   484→                            Acca
   485→                          </Badge>
   486→                        )}
   487→                        <span className="text-sm font-medium text-foreground">
   488→                          {bet.match?.homeTeam} vs {bet.match?.awayTeam}
   489→                        </span>
   490→                      </div>
   491→                      {isSelectedOnLiveMatch && (
   492→                        <Badge
   493→                          className={`text-[10px] ${
   494→                            isWinning
   495→                              ? "bg-emerald-400/10 text-emerald-400"
   496→                              : "bg-red-400/10 text-red-400"
   497→                          }`}
   498→                        >
   499→                          {isWinning ? "WINNING" : "LOSING"}
   500→                        </Badge>
   501→                      )}
   502→                    </div>
   503→
   504→                    <div className="flex items-center justify-between text-xs mb-2">
   505→                      <span className="text-muted-foreground">
   506→                        {bet.selection} @ {bet.odds}
   507→                      </span>
   508→                      <span className="text-foreground font-medium">
   509→                        ${bet.stake.toFixed(2)} stake
   510→                      </span>
   511→                    </div>
   512→
   513→                    {isSelectedOnLiveMatch && bet.match?.homeScore != null && (
   514→                      <div className="flex items-center justify-center gap-4 my-2">
   515→                        <span className="text-lg font-bold text-foreground">
   516→                          {bet.match.homeScore} - {bet.match.awayScore}
   517→                        </span>
   518→                        <span className="text-xs text-muted-foreground">
   519→                          {bet.match.minute}&apos;
   520→                        </span>
   521→                      </div>
   522→                    )}
   523→
   524→                    {/* Cashout visualization with progress bar */}
