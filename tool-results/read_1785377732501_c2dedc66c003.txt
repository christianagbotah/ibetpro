   143→function factorial(n: number): number {
   144→  if (n <= 1) return 1;
   145→  let result = 1;
   146→  for (let i = 2; i <= n; i++) result *= i;
   147→  return result;
   148→}

   150→function poissonProbability(lambda: number, k: number): number {
   151→  return (Math.pow(lambda, k) * Math.exp(-lambda)) / factorial(k);
   152→}

   154→function normalCDF(x: number): number {
   155→  const a1 = 0.254829592;
   156→  const a2 = -0.284496736;
   157→  const a3 = 1.421413741;
   158→  const a4 = -1.453152027;
   159→  const a5 = 1.061405429;
   160→  const p = 0.3275911;
   161→  const sign = x < 0 ? -1 : 1;
   162→  x = Math.abs(x) / Math.sqrt(2);
   163→  const t = 1.0 / (1.0 + p * x);
   164→  const y = 1.0 - ((((a5 * t + a4) * t + a3) * t + a2) * t + a1) * t * Math.exp(-x * x);
   165→  return 0.5 * (1.0 + sign * y);
   166→}

   168→function gaussianRandom(mean: number = 0, stdev: number = 1): number {
   169→  const u1 = Math.random();
   170→  const u2 = Math.random();
   171→  const z = Math.sqrt(-2.0 * Math.log(u1)) * Math.cos(2.0 * Math.PI * u2);
   172→  return z * stdev + mean;
   173→}

   177→function parseForm(form: string | null): number {
   178→  if (!form) return 0.5;
   179→  const chars = form.replace(/[^WDL]/g, "").split("");
   180→  if (chars.length === 0) return 0.5;
   181→  let score = 0;
   182→  let totalWeight = 0;
   183→  // Recent matches weighted more heavily
   184→  for (let i = 0; i < chars.length; i++) {
   185→    const weight = 1 + (i / chars.length) * 0.5; // More recent = higher weight
   186→    if (chars[i] === "W") score += 1 * weight;
   187→    else if (chars[i] === "D") score += 0.4 * weight;
   188→    totalWeight += weight;
   189→  }
   190→  return totalWeight > 0 ? score / totalWeight : 0.5;
   191→}

   193→function parseRecord(record: string | null): { wins: number; draws: number; losses: number; rate: number } {
   194→  if (!record) return { wins: 0, draws: 0, losses: 0, rate: 0.5 };
   195→  const parts = record.split("-").map(Number);
   196→  if (parts.length < 3 || parts.some(isNaN)) return { wins: 0, draws: 0, losses: 0, rate: 0.5 };
   197→  const [wins, draws, losses] = parts;
   198→  const total = wins + draws + losses;
   199→  return { wins, draws, losses, rate: total > 0 ? (wins + draws * 0.4) / total : 0.5 };
   200→}

   204→function calculateEloProbability(homeElo: number, awayElo: number, isHome: boolean = true): number {
   205→  // Home advantage bonus of 65 Elo points (well-established in football analytics)
   206→  const homeAdvantage = isHome ? 65 : 0;
   207→  const eloDiff = (homeElo + homeAdvantage) - awayElo;
   208→  return 1 / (1 + Math.pow(10, -eloDiff / 400));
   209→}
   210→
   211→function calculateEloDrawProbability(homeElo: number, awayElo: number): number {
   212→  // Draw probability is highest when teams are close in rating
   213→  const eloDiff = Math.abs(homeElo - awayElo);
   214→  // Base draw probability ~26% for football, decreasing with rating gap
   215→  const baseDrawProb = 0.26;
   216→  const gapPenalty = eloDiff / 1000; // Larger gap = fewer draws
   217→  return Math.max(0.12, Math.min(0.35, baseDrawProb - gapPenalty));
   218→}
   219→
   220→function predictWithElo(homeElo: number, awayElo: number): { homeWin: number; draw: number; awayWin: number } {
   221→  const homeWinRaw = calculateEloProbability(homeElo, awayElo, true);
   222→  const drawProb = calculateEloDrawProbability(homeElo, awayElo);
   223→  const awayWinRaw = 1 - homeWinRaw;
   224→
   225→  // Adjust for draw probability
   226→  const remaining = 1 - drawProb;
   227→  const homeWin = homeWinRaw * remaining;
   228→  const awayWin = awayWinRaw * remaining;
   229→
   230→  return { homeWin, draw: drawProb, awayWin };
   231→}

   233→// ==================== POISSON MODEL ====================
   234→
   235→export function calculatePoissonProbabilities(
   236→  homeStats: TeamStatsData | null,
   237→  awayStats: TeamStatsData | null,
   238→  leagueAvgGoals: number = 1.35
   239→): PoissonResult {
   240→  let expectedHomeGoals = leagueAvgGoals;
   241→  let expectedAwayGoals = leagueAvgGoals;
   242→
   243→  if (homeStats && awayStats) {
   244→    // Calculate attack strength and defense weakness relative to league average
   245→    const homeAttackStrength = homeStats.goalsFor / Math.max(1, homeStats.matchesPlayed) / leagueAvgGoals;
   246→    const awayDefenseStrength = awayStats.goalsAgainst / Math.max(1, awayStats.matchesPlayed) / leagueAvgGoals;
   247→    const awayAttackStrength = awayStats.goalsFor / Math.max(1, awayStats.matchesPlayed) / leagueAvgGoals;
   248→    const homeDefenseStrength = homeStats.goalsAgainst / Math.max(1, homeStats.matchesPlayed) / leagueAvgGoals;
   249→
   250→    // Use xG data if available for more accurate expected goals
   251→    const homeXgPerGame = homeStats.xgFor > 0 ? homeStats.xgFor / Math.max(1, homeStats.matchesPlayed) : 0;
   252→    const awayXgPerGame = awayStats.xgFor > 0 ? awayStats.xgFor / Math.max(1, awayStats.matchesPlayed) : 0;
   253→
   254→    // Blend actual goals and xG for more robust estimation
   255→    const homeGoalRate = homeStats.goalsFor / Math.max(1, homeStats.matchesPlayed);
   256→    const awayGoalRate = awayStats.goalsFor / Math.max(1, awayStats.matchesPlayed);
   257→
   258→    const homeAttackRate = homeXgPerGame > 0 ? (homeGoalRate * 0.6 + homeXgPerGame * 0.4) : homeGoalRate;
   259→    const awayAttackRate = awayXgPerGame > 0 ? (awayGoalRate * 0.6 + awayXgPerGame * 0.4) : awayGoalRate;
   260→
   261→    // Expected goals = attack strength * opponent defense weakness * league avg
   262→    expectedHomeGoals = homeAttackStrength * awayDefenseStrength * leagueAvgGoals;
   263→    expectedAwayGoals = awayAttackStrength * homeDefenseStrength * leagueAvgGoals;
   264→
   265→    // Override with blended xG model if available
   266→    if (homeXgPerGame > 0 || awayXgPerGame > 0) {
   267→      const homeConcedeRate = homeStats.goalsAgainst / Math.max(1, homeStats.matchesPlayed);
   268→      const awayConcedeRate = awayStats.goalsAgainst / Math.max(1, awayStats.matchesPlayed);
   269→      const homeXgAgainst = homeStats.xgAgainst > 0 ? homeStats.xgAgainst / Math.max(1, homeStats.matchesPlayed) : homeConcedeRate;
   270→      const awayXgAgainst = awayStats.xgAgainst > 0 ? awayStats.xgAgainst / Math.max(1, awayStats.matchesPlayed) : awayConcedeRate;
   271→
   272→      // Blend Poisson expected with xG model
   273→      const xgHomeExpected = (homeAttackRate + awayXgAgainst) / 2;
   274→      const xgAwayExpected = (awayAttackRate + homeXgAgainst) / 2;
   275→
   276→      expectedHomeGoals = expectedHomeGoals * 0.4 + xgHomeExpected * 0.6;
   277→      expectedAwayGoals = expectedAwayGoals * 0.4 + xgAwayExpected * 0.6;
   278→    }
   279→
   280→    // Home advantage boost (1.15x for home team attack)
   281→    expectedHomeGoals *= 1.15;
   282→    expectedAwayGoals *= 0.90;
   283→  }
   284→
   285→  // Ensure reasonable bounds
   286→  expectedHomeGoals = Math.max(0.3, Math.min(4.0, expectedHomeGoals));
   287→  expectedAwayGoals = Math.max(0.2, Math.min(3.5, expectedAwayGoals));
   288→
   289→  // Calculate score matrix (0-7 goals each)
   290→  const maxGoals = 7;
   291→  const scoreProbabilities: Record<string, number> = {};
   292→  let homeWinProb = 0;
   293→  let drawProb = 0;
   294→  let awayWinProb = 0;
   295→
   296→  for (let i = 0; i <= maxGoals; i++) {
   297→    for (let j = 0; j <= maxGoals; j++) {
   298→      const prob = poissonProbability(expectedHomeGoals, i) * poissonProbability(expectedAwayGoals, j);
   299→      const key = `${i}-${j}`;
   300→      scoreProbabilities[key] = Math.round(prob * 10000) / 10000;
   301→
   302→      if (i > j) homeWinProb += prob;
   303→      else if (i === j) drawProb += prob;
   304→      else awayWinProb += prob;
   305→    }
   306→  }
   307→
   308→  // Normalize (account for probabilities beyond maxGoals)
   309→  const total = homeWinProb + drawProb + awayWinProb;
   310→  homeWinProb /= total;
   311→  drawProb /= total;
   312→  awayWinProb /= total;
   313→
   314→  // Calculate over/under probabilities for common lines
   315→  const overUnderProbabilities: { line: number; overProb: number; underProb: number }[] = [];
   316→  for (const line of [0.5, 1.5, 2.5, 3.5, 4.5]) {
   317→    let underProb = 0;
   318→    const totalExpected = expectedHomeGoals + expectedAwayGoals;
   319→
   320→    for (let i = 0; i <= 8; i++) {
   321→      const prob = poissonProbability(totalExpected, i);
   322→      if (i < line) {
   323→        underProb += prob;
   324→      }
   325→    }
   326→
   327→    overUnderProbabilities.push({
   328→      line,
   329→      overProb: Math.round((1 - underProb) * 100) / 100,
   330→      underProb: Math.round(underProb * 100) / 100,
   331→    });
   332→  }
   333→
   334→  return {
   335→    expectedHomeGoals: Math.round(expectedHomeGoals * 100) / 100,
   336→    expectedAwayGoals: Math.round(expectedAwayGoals * 100) / 100,
   337→    homeWinProb: Math.round(homeWinProb * 100) / 100,
   338→    drawProb: Math.round(drawProb * 100) / 100,
   339→    awayWinProb: Math.round(awayWinProb * 100) / 100,
   340→    scoreProbabilities,
   341→    overUnderProbabilities,
   342→  };
   343→}

   345→// ==================== MONTE CARLO SIMULATION ====================
   346→
   347→function runMonteCarloSimulation(
   348→  expectedHomeGoals: number,
   349→  expectedAwayGoals: number,
   350→  iterations: number = 10000
   351→): MonteCarloResult {
   352→  let homeWins = 0;
   353→  let draws = 0;
   354→  let awayWins = 0;
   355→  let totalHomeGoals = 0;
   356→  let totalAwayGoals = 0;
   357→  const scoreCounts: Record<string, number> = {};
   358→
   359→  for (let i = 0; i < iterations; i++) {
   360→    // Simulate goals using Poisson with random variation
   361→    const homeGoals = poissonRandom(expectedHomeGoals);
   362→    const awayGoals = poissonRandom(expectedAwayGoals);
   363→
   364→    totalHomeGoals += homeGoals;
   365→    totalAwayGoals += awayGoals;
   366→
   367→    const scoreKey = `${homeGoals}-${awayGoals}`;
   368→    scoreCounts[scoreKey] = (scoreCounts[scoreKey] || 0) + 1;
   369→
   370→    if (homeGoals > awayGoals) homeWins++;
   371→    else if (homeGoals === awayGoals) draws++;
   372→    else awayWins++;
   373→  }
   374→
   375→  // Find most likely score
   376→  let mostLikelyScore = "1-0";
   377→  let maxCount = 0;
   378→  for (const [score, count] of Object.entries(scoreCounts)) {
   379→    if (count > maxCount) {
   380→      maxCount = count;
   381→      mostLikelyScore = score;
   382→    }
   383→  }
   384→
   385→  // Calculate confidence interval for total goals
   386→  const totalGoals = [];
   387→  for (let i = 0; i < iterations; i++) {
   388→    totalGoals.push(poissonRandom(expectedHomeGoals) + poissonRandom(expectedAwayGoals));
   389→  }
   390→  totalGoals.sort((a, b) => a - b);
   391→  const lowIdx = Math.floor(iterations * 0.1);
   392→  const highIdx = Math.floor(iterations * 0.9);
   393→
   394→  return {
   395→    homeWinProb: Math.round((homeWins / iterations) * 100) / 100,
   396→    drawProb: Math.round((draws / iterations) * 100) / 100,
   397→    awayWinProb: Math.round((awayWins / iterations) * 100) / 100,
   398→    avgHomeGoals: Math.round((totalHomeGoals / iterations) * 100) / 100,
   399→    avgAwayGoals: Math.round((totalAwayGoals / iterations) * 100) / 100,
   400→    mostLikelyScore,
   401→    confidenceInterval: {
   402→      low: totalGoals[lowIdx] || 1,
   403→      high: totalGoals[highIdx] || 4,
   404→    },
   405→  };
   406→}

   408→function poissonRandom(lambda: number): number {
   409→  const L = Math.exp(-lambda);
   410→  let k = 0;
   411→  let p = 1;
   412→  do {
   413→    k++;
   414→    p *= Math.random();
   415→  } while (p > L);
   416→  return k - 1;
   417→}

   419→// ==================== KELLY CRITERION ====================
   420→
   421→export function calculateKellyCriterion(
   422→  probability: number,
   423→  odds: number,
   424→  bankroll: number = 1000,
   425→  fractionalKelly: number = 0.25 // Use quarter-Kelly for safety
   426→): KellyResult {
   427→  const b = odds - 1; // Net odds
   428→  const q = 1 - probability; // Probability of losing
   429→
   430→  // Full Kelly fraction
   431→  const fullKelly = (b * probability - q) / b;
   432→
   433→  // If negative, no bet should be placed
   434→  if (fullKelly <= 0) {
   435→    return {
   436→      fraction: 0,
   437→      recommendedStake: 0,
   438→      maxStake: 0,
   439→      expectedValue: 0,
   440→      riskOfRuin: 0,
   441→    };
   442→  }
   443→
   444→  // Apply fractional Kelly for risk management
   445→  const adjustedFraction = fullKelly * fractionalKelly;
   446→
   447→  // Cap at 10% of bankroll for safety
   448→  const cappedFraction = Math.min(adjustedFraction, 0.10);
   449→
   450→  // Expected value
   451→  const expectedValue = probability * b - q;
   452→
   453→  // Risk of ruin approximation (simplified)
   454→  const riskOfRuin = Math.pow(q / probability, Math.floor(bankroll / (cappedFraction * bankroll)));
   455→
   456→  return {
   457→    fraction: Math.round(cappedFraction * 10000) / 10000,
   458→    recommendedStake: Math.round(cappedFraction * bankroll * 100) / 100,
   459→    maxStake: Math.round(Math.min(fullKelly * bankroll, bankroll * 0.10) * 100) / 100,
   460→    expectedValue: Math.round(expectedValue * 1000) / 1000,
   461→    riskOfRuin: Math.round(Math.min(riskOfRuin, 1) * 10000) / 10000,
   462→  };
   463→}

   465→// ==================== OVER/UNDER ANALYSIS ====================
   466→
   467→export function calculateOverUnderProbabilities(
   468→  expectedGoals: number,
   469→  line: number
   470→): OverUnderResult {
   471→  let underProb = 0;
   472→  const maxGoals = 8;
   473→
   474→  for (let i = 0; i <= maxGoals; i++) {
   475→    const prob = poissonProbability(expectedGoals, i);
   476→    if (i < line) {
   477→      underProb += prob;
   478→    }
   479→  }
   480→
   481→  const overProb = 1 - underProb;
   482→  const recommendation = overProb > underProb ? "over" : "under";
   483→  const value = Math.abs(overProb - underProb);
   484→
   485→  return {
   486→    line,
   487→    overProb: Math.round(overProb * 100) / 100,
   488→    underProb: Math.round(underProb * 100) / 100,
   489→    recommendation,
   490→    value: Math.round(value * 100) / 100,
   491→  };
   492→}

   494→// ==================== VALUE BET DETECTION ====================
   495→
   496→function detectValueBets(
   497→  match: MatchData,
   498→  prediction: { homeWinProb: number; drawProb: number; awayWinProb: number },
   499→  bankroll: number = 1000
   500→): ValueBet[] {
   501→  const valueBets: ValueBet[] = [];
   502→
   503→  // Home win value
   504→  const homeImplied = 1 / match.homeOdds;
   505→  const homeEdge = prediction.homeWinProb - homeImplied;
   506→  if (homeEdge > 0.03) { // Minimum 3% edge
   507→    const kelly = calculateKellyCriterion(prediction.homeWinProb, match.homeOdds, bankroll);
   508→    valueBets.push({
   509→      selection: `${match.homeTeam} Win`,
   510→      aiProb: prediction.homeWinProb,
   511→      impliedProb: homeImplied,
   512→      odds: match.homeOdds,
   513→      edge: Math.round(homeEdge * 100) / 100,
   514→      kellyFraction: kelly.fraction,
   515→      expectedValue: kelly.expectedValue,
   516→    });
   517→  }
   518→
   519→  // Away win value
   520→  const awayImplied = 1 / match.awayOdds;
   521→  const awayEdge = prediction.awayWinProb - awayImplied;
   522→  if (awayEdge > 0.03) {
   523→    const kelly = calculateKellyCriterion(prediction.awayWinProb, match.awayOdds, bankroll);
   524→    valueBets.push({
   525→      selection: `${match.awayTeam} Win`,
   526→      aiProb: prediction.awayWinProb,
   527→      impliedProb: awayImplied,
   528→      odds: match.awayOdds,
   529→      edge: Math.round(awayEdge * 100) / 100,
   530→      kellyFraction: kelly.fraction,
   531→      expectedValue: kelly.expectedValue,
   532→    });
   533→  }
   534→
   535→  // Draw value
   536→  if (match.drawOdds) {
   537→    const drawImplied = 1 / match.drawOdds;
   538→    const drawEdge = prediction.drawProb - drawImplied;
   539→    if (drawEdge > 0.03) {
   540→      const kelly = calculateKellyCriterion(prediction.drawProb, match.drawOdds, bankroll);
   541→      valueBets.push({
   542→        selection: "Draw",
   543→        aiProb: prediction.drawProb,
   544→        impliedProb: drawImplied,
   545→        odds: match.drawOdds,
   546→        edge: Math.round(drawEdge * 100) / 100,
   547→        kellyFraction: kelly.fraction,
   548→        expectedValue: kelly.expectedValue,
   549→      });
   550→    }
   551→  }
   552→
   553→  // Over/Under value
   554→  if (match.overUnderLine && match.overOdds && match.underOdds) {
   555→    const overImplied = 1 / match.overOdds;
   556→    const underImplied = 1 / match.underOdds;
   557→    // Use Poisson model for over/under
   558→    const totalExpected = 2.5; // Will be overridden by actual expected goals
   559→    const overProb = 1 - normalCDF((match.overUnderLine - totalExpected) / Math.sqrt(totalExpected));
   560→    const underProb = 1 - overProb;
   561→
   562→    const overEdge = overProb - overImplied;
   563→    const underEdge = underProb - underImplied;
   564→
   565→    if (overEdge > 0.03) {
   566→      const kelly = calculateKellyCriterion(overProb, match.overOdds, bankroll);
   567→      valueBets.push({
   568→        selection: `Over ${match.overUnderLine}`,
   569→        aiProb: Math.round(overProb * 100) / 100,
   570→        impliedProb: overImplied,
   571→        odds: match.overOdds,
   572→        edge: Math.round(overEdge * 100) / 100,
   573→        kellyFraction: kelly.fraction,
   574→        expectedValue: kelly.expectedValue,
   575→      });
   576→    }
   577→    if (underEdge > 0.03) {
   578→      const kelly = calculateKellyCriterion(underProb, match.underOdds, bankroll);
   579→      valueBets.push({
   580→        selection: `Under ${match.overUnderLine}`,
   581→        aiProb: Math.round(underProb * 100) / 100,
   582→        impliedProb: underImplied,
   583→        odds: match.underOdds,
   584→        edge: Math.round(underEdge * 100) / 100,
   585→        kellyFraction: kelly.fraction,
   586→        expectedValue: kelly.expectedValue,
   587→      });
   588→    }
   589→  }
   590→
   591→  return valueBets.sort((a, b) => b.edge - a.edge);
   592→}
   593→
   594→// ==================== MAIN ANALYSIS ENGINE ====================
   595→
   596→export function analyzeMatch(
   597→  match: MatchData,
   598→  homeTeamStats?: TeamStatsData | null,
   599→  awayTeamStats?: TeamStatsData | null,
   600→  bankroll: number = 1000
   601→): Prediction {
   602→  const modelResults: ModelResult[] = [];
   603→
   604→  // ---- MODEL 1: Bookmaker Odds Implied Probability ----
   605→  const homeImplied = 1 / match.homeOdds;
   606→  const awayImplied = 1 / match.awayOdds;
   607→  const drawImplied = match.drawOdds ? 1 / match.drawOdds : 0.25;
   608→  const totalImplied = homeImplied + drawImplied + awayImplied;
   609→  const margin = 1 / totalImplied;
   610→
   611→  const oddsModelHome = homeImplied * margin;
   612→  const oddsModelDraw = drawImplied * margin;
   613→  const oddsModelAway = awayImplied * margin;
   614→
   615→  modelResults.push({
   616→    modelName: "Bookmaker Odds Model",
   617→    homeWinProb: oddsModelHome,
   618→    drawProb: oddsModelDraw,
   619→    awayWinProb: oddsModelAway,
   620→    weight: 0.25,
   621→    confidence: 0.6,
   622→  });
   623→
   624→  // ---- MODEL 2: Poisson Distribution Model ----
   625→  const poissonResult = calculatePoissonProbabilities(homeTeamStats ?? null, awayTeamStats ?? null);
   626→
   627→  modelResults.push({
   628→    modelName: "Poisson Distribution",
   629→    homeWinProb: poissonResult.homeWinProb,
   630→    drawProb: poissonResult.drawProb,
   631→    awayWinProb: poissonResult.awayWinProb,
   632→    weight: 0.30,
   633→    confidence: homeTeamStats && awayTeamStats ? 0.8 : 0.5,
   634→  });
   635→
   636→  // ---- MODEL 3: ELO Rating Model ----
   637→  const homeElo = homeTeamStats?.eloRating || 1500;
   638→  const awayElo = awayTeamStats?.eloRating || 1500;
   639→  const eloResult = predictWithElo(homeElo, awayElo);
   640→
   641→  modelResults.push({
   642→    modelName: "ELO Rating System",
   643→    homeWinProb: eloResult.homeWin,
   644→    drawProb: eloResult.draw,
   645→    awayWinProb: eloResult.awayWin,
   646→    weight: 0.25,
   647→    confidence: homeTeamStats && awayTeamStats ? 0.75 : 0.4,
   648→  });
   649→
   650→  // ---- MODEL 4: Monte Carlo Simulation ----
   651→  const mcResult = runMonteCarloSimulation(
   652→    poissonResult.expectedHomeGoals,
   653→    poissonResult.expectedAwayGoals,
   654→    10000
   655→  );
   656→
   657→  modelResults.push({
   658→    modelName: "Monte Carlo Simulation",
   659→    homeWinProb: mcResult.homeWinProb,
   660→    drawProb: mcResult.drawProb,
   661→    awayWinProb: mcResult.awayWinProb,
   662→    weight: 0.20,
   663→    confidence: 0.7,
   664→  });
   665→
   666→  // ---- ENSEMBLE: Weighted Average of All Models ----
   667→  let totalWeight = 0;
   668→  let ensembleHome = 0;
   669→  let ensembleDraw = 0;
   670→  let ensembleAway = 0;
   671→
   672→  for (const model of modelResults) {
   673→    // Adjust weight based on confidence and data availability
   674→    const adjustedWeight = model.weight * model.confidence;
   675→    ensembleHome += model.homeWinProb * adjustedWeight;
   676→    ensembleDraw += model.drawProb * adjustedWeight;
   677→    ensembleAway += model.awayWinProb * adjustedWeight;
   678→    totalWeight += adjustedWeight;
   679→  }
   680→
   681→  if (totalWeight > 0) {
   682→    ensembleHome /= totalWeight;
   683→    ensembleDraw /= totalWeight;
   684→    ensembleAway /= totalWeight;
   685→  }
   686→
   687→  // ---- FORM-BASED ADJUSTMENTS ----
   688→  if (homeTeamStats && awayTeamStats) {
   689→    const homeForm = parseForm(homeTeamStats.form);
   690→    const awayForm = parseForm(awayTeamStats.form);
   691→    const formAdjust = (homeForm - awayForm) * 0.04;
   692→
   693→    // Home/away record adjustment
   694→    const homeHomeRecord = parseRecord(homeTeamStats.homeRecord);
   695→    const awayAwayRecord = parseRecord(awayTeamStats.awayRecord);
   696→    const recordAdjust = (homeHomeRecord.rate - awayAwayRecord.rate) * 0.03;
   697→
   698→    // Attack vs defense matchup
   699→    const attackDefenseFactor = ((homeTeamStats.attackRating - awayTeamStats.defenseRating) -
   700→      (awayTeamStats.attackRating - homeTeamStats.defenseRating)) / 200;
   701→
   702→    // Apply adjustments
   703→    ensembleHome += formAdjust + recordAdjust + attackDefenseFactor * 0.05;
   704→    ensembleAway -= formAdjust + recordAdjust + attackDefenseFactor * 0.05;
   705→
   706→    // xG-based adjustment if available
   707→    if (homeTeamStats.xgFor > 0 && awayTeamStats.xgFor > 0) {
   708→      const homeXgRate = homeTeamStats.xgFor / Math.max(1, homeTeamStats.matchesPlayed);
   709→      const awayXgRate = awayTeamStats.xgFor / Math.max(1, awayTeamStats.matchesPlayed);
   710→      const xgDiff = (homeXgRate - awayXgRate) * 0.05;
   711→      ensembleHome += xgDiff;
   712→      ensembleAway -= xgDiff;
   713→    }
   714→  }
   715→
   716→  // ---- Normalize ----
   717→  ensembleHome = Math.max(0.03, ensembleHome);
   718→  ensembleDraw = Math.max(0.03, ensembleDraw);
   719→  ensembleAway = Math.max(0.03, ensembleAway);
   720→
   721→  const normTotal = ensembleHome + ensembleDraw + ensembleAway;
   722→  ensembleHome /= normTotal;
   723→  ensembleDraw /= normTotal;
   724→  ensembleAway /= normTotal;
   725→
   726→  // ---- Determine Recommendation ----
   727→  let recommended: Prediction["recommended"] = "home";
   728→  let maxProb = ensembleHome;
   729→
   730→  if (ensembleAway > maxProb) {
   731→    recommended = "away";
   732→    maxProb = ensembleAway;
   733→  }
   734→  if (ensembleDraw > maxProb && match.sport === "football") {
   735→    recommended = "draw";
   736→    maxProb = ensembleDraw;
   737→  }
   738→
   739→  // Check if over/under is better value
   740→  if (match.overUnderLine && match.sport === "football") {
   741→    const totalExpected = poissonResult.expectedHomeGoals + poissonResult.expectedAwayGoals;
   742→    const ouResult = calculateOverUnderProbabilities(totalExpected, match.overUnderLine);
   743→    const ouValue = Math.abs(ouResult.overProb - (match.overOdds ? 1 / match.overOdds : 0.5));
   744→    const matchWinnerValue = Math.abs(maxProb - (recommended === "home" ? 1 / match.homeOdds : 1 / match.awayOdds));
   745→
   746→    if (ouValue > matchWinnerValue && ouValue > 0.08) {
   747→      recommended = ouResult.recommendation as "over" | "under";
   748→    }
   749→  }
   750→
   751→  // ---- Calculate Confidence ----
   752→  const modelAgreement = 1 - Math.sqrt(
   753→    modelResults.reduce((sum, m) => {
   754→      return sum + Math.pow(m.homeWinProb - ensembleHome, 2) +
   755→        Math.pow(m.drawProb - ensembleDraw, 2) +
   756→        Math.pow(m.awayWinProb - ensembleAway, 2);
   757→    }, 0) / modelResults.length
   758→  );
   759→
   760→  const confidence = Math.max(0.3, Math.min(0.95,
   761→    maxProb * 0.6 + modelAgreement * 0.3 + (homeTeamStats && awayTeamStats ? 0.1 : 0)
   762→  ));
   763→
   764→  // ---- Risk Assessment ----
   765→  let riskScore = 50;
   766→  if (confidence < 0.5) riskScore += 25;
   767→  if (confidence > 0.7) riskScore -= 20;
   768→  if (Math.abs(ensembleHome - ensembleAway) < 0.1) riskScore += 15;
   769→  if (match.homeOdds > 3.0 || match.awayOdds > 3.0) riskScore += 10;
   770→  if (!homeTeamStats || !awayTeamStats) riskScore += 20;
   771→  if (modelAgreement < 0.7) riskScore += 10;
   772→  riskScore = Math.max(0, Math.min(100, riskScore));
   773→
   774→  const riskLevel: "low" | "medium" | "high" =
   775→    riskScore <= 35 ? "low" : riskScore <= 65 ? "medium" : "high";
   776→
   777→  // ---- Value Bets ----
   778→  const valueBets = detectValueBets(
   779→    match,
   780→    { homeWinProb: ensembleHome, drawProb: ensembleDraw, awayWinProb: ensembleAway },
   781→    bankroll
   782→  );
   783→
   784→  // ---- Kelly Criterion for main recommendation ----
   785→  const recOdds = recommended === "home" ? match.homeOdds :
   786→    recommended === "away" ? match.awayOdds :
   787→    match.drawOdds || 3.0;
   788→  const recProb = recommended === "home" ? ensembleHome :
   789→    recommended === "away" ? ensembleAway :
   790→    ensembleDraw;
   791→  const kellyStake = calculateKellyCriterion(recProb, recOdds, bankroll);
   792→
   793→  // ---- Generate Analysis Text ----
   794→  const analysis = generateAnalysisText(match, {
   795→    homeWinProb: ensembleHome,
   796→    drawProb: ensembleDraw,
   797→    awayWinProb: ensembleAway,
   798→    confidence,
   799→    recommended,
   800→    analysis: "",
   801→    modelResults,
   802→    valueBets,
   803→    kellyStake,
   804→    riskScore,
   805→    riskLevel,
   806→  }, homeTeamStats, awayTeamStats, poissonResult, mcResult);
   807→
   808→  return {
   809→    homeWinProb: Math.round(ensembleHome * 100) / 100,
   810→    drawProb: Math.round(ensembleDraw * 100) / 100,
   811→    awayWinProb: Math.round(ensembleAway * 100) / 100,
   812→    confidence: Math.round(confidence * 100) / 100,
   813→    recommended,
   814→    analysis,
   815→    modelResults: modelResults.map(m => ({
   816→      ...m,
   817→      homeWinProb: Math.round(m.homeWinProb * 100) / 100,
   818→      drawProb: Math.round(m.drawProb * 100) / 100,
   819→      awayWinProb: Math.round(m.awayWinProb * 100) / 100,
   820→    })),
   821→    valueBets,
   822→    kellyStake,
   823→    riskScore,
   824→    riskLevel,
   825→  };
   826→}

   828→// ==================== ANALYSIS TEXT GENERATION ====================
   829→
   830→function generateAnalysisText(
   831→  match: MatchData,
   832→  prediction: Prediction,
   833→  homeStats?: TeamStatsData | null,
   834→  awayStats?: TeamStatsData | null,
   835→  poissonResult?: PoissonResult,
   836→  mcResult?: MonteCarloResult
   837→): string {
   838→  const parts: string[] = [];
   839→
   840→  // Model consensus
   841→  const modelCount = prediction.modelResults.length;
   842→  const agreeingModels = prediction.modelResults.filter(m => {
   843→    const maxP = Math.max(m.homeWinProb, m.drawProb, m.awayWinProb);
   844→    if (maxP === m.homeWinProb && prediction.recommended === "home") return true;
   845→    if (maxP === m.awayWinProb && prediction.recommended === "away") return true;
   846→    if (maxP === m.drawProb && prediction.recommended === "draw") return true;
   847→    return false;
   848→  }).length;
   849→
   850→  parts.push(`Multi-model analysis (${modelCount} models, ${agreeingModels} agree): ` +
   851→    `Home ${Math.round(prediction.homeWinProb * 100)}% | Draw ${Math.round(prediction.drawProb * 100)}% | Away ${Math.round(prediction.awayWinProb * 100)}%.`);
   852→
   853→  // Poisson insights
   854→  if (poissonResult) {
   855→    parts.push(`Poisson model expects ${poissonResult.expectedHomeGoals}-${poissonResult.expectedAwayGoals} ` +
   856→      `(total ${Math.round((poissonResult.expectedHomeGoals + poissonResult.expectedAwayGoals) * 10) / 10} goals).`);
   857→  }
   858→
   859→  // Monte Carlo insights
   860→  if (mcResult) {
   861→    parts.push(`Monte Carlo simulation (10,000 iterations): Most likely score ${mcResult.mostLikelyScore}, ` +
   862→      `total goals range ${mcResult.confidenceInterval.low}-${mcResult.confidenceInterval.high}.`);
   863→  }
   864→
   865→  // Team stats analysis
   866→  if (homeStats && awayStats) {
   867→    // Form
   868→    const homeForm = homeStats.form || "N/A";
   869→    const awayForm = awayStats.form || "N/A";
   870→    parts.push(`Form: ${match.homeTeam} (${homeForm}) vs ${match.awayTeam} (${awayForm}).`);
   871→
   872→    // Rating comparison
   873→    const ratingDiff = homeStats.overallRating - awayStats.overallRating;
   874→    if (Math.abs(ratingDiff) > 5) {
   875→      parts.push(`${ratingDiff > 0 ? match.homeTeam : match.awayTeam} has a ${Math.abs(ratingDiff)}-point rating advantage.`);
   876→    }
   877→
   878→    // ELO
   879→    if (homeStats.eloRating > 0 && awayStats.eloRating > 0) {
   880→      parts.push(`ELO ratings: ${match.homeTeam} (${homeStats.eloRating}) vs ${match.awayTeam} (${awayStats.eloRating}).`);
   881→    }
   882→
   883→    // xG
   884→    if (homeStats.xgFor > 0 && awayStats.xgFor > 0) {
   885→      const homeXgPerGame = Math.round(homeStats.xgFor / Math.max(1, homeStats.matchesPlayed) * 100) / 100;
   886→      const awayXgPerGame = Math.round(awayStats.xgFor / Math.max(1, awayStats.matchesPlayed) * 100) / 100;
   887→      parts.push(`xG per game: ${match.homeTeam} (${homeXgPerGame}) vs ${match.awayTeam} (${awayXgPerGame}).`);
   888→    }
   889→
   890→    // Attack vs defense
   891→    if (homeStats.attackRating > awayStats.defenseRating + 10) {
   892→      parts.push(`${match.homeTeam}'s attack (${homeStats.attackRating}) significantly exceeds ${match.awayTeam}'s defense (${awayStats.defenseRating}).`);
   893→    }
   894→    if (awayStats.attackRating > homeStats.defenseRating + 10) {
   895→      parts.push(`${match.awayTeam}'s attack (${awayStats.attackRating}) significantly exceeds ${match.homeTeam}'s defense (${homeStats.defenseRating}).`);
   896→    }
   897→  }
   898→
   899→  // Value bet info
   900→  if (prediction.valueBets.length > 0) {
   901→    const topValue = prediction.valueBets[0];
   902→    parts.push(`Best value: ${topValue.selection} with ${Math.round(topValue.edge * 100)}% edge ` +
   903→      `(AI: ${Math.round(topValue.aiProb * 100)}% vs implied: ${Math.round(topValue.impliedProb * 100)}%).`);
   904→  }
   905→
   906→  // Kelly recommendation
   907→  if (prediction.kellyStake.recommendedStake > 0) {
   908→    parts.push(`Kelly Criterion recommends $${prediction.kellyStake.recommendedStake} stake ` +
   909→      `(${Math.round(prediction.kellyStake.fraction * 100)}% of bankroll).`);
   910→  }
   911→
   912→  // Risk
   913→  parts.push(`Risk level: ${prediction.riskLevel.toUpperCase()} (${prediction.riskScore}/100). ` +
   914→    `Confidence: ${Math.round(prediction.confidence * 100)}%.`);
   915→
   916→  return parts.join(" ");
   917→}

   919→// ==================== CASHOUT ENGINE ====================
   920→
   921→export function shouldCashout(
   922→  bet: {
   923→    selection: string;
   924→    odds: number;
   925→    stake: number;
   926→    potentialWin: number;
   927→    status: string;
   928→  },
   929→  currentMatchState: {
   930→    homeScore: number;
   931→    awayScore: number;
   932→    minute: number;
   933→    homeTeam: string;
   934→    awayTeam: string;
   935→    sport: string;
   936→  },
   937→  homeTeamStats?: TeamStatsData | null,
   938→  awayTeamStats?: TeamStatsData | null
   939→): CashoutRecommendation {
   940→  const { stake, potentialWin, odds } = bet;
   941→  const { homeScore, awayScore, minute, sport } = currentMatchState;
   942→
   943→  const isFootball = sport === "football";
   944→  const isBasketball = sport === "basketball";
   945→  const isTennis = sport === "tennis";
   946→  const totalMinutes = isFootball ? 90 : isBasketball ? 48 : isTennis ? 180 : 90;
   947→  const progress = minute / totalMinutes;
   948→
   949→  // Determine if the bet is currently winning
   950→  const selectionIsHome = bet.selection === currentMatchState.homeTeam;
   951→  const selectionIsAway = bet.selection === currentMatchState.awayTeam;
   952→  const currentLead = selectionIsHome
   953→    ? homeScore - awayScore
   954→    : selectionIsAway
   955→    ? awayScore - homeScore
   956→    : 0;
   957→
   958→  // Calculate probability of current result holding
   959→  let probabilityOfWinning = 0.5;
   960→
   961→  if (currentLead > 0) {
   962→    // Bet is winning - calculate probability of holding
   963→    if (isFootball) {
   964→      // Football: use Poisson model for remaining time
   965→      const remainingMins = 90 - minute;
   966→      const goalRate = remainingMins / 90 * 2.7; // ~2.7 goals per match average
   967→      const probOfConceding = 1 - Math.exp(-goalRate * 0.4); // Approximate
   968→
   969→      if (currentLead >= 2) {
   970→        probabilityOfWinning = 0.85 + (progress * 0.10);
   971→      } else if (currentLead === 1) {
   972→        probabilityOfWinning = 0.55 + (progress * 0.25);
   973→      }
   974→
   975→      // Adjust for team quality
   976→      if (homeTeamStats && awayTeamStats) {
   977→        const defendingTeam = selectionIsHome ? homeTeamStats : awayTeamStats;
   978→        const attackingTeam = selectionIsHome ? awayTeamStats : homeTeamStats;
   979→        if (defendingTeam.defenseRating > 75) probabilityOfWinning += 0.05;
   980→        if (attackingTeam.attackRating < 65) probabilityOfWinning += 0.05;
   981→      }
   982→    } else if (isBasketball) {
   983→      // Basketball: use normal distribution approximation
   984→      const remainingPossessions = (48 - minute) / 48 * 100;
   985→      const standardDev = Math.sqrt(remainingPossessions) * 0.4;
   986→      probabilityOfWinning = normalCDF(currentLead / Math.max(1, standardDev));
   987→    } else if (isTennis) {
   988→      // Tennis: simpler model based on lead and sets
   989→      probabilityOfWinning = 0.5 + currentLead * 0.15 + progress * 0.2;
   990→    }
   991→
   992→    probabilityOfWinning = Math.min(0.98, probabilityOfWinning);
   993→  } else if (currentLead === 0) {
   994→    // Draw - bet is at risk
   995→    probabilityOfWinning = 0.3 - progress * 0.1;
   996→  } else {
   997→    // Losing
   998→    probabilityOfWinning = Math.max(0.02, 0.2 - Math.abs(currentLead) * 0.08);
   999→  }
  1000→
  1001→  // Calculate expected values
  1002→  const profitIfWin = potentialWin - stake;
  1003→  const fairCashout = probabilityOfWinning * profitIfWin + (1 - probabilityOfWinning) * 0;
  1004→
  1005→  // Bookmaker cashout is typically 80-90% of fair value
  1006→  const bookmakerCashout = fairCashout * 0.85;
  1007→
  1008→  // Expected value if holding
  1009→  const expectedValueIfHold = probabilityOfWinning * profitIfWin - (1 - probabilityOfWinning) * stake;
  1010→  // Expected value if cashing out (after bookmaker margin)
  1011→  const expectedValueIfCashout = bookmakerCashout - stake;
  1012→
  1013→  let cashoutAmount = stake + bookmakerCashout;
  1014→  let shouldCashout = false;
  1015→  let urgency: CashoutRecommendation["urgency"] = "low";
  1016→
  1017→  // Cashout decision logic
  1018→  if (currentLead > 0) {
  1019→    // Currently winning
  1020→    if (progress > 0.75 && currentLead >= 1) {
  1021→      // Late in game, winning - strong cashout case
  1022→      shouldCashout = true;
  1023→      urgency = "medium";
  1024→      if (progress > 0.85) urgency = "low"; // Very late, less urgency needed
  1025→    } else if (progress > 0.5 && currentLead >= 2) {
  1026→      // Comfortable lead, second half
  1027→      shouldCashout = true;
  1028→      urgency = "low";
  1029→    } else if (expectedValueIfCashout > expectedValueIfHold * 0.9 && progress > 0.6) {
  1030→      // Cashout value is close to hold value with less risk
  1031→      shouldCashout = true;
  1032→      urgency = "medium";
  1033→    }
  1034→
  1035→    // Minimum cashout should be above stake
  1036→    if (cashoutAmount < stake * 1.05) {
  1037→      shouldCashout = false;
  1038→    }
  1039→  } else if (currentLead === 0) {
  1040→    // Draw
  1041→    if (progress > 0.7) {
  1042→      shouldCashout = true;
  1043→      urgency = "high";
  1044→      cashoutAmount = stake * 0.5;
  1045→    }
  1046→  } else {
  1047→    // Losing
  1048→    if (progress > 0.5) {
  1049→      shouldCashout = true;
  1050→      urgency = "high";
  1051→      cashoutAmount = stake * 0.15;
  1052→    }
  1053→  }
  1054→
  1055→  // Generate reasoning
  1056→  const reasoning = generateCashoutReasoning(
  1057→    bet, currentMatchState, shouldCashout, cashoutAmount, urgency,
  1058→    probabilityOfWinning, expectedValueIfHold, expectedValueIfCashout
  1059→  );
  1060→
  1061→  return {
  1062→    shouldCashout,
  1063→    cashoutAmount: Math.round(cashoutAmount * 100) / 100,
  1064→    reasoning,
  1065→    urgency,
  1066→    probabilityOfWinning: Math.round(probabilityOfWinning * 100) / 100,
  1067→    expectedValueIfHold: Math.round(expectedValueIfHold * 100) / 100,
  1068→    expectedValueIfCashout: Math.round(expectedValueIfCashout * 100) / 100,
  1069→  };
  1070→}
  1071→
  1072→function generateCashoutReasoning(
  1073→  bet: { selection: string; odds: number; stake: number; potentialWin: number },
  1074→  match: { homeScore: number; awayScore: number; minute: number; homeTeam: string; awayTeam: string; sport: string },
  1075→  shouldCashout: boolean,
  1076→  cashoutAmount: number,
  1077→  urgency: string,
  1078→  probOfWinning: number,
  1079→  evHold: number,
  1080→  evCashout: number
  1081→): string {
  1082→  const selectionIsHome = bet.selection === match.homeTeam;
  1083→  const currentLead = selectionIsHome
  1084→    ? match.homeScore - match.awayScore
  1085→    : match.awayScore - match.homeScore;
  1086→
  1087→  if (shouldCashout) {
  1088→    if (currentLead > 0) {
  1089→      return `Your bet on ${bet.selection} is winning (${match.homeScore}-${match.awayScore} at ${match.minute}'). ` +
  1090→        `Win probability: ${Math.round(probOfWinning * 100)}%. Cashout of $${Math.round(cashoutAmount * 100) / 100} secures ` +
  1091→        `$${Math.round((cashoutAmount - bet.stake) * 100) / 100} profit. EV hold: $${Math.round(evHold * 100) / 100} vs EV cashout: $${Math.round(evCashout * 100) / 100}. ` +
  1092→        `${urgency === "high" ? "ACT NOW - position could deteriorate!" : "Moderate urgency - consider cashing out soon."}`;
  1093→    } else if (currentLead === 0) {
  1094→      return `Match is drawn (${match.homeScore}-${match.awayScore} at ${match.minute}'). ` +
  1095→        `Win probability: ${Math.round(probOfWinning * 100)}%. Cashout to recover $${Math.round(cashoutAmount * 100) / 100}. ` +
  1096→        `High urgency - your bet is at significant risk.`;
  1097→    } else {
  1098→      return `${bet.selection} is losing (${match.homeScore}-${match.awayScore} at ${match.minute}'). ` +
  1099→        `Win probability: ${Math.round(probOfWinning * 100)}%. Cashout to recover $${Math.round(cashoutAmount * 100) / 100}. ` +
  1100→        `Recovery unlikely at this stage.`;
  1101→    }
  1102→  } else {
  1103→    if (currentLead > 0) {
  1104→      return `Your bet on ${bet.selection} is winning (${match.homeScore}-${match.awayScore} at ${match.minute}'). ` +
  1105→        `Win probability: ${Math.round(probOfWinning * 100)}%. No cashout recommended - let it ride for maximum profit. ` +
  1106→        `EV hold ($${Math.round(evHold * 100) / 100}) exceeds EV cashout ($${Math.round(evCashout * 100) / 100}).`;
  1107→    }
  1108→    return `Match is level at ${match.homeScore}-${match.awayScore}. ` +
  1109→      `Win probability: ${Math.round(probOfWinning * 100)}%. Hold position - ` +
  1110→      `sufficient time remains for the match to turn in your favor.`;
  1111→  }
  1112→}

  1114→// ==================== DETAILED ANALYSIS GENERATOR ====================
  1115→
  1116→export function generateDetailedAnalysis(
  1117→  match: MatchData,
  1118→  homeStats: TeamStatsData | null,
  1119→  awayStats: TeamStatsData | null,
  1120→  prediction: Prediction
  1121→): DetailedAnalysis {
  1122→  const keyFactors: string[] = [];
  1123→  const homeStrengths: string[] = [];
  1124→  const homeWeaknesses: string[] = [];
  1125→  const awayStrengths: string[] = [];
  1126→  const awayWeaknesses: string[] = [];
  1127→  const riskFactors: string[] = [];
  1128→
  1129→  if (homeStats && awayStats) {
  1130→    // Rating differential
  1131→    const ratingDiff = homeStats.overallRating - awayStats.overallRating;
  1132→    if (Math.abs(ratingDiff) > 10) {
  1133→      keyFactors.push(
  1134→        `${ratingDiff > 0 ? match.homeTeam : match.awayTeam} has a significant rating advantage (${Math.abs(ratingDiff)} points)`
  1135→      );
  1136→    }
  1137→
  1138→    // Form analysis
  1139→    if (homeStats.form && awayStats.form) {
  1140→      const homeFormScore = parseForm(homeStats.form);
  1141→      const awayFormScore = parseForm(awayStats.form);
  1142→      if (Math.abs(homeFormScore - awayFormScore) > 0.3) {
  1143→        keyFactors.push(
  1144→          `${homeFormScore > awayFormScore ? match.homeTeam : match.awayTeam} is in significantly better form`
  1145→        );
  1146→      }
  1147→    }
  1148→
  1149→    // Home advantage
  1150→    keyFactors.push(`${match.homeTeam} has home advantage`);
  1151→
  1152→    // ELO difference
  1153→    if (homeStats.eloRating > 0 && awayStats.eloRating > 0) {
  1154→      const eloDiff = Math.abs(homeStats.eloRating - awayStats.eloRating);
  1155→      if (eloDiff > 100) {
  1156→        keyFactors.push(`Significant ELO gap (${eloDiff} points) between teams`);
  1157→      }
  1158→    }
  1159→
  1160→    // Attack vs defense matchup
  1161→    if (homeStats.attackRating > awayStats.defenseRating + 10) {
  1162→      keyFactors.push(`${match.homeTeam}'s strong attack (${homeStats.attackRating}) vs ${match.awayTeam}'s weaker defense (${awayStats.defenseRating})`);
  1163→    }
  1164→    if (awayStats.attackRating > homeStats.defenseRating + 10) {
  1165→      keyFactors.push(`${match.awayTeam}'s strong attack (${awayStats.attackRating}) vs ${match.homeTeam}'s weaker defense (${homeStats.defenseRating})`);
  1166→    }
  1167→
  1168→    // xG insights
  1169→    if (homeStats.xgFor > 0 && awayStats.xgFor > 0) {
  1170→      const homeXgPerGame = homeStats.xgFor / Math.max(1, homeStats.matchesPlayed);
  1171→      const awayXgPerGame = awayStats.xgFor / Math.max(1, awayStats.matchesPlayed);
  1172→      if (Math.abs(homeXgPerGame - awayXgPerGame) > 0.5) {
  1173→        keyFactors.push(`xG differential: ${homeXgPerGame > awayXgPerGame ? match.homeTeam : match.awayTeam} generates significantly more expected goals`);
  1174→      }
  1175→    }
  1176→
  1177→    // Strengths
  1178→    if (homeStats.attackRating > 75) homeStrengths.push(`Strong attack (${homeStats.attackRating})`);
  1179→    if (homeStats.defenseRating > 75) homeStrengths.push(`Solid defense (${homeStats.defenseRating})`);
  1180→    if (homeStats.form && parseForm(homeStats.form) > 0.7) homeStrengths.push(`Excellent recent form (${homeStats.form})`);
  1181→    if (homeStats.homeRecord) {
  1182→      const homeRec = parseRecord(homeStats.homeRecord);
  1183→      if (homeRec.rate > 0.65) homeStrengths.push(`Strong home record (${homeStats.homeRecord})`);
  1184→    }
  1185→    if (homeStats.eloRating > 1600) homeStrengths.push(`High ELO rating (${homeStats.eloRating})`);
  1186→    if (homeStats.xgFor > 0 && homeStats.xgFor / Math.max(1, homeStats.matchesPlayed) > 1.8) {
  1187→      homeStrengths.push(`High xG output (${(homeStats.xgFor / Math.max(1, homeStats.matchesPlayed)).toFixed(2)} per game)`);
  1188→    }
  1189→
  1190→    if (awayStats.attackRating > 75) awayStrengths.push(`Strong attack (${awayStats.attackRating})`);
  1191→    if (awayStats.defenseRating > 75) awayStrengths.push(`Solid defense (${awayStats.defenseRating})`);
  1192→    if (awayStats.form && parseForm(awayStats.form) > 0.7) awayStrengths.push(`Excellent recent form (${awayStats.form})`);
  1193→    if (awayStats.awayRecord) {
  1194→      const awayRec = parseRecord(awayStats.awayRecord);
  1195→      if (awayRec.rate > 0.55) awayStrengths.push(`Strong away record (${awayStats.awayRecord})`);
  1196→    }
  1197→    if (awayStats.eloRating > 1600) awayStrengths.push(`High ELO rating (${awayStats.eloRating})`);
  1198→
  1199→    // Weaknesses
  1200→    if (homeStats.defenseRating < 55) homeWeaknesses.push(`Weak defense (${homeStats.defenseRating})`);
  1201→    if (homeStats.form && parseForm(homeStats.form) < 0.4) homeWeaknesses.push(`Poor recent form (${homeStats.form})`);
  1202→    if (homeStats.losses > homeStats.wins) homeWeaknesses.push(`More losses than wins this season`);
  1203→    if (homeStats.xgAgainst > 0 && homeStats.xgAgainst / Math.max(1, homeStats.matchesPlayed) > 1.5) {
  1204→      homeWeaknesses.push(`High xG conceded (${(homeStats.xgAgainst / Math.max(1, homeStats.matchesPlayed)).toFixed(2)} per game)`);
  1205→    }
  1206→
  1207→    if (awayStats.defenseRating < 55) awayWeaknesses.push(`Weak defense (${awayStats.defenseRating})`);
  1208→    if (awayStats.form && parseForm(awayStats.form) < 0.4) awayWeaknesses.push(`Poor recent form (${awayStats.form})`);
  1209→    if (awayStats.losses > awayStats.wins) awayWeaknesses.push(`More losses than wins this season`);
  1210→  } else {
  1211→    keyFactors.push("Limited team statistics available - analysis based on odds and model consensus");
  1212→    riskFactors.push("No team stats available for deep analysis");
  1213→  }
  1214→
  1215→  // Value bet
  1216→  const recTeam = prediction.recommended === "home" ? match.homeTeam
  1217→    : prediction.recommended === "away" ? match.awayTeam
  1218→    : prediction.recommended === "draw" ? "Draw"
  1219→    : prediction.recommended === "over" ? `Over ${match.overUnderLine || 2.5}`
  1220→    : `Under ${match.overUnderLine || 2.5}`;
  1221→  const recOdds = prediction.recommended === "home" ? match.homeOdds
  1222→    : prediction.recommended === "away" ? match.awayOdds
  1223→    : match.drawOdds || 3.0;
  1224→  const recProb = prediction.recommended === "home" ? prediction.homeWinProb
  1225→    : prediction.recommended === "away" ? prediction.awayWinProb
  1226→    : prediction.drawProb;
  1227→  const edge = recProb - (1 / recOdds);
  1228→
  1229→  // Model consensus
  1230→  const modelAgreement = prediction.modelResults.filter(m => {
  1231→    const maxP = Math.max(m.homeWinProb, m.drawProb, m.awayWinProb);
  1232→    if (maxP === m.homeWinProb && prediction.recommended === "home") return true;
  1233→    if (maxP === m.awayWinProb && prediction.recommended === "away") return true;
  1234→    if (maxP === m.drawProb && prediction.recommended === "draw") return true;
  1235→    return false;
  1236→  }).length;
  1237→  const agreementPct = modelAgreement / prediction.modelResults.length;
  1238→
  1239→  // Risk factors
  1240→  if (prediction.riskScore > 60) riskFactors.push("High uncertainty in prediction");
  1241→  if (Math.abs(prediction.homeWinProb - prediction.awayWinProb) < 0.1) riskFactors.push("Teams are closely matched");
  1242→  if (prediction.confidence < 0.5) riskFactors.push("Low AI confidence level");
  1243→  if (agreementPct < 0.5) riskFactors.push("Models disagree on outcome");
  1244→  if (!homeStats || !awayStats) riskFactors.push("Insufficient team data for deep analysis");
  1245→
  1246→  // xG analysis
  1247→  const homeXg = homeStats?.xgFor ? homeStats.xgFor / Math.max(1, homeStats.matchesPlayed) : 0;
  1248→  const awayXg = awayStats?.xgFor ? awayStats.xgFor / Math.max(1, awayStats.matchesPlayed) : 0;
  1249→
  1250→  return {
  1251→    keyFactors: keyFactors.length > 0 ? keyFactors : ["Standard match analysis"],
  1252→    strengths: { team: match.homeTeam, points: homeStrengths.length > 0 ? homeStrengths : ["Average performance"] },
  1253→    weaknesses: { team: match.homeTeam, points: homeWeaknesses.length > 0 ? homeWeaknesses : ["No significant weaknesses identified"] },
  1254→    valueBet: {
  1255→      selection: recTeam,
  1256→      reason: edge > 0.05
  1257→        ? `AI detects ${Math.round(edge * 100)}% edge over bookmaker odds. ${recTeam} has ${Math.round(recProb * 100)}% probability vs ${Math.round((1 / recOdds) * 100)}% implied. Kelly Criterion suggests $${prediction.kellyStake.recommendedStake} stake.`
  1258→        : `Marginal value on ${recTeam}. Kelly Criterion suggests minimal or no stake. Proceed with caution.`,
  1259→      edge: Math.round(edge * 100) / 100,
  1260→    },
  1261→    riskAssessment: {
  1262→      level: prediction.riskLevel,
  1263→      score: prediction.riskScore,
  1264→      factors: riskFactors.length > 0 ? riskFactors : ["Standard risk level"],
  1265→    },
  1266→    modelConsensus: {
  1267→      agreement: Math.round(agreementPct * 100) / 100,
  1268→      dominantModel: prediction.modelResults.sort((a, b) => b.weight - a.weight)[0]?.modelName || "Unknown",
  1269→      spread: Math.round(Math.abs(prediction.homeWinProb - prediction.awayWinProb) * 100) / 100,
  1270→    },
  1271→    xgAnalysis: {
  1272→      homeXg: Math.round(homeXg * 100) / 100,
  1273→      awayXg: Math.round(awayXg * 100) / 100,
  1274→      totalExpected: Math.round((homeXg + awayXg) * 100) / 100,
  1275→    },
  1276→  };
  1277→}

  1281→export function calculateOddsValue(aiProb: number, bookmakerOdds: number): number {
  1282→  const impliedProb = 1 / bookmakerOdds;
  1283→  const value = aiProb - impliedProb;
  1284→  return Math.round(value * 100) / 100;
