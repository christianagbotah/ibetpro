     1→"use client";
     2→
     3→import { useState, useCallback, useMemo } from "react";
     4→import { useFetch } from "@/lib/hooks";
     5→import { useToast } from "@/components/ui/toast";
     6→import { Card, CardContent, CardHeader, CardTitle } from "@/components/ui/card";
     7→import { Button } from "@/components/ui/button";
     8→import { Badge } from "@/components/ui/badge";
     9→import { Input } from "@/components/ui/input";
    10→import { Label } from "@/components/ui/label";
    11→import { Separator } from "@/components/ui/separator";
    12→import {
    13→  Dialog,
    14→  DialogContent,
    15→  DialogHeader,
    16→  DialogTitle,
    17→} from "@/components/ui/dialog";
    18→import {
    19→  Link2, RefreshCw, Wallet, Shield, CheckCircle2,
    20→  AlertCircle, Globe, Key, User, Lock, ChevronRight,
    21→  Search, MapPin, Smartphone, Star, ArrowLeft,
    22→} from "lucide-react";
    23→import { getBrokerLogoUrl } from "@/lib/broker-logos";
    24→import {
    25→  REGIONS,
    26→  getPlatformsForRegion,
    27→  getCurrencyForRegion,
    28→  getContinents,
    29→  searchRegions,
    30→  type RegionInfo,
    31→  type BrokerPlatformInfo,
    32→} from "@/lib/regions";
    33→
    34→interface ConnectedAccount {
    35→  id: string;
    36→  platform: string;
    37→  platformName: string;
    38→  accountName: string;
    39→  isConnected: boolean;
    40→  sessionValid: boolean;
    41→  needsRefresh: boolean;
    42→  sessionExpiry: string | null;
    43→  brokerType: string;
    44→  brokerRegion: string | null;
    45→  balance: number;
    46→  currency: string;
    47→  allocatedAmount: number;
    48→  allocationLock: boolean;
    49→  totalBrokerBets: number;
    50→  totalBrokerProfit: number;
    51→  lastBetPlacedAt: string | null;
    52→  features: {
    53→    liveBetting: boolean;
    54→    cashout: boolean;
    55→    partialCashout: boolean;
    56→    accumulators: boolean;
    57→    maxAccumulatorLegs: number;
    58→    minStake: number;
    59→    maxStake: number;
    60→  } | null;
    61→}
    62→
    63→type ConnectStep = "region" | "platform" | "credentials";
    64→
    65→export function BrokerConnect() {
    66→  const { toast } = useToast();
    67→  const [connectDialogOpen, setConnectDialogOpen] = useState(false);
    68→  const [step, setStep] = useState<ConnectStep>("region");
    69→  const [selectedRegion, setSelectedRegion] = useState<string>("");
    70→  const [selectedPlatform, setSelectedPlatform] = useState<string | null>(null);
    71→  const [connecting, setConnecting] = useState(false);
    72→  const [credentials, setCredentials] = useState({
    73→    username: "",
    74→    password: "",
    75→    apiKey: "",
    76→    token: "",
    77→  });
    78→  const [accountName, setAccountName] = useState("");
    79→  const [regionSearch, setRegionSearch] = useState("");
    80→  const [selectedContinent, setSelectedContinent] = useState<string>("all");
    81→
    82→  const { data, loading, refresh } = useFetch<{
    83→    accounts: ConnectedAccount[];
    84→    availablePlatforms: BrokerPlatformInfo[];
    85→  }>("/api/broker/connect", { accounts: [], availablePlatforms: [] });
    86→
    87→  const accounts = data?.accounts || [];
    88→
    89→  // Get filtered regions based on search and continent
    90→  const filteredRegions = useMemo(() => {
    91→    let regions = regionSearch ? searchRegions(regionSearch) : REGIONS;
    92→    if (selectedContinent !== "all") {
    93→      regions = regions.filter((r) => r.continent === selectedContinent);
    94→    }
    95→    // Deduplicate by region code to prevent React key warnings
    96→    const seen = new Set<string>();
    97→    return regions.filter((r) => {
    98→      if (seen.has(r.code)) return false;
    99→      seen.add(r.code);
   100→      return true;
   101→    });
   102→  }, [regionSearch, selectedContinent]);
   103→
   104→  // Get platforms for selected region
   105→  const platformsForRegion = useMemo(() => {
   106→    if (!selectedRegion) return [];
   107→    return getPlatformsForRegion(selectedRegion);
   108→  }, [selectedRegion]);
   109→
   110→  const selectedRegionInfo = useMemo(() => {
   111→    return REGIONS.find((r) => r.code === selectedRegion);
   112→  }, [selectedRegion]);
   113→
   114→  const selectedPlatformData = platformsForRegion.find((p) => p.id === selectedPlatform);
   115→
   116→  const handleConnect = useCallback(async () => {
   117→    if (!selectedPlatform) return;
   118→    setConnecting(true);
   119→
   120→    try {
   121→      const res = await fetch("/api/broker/connect", {
   122→        method: "POST",
   123→        headers: { "Content-Type": "application/json" },
   124→        body: JSON.stringify({
   125→          platformId: selectedPlatform,
   126→          credentials,
   127→          region: selectedRegion,
   128→          accountName: accountName || undefined,
   129→        }),
   130→      });
   131→
   132→      const result = await res.json();
   133→
   134→      if (result.success) {
   135→        toast({
   136→          title: "Broker Connected!",
   137→          description: `${result.account.accountName} connected successfully`,
   138→        });
   139→        setConnectDialogOpen(false);
   140→        resetForm();
   141→        refresh();
   142→      } else {
   143→        toast({
   144→          title: "Connection Failed",
   145→          description: result.error || "Failed to connect broker",
   146→          variant: "destructive",
   147→        });
   148→      }
   149→    } catch {
   150→      toast({
   151→        title: "Error",
   152→        description: "Failed to connect broker",
   153→        variant: "destructive",
   154→      });
   155→    } finally {
   156→      setConnecting(false);
   157→    }
   158→  }, [selectedPlatform, credentials, accountName, selectedRegion, toast, refresh]);
   159→
   160→  const handleSync = useCallback(async (accountId: string) => {
   161→    try {
   162→      const res = await fetch("/api/broker/sync", {
   163→        method: "POST",
   164→        headers: { "Content-Type": "application/json" },
   165→        body: JSON.stringify({ bettingAccountId: accountId }),
   166→      });
   167→
   168→      const result = await res.json();
   169→
   170→      if (result.success) {
   171→        const currency = result.balance?.currency || "USD";
   172→        toast({
   173→          title: "Account Synced",
   174→          description: `Balance: ${currency} ${result.balance.total.toFixed(2)}`,
   175→        });
   176→        refresh();
   177→      } else {
   178→        toast({
   179→          title: "Sync Failed",
   180→          description: result.error,
   181→          variant: "destructive",
   182→        });
   183→      }
   184→    } catch {
   185→      toast({ title: "Sync Error", variant: "destructive" });
   186→    }
   187→  }, [toast, refresh]);
   188→
   189→  const resetForm = () => {
   190→    setCredentials({ username: "", password: "", apiKey: "", token: "" });
   191→    setAccountName("");
   192→    setSelectedPlatform(null);
   193→    setStep("region");
   194→    setRegionSearch("");
   195→    setSelectedContinent("all");
   196→  };
   197→
   198→  const handleDialogClose = (open: boolean) => {
   199→    setConnectDialogOpen(open);
   200→    if (!open) resetForm();
   201→  };
   202→
   203→  const formatBalance = (amount: number, currency: string) => {
   204→    const region = REGIONS.find((r) => r.currencyCode === currency);
   205→    const symbol = region?.currencySymbol || "$";
   206→    return `${symbol}${amount.toLocaleString(undefined, { minimumFractionDigits: 2, maximumFractionDigits: 2 })}`;
   207→  };
   208→
   209→  const continents = getContinents();
   210→
   211→  return (
   212→    <div className="space-y-6">
   213→      <Card className="bg-card border-border">
   214→        <CardHeader className="pb-3">
   215→          <div className="flex items-center justify-between">
   216→            <CardTitle className="flex items-center gap-2 text-base">
   217→              <Link2 className="h-4 w-4 text-primary" />
   218→              Connected Brokers
   219→            </CardTitle>
   220→            <Dialog open={connectDialogOpen} onOpenChange={handleDialogClose}>
   221→              <Button
   222→                size="sm"
   223→                className="bg-primary text-primary-foreground"
   224→                onClick={() => setConnectDialogOpen(true)}
   225→              >
   226→                <Link2 className="h-3.5 w-3.5 mr-1.5" />
   227→                Connect Broker
   228→              </Button>
   229→
   230→              <DialogContent className="max-w-lg max-h-[90vh] overflow-y-auto sm:max-h-[90vh] max-[640px]:h-screen max-[640px]:max-h-screen max-[640px]:w-screen max-[640px]:max-w-screen max-[640px]:rounded-none max-[640px]:p-4">
   231→                <DialogHeader>
   232→                  <DialogTitle>Connect Your Broker Account</DialogTitle>
   233→                </DialogHeader>
   234→
   235→                {/* Step indicator */}
   236→                <div className="flex items-center gap-2 mb-2">
   237→                  <div className={`flex items-center gap-1.5 text-xs ${step === "region" ? "text-primary font-medium" : "text-muted-foreground"}`}>
   238→                    <div className={`h-5 w-5 rounded-full flex items-center justify-center text-[10px] font-bold ${step === "region" ? "bg-primary text-primary-foreground" : "bg-secondary text-muted-foreground"}`}>
   239→                      {step !== "region" ? "✓" : "1"}
   240→                    </div>
   241→                    Region
   242→                  </div>
   243→                  <div className="h-px flex-1 bg-border" />
   244→                  <div className={`flex items-center gap-1.5 text-xs ${step === "platform" ? "text-primary font-medium" : "text-muted-foreground"}`}>
   245→                    <div className={`h-5 w-5 rounded-full flex items-center justify-center text-[10px] font-bold ${step === "platform" ? "bg-primary text-primary-foreground" : step === "credentials" ? "bg-primary text-primary-foreground" : "bg-secondary text-muted-foreground"}`}>
   246→                      {step === "credentials" ? "✓" : "2"}
   247→                    </div>
   248→                    Platform
   249→                  </div>
   250→                  <div className="h-px flex-1 bg-border" />
   251→                  <div className={`flex items-center gap-1.5 text-xs ${step === "credentials" ? "text-primary font-medium" : "text-muted-foreground"}`}>
   252→                    <div className={`h-5 w-5 rounded-full flex items-center justify-center text-[10px] font-bold ${step === "credentials" ? "bg-primary text-primary-foreground" : "bg-secondary text-muted-foreground"}`}>
   253→                      3
   254→                    </div>
   255→                    Connect
   256→                  </div>
   257→                </div>
   258→
   259→                {/* ====== STEP 1: REGION SELECTION ====== */}
   260→                {step === "region" && (
   261→                  <div className="space-y-3">
   262→                    <p className="text-sm text-muted-foreground">
   263→                      Select your country or region. This helps us show betting platforms available in your area and set the right currency.
   264→                    </p>
   265→
   266→                    <div className="relative">
   267→                      <Search className="absolute left-3 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
   268→                      <Input
   269→                        value={regionSearch}
   270→                        onChange={(e) => setRegionSearch(e.target.value)}
   271→                        placeholder="Search country or currency..."
   272→                        className="bg-secondary border-border pl-9"
   273→                      />
   274→                    </div>
   275→
   276→                    <div className="flex flex-wrap gap-1.5">
   277→                      <button
   278→                        onClick={() => setSelectedContinent("all")}
   279→                        className={`rounded-full px-3 py-1 text-xs font-medium transition-all ${
   280→                          selectedContinent === "all"
   281→                            ? "bg-primary/10 text-primary border border-primary/30"
   282→                            : "bg-secondary/50 text-muted-foreground border border-border"
   283→                        }`}
   284→                      >
   285→                        All
   286→                      </button>
   287→                      {continents.map((c) => (
   288→                        <button
   289→                          key={c}
   290→                          onClick={() => setSelectedContinent(c)}
   291→                          className={`rounded-full px-3 py-1 text-xs font-medium transition-all ${
   292→                            selectedContinent === c
   293→                              ? "bg-primary/10 text-primary border border-primary/30"
   294→                              : "bg-secondary/50 text-muted-foreground border border-border"
   295→                          }`}
   296→                        >
   297→                          {c}
   298→                        </button>
   299→                      ))}
   300→                    </div>
   301→
   302→                    <div className="grid grid-cols-2 sm:grid-cols-3 gap-2 max-h-64 overflow-y-auto -webkit-overflow-scrolling-touch">
   303→                      {filteredRegions.map((region) => {
   304→                        const platformCount = getPlatformsForRegion(region.code).length;
   305→                        return (
   306→                          <button
   307→                            key={region.code}
   308→                            onClick={() => {
   309→                              setSelectedRegion(region.code);
   310→                              setStep("platform");
   311→                            }}
   312→                            className={`flex items-center gap-2 rounded-lg border p-2.5 transition-all text-left hover:bg-primary/5 active:scale-[0.98] ${
   313→                              selectedRegion === region.code
   314→                                ? "border-primary/30 bg-primary/5"
   315→                                : "border-border"
   316→                            }`}
   317→                          >
   318→                            <span className="text-lg">{region.flag}</span>
   319→                            <div className="min-w-0 flex-1">
   320→                              <p className="text-xs font-medium text-foreground truncate">{region.name}</p>
   321→                              <p className="text-[10px] text-muted-foreground">
   322→                                {region.currencySymbol} {region.currencyCode} · {platformCount} platforms
   323→                              </p>
   324→                            </div>
   325→                          </button>
   326→                        );
   327→                      })}
   328→                    </div>
   329→
   330→                    {filteredRegions.length === 0 && (
   331→                      <div className="text-center py-4">
   332→                        <p className="text-sm text-muted-foreground">No regions found matching your search</p>
   333→                      </div>
   334→                    )}
   335→                  </div>
   336→                )}
   337→
   338→                {/* ====== STEP 2: PLATFORM SELECTION ====== */}
   339→                {step === "platform" && selectedRegionInfo && (
   340→                  <div className="space-y-3">
   341→                    <button
   342→                      onClick={() => {
   343→                        setStep("region");
   344→                        setSelectedPlatform(null);
   345→                      }}
   346→                      className="flex items-center gap-1 text-xs text-muted-foreground hover:text-primary transition-colors"
   347→                    >
   348→                      <ArrowLeft className="h-3 w-3" /> Change region
   349→                    </button>
   350→
   351→                    <div className="flex items-center gap-2 p-2.5 rounded-lg bg-primary/5 border border-primary/10">
   352→                      <span className="text-lg">{selectedRegionInfo.flag}</span>
   353→                      <div>
   354→                        <p className="text-sm font-medium text-foreground">{selectedRegionInfo.name}</p>
   355→                        <p className="text-xs text-muted-foreground">
   356→                          Currency: {selectedRegionInfo.currencySymbol} {selectedRegionInfo.currencyCode} ({selectedRegionInfo.currencyName})
   357→                        </p>
   358→                      </div>
   359→                    </div>
   360→
   361→                    <p className="text-sm text-muted-foreground">
   362→                      {platformsForRegion.length} platforms available in {selectedRegionInfo.name}. Select your betting platform.
   363→                    </p>
   364→
   365→                    <div className="space-y-2 max-h-64 overflow-y-auto -webkit-overflow-scrolling-touch">
   366→                      {platformsForRegion.map((platform) => {
   367→                        const isPopular = platform.popularIn.includes(selectedRegion);
   368→                        return (
   369→                          <button
   370→                            key={platform.id}
   371→                            onClick={() => {
   372→                              setSelectedPlatform(platform.id);
   373→                              setStep("credentials");
   374→                            }}
   375→                            className="flex items-center justify-between rounded-lg border border-border p-3 hover:bg-primary/5 active:scale-[0.98] transition-colors text-left w-full"
   376→                          >
   377→                            <div className="flex items-center gap-3">
   378→                              <div
   379→                                className="flex h-10 w-10 items-center justify-center rounded-lg shrink-0 overflow-hidden bg-secondary/50 border border-border"
   380→                              >
   381→                                <img
   382→                                  src={getBrokerLogoUrl(platform.id, platform.name, platform.color || "#10b981")}
   383→                                  alt={platform.name}
   384→                                  className="h-10 w-10 object-contain p-1 rounded-lg"
   385→                                  onError={(e) => {
   386→                                    const target = e.currentTarget;
   387→                                    target.style.display = "none";
   388→                                    target.parentElement!.innerHTML = `<div style="background:${platform.color || "#10b981"};width:100%;height:100%;display:flex;align-items:center;justify-content:center;border-radius:0.5rem;color:#fff;font-size:0.65rem;font-weight:700;">${platform.name.substring(0, 2).toUpperCase()}</div>`;
   389→                                  }}
   390→                                />
   391→                              </div>
   392→                              <div>
   393→                                <div className="flex items-center gap-1.5">
   394→                                  <p className="text-sm font-medium text-foreground">{platform.name}</p>
   395→                                  {isPopular && (
   396→                                    <Badge className="bg-amber-400/10 text-amber-400 border-amber-400/30 text-[9px] px-1 py-0">
   397→                                      <Star className="h-2.5 w-2.5 mr-0.5" /> Popular
   398→                                    </Badge>
   399→                                  )}
   400→                                </div>
   401→                                <p className="text-xs text-muted-foreground">
   402→                                  {platform.authType.replace(/_/g, " ")} · {platform.mobileApp ? "📱 App" : "Web only"}
   403→                                </p>
   404→                              </div>
   405→                            </div>
   406→                            <ChevronRight className="h-4 w-4 text-muted-foreground shrink-0" />
   407→                          </button>
   408→                        );
   409→                      })}
   410→                    </div>
   411→                  </div>
   412→                )}
   413→
   414→                {/* ====== STEP 3: CREDENTIALS ====== */}
   415→                {step === "credentials" && selectedPlatformData && selectedRegionInfo && (
   416→                  <div className="space-y-4">
   417→                    <button
   418→                      onClick={() => {
   419→                        setStep("platform");
   420→                        setSelectedPlatform(null);
   421→                      }}
   422→                      className="flex items-center gap-1 text-xs text-muted-foreground hover:text-primary transition-colors"
   423→                    >
   424→                      <ArrowLeft className="h-3 w-3" /> Change platform
   425→                    </button>
   426→
   427→                    <div className="flex items-center gap-3 p-3 rounded-lg bg-primary/5 border border-primary/10">
   428→                      <div
   429→                        className="flex h-10 w-10 items-center justify-center rounded-lg shrink-0 overflow-hidden bg-secondary/50 border border-border"
   430→                      >
   431→                        <img
   432→                          src={getBrokerLogoUrl(selectedPlatformData.id, selectedPlatformData.name, selectedPlatformData.color || "#10b981")}
   433→                          alt={selectedPlatformData.name}
   434→                          className="h-10 w-10 object-contain p-1 rounded-lg"
   435→                          onError={(e) => {
   436→                            const target = e.currentTarget;
   437→                            target.style.display = "none";
   438→                            target.parentElement!.innerHTML = `<div style="background:${selectedPlatformData.color || "#10b981"};width:100%;height:100%;display:flex;align-items:center;justify-content:center;border-radius:0.5rem;color:#fff;font-size:0.65rem;font-weight:700;">${selectedPlatformData.name.substring(0, 2).toUpperCase()}</div>`;
   439→                          }}
   440→                        />
   441→                      </div>
   442→                      <div className="flex-1 min-w-0">
   443→                        <p className="text-sm font-medium text-foreground">{selectedPlatformData.name}</p>
   444→                        <p className="text-xs text-muted-foreground">
   445→                          {selectedPlatformData.authType.replace(/_/g, " ")} authentication
   446→                        </p>
   447→                      </div>
   448→                      <div className="text-right shrink-0">
   449→                        <p className="text-xs text-muted-foreground">{selectedRegionInfo.flag} {selectedRegionInfo.name}</p>
   450→                        <p className="text-xs font-medium text-primary">{selectedRegionInfo.currencySymbol} {selectedRegionInfo.currencyCode}</p>
   451→                      </div>
   452→                    </div>
   453→
   454→                    <div className="space-y-3">
   455→                      <div>
   456→                        <Label className="text-xs text-muted-foreground">Account Name</Label>
   457→                        <Input
   458→                          value={accountName}
   459→                          onChange={(e) => setAccountName(e.target.value)}
   460→                          placeholder={`My ${selectedPlatformData?.name} Account`}
   461→                          className="bg-secondary border-border mt-1"
   462→                        />
   463→                      </div>
   464→
   465→                      <Separator />
   466→
   467→                      {selectedPlatformData?.authType === "web_session" && (
   468→                        <>
   469→                          <div>
   470→                            <Label className="text-xs text-muted-foreground flex items-center gap-1">
   471→                              <User className="h-3 w-3" /> Username / Email
   472→                            </Label>
   473→                            <Input
   474→                              value={credentials.username}
   475→                              onChange={(e) => setCredentials({ ...credentials, username: e.target.value })}
   476→                              placeholder="Your broker username"
   477→                              className="bg-secondary border-border mt-1"
   478→                            />
   479→                          </div>
   480→                          <div>
   481→                            <Label className="text-xs text-muted-foreground flex items-center gap-1">
   482→                              <Lock className="h-3 w-3" /> Password
   483→                            </Label>
   484→                            <Input
   485→                              type="password"
   486→                              value={credentials.password}
   487→                              onChange={(e) => setCredentials({ ...credentials, password: e.target.value })}
   488→                              placeholder="Your broker password"
   489→                              className="bg-secondary border-border mt-1"
   490→                            />
   491→                          </div>
   492→                        </>
   493→                      )}
   494→
   495→                      {selectedPlatformData?.authType === "api_key" && (
   496→                        <div>
   497→                          <Label className="text-xs text-muted-foreground flex items-center gap-1">
   498→                            <Key className="h-3 w-3" /> API Key
   499→                          </Label>
   500→                          <Input
   501→                            value={credentials.apiKey}
   502→                            onChange={(e) => setCredentials({ ...credentials, apiKey: e.target.value })}
   503→                            placeholder="Your broker API key"
   504→                            className="bg-secondary border-border mt-1"
   505→                          />
   506→                        </div>
   507→                      )}
   508→
   509→                      {selectedPlatformData?.authType === "oauth" && (
   510→                        <div>
   511→                          <Label className="text-xs text-muted-foreground flex items-center gap-1">
   512→                            <Shield className="h-3 w-3" /> OAuth Token
   513→                          </Label>
   514→                          <Input
   515→                            value={credentials.token}
   516→                            onChange={(e) => setCredentials({ ...credentials, token: e.target.value })}
   517→                            placeholder="OAuth access token"
   518→                            className="bg-secondary border-border mt-1"
   519→                          />
   520→                        </div>
   521→                      )}
   522→
   523→                      {selectedPlatformData?.authType === "manual" && (
   524→                        <div className="rounded-lg bg-amber-400/5 border border-amber-400/10 p-3">
   525→                          <p className="text-xs text-muted-foreground">
   526→                            Manual connection: You provide your account details and manage bets manually.
   527→                            The app will track your allocation and commission.
   528→                          </p>
   529→                        </div>
   530→                      )}
   531→                    </div>
