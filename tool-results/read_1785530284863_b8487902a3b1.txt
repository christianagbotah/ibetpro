     1→     1→// ============================================================================
     2→     2→// iBetPro Bot Engine - Background Auto-Betting Engine
     3→     3→// Runs continuously on the server using setInterval, independent of frontend
     4→     4→// polling. Each user's bot gets its own timer that scans matches and places
     5→     5→// bets automatically according to their settings.
     6→     6→// ============================================================================
     7→     7→
     8→     8→import { prisma } from "./db";
     9→     9→import { analyzeMatch, shouldAutoBet, checkRiskLimits, isWithinBetSchedule } from "./ai-engine-v2";
    10→    10→import { placeBetOnBroker } from "./broker-integration";
    11→    11→
    12→    12→// ==================== TYPES ====================
    13→    13→
    14→    14→interface ScanResult {
    15→    15→  betsPlaced: number;
    16→    16→  totalStake: number;
    17→    17→  matches: number;
    18→    18→  skipped: number;
    19→    19→  error?: string;
    20→    20→}
    21→    21→
    22→    22→interface BotEngineStats {
    23→    23→  userId: string;
    24→    24→  status: "running" | "stopped" | "paused";
    25→    25→  scanIntervalSec: number;
    26→    26→  totalScans: number;
    27→    27→  totalBetsPlaced: number;
    28→    28→  totalStakeUsed: number;
    29→    29→  totalProfit: number;
    30→    30→  lastScanAt: Date | null;
    31→    31→  lastBetAt: Date | null;
    32→    32→  startedAt: Date | null;
    33→    33→  errorCount: number;
    34→    34→  lastError: string | null;
    35→    35→}
    36→    36→
    37→    37→// ==================== SINGLETON ENGINE ====================
    38→    38→
    39→    39→/**
    40→    40→ * BotEngine is a singleton that manages all running bot instances.
    41→    41→ * Each user has their own timer that runs scan cycles on a configurable interval.
    42→    42→ *
    43→    43→ * The engine is designed to:
    44→    44→ * - Run independently of the frontend (no polling required)
    45→    45→ * - Auto-recover from errors (continues running after failed scans)
    46→    46→ * - Auto-stop on risk limits, schedule violations, or no allocation
    47→    47→ * - Persist state in the database (BotSession) for recovery after restarts
    48→    48→ * - Support graceful shutdown
    49→    49→ */
    50→    50→class BotEngine {
    51→    51→  private static instance: BotEngine | null = null;
    52→    52→  private timers: Map<string, NodeJS.Timeout> = new Map();
    53→    53→  private runningUsers: Map<string, BotEngineStats> = new Map();
    54→    54→  private isShuttingDown = false;
    55→    55→
    56→    56→  private constructor() {
    57→    57→    // Register graceful shutdown handlers
    58→    58→    process.on("SIGTERM", () => this.shutdown());
    59→    59→    process.on("SIGINT", () => this.shutdown());
    60→    60→  }
    61→    61→
    62→    62→  static getInstance(): BotEngine {
    63→    63→    if (!BotEngine.instance) {
    64→    64→      BotEngine.instance = new BotEngine();
    65→    65→    }
    66→    66→    return BotEngine.instance;
    67→    67→  }
    68→    68→
    69→    69→  // ==================== LIFECYCLE ====================
    70→    70→
    71→    71→  /**
    72→    72→   * Start the bot for a specific user. Creates a timer that runs scan cycles
    73→    73→   * on the configured interval (default 30 seconds).
    74→    74→   */
    75→    75→  async start(userId: string, scanIntervalSec: number = 30): Promise<{ success: boolean; message: string }> {
    76→    76→    // Don't start if already running
    77→    77→    if (this.timers.has(userId)) {
    78→    78→      return { success: false, message: "Bot is already running for this user" };
    79→    79→    }
    80→    80→
    81→    81→    // Validate prerequisites
    82→    82→    const validation = await this.validatePrerequisites(userId);
    83→    83→    if (!validation.valid) {
    84→    84→      return { success: false, message: validation.error! };
    85→    85→    }
    86→    86→
    87→    87→    // Initialize stats tracking
    88→    88→    const stats: BotEngineStats = {
    89→    89→      userId,
    90→    90→      status: "running",
    91→    91→      scanIntervalSec,
    92→    92→      totalScans: 0,
    93→    93→      totalBetsPlaced: 0,
    94→    94→      totalStakeUsed: 0,
    95→    95→      totalProfit: 0,
    96→    96→      lastScanAt: null,
    97→    97→      lastBetAt: null,
    98→    98→      startedAt: new Date(),
    99→    99→      errorCount: 0,
   100→   100→      lastError: null,
   101→   101→    };
   102→   102→    this.runningUsers.set(userId, stats);
   103→   103→
   104→   104→    // Run the first scan immediately
   105→   105→    try {
   106→   106→      const result = await this.runScanCycle(userId);
   107→   107→      stats.totalScans++;
   108→   108→      stats.totalBetsPlaced += result.betsPlaced;
   109→   109→      stats.totalStakeUsed += result.totalStake;
   110→   110→      stats.lastScanAt = new Date();
   111→   111→      if (result.betsPlaced > 0) {
   112→   112→        stats.lastBetAt = new Date();
   113→   113→      }
   114→   114→      if (result.error) {
   115→   115→        stats.errorCount++;
   116→   116→        stats.lastError = result.error;
   117→   117→      }
   118→   118→    } catch (error) {
   119→   119→      console.error(`[BotEngine] First scan failed for user ${userId}:`, error);
   120→   120→      stats.errorCount++;
   121→   121→      stats.lastError = error instanceof Error ? error.message : "Unknown error";
   122→   122→    }
   123→   123→
   124→   124→    // Set up the recurring timer
   125→   125→    const timer = setInterval(async () => {
   126→   126→      if (this.isShuttingDown) return;
   127→   127→
   128→   128→      try {
   129→   129→        const currentStats = this.runningUsers.get(userId);
   130→   130→        if (!currentStats || currentStats.status !== "running") {
   131→   131→          // Bot was stopped externally, clean up
   132→   132→          this.stop(userId, "auto_stopped");
   133→   133→          return;
   134→   134→        }
   135→   135→
   136→   136→        // Check if the session is still running in the DB
   137→   137→        const session = await prisma.botSession.findUnique({ where: { userId } });
   138→   138→        if (!session || session.status !== "running") {
   139→   139→          this.stop(userId, "session_expired");
   140→   140→          return;
   141→   141→        }
   142→   142→
   143→   143→        const result = await this.runScanCycle(userId);
   144→   144→
   145→   145→        currentStats.totalScans++;
   146→   146→        currentStats.totalBetsPlaced += result.betsPlaced;
   147→   147→        currentStats.totalStakeUsed += result.totalStake;
   148→   148→        currentStats.lastScanAt = new Date();
   149→   149→        currentStats.lastError = null; // Clear error on success
   150→   150→
   151→   151→        if (result.betsPlaced > 0) {
   152→   152→          currentStats.lastBetAt = new Date();
   153→   153→        }
   154→   154→
   155→   155→        if (result.error) {
   156→   156→          currentStats.errorCount++;
   157→   157→          currentStats.lastError = result.error;
   158→   158→        }
   159→   159→
   160→   160→        // Update session stats in DB
   161→   161→        await prisma.botSession.update({
   162→   162→          where: { userId },
   163→   163→          data: {
   164→   164→            totalScans: { increment: 1 },
   165→   165→            totalBetsPlaced: { increment: result.betsPlaced },
   166→   166→            totalStakeUsed: { increment: result.totalStake },
   167→   167→            lastScanAt: new Date(),
   168→   168→            lastBetAt: result.betsPlaced > 0 ? new Date() : undefined,
   169→   169→          },
   170→   170→        });
   171→   171→
   172→   172→        // Log scan activity periodically (every 10 scans)
   173→   173→        if (currentStats.totalScans % 10 === 0) {
   174→   174→          await prisma.botLog.create({
   175→   175→            data: {
   176→   176→              userId,
   177→   177→              action: "bot_scan",
   178→   178→              reasoning: `Auto-scan #${currentStats.totalScans}: ${result.matches} matches found, ${result.betsPlaced} bets placed, ${result.skipped} skipped`,
   179→   179→              details: JSON.stringify({
   180→   180→                scanNumber: currentStats.totalScans,
   181→   181→                betsPlaced: result.betsPlaced,
   182→   182→                totalStake: result.totalStake,
   183→   183→                matches: result.matches,
   184→   184→                skipped: result.skipped,
   185→   185→              }),
   186→   186→            },
   187→   187→          });
   188→   188→        }
   189→   189→
   190→   190→        // Auto-settle finished matches every 3rd scan
   191→   191→        if (currentStats.totalScans % 3 === 0) {
   192→   192→          try {
   193→   193→            await this.autoSettle(userId);
   194→   194→          } catch (settleError) {
   195→   195→            console.error(`[BotEngine] Auto-settle error for user ${userId}:`, settleError);
   196→   196→          }
   197→   197→        }
   198→   198→
   199→   199→        // Auto-cashout evaluation every 5th scan
   200→   200→        if (currentStats.totalScans % 5 === 0) {
   201→   201→          try {
   202→   202→            await this.autoCashoutEvaluation(userId);
   203→   203→          } catch (cashoutError) {
   204→   204→            console.error(`[BotEngine] Auto-cashout error for user ${userId}:`, cashoutError);
   205→   205→          }
   206→   206→        }
   207→   207→      } catch (error) {
   208→   208→        console.error(`[BotEngine] Scan cycle error for user ${userId}:`, error);
   209→   209→        const currentStats = this.runningUsers.get(userId);
   210→   210→        if (currentStats) {
   211→   211→          currentStats.errorCount++;
   212→   212→          currentStats.lastError = error instanceof Error ? error.message : "Unknown error";
   213→   213→
   214→   214→          // Auto-stop after too many consecutive errors
   215→   215→          if (currentStats.errorCount >= 10) {
   216→   216→            console.error(`[BotEngine] Too many errors for user ${userId}, stopping bot`);
   217→   217→            await this.stop(userId, "too_many_errors");
   218→   218→          }
   219→   219→        }
   220→   220→      }
   221→   221→    }, scanIntervalSec * 1000);
   222→   222→
   223→   223→    // Prevent Node.js from keeping the process alive just for this timer
   224→   224→    if (timer.unref) {
   225→   225→      timer.unref();
   226→   226→    }
   227→   227→
   228→   228→    this.timers.set(userId, timer);
   229→   229→
   230→   230→    console.log(`[BotEngine] Bot started for user ${userId} with interval ${scanIntervalSec}s`);
   231→   231→
   232→   232→    // Log bot start
   233→   233→    await prisma.botLog.create({
   234→   234→      data: {
   235→   235→        userId,
   236→   236→        action: "bot_started",
   237→   237→        reasoning: `Bot engine started in background mode. Scanning every ${scanIntervalSec}s.`,
   238→   238→        details: JSON.stringify({
   239→   239→          mode: "background",
   240→   240→          scanIntervalSec,
   241→   241→          broker: validation.connectedAccount?.platform,
   242→   242→          allocation: validation.connectedAccount?.allocatedAmount,
   243→   243→        }),
   244→   244→      },
   245→   245→    });
   246→   246→
   247→   247→    return {
   248→   248→      success: true,
   249→   249→      message: stats.totalBetsPlaced > 0
   250→   250→        ? `Bot started! Placed ${stats.totalBetsPlaced} bet(s) in first scan. Running in background.`
   251→   251→        : "Bot started! Scanning for matches in background. Will place bets automatically.",
   252→   252→    };
   253→   253→  }
   254→   254→
   255→   255→  /**
   256→   256→   * Stop the bot for a specific user. Clears the timer and updates the session.
   257→   257→   */
   258→   258→  async stop(userId: string, reason: string = "user_stopped"): Promise<void> {
   259→   259→    const timer = this.timers.get(userId);
   260→   260→    if (timer) {
   261→   261→      clearInterval(timer);
   262→   262→      this.timers.delete(userId);
   263→   263→    }
   264→   264→
   265→   265→    const stats = this.runningUsers.get(userId);
   266→   266→    if (stats) {
   267→   267→      stats.status = "stopped";
   268→   268→      this.runningUsers.delete(userId);
   269→   269→    }
   270→   270→
   271→   271→    // Update the database session
   272→   272→    try {
   273→   273→      await prisma.botSession.upsert({
   274→   274→        where: { userId },
   275→   275→        update: {
   276→   276→          status: "stopped",
   277→   277→          stoppedAt: new Date(),
   278→   278→          stopReason: reason,
   279→   279→        },
   280→   280→        create: {
   281→   281→          userId,
   282→   282→          status: "stopped",
   283→   283→          stoppedAt: new Date(),
   284→   284→          stopReason: reason,
   285→   285→        },
   286→   286→      });
   287→   287→
   288→   288→      // Log bot stop
   289→   289→      await prisma.botLog.create({
   290→   290→        data: {
   291→   291→          userId,
   292→   292→          action: "bot_stopped",
   293→   293→          reasoning: `Bot engine stopped. Reason: ${reason}`,
   294→   294→          details: JSON.stringify({
   295→   295→            reason,
   296→   296→            totalScans: stats?.totalScans || 0,
   297→   297→            totalBetsPlaced: stats?.totalBetsPlaced || 0,
   298→   298→            totalStakeUsed: stats?.totalStakeUsed || 0,
   299→   299→            totalProfit: stats?.totalProfit || 0,
   300→   300→            runDuration: stats?.startedAt
   301→   301→              ? Math.round((Date.now() - stats.startedAt.getTime()) / 60000)
   302→   302→              : 0,
   303→   303→            errorCount: stats?.errorCount || 0,
   304→   304→          }),
   305→   305→        },
   306→   306→      });
   307→   307→    } catch (error) {
   308→   308→      console.error(`[BotEngine] Error stopping bot for user ${userId}:`, error);
   309→   309→    }
   310→   310→
   311→   311→    console.log(`[BotEngine] Bot stopped for user ${userId}. Reason: ${reason}`);
   312→   312→  }
   313→   313→
   314→   314→  /**
   315→   315→   * Gracefully shut down all running bots.
   316→   316→   */
   317→   317→  async shutdown(): Promise<void> {
   318→   318→    if (this.isShuttingDown) return;
   319→   319→    this.isShuttingDown = true;
   320→   320→
   321→   321→    console.log(`[BotEngine] Shutting down ${this.timers.size} running bots...`);
   322→   322→
   323→   323→    for (const [userId] of this.timers) {
   324→   324→      await this.stop(userId, "server_shutdown");
   325→   325→    }
   326→   326→
   327→   327→    this.timers.clear();
   328→   328→    this.runningUsers.clear();
   329→   329→  }
   330→   330→
   331→   331→  /**
   332→   332→   * Recover bots that were running before a server restart.
   333→   333→   * Checks the database for sessions with status "running" and restarts them.
   334→   334→   */
   335→   335→  async recoverRunningBots(): Promise<number> {
   336→   336→    try {
   337→   337→      const runningSessions = await prisma.botSession.findMany({
   338→   338→        where: { status: "running" },
   339→   339→      });
   340→   340→
   341→   341→      console.log(`[BotEngine] Found ${runningSessions.length} running sessions to recover`);
   342→   342→
   343→   343→      let recovered = 0;
   344→   344→      for (const session of runningSessions) {
   345→   345→        try {
   346→   346→          const result = await this.start(session.userId, session.scanIntervalSec || 30);
   347→   347→          if (result.success) {
   348→   348→            recovered++;
   349→   349→            console.log(`[BotEngine] Recovered bot for user ${session.userId}`);
   350→   350→          } else {
   351→   351→            console.warn(`[BotEngine] Could not recover bot for user ${session.userId}: ${result.message}`);
   352→   352→          }
   353→   353→        } catch (error) {
   354→   354→          console.error(`[BotEngine] Error recovering bot for user ${session.userId}:`, error);
   355→   355→        }
   356→   356→      }
   357→   357→
   358→   358→      return recovered;
   359→   359→    } catch (error) {
   360→   360→      console.error("[BotEngine] Error recovering bots:", error);
   361→   361→      return 0;
   362→   362→    }
   363→   363→  }

   368→   368→   * Get the current status of a user's bot.
   369→   369→   */
   370→   370→  getStatus(userId: string): BotEngineStats | null {
   371→   371→    return this.runningUsers.get(userId) || null;
   372→   372→  }

   379→   379→  }
   380→   380→
   381→   381→  /**
   382→   382→   * Get all running bot stats.

   386→   386→  }

   388→   388→  /**
   389→   389→   * Get the number of currently running bots.

   391→   391→  getRunningCount(): number {
   392→   392→    return this.timers.size;
   393→   393→  }
   394→   394→
   395→   395→  // ==================== SCAN CYCLE ====================
   396→   396→
   397→   397→  /**
   398→   398→   * Run a single scan cycle - analyze matches and place bets.
   399→   399→   * This is the core logic that runs on each interval tick.
   400→   400→   */
   401→   401→  private async runScanCycle(userId: string): Promise<ScanResult> {
   402→   402→    const result: ScanResult = { betsPlaced: 0, totalStake: 0, matches: 0, skipped: 0 };
   403→   403→
   404→   404→    try {
   405→   405→      // Get user settings
   406→   406→      const settings = await prisma.userSettings.findUnique({ where: { userId } });
   407→   407→      if (!settings) {
   408→   408→        result.error = "Settings not found";
   409→   409→        return result;
   410→   410→      }
   411→   411→
   412→   412→      if (!settings.autoBettingEnabled) {
   413→   413→        result.error = "Auto-betting is disabled";
   414→   414→        await this.stop(userId, "auto_betting_disabled");
   415→   415→        return result;
   416→   416→      }
   417→   417→
   418→   418→      // Check schedule
   419→   419→      if (!isWithinBetSchedule(settings.betScheduleStart, settings.betScheduleEnd)) {
   420→   420→        // Don't stop the bot, just skip this scan
   421→   421→        return result;
   422→   422→      }
   423→   423→
   424→   424→      // Check risk limits
   425→   425→      const user = await prisma.user.findUnique({ where: { id: userId } });
   426→   426→      const riskCheck = checkRiskLimits(user?.dailyPnl || 0, user?.weeklyPnl || 0, {
   427→   427→        stopLossDaily: settings.stopLossDaily,
   428→   428→        stopLossWeekly: settings.stopLossWeekly,
   429→   429→        profitTargetDaily: settings.profitTargetDaily,
   430→   430→        profitTargetWeekly: settings.profitTargetWeekly,
   431→   431→      });
   432→   432→
   433→   433→      if (!riskCheck.canBet) {
   434→   434→        // Auto-stop the bot if risk limits hit
   435→   435→        const reason = (user?.dailyPnl ?? 0) <= -settings.stopLossDaily ? "stop_loss" : "profit_target";
   436→   436→        await this.stop(userId, reason);
   437→   437→        await prisma.botLog.create({
   438→   438→          data: {
   439→   439→            userId,
   440→   440→            action: reason === "stop_loss" ? "stop_loss_hit" : "profit_target_hit",
   441→   441→            reasoning: riskCheck.reason,
   442→   442→          },
   443→   443→        });
   444→   444→        return result;
   445→   445→      }
   446→   446→
   447→   447→      // Get connected betting account
   448→   448→      const bettingAccount = await prisma.bettingAccount.findFirst({
   449→   449→        where: { userId, isConnected: true },
   450→   450→        orderBy: { allocatedAmount: "desc" },
   451→   451→      });
   452→   452→
   453→   453→      if (!bettingAccount || bettingAccount.allocatedAmount <= 0) {
   454→   454→        await this.stop(userId, "no_allocation");
   455→   455→        await prisma.botLog.create({
   456→   456→          data: {
   457→   457→            userId,
   458→   458→            action: "bot_scan",
   459→   459→            reasoning: "No connected broker or allocation available. Stopping bot.",
   460→   460→          },
   461→   461→        });
   462→   462→        return result;
   463→   463→      }
   464→   464→
   465→   465→      // Check daily limit
   466→   466→      const todayStart = new Date();
   467→   467→      todayStart.setHours(0, 0, 0, 0);
   468→   468→      const todayBets = await prisma.bet.findMany({
   469→   469→        where: {
   470→   470→          userId,
   471→   471→          placedAt: { gte: todayStart },
   472→   472→          status: { in: ["pending", "won", "lost", "cashed_out", "partial_cashout"] },
   473→   473→        },
   474→   474→      });
   475→   475→      const dailyStake = todayBets.reduce((sum, b) => sum + b.stake, 0);
   476→   476→      const existingBetMatchIds = todayBets.map((b) => b.matchId);
   477→   477→
   478→   478→      if (dailyStake >= settings.dailyBetLimit) {
   479→   479→        // Don't stop the bot, just skip this scan
   480→   480→        return result;
   481→   481→      }
   482→   482→
   483→   483→      const remainingDailyLimit = Math.min(
   484→   484→        settings.dailyBetLimit - dailyStake,
   485→   485→        bettingAccount.allocatedAmount
   486→   486→      );
   487→   487→
   488→   488→      if (remainingDailyLimit < 5) {
   489→   489→        return result;
   490→   490→      }
   491→   491→
   492→   492→      // Get active allocation
   493→   493→      const activeAllocation = await prisma.allocation.findFirst({
   494→   494→        where: { userId, bettingAccountId: bettingAccount.id, status: "active" },
   495→   495→      });
   496→   496→
   497→   497→      // Get upcoming matches
   498→   498→      const preferredSports = settings.preferredSports?.split(",") || ["football"];
   499→   499→      const upcomingMatches = await prisma.match.findMany({
   500→   500→        where: {
   501→   501→          status: "upcoming",
   502→   502→          sport: { in: preferredSports },
   503→   503→          commenceTime: { gte: new Date() },
   504→   504→          id: { notIn: existingBetMatchIds },
   505→   505→        },
   506→   506→        orderBy: { commenceTime: "asc" },
   507→   507→        take: 20,
   508→   508→      });
   509→   509→
   510→   510→      result.matches = upcomingMatches.length;
   511→   511→

   944→   944→  // ==================== AUTO CASHOUT EVALUATION ====================
   945→   945→
   946→   946→  /**
   947→   947→   * Evaluate live bets for cashout opportunities.
   948→   948→   * Called periodically by the engine (every 5th scan cycle).
   949→   949→   * Uses the AI engine's cashout recommendation system.
   950→   950→   */
   951→   951→  private async autoCashoutEvaluation(userId: string): Promise<number> {
   952→   952→    const settings = await prisma.userSettings.findUnique({ where: { userId } });
   953→   953→    if (!settings?.autoCashoutEnabled) return 0;
   954→   954→
   955→   955→    // Find live bets that could be cashed out
   956→   956→    const liveBets = await prisma.bet.findMany({
   957→   957→      where: {
   958→   958→        userId,
   959→   959→        status: "pending",
   960→   960→        isAutoPlaced: true,
   961→   961→        match: { status: "live" },
   962→   962→      },
   963→   963→      include: { match: true },
   964→   964→    });
   965→   965→
   966→   966→    if (liveBets.length === 0) return 0;
   967→   967→
   968→   968→    let cashoutCount = 0;
   969→   969→
   970→   970→    for (const bet of liveBets) {
   971→   971→      if (!bet.match) continue;
   972→   972→
   973→   973→      // Simple cashout evaluation based on match state
   974→   974→      const match = bet.match;
   975→   975→      const homeScore = match.homeScore ?? 0;
   976→   976→      const awayScore = match.awayScore ?? 0;
   977→   977→      const minute = match.minute ?? 0;
   978→   978→
   979→   979→      // Determine if the bet is winning
   980→   980→      let isWinning = false;
   981→   981→      if (bet.selection === match.homeTeam) {
   982→   982→        isWinning = homeScore > awayScore;
   983→   983→      } else if (bet.selection === match.awayTeam) {
   984→   984→        isWinning = awayScore > homeScore;
   985→   985→      } else if (bet.selection === "Draw") {
   986→   986→        isWinning = homeScore === awayScore;
   987→   987→      } else if (bet.selection === "Over 2.5") {
   988→   988→        isWinning = (homeScore + awayScore) > 2.5;
   989→   989→      } else if (bet.selection === "Under 2.5") {
   990→   990→        isWinning = (homeScore + awayScore) < 2.5;
   991→   991→      }
   992→   992→
   993→   993→      // Cashout logic: if the bet is winning and we're past 70 minutes,
   994→   994→      // and the cashout threshold is met, execute cashout
   995→   995→      if (isWinning && minute >= 70) {
   996→   996→        const cashoutOdds = 1 + (1 / bet.odds) * (minute / 90);
   997→   997→        const cashoutAmount = Math.round(bet.stake * cashoutOdds * 100) / 100;
   998→   998→
   999→   999→        if (cashoutAmount > bet.stake * (1 + settings.cashoutThreshold)) {
  1000→  1000→          // Execute partial or full cashout based on settings
