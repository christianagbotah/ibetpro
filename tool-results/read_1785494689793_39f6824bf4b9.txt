   934→   934→   934→// ==================== ADAPTER FACTORY ====================
   935→   935→   935→
   936→   936→   936→const adapterCache = new Map<string, BrokerAdapter>();
   937→   937→   937→
   938→   938→   938→/**
   939→   939→   939→ * Get or create a broker adapter for the given platform
   940→   940→   940→ * @param platformId - The platform ID
   941→   941→   941→ * @param demoMode - If true, the adapter will use demo/sandbox mode (no real API calls)
   942→   942→   942→ */
   943→   943→   943→export function getBrokerAdapter(platformId: string, demoMode?: boolean): BrokerAdapter | null {
   944→   944→   944→  // Check cache (but demo mode changes per-user, so we need a keyed cache)
   945→   945→   945→  const cacheKey = demoMode !== undefined ? `${platformId}_${demoMode ? "demo" : "real"}` : platformId;
   946→   946→   946→  if (adapterCache.has(cacheKey)) {
   947→   947→   947→    return adapterCache.get(cacheKey)!;
   948→   948→   948→  }
   949→   949→   949→
   950→   950→   950→  // Manual adapter
   951→   951→   951→  if (platformId === "manual") {
   952→   952→   952→    const adapter = new ManualBrokerAdapter();
   953→   953→   953→    adapterCache.set(cacheKey, adapter);
   954→   954→   954→    return adapter;
   955→   955→   955→  }
   956→   956→   956→
   957→   957→   957→  // Get platform config
   958→   958→   958→  const platform = getBrokerPlatform(platformId);
   959→   959→   959→  if (!platform) {
   960→   960→   960→    console.error(`Unknown broker platform: ${platformId}`);
   961→   961→   961→    return null;
   962→   962→   962→  }
   963→   963→   963→
   964→   964→   964→  // Create appropriate adapter based on auth type
   965→   965→   965→  let adapter: BrokerAdapter;
   966→   966→   966→
   967→   967→   967→  switch (platform.authType) {
   968→   968→   968→    case "oauth":
   969→   969→   969→      adapter = new OAuthBrokerAdapter(platform);
   970→   970→   970→      break;
   971→   971→   971→    case "api_key":
   972→   972→   972→      adapter = new ApiKeyBrokerAdapter(platform);
   973→   973→   973→      break;
   974→   974→   974→    case "web_session":
   975→   975→   975→      adapter = new WebSessionBrokerAdapter(platform);
   976→   976→   976→      break;
   977→   977→   977→    case "manual":
   978→   978→   978→      adapter = new ManualBrokerAdapter();
   979→   979→   979→      break;
   980→   980→   980→    default:
   981→   981→   981→      console.error(`Unsupported auth type: ${platform.authType}`);
   982→   982→   982→      return null;
   983→   983→   983→  }
   984→   984→   984→
   985→   985→   985→  // Override demo mode if specified
   986→   986→   986→  if (demoMode !== undefined) {
   987→   987→   987→    (adapter as any)._demoMode = demoMode;
   988→   988→   988→  }
   989→   989→   989→
   990→   990→   990→  adapterCache.set(cacheKey, adapter);
   991→   991→   991→  return adapter;
   992→   992→   992→}
