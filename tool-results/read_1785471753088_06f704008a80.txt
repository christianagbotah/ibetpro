   127→// ==================== RETRY LOGIC ====================
   128→
   129→interface RetryConfig {
   130→  maxRetries: number;
   131→  baseDelayMs: number;
   132→  maxDelayMs: number;
   133→  retryableErrors: string[];
   134→}
   135→
   136→const DEFAULT_RETRY_CONFIG: RetryConfig = {
   137→  maxRetries: 3,
   138→  baseDelayMs: 1000,
   139→  maxDelayMs: 10000,
   140→  retryableErrors: [
   141→    "ETIMEDOUT",
   142→    "ECONNRESET",
   143→    "ECONNREFUSED",
   144→    "rate_limit",
   145→    "server_error",
   146→    "session_expired",
   147→    "temporarily_unavailable",
   148→  ],
   149→};
   150→
   151→async function withRetry<T>(
   152→  fn: () => Promise<T>,
   153→  retryConfig: Partial<RetryConfig> = {}
   154→): Promise<T> {
   155→  const cfg = { ...DEFAULT_RETRY_CONFIG, ...retryConfig };
   156→  let lastError: Error | null = null;
   157→
   158→  for (let attempt = 0; attempt <= cfg.maxRetries; attempt++) {
   159→    try {
   160→      return await fn();
   161→    } catch (error) {
   162→      lastError = error instanceof Error ? error : new Error(String(error));
   163→
   164→      // Check if error is retryable
   165→      const isRetryable = cfg.retryableErrors.some(
   166→        (err) => lastError!.message.toLowerCase().includes(err.toLowerCase())
   167→      );
   168→
   169→      if (!isRetryable || attempt === cfg.maxRetries) {
   170→        throw lastError;
   171→      }
   172→
   173→      // Exponential backoff with jitter
   174→      const delay = Math.min(
   175→        cfg.baseDelayMs * Math.pow(2, attempt) + Math.random() * 1000,
   176→        cfg.maxDelayMs
   177→      );
   178→
   179→      await new Promise((resolve) => setTimeout(resolve, delay));
   180→    }
   181→  }
   182→
   183→  throw lastError;
   184→}

   186→// ==================== IDEMPOTENCY ====================
   187→
   188→// In-memory idempotency store (use Redis in production)
   189→const idempotencyStore = new Map<string, { result: unknown; timestamp: number }>();
   190→const IDEMPOTENCY_TTL = 3600000; // 1 hour
   191→
   192→function checkIdempotency<T>(key: string): T | null {
   193→  const entry = idempotencyStore.get(key);
   194→  if (!entry) return null;
   195→
   196→  if (Date.now() - entry.timestamp > IDEMPOTENCY_TTL) {
   197→    idempotencyStore.delete(key);
   198→    return null;
   199→  }
   200→
   201→  return entry.result as T;
   202→}
   203→
   204→function setIdempotencyResult(key: string, result: unknown): void {
   205→  idempotencyStore.set(key, { result, timestamp: Date.now() });
   206→
   207→  // Clean up old entries periodically
   208→  if (idempotencyStore.size > 10000) {
   209→    const now = Date.now();
   210→    for (const [k, v] of idempotencyStore) {
   211→      if (now - v.timestamp > IDEMPOTENCY_TTL) {
   212→        idempotencyStore.delete(k);
   213→      }
   214→    }
   215→  }
   216→}

   218→// ==================== BASE ADAPTER ====================
   219→
   220→abstract class BaseBrokerAdapter implements BrokerAdapter {
   221→  abstract platformId: string;
   222→  abstract platform: BrokerPlatform;
   223→
   224→  abstract authenticate(credentials: BrokerCredentials): Promise<BrokerAuthResult>;
   225→  abstract refreshSession(refreshToken: string): Promise<BrokerAuthResult>;
   226→  abstract getBalance(accessToken: string, regionCode?: string): Promise<BrokerBalance>;
   227→  abstract getProfile(accessToken: string): Promise<BrokerUserProfile>;
   228→  abstract placeBet(accessToken: string, bet: BrokerBetRequest): Promise<BrokerBetResult>;
   229→  abstract getBetStatus(accessToken: string, brokerBetId: string): Promise<BrokerBetStatus>;
   230→  abstract cashout(accessToken: string, brokerBetId: string, type: "full" | "partial", partialPercent?: number): Promise<BrokerCashoutResult>;
   231→  abstract transferToWallet(accessToken: string, amount: number, walletAddress: string, reference: string): Promise<BrokerTransferResult>;
   232→  abstract getAvailableMarkets(accessToken: string, matchId: string): Promise<BrokerMarket[]>;
   233→  abstract getLiveEvents(accessToken: string): Promise<BrokerLiveEvent[]>;
   234→
   235→  validateSession(sessionToken: string, sessionExpiry: Date): BrokerSession {
   236→    return validateBrokerSession(sessionToken, sessionExpiry);
   237→  }
   238→
   239→  // Common helper: make authenticated API call with retry
   240→  protected async apiCall<T>(
   241→    accessToken: string,
   242→    method: string,
   243→    path: string,
   244→    body?: unknown,
   245→    options?: { idempotencyKey?: string }
   246→  ): Promise<T> {
   247→    const url = `${this.platform.baseUrl}${path}`;
   248→
   249→    // Check idempotency
   250→    if (options?.idempotencyKey) {
   251→      const cached = checkIdempotency<T>(options.idempotencyKey);
   252→      if (cached) return cached;
   253→    }
   254→
   255→    const result = await withRetry(async () => {
   256→      const response = await fetch(url, {
   257→        method,
   258→        headers: {
   259→          "Authorization": `Bearer ${accessToken}`,
   260→          "Content-Type": "application/json",
   261→          "X-Idempotency-Key": options?.idempotencyKey || "",
   262→          "X-Platform": this.platformId,
   263→          "X-App-Version": "1.0.0",
   264→        },
   265→        body: body ? JSON.stringify(body) : undefined,
   266→      });
   267→
   268→      if (!response.ok) {
   269→        const errorData = await response.json().catch(() => ({}));
   270→        const errorCode = errorData.code || errorData.error || `HTTP_${response.status}`;
   271→        const errorMessage = errorData.message || errorData.error || `API error: ${response.status}`;
   272→
   273→        // Map common error codes
   274→        if (response.status === 401) {
   275→          throw new Error(`session_expired: ${errorMessage}`);
   276→        }
   277→        if (response.status === 429) {
   278→          throw new Error(`rate_limit: ${errorMessage}`);
   279→        }
   280→        if (response.status >= 500) {
   281→          throw new Error(`server_error: ${errorMessage}`);
   282→        }
   283→
   284→        throw new Error(`${errorCode}: ${errorMessage}`);
   285→      }
   286→
   287→      return response.json() as Promise<T>;
   288→    });
   289→
   290→    // Store idempotency result
   291→    if (options?.idempotencyKey) {
   292→      setIdempotencyResult(options.idempotencyKey, result);
   293→    }
   294→
   295→    return result;
   296→  }
   297→}

   301→class OAuthBrokerAdapter extends BaseBrokerAdapter {
   302→  platformId: string;
   303→  platform: BrokerPlatform;
   304→
   305→  constructor(platform: BrokerPlatform) {
   306→    super();
   307→    this.platformId = platform.id;

   310→
   311→  async authenticate(credentials: BrokerCredentials): Promise<BrokerAuthResult> {
   312→    if (!credentials.token) {
   313→      return { success: false, error: "OAuth authorization code required" };
   314→    }
   315→
   316→    try {
   317→      // In production, exchange authorization code for access token
   318→      // POST /oauth/token with client_id, client_secret, code, redirect_uri
   319→      const result = await withRetry(async () => {
   320→        const response = await fetch(`${this.platform.baseUrl}/oauth/token`, {
   321→          method: "POST",
   322→          headers: { "Content-Type": "application/json" },
   323→          body: JSON.stringify({
   324→            grant_type: "authorization_code",
   325→            code: credentials.token,
   326→            client_id: process.env[`${this.platformId.toUpperCase()}_CLIENT_ID`],
   327→            client_secret: process.env[`${this.platformId.toUpperCase()}_CLIENT_SECRET`],
   328→            redirect_uri: `${config.platform.url}/api/broker/callback/${this.platformId}`,
   329→          }),
   330→        });
   331→
   332→        if (!response.ok) {
   333→          throw new Error(`OAuth token exchange failed: ${response.status}`);
   334→        }
   335→
   336→        return response.json();
   337→      });
   338→
   339→      return {
   340→        success: true,
   341→        accessToken: result.access_token,
   342→        refreshToken: result.refresh_token,
   343→        sessionExpiry: new Date(Date.now() + (result.expires_in || 3600) * 1000),
   344→        brokerUserId: result.user_id,
   345→      };
   346→    } catch (error) {
   347→      return {
   348→        success: false,
   349→        error: error instanceof Error ? error.message : "OAuth authentication failed",
   350→      };
   351→    }
   352→  }
   353→
   354→  async refreshSession(refreshToken: string): Promise<BrokerAuthResult> {
   355→    try {
   356→      const result = await withRetry(async () => {
   357→        const response = await fetch(`${this.platform.baseUrl}/oauth/token`, {
   358→          method: "POST",
   359→          headers: { "Content-Type": "application/json" },
   360→          body: JSON.stringify({
   361→            grant_type: "refresh_token",
   362→            refresh_token: refreshToken,
   363→            client_id: process.env[`${this.platformId.toUpperCase()}_CLIENT_ID`],
   364→            client_secret: process.env[`${this.platformId.toUpperCase()}_CLIENT_SECRET`],
   365→          }),
   366→        });
   367→
   368→        if (!response.ok) {
   369→          throw new Error(`Token refresh failed: ${response.status}`);
   370→        }
   371→
   372→        return response.json();
   373→      });
   374→
   375→      return {
   376→        success: true,
   377→        accessToken: result.access_token,
   378→        refreshToken: result.refresh_token || refreshToken,
   379→        sessionExpiry: new Date(Date.now() + (result.expires_in || 3600) * 1000),
   380→      };
   381→    } catch (error) {
   382→      return {
   383→        success: false,
   384→        error: error instanceof Error ? error.message : "Session refresh failed",
   385→      };
   386→    }

   620→  async authenticate(credentials: BrokerCredentials): Promise<BrokerAuthResult> {
   621→    if (!credentials.username || !credentials.password) {
   622→      return { success: false, error: "Username and password required" };
   623→    }
   624→
   625→    try {
   626→      const result = await withRetry(async () => {
   627→        const response = await fetch(`${this.platform.baseUrl}/api/v1/auth/login`, {
   628→          method: "POST",
   629→          headers: { "Content-Type": "application/json" },
   630→          body: JSON.stringify({
   631→            username: credentials.username,
   632→            password: credentials.password,
   633→            otp: credentials.otp,
   634→          }),
   635→        });
   636→
   637→        if (!response.ok) {
   638→          const data = await response.json().catch(() => ({}));
   639→          throw new Error(data.message || `Login failed: ${response.status}`);
   640→        }
   641→
   642→        return response.json();
   643→      });
   644→
   645→      return {
   646→        success: true,
   647→        accessToken: result.access_token || result.token,
   648→        sessionToken: result.session_token || result.access_token,
   649→        refreshToken: result.refresh_token,
   650→        sessionExpiry: new Date(Date.now() + (result.expires_in || 7200) * 1000),
   651→        brokerUserId: result.user_id || credentials.username,
   652→      };
   653→    } catch (error) {
   654→      return {
   655→        success: false,
   656→        error: error instanceof Error ? error.message : "Authentication failed",
   657→      };
   658→    }
   659→  }

   667→          body: JSON.stringify({ refresh_token: refreshToken }),
   668→        });
   669→
   670→        if (!response.ok) {
   671→          throw new Error(`Session refresh failed: ${response.status}`);
   672→        }

   675→      });
   676→
   677→      return {
   678→        success: true,
   679→        accessToken: result.access_token || result.token,

   846→// ==================== ADAPTER FACTORY ====================
   847→
   848→const adapterCache = new Map<string, BrokerAdapter>();
   849→
   850→/**
   851→ * Get or create a broker adapter for the given platform
   852→ */
   853→export function getBrokerAdapter(platformId: string): BrokerAdapter | null {
   854→  // Check cache
   855→  if (adapterCache.has(platformId)) {
   856→    return adapterCache.get(platformId)!;
   857→  }
   858→
   859→  // Manual adapter
   860→  if (platformId === "manual") {
   861→    const adapter = new ManualBrokerAdapter();
   862→    adapterCache.set(platformId, adapter);
   863→    return adapter;
   864→  }
   865→
   866→  // Get platform config
   867→  const platform = getBrokerPlatform(platformId);
   868→  if (!platform) {
   869→    console.error(`Unknown broker platform: ${platformId}`);
   870→    return null;
   871→  }
   872→
   873→  // Create appropriate adapter based on auth type
   874→  let adapter: BrokerAdapter;
   875→
   876→  switch (platform.authType) {
   877→    case "oauth":
   878→      adapter = new OAuthBrokerAdapter(platform);
   879→      break;
   880→    case "api_key":
   881→      adapter = new ApiKeyBrokerAdapter(platform);
   882→      break;
   883→    case "web_session":
   884→      adapter = new WebSessionBrokerAdapter(platform);
   885→      break;
   886→    case "manual":
   887→      adapter = new ManualBrokerAdapter();
   888→      break;
   889→    default:
   890→      console.error(`Unsupported auth type: ${platform.authType}`);
   891→      return null;
   892→  }
   893→
   894→  adapterCache.set(platformId, adapter);
   895→  return adapter;
   896→}
   897→
   898→/**
   899→ * Get all available broker adapters
   900→ */
   901→export function getAvailableAdapters(): BrokerAdapter[] {
   902→  const adapters: BrokerAdapter[] = [new ManualBrokerAdapter()];
   903→
   904→  // Dynamically create adapters for all registered platforms
   905→  const platforms = [
   906→    ...new Map(
   907→      // Import from regions to avoid circular dependency
   908→      require("./regions").BROKER_PLATFORMS as Array<{ id: string }>
   909→    ).values(),
   910→  ];
   911→
   912→  for (const platform of platforms) {
   913→    const adapter = getBrokerAdapter(platform.id);
   914→    if (adapter) {
   915→      adapters.push(adapter);
   916→    }
   917→  }
   918→
   919→  return adapters;
   920→}

   924→/**
   925→ * Place a bet on a broker with full safety checks
   926→ */
   927→export async function safePlaceBet(
   928→  platformId: string,
   929→  accessToken: string,
   930→  bet: BrokerBetRequest,
   931→  maxStake: number
   932→): Promise<BrokerBetResult> {
   933→  // Validate stake limits
   934→  if (bet.stake <= 0) {
   935→    return { success: false, error: "Stake must be positive" };
   936→  }
   937→  if (bet.stake > maxStake) {
   938→    return { success: false, error: `Stake exceeds maximum allowed (${maxStake})` };
   939→  }
   940→
   941→  // Validate odds
   942→  if (bet.odds < 1.01) {
   943→    return { success: false, error: "Invalid odds" };
   944→  }
   945→
   946→  const adapter = getBrokerAdapter(platformId);
   947→  if (!adapter) {
   948→    return { success: false, error: `No adapter for platform: ${platformId}` };
   949→  }
   950→
   951→  // Generate idempotency key
   952→  const idempotencyKey = `${platformId}_${bet.matchId}_${bet.selection}_${bet.stake}_${Date.now()}`;
   953→
   954→  return adapter.placeBet(accessToken, {
   955→    ...bet,
   956→    idempotencyKey,
   957→  });
   958→}
