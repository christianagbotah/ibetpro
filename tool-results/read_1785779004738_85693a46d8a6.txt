    56→function expandSportKeys(sports: string[]): string[] {
    57→  const expanded = new Set<string>();
    58→  for (const sport of sports) {
    59→    const lower = sport.toLowerCase().trim();
    60→    if (SPORT_CATEGORY_MAP[lower]) {
    61→      for (const key of SPORT_CATEGORY_MAP[lower]) {
    62→        expanded.add(key);
    63→      }
    64→    } else {
    65→      // Keep the original key as-is (might be a specific key like "soccer_epl")
    66→      expanded.add(lower);
    67→    }
    68→  }
    69→  return Array.from(expanded);
    70→}

    98→// ==================== SINGLETON ENGINE ====================
    99→
   100→/**
   101→ * BotEngine is a singleton that manages all running bot instances.
   102→ * Each user has their own timer that runs scan cycles on a configurable interval.
   103→ *
   104→ * The engine is designed to:
   105→ * - Run independently of the frontend (no polling required)
   106→ * - Auto-recover from errors (continues running after failed scans)
   107→ * - Auto-stop on risk limits, schedule violations, or no allocation
   108→ * - Persist state in the database (BotSession) for recovery after restarts
   109→ * - Support graceful shutdown
   110→ */
   111→class BotEngine {
   112→  private static instance: BotEngine | null = null;
   113→  private timers: Map<string, NodeJS.Timeout> = new Map();
   114→  private runningUsers: Map<string, BotEngineStats> = new Map();
   115→  private isShuttingDown = false;
   116→
   117→  private constructor() {
   118→    // Register graceful shutdown handlers
   119→    process.on("SIGTERM", () => this.shutdown());
   120→    process.on("SIGINT", () => this.shutdown());
   121→  }
   122→
   123→  static getInstance(): BotEngine {
   124→    if (!BotEngine.instance) {
   125→      BotEngine.instance = new BotEngine();
   126→    }
   127→    return BotEngine.instance;
   128→  }

   130→  // ==================== LIFECYCLE ====================
   131→
   132→  /**
   133→   * Start the bot for a specific user. Creates a timer that runs scan cycles
   134→   * on the configured interval (default 30 seconds).
   135→   */
   136→  async start(userId: string, scanIntervalSec: number = 30): Promise<{ success: boolean; message: string }> {
   137→    // Don't start if already running
   138→    if (this.timers.has(userId)) {
   139→      return { success: false, message: "Bot is already running for this user" };
   140→    }
   141→
   142→    // Validate prerequisites
   143→    const validation = await this.validatePrerequisites(userId);
   144→    if (!validation.valid) {
   145→      return { success: false, message: validation.error! };
   146→    }
   147→
   148→    // Initialize stats tracking
   149→    const stats: BotEngineStats = {
   150→      userId,
   151→      status: "running",
   152→      scanIntervalSec,
   153→      totalScans: 0,
   154→      totalBetsPlaced: 0,
   155→      totalStakeUsed: 0,
   156→      totalProfit: 0,
   157→      lastScanAt: null,
   158→      lastBetAt: null,
   159→      startedAt: new Date(),
   160→      errorCount: 0,
   161→      lastError: null,
   162→    };
   163→    this.runningUsers.set(userId, stats);
   164→
   165→    // Run the first scan immediately
   166→    try {
   167→      const result = await this.runScanCycle(userId);
   168→      stats.totalScans++;
   169→      stats.totalBetsPlaced += result.betsPlaced;
   170→      stats.totalStakeUsed += result.totalStake;
   171→      stats.lastScanAt = new Date();
   172→      if (result.betsPlaced > 0 || result.tipsGenerated > 0) {
   173→        stats.lastBetAt = new Date();
   174→      }
   175→      if (result.error) {
   176→        stats.errorCount++;
   177→        stats.lastError = result.error;
   178→      }
   179→    } catch (error) {
   180→      console.error(`[BotEngine] First scan failed for user ${userId}:`, error);
   181→      stats.errorCount++;
   182→      stats.lastError = error instanceof Error ? error.message : "Unknown error";
   183→    }
   184→
   185→    // Set up the recurring timer
   186→    const timer = setInterval(async () => {
   187→      if (this.isShuttingDown) return;
   188→
   189→      try {
   190→        const currentStats = this.runningUsers.get(userId);
   191→        if (!currentStats || currentStats.status !== "running") {
   192→          // Bot was stopped externally, clean up
   193→          this.stop(userId, "auto_stopped");
   194→          return;
   195→        }
   196→
   197→        // Check if the session is still running in the DB
   198→        const session = await prisma.botSession.findUnique({ where: { userId } });
   199→        if (!session || session.status !== "running") {
   200→          this.stop(userId, "session_expired");
   201→          return;
   202→        }
   203→
   204→        const result = await this.runScanCycle(userId);
   205→
   206→        currentStats.totalScans++;
   207→        currentStats.totalBetsPlaced += result.betsPlaced;
   208→        currentStats.totalStakeUsed += result.totalStake;
   209→        currentStats.lastScanAt = new Date();
   210→        currentStats.lastError = null; // Clear error on success
   211→
   212→        if (result.betsPlaced > 0) {
   213→          currentStats.lastBetAt = new Date();
   214→        }
   215→
   216→        if (result.error) {
   217→          currentStats.errorCount++;
   218→          currentStats.lastError = result.error;
   219→        }
   220→
   221→        // Update session stats in DB
   222→        await prisma.botSession.update({
   223→          where: { userId },
   224→          data: {
   225→            totalScans: { increment: 1 },
   226→            totalBetsPlaced: { increment: result.betsPlaced },
   227→            totalStakeUsed: { increment: result.totalStake },
   228→            lastScanAt: new Date(),
   229→            lastBetAt: result.betsPlaced > 0 ? new Date() : undefined,
   230→          },
   231→        });
   232→
   233→        // Log scan activity periodically (every 10 scans)
   234→        if (currentStats.totalScans % 10 === 0) {
   235→          await prisma.botLog.create({
   236→            data: {
   237→              userId,
   238→              action: "bot_scan",
   239→              reasoning: `Auto-scan #${currentStats.totalScans}: ${result.matches} matches found, ${result.betsPlaced} bets placed, ${result.skipped} skipped`,
   240→              details: JSON.stringify({
   241→                scanNumber: currentStats.totalScans,
   242→                betsPlaced: result.betsPlaced,
   243→                totalStake: result.totalStake,
   244→                matches: result.matches,
   245→                skipped: result.skipped,
   246→              }),
   247→            },
   248→          });
   249→        }
   250→
   251→        // Auto-settle finished matches every 3rd scan
   252→        if (currentStats.totalScans % 3 === 0) {
   253→          try {
   254→            await this.autoSettle(userId);
   255→          } catch (settleError) {
   256→            console.error(`[BotEngine] Auto-settle error for user ${userId}:`, settleError);
   257→          }
   258→        }
   259→
   260→        // Auto-cashout evaluation every 5th scan
   261→        if (currentStats.totalScans % 5 === 0) {
   262→          try {
   263→            await this.autoCashoutEvaluation(userId);
   264→          } catch (cashoutError) {
   265→            console.error(`[BotEngine] Auto-cashout error for user ${userId}:`, cashoutError);
   266→          }
   267→        }
   268→      } catch (error) {
   269→        console.error(`[BotEngine] Scan cycle error for user ${userId}:`, error);
   270→        const currentStats = this.runningUsers.get(userId);
   271→        if (currentStats) {
   272→          currentStats.errorCount++;
   273→          currentStats.lastError = error instanceof Error ? error.message : "Unknown error";
   274→
   275→          // Auto-stop after too many consecutive errors
   276→          if (currentStats.errorCount >= 10) {
   277→            console.error(`[BotEngine] Too many errors for user ${userId}, stopping bot`);
   278→            await this.stop(userId, "too_many_errors");
   279→          }
   280→        }
   281→      }
   282→    }, scanIntervalSec * 1000);
   283→
   284→    // Prevent Node.js from keeping the process alive just for this timer
   285→    if (timer.unref) {
   286→      timer.unref();
   287→    }
   288→
   289→    this.timers.set(userId, timer);
   290→
   291→    console.log(`[BotEngine] Bot started for user ${userId} with interval ${scanIntervalSec}s`);
   292→
   293→    // Log bot start
   294→    await prisma.botLog.create({
   295→      data: {
   296→        userId,
   297→        action: "bot_started",
   298→        reasoning: `Bot engine started in background mode. Scanning every ${scanIntervalSec}s.`,
   299→        details: JSON.stringify({
   300→          mode: "background",
   301→          scanIntervalSec,
   302→          broker: validation.connectedAccount?.platform,
   303→          allocation: validation.connectedAccount?.allocatedAmount,
   304→        }),
   305→      },
   306→    });
   307→
   308→    return {
   309→      success: true,
   310→      message: stats.tipsGenerated > 0
   311→        ? `Advisor started! Generated ${stats.tipsGenerated} tip(s) in first scan. Running in background.`
   312→        : stats.betsPlaced > 0
   313→        ? `Bot started! Placed ${stats.betsPlaced} bet(s) in first scan. Running in background.`
   314→        : "Bot started! Scanning for matches in background. Will generate tips automatically.",
   315→    };
   316→  }

   318→  /**
   319→   * Stop the bot for a specific user. Clears the timer and updates the session.
   320→   */
   321→  async stop(userId: string, reason: string = "user_stopped"): Promise<void> {
   322→    const timer = this.timers.get(userId);
   323→    if (timer) {
   324→      clearInterval(timer);
   325→      this.timers.delete(userId);
   326→    }
   327→
   328→    const stats = this.runningUsers.get(userId);
   329→    if (stats) {
   330→      stats.status = "stopped";
   331→      this.runningUsers.delete(userId);
   332→    }
   333→
   334→    // Update the database session
   335→    try {
   336→      await prisma.botSession.upsert({
   337→        where: { userId },
   338→        update: {
   339→          status: "stopped",
   340→          stoppedAt: new Date(),
   341→          stopReason: reason,
   342→        },
   343→        create: {
   344→          userId,
   345→          status: "stopped",
   346→          stoppedAt: new Date(),
   347→          stopReason: reason,
   348→        },
   349→      });
   350→
   351→      // Log bot stop
   352→      await prisma.botLog.create({
   353→        data: {
   354→          userId,
   355→          action: "bot_stopped",
   356→          reasoning: `Bot engine stopped. Reason: ${reason}`,
   357→          details: JSON.stringify({
   358→            reason,
   359→            totalScans: stats?.totalScans || 0,
   360→            totalBetsPlaced: stats?.totalBetsPlaced || 0,
   361→            totalStakeUsed: stats?.totalStakeUsed || 0,
   362→            totalProfit: stats?.totalProfit || 0,
   363→            runDuration: stats?.startedAt
   364→              ? Math.round((Date.now() - stats.startedAt.getTime()) / 60000)
   365→              : 0,
   366→            errorCount: stats?.errorCount || 0,
   367→          }),
   368→        },
   369→      });
   370→    } catch (error) {
   371→      console.error(`[BotEngine] Error stopping bot for user ${userId}:`, error);
   372→    }
   373→
   374→    console.log(`[BotEngine] Bot stopped for user ${userId}. Reason: ${reason}`);
   375→  }

   377→  /**
   378→   * Gracefully shut down all running bots.
   379→   */
   380→  async shutdown(): Promise<void> {
   381→    if (this.isShuttingDown) return;
   382→    this.isShuttingDown = true;
   383→
   384→    console.log(`[BotEngine] Shutting down ${this.timers.size} running bots...`);
   385→
   386→    for (const [userId] of this.timers) {
   387→      await this.stop(userId, "server_shutdown");
   388→    }
   389→
   390→    this.timers.clear();
   391→    this.runningUsers.clear();
   392→  }

   458→  // ==================== SCAN CYCLE ====================
   459→
   460→  /**
   461→   * Run a single scan cycle - analyze matches and generate tips or place bets.
   462→   * In advisor mode: AI finds value bets → creates Tip → sends Telegram alert.
   463→   * In auto mode: AI finds value bets → places bet via broker → creates Tip.
   464→   */
   465→  private async runScanCycle(userId: string): Promise<ScanResult> {
   466→    const result: ScanResult = { betsPlaced: 0, tipsGenerated: 0, totalStake: 0, matches: 0, skipped: 0 };
   467→
   468→    try {
   469→      // Refresh match data from external APIs (throttled to avoid rate limits)
   470→      try {
   471→        await syncMatchData(false);
   472→      } catch (syncErr) {
   473→        console.warn("[BotEngine] Match sync failed, continuing with existing data:", syncErr);
   474→      }
   475→
   476→      // Get user settings
   477→      const settings = await prisma.userSettings.findUnique({ where: { userId } });
   478→      if (!settings) {
   479→        result.error = "Settings not found";
   480→        return result;
   481→      }
   482→
   483→      const isAdvisorMode = settings.botMode === "advisor";
   484→
   485→      if (!settings.autoBettingEnabled && !isAdvisorMode) {
   486→        result.error = "Auto-betting is disabled";
   487→        await this.stop(userId, "auto_betting_disabled");
   488→        return result;
   489→      }
   490→
   491→      // Check schedule (using user's timezone setting)
   492→      if (!isWithinBetSchedule(settings.betScheduleStart, settings.betScheduleEnd, settings.timezone)) {
   493→        return result;
   494→      }
   495→
   496→      // Check risk limits
   497→      const user = await prisma.user.findUnique({ where: { id: userId } });
   498→      const riskCheck = checkRiskLimits(user?.dailyPnl || 0, user?.weeklyPnl || 0, {
   499→        stopLossDaily: settings.stopLossDaily,
   500→        stopLossWeekly: settings.stopLossWeekly,
   501→        profitTargetDaily: settings.profitTargetDaily,
   502→        profitTargetWeekly: settings.profitTargetWeekly,
   503→      });
   504→
   505→      if (!riskCheck.canBet) {
   506→        const reason = (user?.dailyPnl ?? 0) <= -settings.stopLossDaily ? "stop_loss" : "profit_target";
   507→        await this.stop(userId, reason);
   508→        await prisma.botLog.create({
   509→          data: {
   510→            userId,
   511→            action: reason === "stop_loss" ? "stop_loss_hit" : "profit_target_hit",
   512→            reasoning: riskCheck.reason,
   513→          },
   514→        });
   515→        return result;
   516→      }
   517→
   518→      // ---- Advisor mode: no broker required ----
   519→      // ---- Auto mode: need connected broker ----
   520→      let bettingAccount: Awaited<ReturnType<typeof prisma.bettingAccount.findFirst>> = null;
   521→      let activeAllocation: Awaited<ReturnType<typeof prisma.allocation.findFirst>> = null;
   522→      let remainingDailyLimit = settings.dailyBetLimit;
   523→      let existingBetMatchIds: string[] = [];
   524→
   525→      if (!isAdvisorMode) {
   526→        bettingAccount = await prisma.bettingAccount.findFirst({
   527→          where: { userId, isConnected: true },
   528→          orderBy: { allocatedAmount: "desc" },
   529→        });
   530→
   531→        if (!bettingAccount || bettingAccount.allocatedAmount <= 0) {
   532→          await this.stop(userId, "no_allocation");
   533→          await prisma.botLog.create({
   534→            data: {
   535→              userId,
   536→              action: "bot_scan",
   537→              reasoning: "No connected broker or allocation available. Stopping bot.",
   538→            },
   539→          });
   540→          return result;
   541→        }
   542→
   543→        // Check daily limit
   544→        const todayStart = new Date();
   545→        todayStart.setHours(0, 0, 0, 0);
   546→        const todayBets = await prisma.bet.findMany({
   547→          where: {
   548→            userId,
   549→            placedAt: { gte: todayStart },
   550→            status: { in: ["pending", "won", "lost", "cashed_out", "partial_cashout"] },
   551→          },
   552→        });
   553→        const dailyStake = todayBets.reduce((sum, b) => sum + b.stake, 0);
   554→        existingBetMatchIds = todayBets.map((b) => b.matchId);
   555→
   556→        if (dailyStake >= settings.dailyBetLimit) {
   557→          return result;
   558→        }
   559→
   560→        remainingDailyLimit = Math.min(
   561→          settings.dailyBetLimit - dailyStake,
   562→          bettingAccount.allocatedAmount
   563→        );
   564→
   565→        if (remainingDailyLimit < 5) {
   566→          return result;
   567→        }
   568→
   569→        activeAllocation = await prisma.allocation.findFirst({

   909→
   910→              // Send Telegram + in-app notification

   940→        }
   941→      }
   942→
   943→      // Log scan summary

   946→          data: {

   950→          },
   951→        });
   952→      }
   953→
   954→    } catch (error) {
   955→      console.error(`[BotEngine] Scan cycle error for user ${userId}:`, error);
   956→      result.error = error instanceof Error ? error.message : "Unknown error";
   957→    }
   958→
   959→    return result;
   960→  }

   964→  private async validatePrerequisites(userId: string): Promise<{
   965→    valid: boolean;
   966→    error?: string;
   967→    connectedAccount?: { platform: string; allocatedAmount: number };
   968→  }> {
   969→    const settings = await prisma.userSettings.findUnique({ where: { userId } });
   970→    if (!settings) {
   971→      return { valid: false, error: "Settings not found. Please configure your settings first." };
   972→    }
   973→
   974→    const isAdvisorMode = settings.botMode === "advisor";
   975→
   976→    if (!settings.autoBettingEnabled && !isAdvisorMode) {
   977→      return { valid: false, error: "Auto-betting is disabled. Enable it first." };
   978→    }
   979→
   980→    // In advisor mode, no broker is required — just need settings
   981→    if (isAdvisorMode) {
   982→      return { valid: true };
   983→    }
   984→
   985→    // Auto mode requires a connected broker
   986→    const connectedAccount = await prisma.bettingAccount.findFirst({
   987→      where: { userId, isConnected: true },
   988→    });
   989→
   990→    if (!connectedAccount) {
   991→      return { valid: false, error: "No connected broker account. Connect a broker and set allocation first." };
   992→    }
   993→
   994→    if (connectedAccount.allocatedAmount <= 0) {
   995→      return { valid: false, error: "No allocation set. Allocate funds from your broker account first." };
   996→    }
   997→
   998→    return {
   999→      valid: true,
  1000→      connectedAccount: {
  1001→        platform: connectedAccount.platform,
  1002→        allocatedAmount: connectedAccount.allocatedAmount,
  1003→      },
  1004→    };
  1005→  }

  1007→  // ==================== AUTO SETTLEMENT ====================
  1008→
  1009→  /**
  1010→   * Automatically settle bets for finished matches.
  1011→   * Called periodically by the engine (every 3rd scan cycle).
  1012→   */
  1013→  private async autoSettle(userId: string): Promise<number> {
  1014→    const user = await prisma.user.findUnique({
  1015→      where: { id: userId },
  1016→      include: { settings: true },
  1017→    });
  1018→
  1019→    if (!user) return 0;
  1020→
  1021→    const commissionRate = user.settings?.commissionRate || 0.10;
  1022→
  1023→    // Find pending bets for finished matches
  1024→    const unsettledBets = await prisma.bet.findMany({
  1025→      where: {
  1026→        userId,
  1027→        status: "pending",
  1028→        match: { status: "finished" },
  1029→      },
  1030→      include: { match: true, bettingAccount: true },
  1031→    });
  1032→
  1033→    if (unsettledBets.length === 0) return 0;
  1034→
  1035→    let settledCount = 0;
  1036→
  1037→    for (const bet of unsettledBets) {
  1038→      if (!bet.match) continue;
  1039→
  1040→      const match = bet.match;
  1041→      const homeScore = match.homeScore ?? 0;
  1042→      const awayScore = match.awayScore ?? 0;
  1043→
  1044→      let betWon = false;
  1045→      let resultReason = "";
  1046→
  1047→      if (bet.betType === "single" || bet.betType === "accumulator_leg") {
  1048→        if (bet.selection === match.homeTeam) {
  1049→          betWon = homeScore > awayScore;
  1050→          resultReason = betWon
  1051→            ? `${match.homeTeam} won ${homeScore}-${awayScore}`
  1052→            : `${match.homeTeam} did not win (${homeScore}-${awayScore})`;
  1053→        } else if (bet.selection === match.awayTeam) {
  1054→          betWon = awayScore > homeScore;
  1055→          resultReason = betWon
  1056→            ? `${match.awayTeam} won ${awayScore}-${homeScore}`
  1057→            : `${match.awayTeam} did not win (${awayScore}-${homeScore})`;
  1058→        } else if (bet.selection === "Draw") {
  1059→          betWon = homeScore === awayScore;
  1060→          resultReason = betWon
  1061→            ? `Match drawn ${homeScore}-${awayScore}`
  1062→            : `Match not drawn (${homeScore}-${awayScore})`;
  1063→        } else if (bet.selection === "Over 2.5") {
  1064→          const totalGoals = homeScore + awayScore;
  1065→          betWon = totalGoals > 2.5;
  1066→          resultReason = betWon
  1067→            ? `Total goals ${totalGoals} > 2.5`
  1068→            : `Total goals ${totalGoals} <= 2.5`;
  1069→        } else if (bet.selection === "Under 2.5") {
  1070→          const totalGoals = homeScore + awayScore;
  1071→          betWon = totalGoals < 2.5;
  1072→          resultReason = betWon
  1073→            ? `Total goals ${totalGoals} < 2.5`
  1074→            : `Total goals ${totalGoals} >= 2.5`;
  1075→        }
  1076→      }
  1077→
  1078→      const now = new Date();
  1079→
  1080→      if (betWon) {
  1081→        const effectiveStake = bet.partialCashoutAmount ? bet.stake - bet.partialCashoutAmount : bet.stake;
  1082→        const grossProfit = bet.potentialWin - effectiveStake;
  1083→        const commission = grossProfit * commissionRate;
  1084→        const netProfit = grossProfit - commission;
  1085→
  1086→        await prisma.bet.update({
  1087→          where: { id: bet.id },
  1088→          data: {
  1089→            status: "won",
  1090→            profit: Math.round(netProfit * 100) / 100,
  1091→            commission: Math.round(commission * 100) / 100,
  1092→            settledAt: now,
  1093→            settlementReason: "match_finished",
  1094→          },
  1095→        });
  1096→
  1097→        await prisma.user.update({
  1098→          where: { id: userId },
  1099→          data: {
  1100→            balance: { increment: bet.potentialWin },
  1101→            totalProfit: { increment: netProfit },
  1102→            commissionPaid: { increment: commission },
  1103→            dailyPnl: { increment: netProfit },
  1104→            weeklyPnl: { increment: netProfit },
  1105→          },
  1106→        });
  1107→
  1108→        await prisma.transaction.create({
  1109→          data: {
  1110→            userId,
  1111→            type: "bet_won",
  1112→            amount: bet.potentialWin,
  1113→            currency: "USD",
  1114→            status: "completed",
  1115→            description: `Auto-settled: ${match.homeTeam} vs ${match.awayTeam} - ${bet.selection} @ ${bet.odds} (profit: $${netProfit.toFixed(2)})`,
  1116→            betId: bet.id,
  1117→          },
  1118→        });
  1119→
  1120→        if (commission > 0 && bet.bettingAccountId) {
  1121→          await prisma.commissionLedger.create({
  1122→            data: {
  1123→              userId,
  1124→              bettingAccountId: bet.bettingAccountId,
  1125→              betId: bet.id,
  1126→              accumulatorId: bet.accumulatorId,
  1127→              grossProfit,
  1128→              commissionRate,
  1129→              commissionAmount: commission,
  1130→              netProfit,
  1131→              status: "pending",
  1132→            },
  1133→          });
  1134→        }
  1135→
  1136→        await prisma.botLog.create({
  1137→          data: {
  1138→            userId,
  1139→            action: "bet_settled",
  1140→            betId: bet.id,
  1141→            matchId: match.id,
  1142→            accumulatorId: bet.accumulatorId,
  1143→            details: JSON.stringify({ result: "won", profit: netProfit, commission, autoSettled: true }),
  1144→            reasoning: resultReason,
  1145→            profitImpact: netProfit,
  1146→          },
  1147→        });
  1148→      } else {
  1149→        const effectiveStake = bet.partialCashoutAmount ? bet.stake - bet.partialCashoutAmount : bet.stake;
  1150→
  1151→        await prisma.bet.update({
  1152→          where: { id: bet.id },
  1153→          data: {
  1154→            status: "lost",
  1155→            profit: -effectiveStake,
  1156→            settledAt: now,
  1157→            settlementReason: "match_finished",
  1158→          },
  1159→        });
  1160→
  1161→        await prisma.user.update({
  1162→          where: { id: userId },
  1163→          data: {
  1164→            totalLoss: { increment: effectiveStake },
  1165→            dailyPnl: { decrement: effectiveStake },

  1194→  // ==================== AUTO CASHOUT EVALUATION ====================
  1195→
  1196→  /**
  1197→   * Evaluate live bets for cashout opportunities.
  1198→   * Called periodically by the engine (every 5th scan cycle).
  1199→   * Uses the AI engine's cashout recommendation system.
  1200→   */
  1201→  private async autoCashoutEvaluation(userId: string): Promise<number> {
  1202→    const settings = await prisma.userSettings.findUnique({ where: { userId } });
  1203→    if (!settings?.autoCashoutEnabled) return 0;
  1204→
  1205→    // Find live bets that could be cashed out
  1206→    const liveBets = await prisma.bet.findMany({
  1207→      where: {
  1208→        userId,
  1209→        status: "pending",
  1210→        isAutoPlaced: true,
  1211→        match: { status: "live" },
  1212→      },
  1213→      include: { match: true },
  1214→    });
  1215→
  1216→    if (liveBets.length === 0) return 0;
  1217→
  1218→    let cashoutCount = 0;
  1219→
  1220→    for (const bet of liveBets) {
  1221→      if (!bet.match) continue;
  1222→
  1223→      // Simple cashout evaluation based on match state
  1224→      const match = bet.match;
  1225→      const homeScore = match.homeScore ?? 0;
  1226→      const awayScore = match.awayScore ?? 0;
  1227→      const minute = match.minute ?? 0;
  1228→
  1229→      // Determine if the bet is winning
  1230→      let isWinning = false;
  1231→      if (bet.selection === match.homeTeam) {
  1232→        isWinning = homeScore > awayScore;
  1233→      } else if (bet.selection === match.awayTeam) {
  1234→        isWinning = awayScore > homeScore;
  1235→      } else if (bet.selection === "Draw") {
  1236→        isWinning = homeScore === awayScore;
  1237→      } else if (bet.selection === "Over 2.5") {
  1238→        isWinning = (homeScore + awayScore) > 2.5;
  1239→      } else if (bet.selection === "Under 2.5") {
  1240→        isWinning = (homeScore + awayScore) < 2.5;
  1241→      }
  1242→
  1243→      // Cashout logic: if the bet is winning and we're past 70 minutes,
  1244→      // and the cashout threshold is met, execute cashout
  1245→      if (isWinning && minute >= 70) {
  1246→        const cashoutOdds = 1 + (1 / bet.odds) * (minute / 90);
  1247→        const cashoutAmount = Math.round(bet.stake * cashoutOdds * 100) / 100;
  1248→
  1249→        if (cashoutAmount > bet.stake * (1 + settings.cashoutThreshold)) {
  1250→          // Execute partial or full cashout based on settings
  1251→          if (settings.partialCashoutEnabled) {
  1252→            // Partial cashout: take 50% profit, keep the rest riding
  1253→            const partialAmount = Math.round(cashoutAmount * settings.partialCashoutPercent * 100) / 100;
  1254→
  1255→            await prisma.bet.update({
  1256→              where: { id: bet.id },
  1257→              data: {
  1258→                status: "partial_cashout",
  1259→                partialCashoutAmount,
  1260→                partialCashoutPercent: settings.partialCashoutPercent,
  1261→                cashoutOdds,
  1262→                cashedOutAt: new Date(),
  1263→              },
  1264→            });
  1265→
  1266→            await prisma.user.update({
  1267→              where: { id: userId },
  1268→              data: {
  1269→                balance: { increment: partialAmount },
  1270→                dailyPnl: { increment: partialAmount - bet.stake * settings.partialCashoutPercent },
  1271→                weeklyPnl: { increment: partialAmount - bet.stake * settings.partialCashoutPercent },
  1272→              },
  1273→            });
  1274→
  1275→            await prisma.transaction.create({
  1276→              data: {
  1277→                userId,
  1278→                type: "partial_cashout",
  1279→                amount: partialAmount,
  1280→                currency: "USD",
  1281→                status: "completed",
  1282→                description: `Auto partial cashout: ${match.homeTeam} vs ${match.awayTeam} - ${bet.selection} ($${partialAmount.toFixed(2)})`,
  1283→                betId: bet.id,
  1284→              },
  1285→            });
  1286→
  1287→            await prisma.botLog.create({
  1288→              data: {
  1289→                userId,
  1290→                action: "cashout_executed",
  1291→                betId: bet.id,
  1292→                matchId: match.id,
  1293→                details: JSON.stringify({
  1294→                  type: "partial",
  1295→                  amount: partialAmount,
  1296→                  percent: settings.partialCashoutPercent,
  1297→                  minute,
  1298→                  autoCashout: true,
  1299→                }),
  1300→                reasoning: `Auto partial cashout at ${minute}' - bet winning. ${Math.round(settings.partialCashoutPercent * 100)}% cashed out for $${partialAmount.toFixed(2)}`,
  1301→                profitImpact: partialAmount - bet.stake * settings.partialCashoutPercent,
  1302→              },
  1303→            });
  1304→          } else {
  1305→            // Full cashout
  1306→            await prisma.bet.update({
  1307→              where: { id: bet.id },
  1308→              data: {
  1309→                status: "cashed_out",
  1310→                cashoutAmount,
  1311→                cashoutOdds,
  1312→                profit: cashoutAmount - bet.stake,
  1313→                cashedOutAt: new Date(),
  1314→                settlementReason: "cashout",
  1315→              },
  1316→            });
  1317→
  1318→            await prisma.user.update({
  1319→              where: { id: userId },
  1320→              data: {
  1321→                balance: { increment: cashoutAmount },
  1322→                totalProfit: { increment: cashoutAmount - bet.stake },
  1323→                dailyPnl: { increment: cashoutAmount - bet.stake },
  1324→                weeklyPnl: { increment: cashoutAmount - bet.stake },
  1325→              },
  1326→            });
  1327→
  1328→            await prisma.transaction.create({
  1329→              data: {
  1330→                userId,
  1331→                type: "cashout",
  1332→                amount: cashoutAmount,
  1333→                currency: "USD",
  1334→                status: "completed",

  1347→                  type: "full",
  1348→                  amount: cashoutAmount,

  1356→          }
  1357→
  1358→          cashoutCount++;
  1359→        }
  1360→      }
  1361→    }
  1362→
  1363→    if (cashoutCount > 0) {

  1368→  }
