     1→// ============================================================================
     2→// iBetPro Bot Engine - Background Auto-Betting Engine
     3→// Runs continuously on the server using setInterval, independent of frontend
     4→// polling. Each user's bot gets its own timer that scans matches and places
     5→// bets automatically according to their settings.
     6→// ============================================================================
     7→
     8→import { prisma } from "./db";
     9→import { analyzeMatch, shouldAutoBet, checkRiskLimits, isWithinBetSchedule } from "./ai-engine-v2";
    10→import { placeBetOnBroker } from "./broker-integration";
    11→
    12→// ==================== TYPES ====================
    13→
    14→interface ScanResult {
    15→  betsPlaced: number;
    16→  totalStake: number;
    17→  matches: number;
    18→  skipped: number;
    19→  error?: string;
    20→}
    21→
    22→interface BotEngineStats {
    23→  userId: string;
    24→  status: "running" | "stopped" | "paused";
    25→  scanIntervalSec: number;
    26→  totalScans: number;
    27→  totalBetsPlaced: number;
    28→  totalStakeUsed: number;
    29→  totalProfit: number;
    30→  lastScanAt: Date | null;
    31→  lastBetAt: Date | null;
    32→  startedAt: Date | null;
    33→  errorCount: number;
    34→  lastError: string | null;
    35→}
    36→
    37→// ==================== SINGLETON ENGINE ====================
    38→
    39→/**
    40→ * BotEngine is a singleton that manages all running bot instances.
    41→ * Each user has their own timer that runs scan cycles on a configurable interval.
    42→ *
    43→ * The engine is designed to:
    44→ * - Run independently of the frontend (no polling required)
    45→ * - Auto-recover from errors (continues running after failed scans)
    46→ * - Auto-stop on risk limits, schedule violations, or no allocation
    47→ * - Persist state in the database (BotSession) for recovery after restarts
    48→ * - Support graceful shutdown
    49→ */
    50→class BotEngine {
    51→  private static instance: BotEngine | null = null;
    52→  private timers: Map<string, NodeJS.Timeout> = new Map();
    53→  private runningUsers: Map<string, BotEngineStats> = new Map();
    54→  private isShuttingDown = false;
    55→
    56→  private constructor() {
    57→    // Register graceful shutdown handlers
    58→    process.on("SIGTERM", () => this.shutdown());
    59→    process.on("SIGINT", () => this.shutdown());
    60→  }
    61→
    62→  static getInstance(): BotEngine {
    63→    if (!BotEngine.instance) {
    64→      BotEngine.instance = new BotEngine();
    65→    }
    66→    return BotEngine.instance;
    67→  }
    68→
    69→  // ==================== LIFECYCLE ====================
    70→
    71→  /**
    72→   * Start the bot for a specific user. Creates a timer that runs scan cycles
    73→   * on the configured interval (default 30 seconds).
    74→   */
    75→  async start(userId: string, scanIntervalSec: number = 30): Promise<{ success: boolean; message: string }> {
    76→    // Don't start if already running
    77→    if (this.timers.has(userId)) {
    78→      return { success: false, message: "Bot is already running for this user" };
    79→    }
    80→
    81→    // Validate prerequisites
    82→    const validation = await this.validatePrerequisites(userId);
    83→    if (!validation.valid) {
    84→      return { success: false, message: validation.error! };
    85→    }
    86→
    87→    // Initialize stats tracking
    88→    const stats: BotEngineStats = {
    89→      userId,
    90→      status: "running",
    91→      scanIntervalSec,
    92→      totalScans: 0,
    93→      totalBetsPlaced: 0,
    94→      totalStakeUsed: 0,
    95→      totalProfit: 0,
    96→      lastScanAt: null,
    97→      lastBetAt: null,
    98→      startedAt: new Date(),
    99→      errorCount: 0,
   100→      lastError: null,
   101→    };
   102→    this.runningUsers.set(userId, stats);
   103→
   104→    // Run the first scan immediately
   105→    try {
   106→      const result = await this.runScanCycle(userId);
   107→      stats.totalScans++;
   108→      stats.totalBetsPlaced += result.betsPlaced;
   109→      stats.totalStakeUsed += result.totalStake;
   110→      stats.lastScanAt = new Date();
   111→      if (result.betsPlaced > 0) {
   112→        stats.lastBetAt = new Date();
   113→      }
   114→      if (result.error) {
   115→        stats.errorCount++;
   116→        stats.lastError = result.error;
   117→      }
   118→    } catch (error) {
   119→      console.error(`[BotEngine] First scan failed for user ${userId}:`, error);
   120→      stats.errorCount++;
   121→      stats.lastError = error instanceof Error ? error.message : "Unknown error";
   122→    }
   123→
   124→    // Set up the recurring timer
   125→    const timer = setInterval(async () => {
   126→      if (this.isShuttingDown) return;
   127→
   128→      try {
   129→        const currentStats = this.runningUsers.get(userId);
   130→        if (!currentStats || currentStats.status !== "running") {
   131→          // Bot was stopped externally, clean up
   132→          this.stop(userId, "auto_stopped");
   133→          return;
   134→        }
   135→
   136→        // Check if the session is still running in the DB
   137→        const session = await prisma.botSession.findUnique({ where: { userId } });
   138→        if (!session || session.status !== "running") {
   139→          this.stop(userId, "session_expired");
   140→          return;
   141→        }
   142→
   143→        const result = await this.runScanCycle(userId);
   144→
   145→        currentStats.totalScans++;
   146→        currentStats.totalBetsPlaced += result.betsPlaced;
   147→        currentStats.totalStakeUsed += result.totalStake;
   148→        currentStats.lastScanAt = new Date();
   149→        currentStats.lastError = null; // Clear error on success
   150→
   151→        if (result.betsPlaced > 0) {
   152→          currentStats.lastBetAt = new Date();
   153→        }
   154→
   155→        if (result.error) {
   156→          currentStats.errorCount++;
   157→          currentStats.lastError = result.error;
   158→        }
   159→
   160→        // Update session stats in DB
   161→        await prisma.botSession.update({
   162→          where: { userId },
   163→          data: {
   164→            totalScans: { increment: 1 },
   165→            totalBetsPlaced: { increment: result.betsPlaced },
   166→            totalStakeUsed: { increment: result.totalStake },
   167→            lastScanAt: new Date(),
   168→            lastBetAt: result.betsPlaced > 0 ? new Date() : undefined,
   169→          },
   170→        });
   171→
   172→        // Log scan activity periodically (every 10 scans)
   173→        if (currentStats.totalScans % 10 === 0) {
   174→          await prisma.botLog.create({
   175→            data: {
   176→              userId,
   177→              action: "bot_scan",
   178→              reasoning: `Auto-scan #${currentStats.totalScans}: ${result.matches} matches found, ${result.betsPlaced} bets placed, ${result.skipped} skipped`,
   179→              details: JSON.stringify({
   180→                scanNumber: currentStats.totalScans,
   181→                betsPlaced: result.betsPlaced,
   182→                totalStake: result.totalStake,
   183→                matches: result.matches,
   184→                skipped: result.skipped,
   185→              }),
   186→            },
   187→          });
   188→        }
   189→
   190→        // Auto-settle finished matches every 3rd scan
   191→        if (currentStats.totalScans % 3 === 0) {
   192→          try {
   193→            await this.autoSettle(userId);
   194→          } catch (settleError) {
   195→            console.error(`[BotEngine] Auto-settle error for user ${userId}:`, settleError);
   196→          }
   197→        }
   198→
   199→        // Auto-cashout evaluation every 5th scan
   200→        if (currentStats.totalScans % 5 === 0) {
   201→          try {
   202→            await this.autoCashoutEvaluation(userId);
   203→          } catch (cashoutError) {
   204→            console.error(`[BotEngine] Auto-cashout error for user ${userId}:`, cashoutError);
   205→          }
   206→        }
   207→      } catch (error) {
   208→        console.error(`[BotEngine] Scan cycle error for user ${userId}:`, error);
   209→        const currentStats = this.runningUsers.get(userId);
   210→        if (currentStats) {
   211→          currentStats.errorCount++;
   212→          currentStats.lastError = error instanceof Error ? error.message : "Unknown error";
   213→
   214→          // Auto-stop after too many consecutive errors
   215→          if (currentStats.errorCount >= 10) {
   216→            console.error(`[BotEngine] Too many errors for user ${userId}, stopping bot`);
   217→            await this.stop(userId, "too_many_errors");
   218→          }
   219→        }
   220→      }
   221→    }, scanIntervalSec * 1000);
   222→
   223→    // Prevent Node.js from keeping the process alive just for this timer
   224→    if (timer.unref) {
   225→      timer.unref();
   226→    }
   227→
   228→    this.timers.set(userId, timer);
   229→
   230→    console.log(`[BotEngine] Bot started for user ${userId} with interval ${scanIntervalSec}s`);
   231→
   232→    // Log bot start
   233→    await prisma.botLog.create({
   234→      data: {
   235→        userId,
   236→        action: "bot_started",
   237→        reasoning: `Bot engine started in background mode. Scanning every ${scanIntervalSec}s.`,
   238→        details: JSON.stringify({
   239→          mode: "background",
   240→          scanIntervalSec,
   241→          broker: validation.connectedAccount?.platform,
   242→          allocation: validation.connectedAccount?.allocatedAmount,
   243→        }),
   244→      },
   245→    });
   246→
   247→    return {
   248→      success: true,
   249→      message: stats.totalBetsPlaced > 0
   250→        ? `Bot started! Placed ${stats.totalBetsPlaced} bet(s) in first scan. Running in background.`
   251→        : "Bot started! Scanning for matches in background. Will place bets automatically.",
   252→    };
   253→  }
   254→
   255→  /**
   256→   * Stop the bot for a specific user. Clears the timer and updates the session.
   257→   */
   258→  async stop(userId: string, reason: string = "user_stopped"): Promise<void> {
   259→    const timer = this.timers.get(userId);
   260→    if (timer) {
   261→      clearInterval(timer);
   262→      this.timers.delete(userId);
   263→    }
   264→
   265→    const stats = this.runningUsers.get(userId);
   266→    if (stats) {
   267→      stats.status = "stopped";
   268→      this.runningUsers.delete(userId);
   269→    }
   270→
   271→    // Update the database session
   272→    try {
   273→      await prisma.botSession.upsert({
   274→        where: { userId },
   275→        update: {
   276→          status: "stopped",
   277→          stoppedAt: new Date(),
   278→          stopReason: reason,
   279→        },
   280→        create: {
   281→          userId,
   282→          status: "stopped",
   283→          stoppedAt: new Date(),
   284→          stopReason: reason,
   285→        },
   286→      });
   287→
   288→      // Log bot stop
   289→      await prisma.botLog.create({
   290→        data: {
   291→          userId,
   292→          action: "bot_stopped",
   293→          reasoning: `Bot engine stopped. Reason: ${reason}`,
   294→          details: JSON.stringify({
   295→            reason,
   296→            totalScans: stats?.totalScans || 0,
   297→            totalBetsPlaced: stats?.totalBetsPlaced || 0,
   298→            totalStakeUsed: stats?.totalStakeUsed || 0,
   299→            totalProfit: stats?.totalProfit || 0,
   300→            runDuration: stats?.startedAt
   301→              ? Math.round((Date.now() - stats.startedAt.getTime()) / 60000)
   302→              : 0,
   303→            errorCount: stats?.errorCount || 0,
   304→          }),
   305→        },
   306→      });
   307→    } catch (error) {
   308→      console.error(`[BotEngine] Error stopping bot for user ${userId}:`, error);
   309→    }
   310→
   311→    console.log(`[BotEngine] Bot stopped for user ${userId}. Reason: ${reason}`);
   312→  }
   313→
   314→  /**
   315→   * Gracefully shut down all running bots.
   316→   */
   317→  async shutdown(): Promise<void> {
   318→    if (this.isShuttingDown) return;
   319→    this.isShuttingDown = true;
   320→
   321→    console.log(`[BotEngine] Shutting down ${this.timers.size} running bots...`);
   322→
   323→    for (const [userId] of this.timers) {
   324→      await this.stop(userId, "server_shutdown");
   325→    }
   326→
   327→    this.timers.clear();
   328→    this.runningUsers.clear();
   329→  }

   331→  /**
   332→   * Recover bots that were running before a server restart.
   333→   * Checks the database for sessions with status "running" and restarts them.
   334→   */
   335→  async recoverRunningBots(): Promise<number> {
   336→    try {
   337→      const runningSessions = await prisma.botSession.findMany({
   338→        where: { status: "running" },
   339→      });
   340→
   341→      console.log(`[BotEngine] Found ${runningSessions.length} running sessions to recover`);
   342→
   343→      let recovered = 0;
   344→      for (const session of runningSessions) {
   345→        try {
   346→          const result = await this.start(session.userId, session.scanIntervalSec || 30);
   347→          if (result.success) {
   348→            recovered++;
   349→            console.log(`[BotEngine] Recovered bot for user ${session.userId}`);
   350→          } else {
   351→            console.warn(`[BotEngine] Could not recover bot for user ${session.userId}: ${result.message}`);
   352→          }
   353→        } catch (error) {
   354→          console.error(`[BotEngine] Error recovering bot for user ${session.userId}:`, error);
   355→        }
   356→      }
   357→
   358→      return recovered;
   359→    } catch (error) {
   360→      console.error("[BotEngine] Error recovering bots:", error);
   361→      return 0;
   362→    }
   363→  }

   395→  // ==================== SCAN CYCLE ====================
   396→
   397→  /**
   398→   * Run a single scan cycle - analyze matches and place bets.
   399→   * This is the core logic that runs on each interval tick.
   400→   */
   401→  private async runScanCycle(userId: string): Promise<ScanResult> {
   402→    const result: ScanResult = { betsPlaced: 0, totalStake: 0, matches: 0, skipped: 0 };
   403→
   404→    try {
   405→      // Get user settings
   406→      const settings = await prisma.userSettings.findUnique({ where: { userId } });
   407→      if (!settings) {
   408→        result.error = "Settings not found";
   409→        return result;
   410→      }
   411→
   412→      if (!settings.autoBettingEnabled) {
   413→        result.error = "Auto-betting is disabled";
   414→        await this.stop(userId, "auto_betting_disabled");
   415→        return result;
   416→      }
   417→
   418→      // Check schedule
   419→      if (!isWithinBetSchedule(settings.betScheduleStart, settings.betScheduleEnd)) {
   420→        // Don't stop the bot, just skip this scan
   421→        return result;
   422→      }
   423→
   424→      // Check risk limits
   425→      const user = await prisma.user.findUnique({ where: { id: userId } });
   426→      const riskCheck = checkRiskLimits(user?.dailyPnl || 0, user?.weeklyPnl || 0, {
   427→        stopLossDaily: settings.stopLossDaily,
   428→        stopLossWeekly: settings.stopLossWeekly,
   429→        profitTargetDaily: settings.profitTargetDaily,
   430→        profitTargetWeekly: settings.profitTargetWeekly,
   431→      });
   432→
   433→      if (!riskCheck.canBet) {
   434→        // Auto-stop the bot if risk limits hit
   435→        const reason = (user?.dailyPnl ?? 0) <= -settings.stopLossDaily ? "stop_loss" : "profit_target";
   436→        await this.stop(userId, reason);
   437→        await prisma.botLog.create({
   438→          data: {
   439→            userId,
   440→            action: reason === "stop_loss" ? "stop_loss_hit" : "profit_target_hit",
   441→            reasoning: riskCheck.reason,
   442→          },
   443→        });
   444→        return result;
   445→      }
   446→
   447→      // Get connected betting account
   448→      const bettingAccount = await prisma.bettingAccount.findFirst({
   449→        where: { userId, isConnected: true },
   450→        orderBy: { allocatedAmount: "desc" },
   451→      });
   452→
   453→      if (!bettingAccount || bettingAccount.allocatedAmount <= 0) {
   454→        await this.stop(userId, "no_allocation");
   455→        await prisma.botLog.create({
   456→          data: {
   457→            userId,
   458→            action: "bot_scan",
   459→            reasoning: "No connected broker or allocation available. Stopping bot.",
   460→          },
   461→        });
   462→        return result;
   463→      }
   464→
   465→      // Check daily limit
   466→      const todayStart = new Date();
   467→      todayStart.setHours(0, 0, 0, 0);
   468→      const todayBets = await prisma.bet.findMany({
   469→        where: {
   470→          userId,
   471→          placedAt: { gte: todayStart },
   472→          status: { in: ["pending", "won", "lost", "cashed_out", "partial_cashout"] },
   473→        },
   474→      });
   475→      const dailyStake = todayBets.reduce((sum, b) => sum + b.stake, 0);
   476→      const existingBetMatchIds = todayBets.map((b) => b.matchId);
   477→
   478→      if (dailyStake >= settings.dailyBetLimit) {
   479→        // Don't stop the bot, just skip this scan
   480→        return result;
   481→      }
   482→
   483→      const remainingDailyLimit = Math.min(
   484→        settings.dailyBetLimit - dailyStake,
   485→        bettingAccount.allocatedAmount
   486→      );
   487→
   488→      if (remainingDailyLimit < 5) {
   489→        return result;
   490→      }
   491→
   492→      // Get active allocation
   493→      const activeAllocation = await prisma.allocation.findFirst({
   494→        where: { userId, bettingAccountId: bettingAccount.id, status: "active" },
   495→      });
   496→
   497→      // Get upcoming matches
   498→      const preferredSports = settings.preferredSports?.split(",") || ["football"];
   499→      const upcomingMatches = await prisma.match.findMany({
   500→        where: {
   501→          status: "upcoming",
   502→          sport: { in: preferredSports },
   503→          commenceTime: { gte: new Date() },
   504→          id: { notIn: existingBetMatchIds },
   505→        },
   506→        orderBy: { commenceTime: "asc" },
   507→        take: 20,
   508→      });
   509→
   510→      result.matches = upcomingMatches.length;
   511→
   512→      if (upcomingMatches.length === 0) {
   513→        return result;
   514→      }
   515→
   516→      // Analyze and place bets
   517→      for (const match of upcomingMatches) {
   518→        try {
   519→          const homeTeamStats = await prisma.teamStats.findFirst({
   520→            where: { teamName: match.homeTeam, sport: match.sport },
   521→          });
   522→          const awayTeamStats = await prisma.teamStats.findFirst({
   523→            where: { teamName: match.awayTeam, sport: match.sport },
   524→          });
   525→
   526→          const prediction = analyzeMatch(
   527→            {
   528→              homeTeam: match.homeTeam,
   529→              awayTeam: match.awayTeam,
   530→              sport: match.sport,
   531→              league: match.league,
   532→              homeOdds: match.homeOdds,
   533→              drawOdds: match.drawOdds ?? undefined,
   534→              awayOdds: match.awayOdds,
   535→              overOdds: match.overOdds ?? undefined,
   536→              underOdds: match.underOdds ?? undefined,
   537→              overUnderLine: match.overUnderLine ?? undefined,
   538→              status: match.status,
   539→              commenceTime: match.commenceTime.toISOString(),
   540→            },
   541→            homeTeamStats,
   542→            awayTeamStats,
   543→            user?.bankroll || 1000,
   544→            settings.kellyFraction
   545→          );
   546→
   547→          const autoBetCheck = shouldAutoBet(prediction, {
   548→            minOddsThreshold: settings.minOddsThreshold,
   549→            maxOddsThreshold: settings.maxOddsThreshold,
   550→            minAiConfidence: settings.minAiConfidence,
   551→            minEdgeThreshold: settings.minEdgeThreshold,
   552→            riskLevel: settings.riskLevel,
   553→            preferredSports: settings.preferredSports,
   554→          }, match.sport);
   555→
   556→          const recOdds = prediction.recommended === "home" ? match.homeOdds
   557→            : prediction.recommended === "away" ? match.awayOdds
   558→            : match.drawOdds || 3.0;
   559→
   560→          // Check odds range
   561→          if (recOdds < settings.minOddsThreshold || recOdds > settings.maxOddsThreshold) {
   562→            result.skipped++;
   563→            continue;
   564→          }
   565→
   566→          if (!autoBetCheck.shouldPlace) {
   567→            result.skipped++;
   568→            continue;
   569→          }
   570→
   571→          // Calculate stake
   572→          const stake = Math.min(
   573→            autoBetCheck.suggestedStake || settings.maxBetAmount * 0.5,
   574→            settings.maxBetAmount,
   575→            remainingDailyLimit - result.totalStake
   576→          );
   577→
   578→          if (stake < 5) continue;
   579→
   580→          const selection = prediction.recommended === "home" ? match.homeTeam
   581→            : prediction.recommended === "away" ? match.awayTeam
   582→            : prediction.recommended === "draw" ? "Draw"
   583→            : prediction.recommended === "over" ? "Over 2.5"
   584→            : "Under 2.5";
   585→
   586→          const potentialWin = Math.round(stake * recOdds * 100) / 100;
   587→
   588→          // Place bet on broker
   589→          const brokerResult = await placeBetOnBroker(
   590→            bettingAccount.platform,
   591→            bettingAccount.accessToken || "",
   592→            {
   593→              matchId: match.id,
   594→              selection,
   595→              odds: recOdds,
   596→              stake,
   597→              betType: "single",
   598→            }
   599→          );
   600→
   601→          // Create bet record
   602→          const bet = await prisma.bet.create({
   603→            data: {
   604→              userId,
   605→              bettingAccountId: bettingAccount.id,
   606→              matchId: match.id,
   607→              betType: "single",
   608→              selection,
   609→              odds: recOdds,
   610→              stake,
   611→              potentialWin,
   612→              isAutoPlaced: true,
   613→              aiConfidence: prediction.confidence,
   614→              aiReasoning: prediction.analysis,
   615→              aiModelUsed: "v2_ensemble",
   616→              kellyStake: prediction.kellyStake,
   617→              valueEdge: prediction.valueEdge,
   618→              riskScore: prediction.riskScore,
   619→            },
   620→          });
   621→
   622→          // Update allocation
   623→          if (activeAllocation) {
   624→            await prisma.allocation.update({
   625→              where: { id: activeAllocation.id },
   626→              data: {
   627→                usedAmount: { increment: stake },
   628→                remainingAmount: { decrement: stake },
   629→              },
   630→            });
   631→          }
   632→
   633→          // Update betting account
   634→          await prisma.bettingAccount.update({
   635→            where: { id: bettingAccount.id },
   636→            data: {
   637→              allocatedAmount: { decrement: stake },
   638→              lastBetPlacedAt: new Date(),
   639→              totalBrokerBets: { increment: 1 },
   640→            },
   641→          });
   642→
   643→          // Create transaction
   644→          await prisma.transaction.create({
   645→            data: {
   646→              userId,
   647→              type: "bet_placed",
   648→              amount: -stake,
   649→              currency: bettingAccount.currency || "USD",
   650→              status: "completed",
   651→              description: `Auto-bet via ${bettingAccount.platform}: ${match.homeTeam} vs ${match.awayTeam} - ${selection} @ ${recOdds}`,
   652→              betId: bet.id,
   653→            },
   654→          });
   655→
   656→          // Update match AI data
   657→          await prisma.match.update({
   658→            where: { id: match.id },
   659→            data: {
   660→              aiHomeWinProb: prediction.homeWinProb,
   661→              aiDrawProb: prediction.drawProb,
   662→              aiAwayWinProb: prediction.awayWinProb,
   663→              aiConfidence: prediction.confidence,
   664→              aiRecommended: prediction.recommended,
   665→              aiAnalysis: prediction.analysis,
   666→              aiRiskScore: prediction.riskScore,
   667→              aiValueEdge: prediction.valueEdge,
   668→              aiKellyStake: prediction.kellyStake,
   669→            },
   670→          });
   671→
   672→          // Log the bet
   673→          await prisma.botLog.create({
   674→            data: {
   675→              userId,
   676→              action: "bet_placed",
   677→              matchId: match.id,
   678→              betId: bet.id,
   679→              details: JSON.stringify({
   680→                stake,
   681→                odds: recOdds,
   682→                selection,
   683→                potentialWin,
   684→                broker: bettingAccount.platform,
   685→                brokerBetId: brokerResult.brokerBetId,
   686→              }),
   687→              reasoning: autoBetCheck.reason,
   688→              confidence: prediction.confidence,
   689→              profitImpact: -stake,
   690→            },
   691→          });
   692→
   693→          result.betsPlaced++;
   694→          result.totalStake += stake;
   695→        } catch (matchError) {
   696→          console.error(`[BotEngine] Error processing match ${match.id}:`, matchError);
   697→          result.skipped++;
   698→        }
   699→      }
   700→
   701→      // Check for accumulator opportunities
   702→      if (result.betsPlaced === 0 && result.skipped > 0) {
   703→        await prisma.botLog.create({
   704→          data: {
   705→            userId,
   706→            action: "bot_scan",
   707→            reasoning: `Scan completed: ${result.matches} matches found, ${result.skipped} skipped (no qualifying bets)`,
   708→          },
   709→        });
   710→      }
   711→
   712→    } catch (error) {
   713→      console.error(`[BotEngine] Scan cycle error for user ${userId}:`, error);
   714→      result.error = error instanceof Error ? error.message : "Unknown error";
   715→    }
   716→
   717→    return result;
   718→  }

   722→  private async validatePrerequisites(userId: string): Promise<{
   723→    valid: boolean;
   724→    error?: string;
   725→    connectedAccount?: { platform: string; allocatedAmount: number };
   726→  }> {
   727→    const settings = await prisma.userSettings.findUnique({ where: { userId } });
   728→    if (!settings) {
   729→      return { valid: false, error: "Settings not found. Please configure your settings first." };
   730→    }
   731→
   732→    if (!settings.autoBettingEnabled) {
   733→      return { valid: false, error: "Auto-betting is disabled. Enable it first." };
   734→    }
   735→
   736→    const connectedAccount = await prisma.bettingAccount.findFirst({
   737→      where: { userId, isConnected: true },
   738→    });
   739→
   740→    if (!connectedAccount) {
   741→      return { valid: false, error: "No connected broker account. Connect a broker and set allocation first." };
   742→    }
   743→
   744→    if (connectedAccount.allocatedAmount <= 0) {
   745→      return { valid: false, error: "No allocation set. Allocate funds from your broker account first." };
   746→    }
   747→
   748→    return {
   749→      valid: true,
   750→      connectedAccount: {
   751→        platform: connectedAccount.platform,
   752→        allocatedAmount: connectedAccount.allocatedAmount,
   753→      },
   754→    };
   755→  }

   757→  // ==================== AUTO SETTLEMENT ====================
   758→
   759→  /**
   760→   * Automatically settle bets for finished matches.
   761→   * Called periodically by the engine (every 3rd scan cycle).
   762→   */
   763→  private async autoSettle(userId: string): Promise<number> {
   764→    const user = await prisma.user.findUnique({
   765→      where: { id: userId },
   766→      include: { settings: true },
   767→    });
   768→
   769→    if (!user) return 0;
   770→
   771→    const commissionRate = user.settings?.commissionRate || 0.10;
   772→
   773→    // Find pending bets for finished matches
   774→    const unsettledBets = await prisma.bet.findMany({
   775→      where: {
   776→        userId,
   777→        status: "pending",
   778→        match: { status: "finished" },
   779→      },
   780→      include: { match: true, bettingAccount: true },
   781→    });
   782→
   783→    if (unsettledBets.length === 0) return 0;
   784→
   785→    let settledCount = 0;
   786→
   787→    for (const bet of unsettledBets) {
   788→      if (!bet.match) continue;
   789→
   790→      const match = bet.match;
   791→      const homeScore = match.homeScore ?? 0;
   792→      const awayScore = match.awayScore ?? 0;
   793→
   794→      let betWon = false;
   795→      let resultReason = "";
   796→
   797→      if (bet.betType === "single" || bet.betType === "accumulator_leg") {
   798→        if (bet.selection === match.homeTeam) {
   799→          betWon = homeScore > awayScore;
   800→          resultReason = betWon
   801→            ? `${match.homeTeam} won ${homeScore}-${awayScore}`
   802→            : `${match.homeTeam} did not win (${homeScore}-${awayScore})`;
   803→        } else if (bet.selection === match.awayTeam) {
   804→          betWon = awayScore > homeScore;
   805→          resultReason = betWon
   806→            ? `${match.awayTeam} won ${awayScore}-${homeScore}`
   807→            : `${match.awayTeam} did not win (${awayScore}-${homeScore})`;
   808→        } else if (bet.selection === "Draw") {
   809→          betWon = homeScore === awayScore;
   810→          resultReason = betWon
   811→            ? `Match drawn ${homeScore}-${awayScore}`
   812→            : `Match not drawn (${homeScore}-${awayScore})`;
   813→        } else if (bet.selection === "Over 2.5") {
   814→          const totalGoals = homeScore + awayScore;
   815→          betWon = totalGoals > 2.5;
   816→          resultReason = betWon
   817→            ? `Total goals ${totalGoals} > 2.5`
   818→            : `Total goals ${totalGoals} <= 2.5`;
   819→        } else if (bet.selection === "Under 2.5") {
   820→          const totalGoals = homeScore + awayScore;
   821→          betWon = totalGoals < 2.5;
   822→          resultReason = betWon
   823→            ? `Total goals ${totalGoals} < 2.5`
   824→            : `Total goals ${totalGoals} >= 2.5`;
   825→        }
   826→      }
   827→
   828→      const now = new Date();
   829→
   830→      if (betWon) {
   831→        const effectiveStake = bet.partialCashoutAmount ? bet.stake - bet.partialCashoutAmount : bet.stake;
   832→        const grossProfit = bet.potentialWin - effectiveStake;
   833→        const commission = grossProfit * commissionRate;
   834→        const netProfit = grossProfit - commission;
   835→
   836→        await prisma.bet.update({
   837→          where: { id: bet.id },
   838→          data: {
   839→            status: "won",
   840→            profit: Math.round(netProfit * 100) / 100,
   841→            commission: Math.round(commission * 100) / 100,
   842→            settledAt: now,
   843→            settlementReason: "match_finished",
   844→          },
   845→        });
   846→
   847→        await prisma.user.update({
   848→          where: { id: userId },
   849→          data: {
   850→            balance: { increment: bet.potentialWin },
   851→            totalProfit: { increment: netProfit },
   852→            commissionPaid: { increment: commission },
   853→            dailyPnl: { increment: netProfit },
   854→            weeklyPnl: { increment: netProfit },
   855→          },
   856→        });
   857→
   858→        await prisma.transaction.create({
   859→          data: {
   860→            userId,
   861→            type: "bet_won",
   862→            amount: bet.potentialWin,
   863→            currency: "USD",
   864→            status: "completed",
   865→            description: `Auto-settled: ${match.homeTeam} vs ${match.awayTeam} - ${bet.selection} @ ${bet.odds} (profit: $${netProfit.toFixed(2)})`,
   866→            betId: bet.id,
   867→          },
   868→        });
   869→
   870→        if (commission > 0 && bet.bettingAccountId) {
   871→          await prisma.commissionLedger.create({
   872→            data: {
   873→              userId,
   874→              bettingAccountId: bet.bettingAccountId,
   875→              betId: bet.id,
   876→              accumulatorId: bet.accumulatorId,
   877→              grossProfit,
   878→              commissionRate,
   879→              commissionAmount: commission,
   880→              netProfit,
   881→              status: "pending",
   882→            },
   883→          });
   884→        }
   885→
   886→        await prisma.botLog.create({
   887→          data: {
   888→            userId,
   889→            action: "bet_settled",
   890→            betId: bet.id,
   891→            matchId: match.id,
   892→            accumulatorId: bet.accumulatorId,
   893→            details: JSON.stringify({ result: "won", profit: netProfit, commission, autoSettled: true }),
   894→            reasoning: resultReason,
   895→            profitImpact: netProfit,
   896→          },
   897→        });
   898→      } else {
   899→        const effectiveStake = bet.partialCashoutAmount ? bet.stake - bet.partialCashoutAmount : bet.stake;
   900→
   901→        await prisma.bet.update({
   902→          where: { id: bet.id },
   903→          data: {
   904→            status: "lost",
   905→            profit: -effectiveStake,
   906→            settledAt: now,
   907→            settlementReason: "match_finished",
   908→          },
   909→        });
   910→
   911→        await prisma.user.update({
   912→          where: { id: userId },
   913→          data: {
   914→            totalLoss: { increment: effectiveStake },
   915→            dailyPnl: { decrement: effectiveStake },
   916→            weeklyPnl: { decrement: effectiveStake },
   917→          },
   918→        });
   919→
   920→        await prisma.botLog.create({
   921→          data: {
   922→            userId,
   923→            action: "bet_settled",
   924→            betId: bet.id,
   925→            matchId: match.id,
   926→            accumulatorId: bet.accumulatorId,
   927→            details: JSON.stringify({ result: "lost", loss: effectiveStake, autoSettled: true }),
   928→            reasoning: resultReason,
   929→            profitImpact: -effectiveStake,
   930→          },
   931→        });
   932→      }
   933→
   934→      settledCount++;
   935→    }
   936→
   937→    if (settledCount > 0) {
   938→      console.log(`[BotEngine] Auto-settled ${settledCount} bet(s) for user ${userId}`);
   939→    }
   940→
   941→    return settledCount;
   942→  }

   944→  // ==================== AUTO CASHOUT EVALUATION ====================
   945→
   946→  /**
   947→   * Evaluate live bets for cashout opportunities.
   948→   * Called periodically by the engine (every 5th scan cycle).
   949→   * Uses the AI engine's cashout recommendation system.
   950→   */
   951→  private async autoCashoutEvaluation(userId: string): Promise<number> {
   952→    const settings = await prisma.userSettings.findUnique({ where: { userId } });
   953→    if (!settings?.autoCashoutEnabled) return 0;
   954→
   955→    // Find live bets that could be cashed out
   956→    const liveBets = await prisma.bet.findMany({
   957→      where: {
   958→        userId,
   959→        status: "pending",
   960→        isAutoPlaced: true,
   961→        match: { status: "live" },
   962→      },
   963→      include: { match: true },
   964→    });
   965→
   966→    if (liveBets.length === 0) return 0;
   967→
   968→    let cashoutCount = 0;
   969→
   970→    for (const bet of liveBets) {
   971→      if (!bet.match) continue;
   972→
   973→      // Simple cashout evaluation based on match state
   974→      const match = bet.match;
   975→      const homeScore = match.homeScore ?? 0;
   976→      const awayScore = match.awayScore ?? 0;
   977→      const minute = match.minute ?? 0;
   978→
   979→      // Determine if the bet is winning
   980→      let isWinning = false;
   981→      if (bet.selection === match.homeTeam) {
   982→        isWinning = homeScore > awayScore;
   983→      } else if (bet.selection === match.awayTeam) {
   984→        isWinning = awayScore > homeScore;
   985→      } else if (bet.selection === "Draw") {
   986→        isWinning = homeScore === awayScore;
   987→      } else if (bet.selection === "Over 2.5") {
   988→        isWinning = (homeScore + awayScore) > 2.5;
   989→      } else if (bet.selection === "Under 2.5") {
   990→        isWinning = (homeScore + awayScore) < 2.5;
   991→      }
   992→
   993→      // Cashout logic: if the bet is winning and we're past 70 minutes,
   994→      // and the cashout threshold is met, execute cashout
   995→      if (isWinning && minute >= 70) {
   996→        const cashoutOdds = 1 + (1 / bet.odds) * (minute / 90);
   997→        const cashoutAmount = Math.round(bet.stake * cashoutOdds * 100) / 100;
   998→
   999→        if (cashoutAmount > bet.stake * (1 + settings.cashoutThreshold)) {
  1000→          // Execute partial or full cashout based on settings
  1001→          if (settings.partialCashoutEnabled) {
  1002→            // Partial cashout: take 50% profit, keep the rest riding
  1003→            const partialAmount = Math.round(cashoutAmount * settings.partialCashoutPercent * 100) / 100;
  1004→
  1005→            await prisma.bet.update({
  1006→              where: { id: bet.id },
  1007→              data: {
  1008→                status: "partial_cashout",
  1009→                partialCashoutAmount,
  1010→                partialCashoutPercent: settings.partialCashoutPercent,
  1011→                cashoutOdds,
  1012→                cashedOutAt: new Date(),
  1013→              },
  1014→            });
  1015→
  1016→            await prisma.user.update({
  1017→              where: { id: userId },
  1018→              data: {
  1019→                balance: { increment: partialAmount },
  1020→                dailyPnl: { increment: partialAmount - bet.stake * settings.partialCashoutPercent },
  1021→                weeklyPnl: { increment: partialAmount - bet.stake * settings.partialCashoutPercent },
  1022→              },
  1023→            });
  1024→
  1025→            await prisma.transaction.create({
  1026→              data: {
  1027→                userId,
  1028→                type: "partial_cashout",
  1029→                amount: partialAmount,
  1030→                currency: "USD",
  1031→                status: "completed",
  1032→                description: `Auto partial cashout: ${match.homeTeam} vs ${match.awayTeam} - ${bet.selection} ($${partialAmount.toFixed(2)})`,
  1033→                betId: bet.id,
  1034→              },
  1035→            });
  1036→
  1037→            await prisma.botLog.create({
  1038→              data: {
  1039→                userId,
  1040→                action: "cashout_executed",
  1041→                betId: bet.id,
  1042→                matchId: match.id,
  1043→                details: JSON.stringify({
  1044→                  type: "partial",
  1045→                  amount: partialAmount,
  1046→                  percent: settings.partialCashoutPercent,
  1047→                  minute,
  1048→                  autoCashout: true,
  1049→                }),
  1050→                reasoning: `Auto partial cashout at ${minute}' - bet winning. ${Math.round(settings.partialCashoutPercent * 100)}% cashed out for $${partialAmount.toFixed(2)}`,
  1051→                profitImpact: partialAmount - bet.stake * settings.partialCashoutPercent,
  1052→              },
  1053→            });
  1054→          } else {
  1055→            // Full cashout
  1056→            await prisma.bet.update({
  1057→              where: { id: bet.id },
  1058→              data: {
  1059→                status: "cashed_out",
  1060→                cashoutAmount,
  1061→                cashoutOdds,
  1062→                profit: cashoutAmount - bet.stake,
  1063→                cashedOutAt: new Date(),
  1064→                settlementReason: "cashout",
  1065→              },
  1066→            });
  1067→
  1068→            await prisma.user.update({
  1069→              where: { id: userId },
  1070→              data: {
  1071→                balance: { increment: cashoutAmount },
  1072→                totalProfit: { increment: cashoutAmount - bet.stake },
  1073→                dailyPnl: { increment: cashoutAmount - bet.stake },
  1074→                weeklyPnl: { increment: cashoutAmount - bet.stake },
  1075→              },
  1076→            });
  1077→
  1078→            await prisma.transaction.create({
  1079→              data: {
  1080→                userId,
  1081→                type: "cashout",
  1082→                amount: cashoutAmount,
  1083→                currency: "USD",
  1084→                status: "completed",
  1085→                description: `Auto cashout: ${match.homeTeam} vs ${match.awayTeam} - ${bet.selection} ($${cashoutAmount.toFixed(2)})`,
  1086→                betId: bet.id,
  1087→              },
  1088→            });
  1089→
  1090→            await prisma.botLog.create({
  1091→              data: {
  1092→                userId,
  1093→                action: "cashout_executed",
  1094→                betId: bet.id,
  1095→                matchId: match.id,
  1096→                details: JSON.stringify({
  1097→                  type: "full",
  1098→                  amount: cashoutAmount,
  1099→                  minute,
  1100→                  autoCashout: true,
  1101→                }),
  1102→                reasoning: `Auto cashout at ${minute}' - bet winning. Full cashout for $${cashoutAmount.toFixed(2)}`,
  1103→                profitImpact: cashoutAmount - bet.stake,
  1104→              },
  1105→            });
  1106→          }
  1107→
  1108→          cashoutCount++;
  1109→        }
  1110→      }
  1111→    }
  1112→
  1113→    if (cashoutCount > 0) {
  1114→      console.log(`[BotEngine] Auto-cashout executed for ${cashoutCount} bet(s) for user ${userId}`);
  1115→    }
  1116→
  1117→    return cashoutCount;
  1118→  }
