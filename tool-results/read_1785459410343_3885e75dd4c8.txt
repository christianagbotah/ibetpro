   119→   119→          credentials,
   120→   120→          region: selectedRegion,

   156→   156→        method: "POST",
   157→   157→        headers: { "Content-Type": "application/json" },
   158→   158→        body: JSON.stringify({ bettingAccountId: accountId }),
   159→   159→      });

   161→   161→      const result = await res.json();

   172→   172→          title: "Sync Failed",
   173→   173→          description: result.error,
   174→   174→          variant: "destructive",
   175→   175→        });
   176→   176→      }
   177→   177→    } catch {
   178→   178→      toast({ title: "Sync Error", variant: "destructive" });
   179→   179→    }
   180→   180→  }, [toast, refresh]);
   181→   181→

   184→   184→    setAccountName("");
   185→   185→    setSelectedPlatform(null);
   186→   186→    setStep("region");

   189→   189→  };

   222→   222→
   223→   223→              <DialogContent className="max-w-lg max-h-[90vh] overflow-y-auto">

   239→   239→                      {step === "credentials" ? "✓" : "2"}
   240→   240→                    </div>
   241→   241→                    Platform

   243→   243→                  <div className="h-px flex-1 bg-border" />
   244→   244→                  <div className={`flex items-center gap-1.5 text-xs ${step === "credentials" ? "text-primary font-medium" : "text-muted-foreground"}`}>

   298→   298→                        return (
   299→   299→                          <button
   300→   300→                            key={region.code}
   301→   301→                            onClick={() => {
   302→   302→                              setSelectedRegion(region.code);
   303→   303→                              setStep("platform");
   304→   304→                            }}
   305→   305→                            className={`flex items-center gap-2 rounded-lg border p-2.5 transition-all text-left hover:bg-primary/5 ${
   306→   306→                              selectedRegion === region.code

   329→   329→                )}
   330→   330→
   331→   331→                {/* ====== STEP 2: PLATFORM SELECTION ====== */}

   333→   333→                  <div className="space-y-3">
   334→   334→                    <button

   336→   336→                        setStep("region");
   337→   337→                        setSelectedPlatform(null);
   338→   338→                      }}
   339→   339→                      className="flex items-center gap-1 text-xs text-muted-foreground hover:text-primary transition-colors"
   340→   340→                    >
   341→   341→                      <ArrowLeft className="h-3 w-3" /> Change region

   392→   392→                          </button>

   396→   396→                  </div>
   397→   397→                )}
   398→   398→
   399→   399→                {/* ====== STEP 3: CREDENTIALS ====== */}
