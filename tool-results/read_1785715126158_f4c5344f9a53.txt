   103→// Parse form string like "WWWDW" into a score
   104→function parseForm(form: string): number {
   105→  if (!form) return 0.5;
   106→  const chars = form.split("");
   107→  let score = 0;
   108→  let total = 0;
   109→  for (const c of chars) {
   110→    if (c === "W") score += 1;
   111→    else if (c === "D") score += 0.5;
   112→    total += 1;
   113→  }
   114→  return total > 0 ? score / total : 0.5;
   115→}
   116→
   117→// Parse record string like "12-1-1" into win rate
   118→function parseRecord(record: string): number {
   119→  if (!record) return 0.5;
   120→  const parts = record.split("-").map(Number);
   121→  if (parts.length < 3 || parts.some(isNaN)) return 0.5;
   122→  const [wins, draws, losses] = parts;
   123→  const total = wins + draws + losses;
   124→  return total > 0 ? (wins + draws * 0.5) / total : 0.5;
   125→}
   126→
   127→// Calculate Poisson-based probability for goals
   128→function poissonProbability(lambda: number, k: number): number {
   129→  return (Math.pow(lambda, k) * Math.exp(-lambda)) / factorial(k);
   130→}
   131→
   132→function factorial(n: number): number {
   133→  if (n <= 1) return 1;
   134→  let result = 1;
   135→  for (let i = 2; i <= n; i++) result *= i;
   136→  return result;
   137→}

   139→// ==================== KELLY CRITERION ====================
   140→
   141→/**
   142→ * Calculate Kelly Criterion optimal stake
   143→ * f* = (bp - q) / b
   144→ * where b = odds-1, p = probability, q = 1-p
   145→ */
   146→export function calculateKellyStake(
   147→  probability: number,
   148→  odds: number,
   149→  bankroll: number,
   150→  fraction: number = 0.25 // quarter-Kelly by default
   151→): { stake: number; fullKelly: number; fractionKelly: number; edge: number } {
   152→  const b = odds - 1; // net odds
   153→  const p = probability;
   154→  const q = 1 - p;
   155→  const fullKelly = (b * p - q) / b;
   156→  const fractionKelly = fullKelly * fraction;
   157→  const edge = p * odds - 1; // expected value per dollar
   158→
   159→  // Cap at 10% of bankroll for safety
   160→  const maxStake = bankroll * 0.10;
   161→  const stake = Math.max(0, Math.min(fractionKelly * bankroll, maxStake));
   162→
   163→  return {
   164→    stake: Math.round(stake * 100) / 100,
   165→    fullKelly: Math.round(fullKelly * 10000) / 10000,
   166→    fractionKelly: Math.round(fractionKelly * 10000) / 10000,
   167→    edge: Math.round(edge * 10000) / 10000,
   168→  };
   169→}

   171→// ==================== ELO RATING SYSTEM ====================
   172→
   173→/**
   174→ * Calculate ELO-based win probability
   175→ */
   176→export function calculateEloProbability(
   177→  homeElo: number,
   178→  awayElo: number,
   179→  homeAdvantage: number = 65 // typical home advantage in ELO points
   180→): { homeWinProb: number; drawProb: number; awayWinProb: number } {
   181→  const effectiveHomeElo = homeElo + homeAdvantage;
   182→  const diff = effectiveHomeElo - awayElo;
   183→
   184→  // Standard ELO expected score formula
   185→  const homeExpected = 1 / (1 + Math.pow(10, -diff / 400));
   186→
   187→  // Approximate draw probability (peaks when teams are equal)
   188→  const drawProb = 0.25 * Math.exp(-Math.abs(diff) / 400);
   189→  const homeWinProb = homeExpected * (1 - drawProb);
   190→  const awayWinProb = (1 - homeExpected) * (1 - drawProb);
   191→
   192→  // Normalize
   193→  const total = homeWinProb + drawProb + awayWinProb;
   194→  return {
   195→    homeWinProb: Math.round((homeWinProb / total) * 100) / 100,
   196→    drawProb: Math.round((drawProb / total) * 100) / 100,
   197→    awayWinProb: Math.round((awayWinProb / total) * 100) / 100,
   198→  };
   199→}

   201→// ==================== MAIN ANALYSIS ====================
   202→
   203→export function analyzeMatch(
   204→  match: MatchData,
   205→  homeTeamStats?: TeamStatsData | null,
   206→  awayTeamStats?: TeamStatsData | null,
   207→  bankroll: number = 1000,
   208→  kellyFraction: number = 0.25
   209→): Prediction {
   210→  let homeWinProb = 0.33;
   211→  let drawProb = 0.33;
   212→  let awayWinProb = 0.33;
   213→
   214→  // Model 1: Bookmaker Odds Implied Probability
   215→  const homeImplied = 1 / match.homeOdds;
   216→  const awayImplied = 1 / match.awayOdds;
   217→  const drawImplied = match.drawOdds ? 1 / match.drawOdds : 0.25;
   218→  const totalImplied = homeImplied + drawImplied + awayImplied;
   219→  const margin = 1 / totalImplied;
   220→
   221→  homeWinProb = homeImplied * margin;
   222→  drawProb = drawImplied * margin;
   223→  awayWinProb = awayImplied * margin;
   224→
   225→  // Model 2: Poisson Distribution (if team stats available)
   226→  let poissonResult: PoissonResult | null = null;
   227→  if (homeTeamStats && awayTeamStats) {
   228→    poissonResult = calculatePoissonProbabilities(
   229→      homeTeamStats.attackRating,
   230→      awayTeamStats.attackRating,
   231→      homeTeamStats.defenseRating,
   232→      awayTeamStats.defenseRating
   233→    );
   234→
   235→    // Model 3: ELO Rating (if available)
   236→    let eloResult = null;
   237→    if (homeTeamStats.eloRating && awayTeamStats.eloRating) {
   238→      eloResult = calculateEloProbability(homeTeamStats.eloRating, awayTeamStats.eloRating);
   239→    }
   240→
   241→    // Form factor (0-1)
   242→    const homeForm = parseForm(homeTeamStats.form ?? "");
   243→    const awayForm = parseForm(awayTeamStats.form ?? "");
   244→
   245→    // Overall rating factor
   246→    const ratingDiff = homeTeamStats.overallRating - awayTeamStats.overallRating;
   247→    const ratingFactor = ratingDiff / 100;
   248→
   249→    // Attack vs Defense
   250→    const homeAttackVsAwayDefense = (homeTeamStats.attackRating - awayTeamStats.defenseRating) / 100;
   251→    const awayAttackVsHomeDefense = (awayTeamStats.attackRating - homeTeamStats.defenseRating) / 100;
   252→
   253→    // Home/away record
   254→    const homeHomeRecord = parseRecord(homeTeamStats.homeRecord ?? "");
   255→    const awayAwayRecord = parseRecord(awayTeamStats.awayRecord ?? "");
   256→
   257→    // Win rate
   258→    const homeWinRate = homeTeamStats.matchesPlayed > 0
   259→      ? homeTeamStats.wins / homeTeamStats.matchesPlayed
   260→      : 0.5;
   261→    const awayWinRate = awayTeamStats.matchesPlayed > 0
   262→      ? awayTeamStats.wins / awayTeamStats.matchesPlayed
   263→      : 0.5;
   264→
   265→    // Composite adjustments
   266→    const homeAdvantage = 0.05;
   267→    const formAdjust = (homeForm - awayForm) * 0.08;
   268→    const ratingAdjust = ratingFactor * 0.15;
   269→    const attackDefenseAdjust = (homeAttackVsAwayDefense - awayAttackVsHomeDefense) * 0.1;
   270→    const recordAdjust = (homeHomeRecord - awayAwayRecord) * 0.06;
   271→    const winRateAdjust = (homeWinRate - awayWinRate) * 0.05;
   272→
   273→    const totalAdjust = homeAdvantage + formAdjust + ratingAdjust + attackDefenseAdjust + recordAdjust + winRateAdjust;
   274→
   275→    // Apply adjustments
   276→    homeWinProb += totalAdjust * 0.5;
   277→    awayWinProb -= totalAdjust * 0.5;
   278→
   279→    // Draw probability adjustment
   280→    const closeness = 1 - Math.abs(homeWinProb - awayWinProb);
   281→    drawProb = 0.2 + closeness * 0.15;
   282→
   283→    // Model 4: Ensemble blending (weighted average)
   284→    // Weights: Implied=40%, Poisson=30%, ELO=20%, Stats=10%
   285→    const weights = { implied: 0.40, poisson: 0.30, elo: 0.20, stats: 0.10 };
   286→
   287→    const statsHome = homeWinProb;
   288→    const statsDraw = drawProb;
   289→    const statsAway = awayWinProb;
   290→
   291→    if (poissonResult && eloResult) {
   292→      homeWinProb = (
   293→        weights.implied * (homeImplied * margin) +
   294→        weights.poisson * poissonResult.homeWinProb +
   295→        weights.elo * eloResult.homeWinProb +
   296→        weights.stats * statsHome
   297→      );
   298→      drawProb = (
   299→        weights.implied * (drawImplied * margin) +
   300→        weights.poisson * poissonResult.drawProb +
   301→        weights.elo * eloResult.drawProb +
   302→        weights.stats * statsDraw
   303→      );
   304→      awayWinProb = (
   305→        weights.implied * (awayImplied * margin) +
   306→        weights.poisson * poissonResult.awayWinProb +
   307→        weights.elo * eloResult.awayWinProb +
   308→        weights.stats * statsAway
   309→      );
   310→    } else if (poissonResult) {
   311→      // Blend implied + poisson + stats
   312→      homeWinProb = 0.45 * (homeImplied * margin) + 0.35 * poissonResult.homeWinProb + 0.20 * statsHome;
   313→      drawProb = 0.45 * (drawImplied * margin) + 0.35 * poissonResult.drawProb + 0.20 * statsDraw;
   314→      awayWinProb = 0.45 * (awayImplied * margin) + 0.35 * poissonResult.awayWinProb + 0.20 * statsAway;
   315→    }
   316→  }
   317→
   318→  // Normalize probabilities
   319→  const total = homeWinProb + drawProb + awayWinProb;
   320→  homeWinProb /= total;
   321→  drawProb /= total;
   322→  awayWinProb /= total;
   323→
   324→  // Clamp values
   325→  homeWinProb = Math.max(0.05, Math.min(0.95, homeWinProb));
   326→  drawProb = Math.max(0.05, Math.min(0.40, drawProb));
   327→  awayWinProb = Math.max(0.05, Math.min(0.95, awayWinProb));
   328→
   329→  // Re-normalize
   330→  const finalTotal = homeWinProb + drawProb + awayWinProb;
   331→  homeWinProb /= finalTotal;
   332→  drawProb /= finalTotal;
   333→  awayWinProb /= finalTotal;
   334→
   335→  // Calculate confidence
   336→  const maxProb = Math.max(homeWinProb, drawProb, awayWinProb);
   337→  const confidence = Math.min(0.95, maxProb + 0.1);
   338→
   339→  // Determine recommendation
   340→  let recommended: Prediction["recommended"] = "home";
   341→  const oddsValue = calculateOddsValue(maxProb, match.homeOdds);
   342→
   343→  if (homeWinProb > awayWinProb && homeWinProb > drawProb) {
   344→    recommended = "home";
   345→  } else if (awayWinProb > homeWinProb && awayWinProb > drawProb) {
   346→    recommended = "away";
   347→  } else {
   348→    recommended = "draw";
   349→  }
   350→
   351→  // Check for better value in over/under
   352→  if (match.sport === "football" && homeTeamStats && awayTeamStats) {
   353→    const avgHomeGoals = homeTeamStats.goalsFor / Math.max(1, homeTeamStats.matchesPlayed);
   354→    const avgAwayGoals = awayTeamStats.goalsFor / Math.max(1, awayTeamStats.matchesPlayed);
   355→    const expectedGoals = avgHomeGoals + avgAwayGoals;
   356→
   357→    if (expectedGoals < 2.2 && match.drawOdds && match.drawOdds < 3.5) {
   358→      if (calculateOddsValue(0.55, 2.1) > calculateOddsValue(maxProb, recommended === "home" ? match.homeOdds : match.awayOdds)) {
   359→        recommended = "under";
   360→      }
   361→    }
   362→  }
   363→
   364→  // Calculate value edge
   365→  const recOdds = recommended === "home" ? match.homeOdds : recommended === "away" ? match.awayOdds : match.drawOdds || 3.0;
   366→  const recProb = recommended === "home" ? homeWinProb : recommended === "away" ? awayWinProb : drawProb;
   367→  const valueEdge = recProb - (1 / recOdds);
   368→
   369→  // Calculate risk score (0-100)
   370→  let riskScore = 50;
   371→  if (confidence < 0.5) riskScore += 20;
   372→  if (confidence > 0.7) riskScore -= 20;
   373→  if (Math.abs(homeWinProb - awayWinProb) < 0.1) riskScore += 15;
   374→  if (match.homeOdds > 3.0 || match.awayOdds > 3.0) riskScore += 10;
   375→  if (!homeTeamStats || !awayTeamStats) riskScore += 15;
   376→  riskScore = Math.max(0, Math.min(100, riskScore));
   377→
   378→  // Calculate Kelly stake
   379→  const kelly = calculateKellyStake(recProb, recOdds, bankroll, kellyFraction);
   380→
   381→  // Generate analysis text
   382→  const analysis = generateBetReasoning(match, { homeWinProb, drawProb, awayWinProb, confidence, recommended, analysis: "", valueEdge, riskScore, kellyStake: kelly.stake }, homeTeamStats, awayTeamStats);
   383→
   384→  return {
   385→    homeWinProb: Math.round(homeWinProb * 100) / 100,
   386→    drawProb: Math.round(drawProb * 100) / 100,
   387→    awayWinProb: Math.round(awayWinProb * 100) / 100,
   388→    confidence: Math.round(confidence * 100) / 100,
   389→    recommended,
   390→    analysis,
   391→    valueEdge: Math.round(valueEdge * 10000) / 10000,
   392→    riskScore,
   393→    kellyStake: kelly.stake,
   394→  };
   395→}

   397→// ==================== SMART CASHOUT ENGINE ====================

   399→export function shouldCashout(
   400→  bet: {
   401→    selection: string;
   402→    odds: number;
   403→    stake: number;
   404→    potentialWin: number;
   405→    status: string;
   406→    partialCashoutAmount?: number | null;
   407→    partialCashoutPercent?: number | null;
   408→  },
   409→  currentMatchState: {
   410→    homeScore: number;
   411→    awayScore: number;
   412→    minute: number;
   413→    homeTeam: string;
   414→    awayTeam: string;
   415→    sport: string;
   416→    status: string;
   417→  },
   418→  settings?: {
   419→    autoCashoutEnabled?: boolean;
   420→    cashoutThreshold?: number;
   421→    waitFullSettlement?: boolean;
   422→    partialCashoutEnabled?: boolean;
   423→    partialCashoutPercent?: number;
   424→  }
   425→): CashoutRecommendation {
   426→  const { stake, potentialWin, odds } = bet;
   427→  const { homeScore, awayScore, minute, sport, status } = currentMatchState;
   428→
   429→  // If match is finished, wait for full settlement
   430→  if (status === "finished") {
   431→    return {
   432→      shouldCashout: false,
   433→      cashoutAmount: potentialWin,
   434→      partialCashoutAmount: 0,
   435→      reasoning: "Match has finished. Waiting for full settlement to collect maximum payout.",
   436→      urgency: "low",
   437→      waitOrCashout: "wait_for_settlement",
   438→      settlementProbability: 1.0,
   439→    };
   440→  }
   441→
   442→  // If match is upcoming, no cashout needed
   443→  if (status === "upcoming") {
   444→    return {
   445→      shouldCashout: false,
   446→      cashoutAmount: stake * 0.85, // slight loss due to margin
   447→      partialCashoutAmount: 0,
   448→      reasoning: "Match hasn't started yet. No cashout needed - hold your position.",
   449→      urgency: "low",
   450→      waitOrCashout: "wait",
   451→      settlementProbability: 0,
   452→    };
   453→  }
   454→
   455→  const isFootball = sport === "football";
   456→  const isBasketball = sport === "basketball";
   457→  const isTennis = sport === "tennis";
   458→  const totalMinutes = isFootball ? 90 : isBasketball ? 48 : isTennis ? 180 : 90;
   459→  const progress = minute / totalMinutes;
   460→
   461→  // Determine if the bet is currently winning
   462→  const selectionIsHome = bet.selection === currentMatchState.homeTeam;
   463→  const selectionIsAway = bet.selection === currentMatchState.awayTeam;
   464→  const currentLead = selectionIsHome
   465→    ? homeScore - awayScore
   466→    : selectionIsAway
   467→    ? awayScore - homeScore
   468→    : 0;
   469→
   470→  let cashoutAmount = 0;
   471→  let partialCashoutAmount = 0;
   472→  let shouldCashout = false;
   473→  let urgency: CashoutRecommendation["urgency"] = "low";
   474→  let waitOrCashout: CashoutRecommendation["waitOrCashout"] = "wait";
   475→  let settlementProbability = 0;
   476→
   477→  // Calculate settlement probability based on match state
   478→  if (currentLead > 0) {
   479→    // Winning - calculate probability of winning at full time
   480→    if (isFootball) {
   481→      if (currentLead >= 2 && progress >= 0.6) settlementProbability = 0.92;
   482→      else if (currentLead >= 2 && progress >= 0.4) settlementProbability = 0.80;
   483→      else if (currentLead >= 1 && progress >= 0.75) settlementProbability = 0.78;
   484→      else if (currentLead >= 1 && progress >= 0.5) settlementProbability = 0.55;
   485→      else settlementProbability = 0.35;
   486→    } else if (isBasketball) {
   487→      if (currentLead >= 10 && progress >= 0.75) settlementProbability = 0.93;
   488→      else if (currentLead >= 5 && progress >= 0.8) settlementProbability = 0.80;
   489→      else settlementProbability = 0.45;
   490→    } else {
   491→      settlementProbability = 0.5 + currentLead * 0.1;
   492→    }
   493→  } else if (currentLead === 0) {
   494→    settlementProbability = 0.25; // Draw situation
   495→  } else {
   496→    // Losing
   497→    settlementProbability = Math.max(0.05, 0.15 + progress * 0.1);
   498→  }
   499→
   500→  const effectiveStake = bet.partialCashoutAmount ? stake - bet.partialCashoutAmount : stake;
   501→
   502→  if (currentLead > 0) {
   503→    // Bet is currently winning
   504→    const remainingTime = 1 - progress;
   505→    const riskFactor = remainingTime * 0.3;
   506→
   507→    if (isFootball) {
   508→      if (currentLead >= 2 && minute >= 60) {
   509→        // Comfortable lead, late in game - WAIT for full settlement
   510→        cashoutAmount = potentialWin * 0.88;
   511→        shouldCashout = false;
   512→        urgency = "low";
   513→        waitOrCashout = settings?.waitFullSettlement ? "wait_for_settlement" : "wait";
   514→      } else if (currentLead >= 1 && minute >= 75) {
   515→        // One goal lead, very late - consider partial cashout
   516→        cashoutAmount = potentialWin * 0.78;
   517→        partialCashoutAmount = (effectiveStake + (potentialWin - effectiveStake) * 0.5) * (settings?.partialCashoutPercent || 0.5);
   518→        shouldCashout = settings?.autoCashoutEnabled && (cashoutAmount / potentialWin) >= (settings?.cashoutThreshold || 0.7);
   519→        urgency = "medium";
   520→        waitOrCashout = shouldCashout ? "cashout_partial" : "wait";
   521→      } else if (currentLead >= 1 && minute >= 60) {
   522→        // One goal lead, late
   523→        cashoutAmount = potentialWin * 0.55;
   524→        partialCashoutAmount = (effectiveStake + (potentialWin - effectiveStake) * 0.3) * (settings?.partialCashoutPercent || 0.5);
   525→        shouldCashout = false;
   526→        urgency = "low";
   527→        waitOrCashout = "wait";
   528→      } else if (currentLead >= 1 && minute >= 45) {
   529→        // One goal lead, second half
   530→        cashoutAmount = potentialWin * 0.4;
   531→        shouldCashout = false;
   532→        urgency = "low";
   533→        waitOrCashout = "wait";
   534→      } else {
   535→        // Early goal lead
   536→        cashoutAmount = potentialWin * 0.25;
   537→        shouldCashout = false;
   538→        urgency = "low";
   539→        waitOrCashout = "wait";
   540→      }
   541→    } else if (isBasketball) {
   542→      if (currentLead >= 10 && minute >= 36) {
   543→        cashoutAmount = potentialWin * 0.88;
   544→        shouldCashout = false;
   545→        urgency = "low";
   546→        waitOrCashout = "wait_for_settlement";
   547→      } else if (currentLead >= 5 && minute >= 40) {
   548→        cashoutAmount = potentialWin * 0.72;
   549→        shouldCashout = settings?.autoCashoutEnabled ?? true;
   550→        urgency = "medium";
   551→        waitOrCashout = shouldCashout ? "cashout_partial" : "wait";
   552→      } else {
   553→        cashoutAmount = potentialWin * 0.4;
   554→        shouldCashout = false;
   555→        urgency = "low";
   556→        waitOrCashout = "wait";
   557→      }
   558→    } else {
   559→      // Generic sport
   560→      cashoutAmount = potentialWin * (0.3 + progress * 0.5);
   561→      shouldCashout = progress > 0.8 && currentLead >= 2;
   562→      urgency = shouldCashout ? "medium" : "low";
   563→      waitOrCashout = shouldCashout ? "cashout_full" : "wait";
   564→    }
   565→
   566→    // Always ensure minimum cashout
   567→    cashoutAmount = Math.max(cashoutAmount, effectiveStake * 1.1);
   568→  } else if (currentLead === 0) {
   569→    // Draw - bet is at risk
   570→    cashoutAmount = effectiveStake * 0.5;
   571→    partialCashoutAmount = 0;
   572→    if (progress > 0.7) {
   573→      shouldCashout = true;
   574→      urgency = "high";
   575→      waitOrCashout = "cashout_full";
   576→    } else {
   577→      shouldCashout = false;
   578→      urgency = "medium";
   579→      waitOrCashout = "wait";
   580→    }
   581→  } else {
   582→    // Bet is losing
   583→    cashoutAmount = effectiveStake * 0.15;
   584→    partialCashoutAmount = 0;
   585→    if (progress > 0.5) {
   586→      shouldCashout = true;
   587→      urgency = "high";
   588→      waitOrCashout = "cashout_full";
   589→    } else {
   590→      shouldCashout = false;
   591→      urgency = "medium";
   592→      waitOrCashout = "wait";
   593→    }
   594→  }
   595→
   596→  // Apply settings thresholds
   597→  if (settings?.autoCashoutEnabled && settings.cashoutThreshold) {
   598→    if (cashoutAmount / potentialWin >= settings.cashoutThreshold && currentLead > 0) {
   599→      if (settings.waitFullSettlement && settlementProbability > 0.8) {
   600→        // Wait for full settlement - AI is confident bet will win
   601→        shouldCashout = false;
   602→        waitOrCashout = "wait_for_settlement";
   603→      } else if (settings.partialCashoutEnabled && partialCashoutAmount > 0) {
   604→        shouldCashout = true;
   605→        waitOrCashout = "cashout_partial";
   606→      } else {
   607→        shouldCashout = true;
   608→        waitOrCashout = "cashout_full";
   609→      }
   610→    }
   611→  }
   612→
   613→  const reasoning = generateCashoutReasoning(
   614→    bet, currentMatchState, shouldCashout, cashoutAmount, urgency,
   615→    waitOrCashout, settlementProbability, partialCashoutAmount
   616→  );
   617→
   618→  return {
   619→    shouldCashout,
   620→    cashoutAmount: Math.round(cashoutAmount * 100) / 100,
   621→    partialCashoutAmount: Math.round(partialCashoutAmount * 100) / 100,
   622→    reasoning,
   623→    urgency,
   624→    waitOrCashout,
   625→    settlementProbability: Math.round(settlementProbability * 100) / 100,
   626→  };
   627→}

   631→/**
   632→ * Analyze an accumulator (parlay) bet with multiple legs
   633→ */
   634→export function analyzeAccumulator(
   635→  legs: Array<{
   636→    match: MatchData;
   637→    homeTeamStats?: TeamStatsData | null;
   638→    awayTeamStats?: TeamStatsData | null;
   639→    recommended: "home" | "away" | "draw" | "over" | "under";
   640→  }>,
   641→  bankroll: number = 1000,
   642→  kellyFraction: number = 0.25,
   643→  bonusThresholds: Array<{ legs: number; bonus: number }> = []
   644→): AccumulatorAnalysis {
   645→  const legAnalyses: Prediction[] = [];
   646→  let totalOdds = 1;
   647→
   648→  for (const leg of legs) {
   649→    const prediction = analyzeMatch(leg.match, leg.homeTeamStats, leg.awayTeamStats, bankroll, kellyFraction);
   650→    legAnalyses.push(prediction);
   651→
   652→    const recOdds = leg.recommended === "home" ? leg.match.homeOdds
   653→      : leg.recommended === "away" ? leg.match.awayOdds
   654→      : leg.match.drawOdds || 3.0;
   655→    totalOdds *= recOdds;
   656→  }
   657→
   658→  // Combined probability (independent events)
   659→  const combinedProbability = legAnalyses.reduce((prod, leg) => {
   660→    const prob = leg.recommended === "home" ? leg.homeWinProb
   661→      : leg.recommended === "away" ? leg.awayWinProb
   662→      : leg.drawProb;
   663→    return prod * prob;
   664→  }, 1);
   665→
   666→  // Risk score increases with more legs
   667→  const avgRisk = legAnalyses.reduce((sum, leg) => sum + leg.riskScore, 0) / legAnalyses.length;
   668→  const riskScore = Math.min(100, avgRisk + (legs.length - 1) * 10);
   669→
   670→  // Calculate accumulator bonus
   671→  let bonusPercent = 0;
   672→  for (const threshold of bonusThresholds) {
   673→    if (legs.length >= threshold.legs) {
   674→      bonusPercent = threshold.bonus;
   675→    }
   676→  }
   677→
   678→  // Expected value
   679→  const expectedValue = combinedProbability * totalOdds * (1 + bonusPercent / 100) - 1;
   680→
   681→  // Recommended stake (fractional Kelly, even more conservative for accumulators)
   682→  const accaKellyFraction = kellyFraction * 0.5; // Half the normal Kelly for accumulators
   683→  const recommendedStake = Math.max(0, Math.min(
   684→    (expectedValue / (totalOdds - 1)) * accaKellyFraction * bankroll,
   685→    bankroll * 0.05 // Max 5% of bankroll for accumulators
   686→  ));
   687→
   688→  // Should place if expected value is positive and risk is acceptable
   689→  const shouldPlace = expectedValue > 0.05 && riskScore < 75 && legs.length <= 6;
   690→
   691→  // Generate reasoning
   692→  const reasoning = generateAccumulatorReasoning(legs, legAnalyses, totalOdds, combinedProbability, expectedValue, riskScore, bonusPercent);
   693→
   694→  return {
   695→    totalOdds: Math.round(totalOdds * 100) / 100,
   696→    combinedProbability: Math.round(combinedProbability * 10000) / 10000,
   697→    riskScore: Math.round(riskScore),
   698→    recommendedStake: Math.round(recommendedStake * 100) / 100,
   699→    expectedValue: Math.round(expectedValue * 10000) / 10000,
   700→    bonusPercent,
   701→    legAnalyses,
   702→    shouldPlace,
   703→    reasoning,
   704→  };
   705→}

   734→  }
   735→
   736→  // Check odds range
   737→  const recOdds = prediction.recommended === "home" ? 0 : prediction.recommended === "away" ? 0 : 0; // We need the actual odds from the match
   738→  if (prediction.recommended === "home" || prediction.recommended === "away" || prediction.recommended === "draw") {
   739→    // Odds checks will be done at the API level where we have match data
   740→  }
   741→
   742→  // Check AI confidence
   743→  if (prediction.confidence < settings.minAiConfidence) {
   744→    return { shouldPlace: false, reason: `AI confidence ${Math.round(prediction.confidence * 100)}% below minimum ${Math.round(settings.minAiConfidence * 100)}%` };
   745→  }
   746→
   747→  // Check value edge
   748→  if (prediction.valueEdge < settings.minEdgeThreshold) {
   749→    return { shouldPlace: false, reason: `Value edge ${(prediction.valueEdge * 100).toFixed(1)}% below minimum ${(settings.minEdgeThreshold * 100).toFixed(1)}%` };
   750→  }
   751→
   752→  // Check risk level
   753→  if (settings.riskLevel === "low" && prediction.riskScore > 40) {

   755→  }
   756→  if (settings.riskLevel === "medium" && prediction.riskScore > 65) {
   757→    return { shouldPlace: false, reason: `Risk score ${prediction.riskScore} too high for medium risk setting` };
   758→  }
   759→
   760→  return { shouldPlace: true, reason: `All criteria met. Confidence: ${Math.round(prediction.confidence * 100)}%, Edge: ${(prediction.valueEdge * 100).toFixed(1)}%, Risk: ${prediction.riskScore}` };
   761→}

   763→/**
   764→ * Check stop-loss and profit targets
   765→ */
   766→export function checkRiskLimits(
   767→  dailyPnl: number,
   768→  weeklyPnl: number,
   769→  settings: {
   770→    stopLossDaily: number;
   771→    stopLossWeekly: number;
   772→    profitTargetDaily: number;
   773→    profitTargetWeekly: number;
   774→  }
   775→): { canBet: boolean; reason: string } {
   776→  // Check daily stop-loss
   777→  if (dailyPnl <= -settings.stopLossDaily) {
   778→    return { canBet: false, reason: `Daily stop-loss reached: -$${Math.abs(dailyPnl).toFixed(2)} / -$${settings.stopLossDaily.toFixed(2)}` };
   779→  }
   780→
   781→  // Check weekly stop-loss
   782→  if (weeklyPnl <= -settings.stopLossWeekly) {
   783→    return { canBet: false, reason: `Weekly stop-loss reached: -$${Math.abs(weeklyPnl).toFixed(2)} / -$${settings.stopLossWeekly.toFixed(2)}` };
   784→  }
   785→
   786→  // Check daily profit target
   787→  if (dailyPnl >= settings.profitTargetDaily) {
   788→    return { canBet: false, reason: `Daily profit target reached: +$${dailyPnl.toFixed(2)} / $${settings.profitTargetDaily.toFixed(2)}` };
   789→  }
   790→
   791→  // Check weekly profit target
   792→  if (weeklyPnl >= settings.profitTargetWeekly) {
   793→    return { canBet: false, reason: `Weekly profit target reached: +$${weeklyPnl.toFixed(2)} / $${settings.profitTargetWeekly.toFixed(2)}` };
   794→  }
   795→
   796→  return { canBet: true, reason: "Within risk limits" };
   797→}

   800→ * Check if current time is within the betting schedule
   801→ */
   802→export function isWithinBetSchedule(
   803→  scheduleStart: string,
   804→  scheduleEnd: string
   805→): boolean {
   806→  const now = new Date();
   807→  const currentMinutes = now.getHours() * 60 + now.getMinutes();
   808→
   809→  const [startH, startM] = scheduleStart.split(":").map(Number);
   810→  const [endH, endM] = scheduleEnd.split(":").map(Number);
   811→
   812→  const startMinutes = startH * 60 + startM;
   813→  const endMinutes = endH * 60 + endM;
   814→
   815→  if (startMinutes <= endMinutes) {
   816→    return currentMinutes >= startMinutes && currentMinutes <= endMinutes;
   817→  } else {
   818→    // Overnight schedule (e.g., 22:00 to 08:00)
   819→    return currentMinutes >= startMinutes || currentMinutes <= endMinutes;
   820→  }
   821→}

   823→// ==================== POISSON & OVER/UNDER ====================
   824→
   825→export function calculatePoissonProbabilities(
   826→  homeAttackRating: number,
   827→  awayAttackRating: number,
   828→  homeDefenseRating: number,
   829→  awayDefenseRating: number
   830→): PoissonResult {
   831→  const avgGoals = 1.3;
   832→
   833→  const homeAttackStrength = homeAttackRating / 70;
   834→  const awayDefenseWeakness = (100 - awayDefenseRating) / 70;
   835→  const expectedHomeGoals = avgGoals * homeAttackStrength * awayDefenseWeakness;
   836→
   837→  const awayAttackStrength = awayAttackRating / 70;
   838→  const homeDefenseWeakness = (100 - homeDefenseRating) / 70;
   839→  const expectedAwayGoals = avgGoals * awayAttackStrength * homeDefenseWeakness;
   840→
   841→  const maxGoals = 5;
   842→  const scoreProbabilities: Record<string, number> = {};
   843→  let homeWinProb = 0;
   844→  let drawProb = 0;
   845→  let awayWinProb = 0;
   846→
   847→  for (let i = 0; i <= maxGoals; i++) {
   848→    for (let j = 0; j <= maxGoals; j++) {
   849→      const prob = poissonProbability(expectedHomeGoals, i) * poissonProbability(expectedAwayGoals, j);
   850→      const key = `${i}-${j}`;
   851→      scoreProbabilities[key] = Math.round(prob * 10000) / 10000;
   852→
   853→      if (i > j) homeWinProb += prob;
   854→      else if (i === j) drawProb += prob;
   855→      else awayWinProb += prob;
   856→    }
   857→  }
   858→
   859→  const total = homeWinProb + drawProb + awayWinProb;
   860→  homeWinProb /= total;
   861→  drawProb /= total;
   862→  awayWinProb /= total;
   863→
   864→  return {
   865→    expectedHomeGoals: Math.round(expectedHomeGoals * 100) / 100,
   866→    expectedAwayGoals: Math.round(expectedAwayGoals * 100) / 100,
   867→    homeWinProb: Math.round(homeWinProb * 100) / 100,
   868→    drawProb: Math.round(drawProb * 100) / 100,
   869→    awayWinProb: Math.round(awayWinProb * 100) / 100,
   870→    scoreProbabilities,
   871→  };
   872→}
   873→
   874→export function calculateOverUnderProbabilities(
   875→  expectedGoals: number,
   876→  line: number
   877→): OverUnderResult {
   878→  let underProb = 0;
   879→  const maxGoals = 8;
   880→
   881→  for (let i = 0; i <= maxGoals; i++) {
   882→    const prob = poissonProbability(expectedGoals, i);
   883→    if (i < line) {
   884→      underProb += prob;
   885→    }
   886→  }
   887→
   888→  const overProb = 1 - underProb;
   889→  const recommendation = overProb > underProb ? "over" : "under";
   890→  const value = Math.abs(overProb - underProb);
   891→
   892→  return {
   893→    line,
   894→    overProb: Math.round(overProb * 100) / 100,
   895→    underProb: Math.round(underProb * 100) / 100,
   896→    recommendation,
   897→    value: Math.round(value * 100) / 100,
   898→  };
   899→}

   903→export function generateDetailedAnalysis(
   904→  match: MatchData,
   905→  homeStats: TeamStatsData | null,
   906→  awayStats: TeamStatsData | null,
   907→  prediction: Prediction
   908→): DetailedAnalysis {
   909→  const keyFactors: string[] = [];
   910→  const homeStrengths: string[] = [];
   911→  const homeWeaknesses: string[] = [];
   912→  const awayStrengths: string[] = [];
   913→  const awayWeaknesses: string[] = [];
   914→  const riskFactors: string[] = [];
   915→
   916→  if (homeStats && awayStats) {
   917→    const ratingDiff = homeStats.overallRating - awayStats.overallRating;
   918→    if (Math.abs(ratingDiff) > 10) {
   919→      keyFactors.push(
   920→        `${ratingDiff > 0 ? match.homeTeam : match.awayTeam} has a significant rating advantage (${Math.abs(ratingDiff)} points)`
   921→      );
   922→    }
   923→
   924→    if (homeStats.form && awayStats.form) {
   925→      const homeFormScore = parseForm(homeStats.form);
   926→      const awayFormScore = parseForm(awayStats.form);
   927→      if (Math.abs(homeFormScore - awayFormScore) > 0.3) {
   928→        keyFactors.push(
   929→          `${homeFormScore > awayFormScore ? match.homeTeam : match.awayTeam} is in significantly better form`
   930→        );
   931→      }
   932→    }
   933→
   934→    keyFactors.push(`${match.homeTeam} has home advantage`);
   935→
   936→    if (homeStats.attackRating > awayStats.defenseRating + 10) {
   937→      keyFactors.push(`${match.homeTeam}'s strong attack (${homeStats.attackRating}) vs ${match.awayTeam}'s weaker defense (${awayStats.defenseRating})`);
   938→    }
   939→    if (awayStats.attackRating > homeStats.defenseRating + 10) {
   940→      keyFactors.push(`${match.awayTeam}'s strong attack (${awayStats.attackRating}) vs ${match.homeTeam}'s weaker defense (${homeStats.defenseRating})`);
   941→    }
   942→
   943→    if (homeStats.attackRating > 70) homeStrengths.push(`Strong attack (${homeStats.attackRating})`);
   944→    if (homeStats.defenseRating > 70) homeStrengths.push(`Solid defense (${homeStats.defenseRating})`);
   945→    if (homeStats.form && parseForm(homeStats.form) > 0.7) homeStrengths.push(`Excellent recent form (${homeStats.form})`);
   946→    if (homeStats.homeRecord) {
   947→      const homeRec = parseRecord(homeStats.homeRecord);
   948→      if (homeRec > 0.6) homeStrengths.push(`Strong home record (${homeStats.homeRecord})`);
   949→    }
   950→
   951→    if (awayStats.attackRating > 70) awayStrengths.push(`Strong attack (${awayStats.attackRating})`);
   952→    if (awayStats.defenseRating > 70) awayStrengths.push(`Solid defense (${awayStats.defenseRating})`);
   953→    if (awayStats.form && parseForm(awayStats.form) > 0.7) awayStrengths.push(`Excellent recent form (${awayStats.form})`);
   954→    if (awayStats.awayRecord) {
   955→      const awayRec = parseRecord(awayStats.awayRecord);
   956→      if (awayRec > 0.6) awayStrengths.push(`Strong away record (${awayStats.awayRecord})`);
   957→    }
   958→
   959→    if (homeStats.defenseRating < 50) homeWeaknesses.push(`Weak defense (${homeStats.defenseRating})`);
   960→    if (homeStats.form && parseForm(homeStats.form) < 0.4) homeWeaknesses.push(`Poor recent form (${homeStats.form})`);
   961→    if (homeStats.losses > homeStats.wins) homeWeaknesses.push(`More losses than wins this season`);
   962→
   963→    if (awayStats.defenseRating < 50) awayWeaknesses.push(`Weak defense (${awayStats.defenseRating})`);
   964→    if (awayStats.form && parseForm(awayStats.form) < 0.4) awayWeaknesses.push(`Poor recent form (${awayStats.form})`);
   965→    if (awayStats.losses > awayStats.wins) awayWeaknesses.push(`More losses than wins this season`);
   966→  } else {
   967→    keyFactors.push("Limited team statistics available - analysis based on odds only");
   968→    riskFactors.push("No team stats available for deep analysis");
   969→  }
   970→
   971→  const recTeam = prediction.recommended === "home" ? match.homeTeam
   972→    : prediction.recommended === "away" ? match.awayTeam
   973→    : prediction.recommended === "draw" ? "Draw"
   974→    : prediction.recommended === "over" ? "Over 2.5"
   975→    : "Under 2.5";
   976→  const recOdds = prediction.recommended === "home" ? match.homeOdds
   977→    : prediction.recommended === "away" ? match.awayOdds
   978→    : match.drawOdds || 3.0;
   979→  const recProb = prediction.recommended === "home" ? prediction.homeWinProb
   980→    : prediction.recommended === "away" ? prediction.awayWinProb
   981→    : prediction.drawProb;
   982→  const edge = recProb - (1 / recOdds);
   983→
   984→  let riskScore = 50;
   985→  if (prediction.confidence < 0.5) riskScore += 20;
   986→  if (prediction.confidence > 0.7) riskScore -= 20;
   987→  if (Math.abs(prediction.homeWinProb - prediction.awayWinProb) < 0.1) riskScore += 15;
   988→  if (match.homeOdds > 3.0 || match.awayOdds > 3.0) riskScore += 10;
   989→  if (!homeStats || !awayStats) riskScore += 15;
   990→  riskScore = Math.max(0, Math.min(100, riskScore));
   991→
   992→  if (riskScore > 60) riskFactors.push("High uncertainty in prediction");
   993→  if (Math.abs(prediction.homeWinProb - prediction.awayWinProb) < 0.1) riskFactors.push("Teams are closely matched");
   994→  if (prediction.confidence < 0.5) riskFactors.push("Low AI confidence level");
   995→
   996→  const riskLevel: "low" | "medium" | "high" =
   997→    riskScore <= 35 ? "low" : riskScore <= 65 ? "medium" : "high";
   998→
   999→  return {
  1000→    keyFactors: keyFactors.length > 0 ? keyFactors : ["Standard match analysis"],
  1001→    strengths: { team: match.homeTeam, points: homeStrengths.length > 0 ? homeStrengths : ["Average performance"] },
  1002→    weaknesses: { team: match.homeTeam, points: homeWeaknesses.length > 0 ? homeWeaknesses : ["No significant weaknesses"] },
  1003→    valueBet: {
  1004→      selection: recTeam,
  1005→      reason: edge > 0.05
  1006→        ? `AI detects ${Math.round(edge * 100)}% edge over bookmaker odds. ${recTeam} has ${Math.round(recProb * 100)}% probability vs ${Math.round((1 / recOdds) * 100)}% implied probability.`
  1007→        : `Marginal value on ${recTeam}. Proceed with caution.`,
  1008→      edge: Math.round(edge * 100) / 100,
  1009→    },
  1010→    riskAssessment: {
  1011→      level: riskLevel,
  1012→      score: riskScore,
  1013→      factors: riskFactors.length > 0 ? riskFactors : ["Standard risk level"],
  1014→    },
  1015→  };
  1016→}

  1020→function generateBetReasoning(
  1021→  match: MatchData,
  1022→  prediction: Prediction,
  1023→  homeTeamStats?: TeamStatsData | null,
  1024→  awayTeamStats?: TeamStatsData | null
  1025→): string {
  1026→  const parts: string[] = [];
  1027→
  1028→  if (homeTeamStats && awayTeamStats) {
  1029→    const homeForm = homeTeamStats.form || "N/A";
  1030→    const awayForm = awayTeamStats.form || "N/A";
  1031→    parts.push(`${match.homeTeam} form: ${homeForm}, ${match.awayTeam} form: ${awayForm}.`);
  1032→
  1033→    if (homeTeamStats.overallRating > awayTeamStats.overallRating) {
  1034→      parts.push(`${match.homeTeam} have a higher overall rating (${homeTeamStats.overallRating} vs ${awayTeamStats.overallRating}).`);
  1035→    } else {
  1036→      parts.push(`${match.awayTeam} have a higher overall rating (${awayTeamStats.overallRating} vs ${homeTeamStats.overallRating}).`);
  1037→    }
  1038→
  1039→    parts.push(`${match.homeTeam} attack (${homeTeamStats.attackRating}) vs ${match.awayTeam} defense (${awayTeamStats.defenseRating}).`);
  1040→
  1041→    if (homeTeamStats.homeRecord) {
  1042→      parts.push(`${match.homeTeam} home record: ${homeTeamStats.homeRecord}.`);
  1043→    }
  1044→    if (awayTeamStats.awayRecord) {
  1045→      parts.push(`${match.awayTeam} away record: ${awayTeamStats.awayRecord}.`);
  1046→    }
  1047→  }
  1048→
  1049→  const recTeam = prediction.recommended === "home" ? match.homeTeam : prediction.recommended === "away" ? match.awayTeam : "Draw";
  1050→  const confidencePct = Math.round(prediction.confidence * 100);
  1051→  parts.push(`AI recommends: ${recTeam} with ${confidencePct}% confidence.`);
  1052→
  1053→  const value = prediction.valueEdge;
  1054→  if (value > 0.05) {
  1055→    parts.push(`Strong value detected (${Math.round(value * 100)}% edge over bookmaker).`);
  1056→  }
  1057→
  1058→  if (prediction.kellyStake > 0) {

  1065→function generateCashoutReasoning(
  1066→  bet: { selection: string; odds: number; stake: number; potentialWin: number; partialCashoutAmount?: number | null },
  1067→  match: { homeScore: number; awayScore: number; minute: number; homeTeam: string; awayTeam: string; sport: string; status: string },
  1068→  shouldCashout: boolean,
  1069→  cashoutAmount: number,
  1070→  urgency: string,
  1071→  waitOrCashout: string,
  1072→  settlementProbability: number,
  1073→  partialCashoutAmount: number
  1074→): string {
  1075→  const selectionIsHome = bet.selection === match.homeTeam;
  1076→  const currentLead = selectionIsHome
  1077→    ? match.homeScore - match.awayScore
  1078→    : match.awayScore - match.homeScore;
  1079→
  1080→  if (waitOrCashout === "wait_for_settlement") {
  1081→    return `Your bet on ${bet.selection} is currently winning (${match.homeScore}-${match.awayScore} at ${match.minute}'). Settlement probability: ${Math.round(settlementProbability * 100)}%. AI recommends waiting for full match settlement to collect maximum payout of $${bet.potentialWin.toFixed(2)}.`;
  1082→  }
  1083→
  1084→  if (shouldCashout) {
  1085→    if (waitOrCashout === "cashout_partial" && partialCashoutAmount > 0) {
  1086→      return `Your bet on ${bet.selection} is winning (${match.homeScore}-${match.awayScore} at ${match.minute}'). AI recommends partial cashout of $${partialCashoutAmount.toFixed(2)} to secure some profit while keeping the remaining stake active for full payout. ${urgency === "high" ? "ACT NOW - match state could change!" : "Moderate urgency."}`;
  1087→    }
  1088→    if (currentLead > 0) {
  1089→      return `Your bet on ${bet.selection} is currently winning (${match.homeScore}-${match.awayScore} at ${match.minute}'). Cashout recommended to secure profit of $${Math.round((cashoutAmount - bet.stake) * 100) / 100}. ${urgency === "high" ? "ACT NOW - match state could change!" : "Moderate urgency - consider cashing out soon."}`;
  1090→    } else if (currentLead === 0) {
  1091→      return `Match is drawn (${match.homeScore}-${match.awayScore} at ${match.minute}'). Cashout recommended to recover partial stake. High urgency - your bet is at risk.`;
  1092→    } else {
  1093→      return `${bet.selection} is currently losing (${match.homeScore}-${match.awayScore} at ${match.minute}'). Cashout to minimize losses. Recovery unlikely at this stage.`;
  1094→    }
  1095→  } else {
  1096→    if (currentLead > 0) {
  1097→      return `Your bet on ${bet.selection} is winning (${match.homeScore}-${match.awayScore} at ${match.minute}'). Settlement probability: ${Math.round(settlementProbability * 100)}%. AI recommends letting it ride for maximum profit of $${bet.potentialWin.toFixed(2)}.`;
  1098→    }
  1099→    return `Match is level at ${match.homeScore}-${match.awayScore}. Hold your position - there's still time for the match to turn in your favor.`;
  1100→  }
  1101→}

  1103→function generateAccumulatorReasoning(
  1104→  legs: Array<{ match: MatchData; recommended: string }>,
  1105→  legAnalyses: Prediction[],
  1106→  totalOdds: number,
  1107→  combinedProbability: number,
  1108→  expectedValue: number,
  1109→  riskScore: number,
  1110→  bonusPercent: number
  1111→): string {
  1112→  const parts: string[] = [];
  1113→
  1114→  parts.push(`${legs.length}-leg accumulator with total odds of ${totalOdds.toFixed(2)}.`);
  1115→  parts.push(`Combined probability: ${Math.round(combinedProbability * 100)}%.`);
  1116→  parts.push(`Expected value: ${(expectedValue * 100).toFixed(1)}%.`);
  1117→  parts.push(`Risk score: ${riskScore}/100.`);
  1118→
  1119→  if (bonusPercent > 0) {
  1120→    parts.push(`Accumulator bonus: ${bonusPercent}% applied.`);
  1121→  }
  1122→
  1123→  for (let i = 0; i < legs.length; i++) {
  1124→    const leg = legs[i];
  1125→    const analysis = legAnalyses[i];
  1126→    const recTeam = leg.recommended === "home" ? leg.match.homeTeam
  1127→      : leg.recommended === "away" ? leg.match.awayTeam
  1128→      : leg.recommended === "draw" ? "Draw" : leg.recommended;
  1129→    parts.push(`Leg ${i + 1}: ${leg.match.homeTeam} vs ${leg.match.awayTeam} - ${recTeam} (${Math.round(analysis.confidence * 100)}% confidence).`);
  1130→  }
  1131→
  1132→  if (riskScore > 65) {
  1133→    parts.push("Warning: High risk accumulator. Consider reducing stake or number of legs.");
  1134→  }
  1135→
  1136→  return parts.join(" ");
  1137→}
