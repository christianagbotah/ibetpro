    57→function expandSportKeys(sports: string[]): string[] {
    58→  const expanded = new Set<string>();
    59→  for (const sport of sports) {
    60→    const lower = sport.toLowerCase().trim();
    61→    if (SPORT_CATEGORY_MAP[lower]) {
    62→      for (const key of SPORT_CATEGORY_MAP[lower]) {
    63→        expanded.add(key);
    64→      }
    65→    } else {
    66→      // Keep the original key as-is (might be a specific key like "soccer_epl")
    67→      expanded.add(lower);
    68→    }
    69→  }
    70→  return Array.from(expanded);
    71→}

    99→// ==================== SINGLETON ENGINE ====================
   100→
   101→/**
   102→ * BotEngine is a singleton that manages all running bot instances.
   103→ * Each user has their own timer that runs scan cycles on a configurable interval.
   104→ *
   105→ * The engine is designed to:
   106→ * - Run independently of the frontend (no polling required)
   107→ * - Auto-recover from errors (continues running after failed scans)
   108→ * - Auto-stop on risk limits, schedule violations, or no allocation
   109→ * - Persist state in the database (BotSession) for recovery after restarts
   110→ * - Support graceful shutdown
   111→ */
   112→class BotEngine {
   113→  private static instance: BotEngine | null = null;
   114→  private timers: Map<string, NodeJS.Timeout> = new Map();
   115→  private runningUsers: Map<string, BotEngineStats> = new Map();
   116→  private isShuttingDown = false;
   117→  private heartbeatTimer: NodeJS.Timeout | null = null;
   118→
   119→  private constructor() {
   120→    // Register graceful shutdown handlers
   121→    process.on("SIGTERM", () => this.shutdown());
   122→    process.on("SIGINT", () => this.shutdown());
   123→
   124→    // Start a heartbeat that runs every 5 minutes to detect and recover
   125→    // zombie sessions (DB says "running" but engine timers are dead).
   126→    // This is critical for Telegram alert reliability — if the process
   127→    // survives but timers die (e.g. event loop stall, unref'd timer),
   128→    // alerts stop silently without this check.
   129→    this.startHeartbeat();
   130→  }
   131→
   132→  /**
   133→   * Periodic heartbeat that checks DB for sessions marked "running" that
   134→   * don't have a live engine timer, and recovers them.
   135→   */
   136→  private startHeartbeat(): void {
   137→    const HEARTBEAT_INTERVAL_MS = 5 * 60 * 1000; // 5 minutes
   138→
   139→    this.heartbeatTimer = setInterval(async () => {
   140→      if (this.isShuttingDown) return;
   141→
   142→      try {
   143→        const runningSessions = await prisma.botSession.findMany({
   144→          where: { status: "running" },
   145→          select: { userId: true, scanIntervalSec: true },
   146→        });
   147→
   148→        for (const session of runningSessions) {
   149→          // If the DB says running but the engine has no timer, it's a zombie
   150→          if (!this.timers.has(session.userId)) {
   151→            console.log(`[BotEngine Heartbeat] Zombie session detected for user ${session.userId}, recovering...`);
   152→            try {
   153→              const result = await this.start(session.userId, session.scanIntervalSec || 30);
   154→              if (result.success) {
   155→                console.log(`[BotEngine Heartbeat] ✅ Recovered zombie for user ${session.userId}`);
   156→              } else {
   157→                console.warn(`[BotEngine Heartbeat] ⚠️ Could not recover zombie for ${session.userId}: ${result.message}`);
   158→                // Mark as stopped so we don't keep retrying every 5 min
   159→                await prisma.botSession.update({
   160→                  where: { userId: session.userId },
   161→                  data: {
   162→                    status: "stopped",
   163→                    stoppedAt: new Date(),
   164→                    stopReason: `heartbeat_recovery_failed: ${result.message}`,
   165→                  },
   166→                }).catch(() => {});
   167→              }
   168→            } catch (err) {
   169→              console.error(`[BotEngine Heartbeat] ❌ Error recovering zombie for ${session.userId}:`, err);
   170→            }
   171→          }
   172→        }
   173→      } catch (err) {
   174→        // Don't let heartbeat errors crash anything
   175→        console.warn("[BotEngine Heartbeat] Check failed:", err);
   176→      }
   177→    }, HEARTBEAT_INTERVAL_MS);
   178→
   179→    // Keep the heartbeat alive — it's our zombie recovery mechanism
   180→  }
   181→
   182→  static getInstance(): BotEngine {
   183→    if (!BotEngine.instance) {
   184→      BotEngine.instance = new BotEngine();
   185→    }
   186→    return BotEngine.instance;
   187→  }

   191→  /**
   192→   * Start the bot for a specific user. Creates a timer that runs scan cycles
   193→   * on the configured interval (default 30 seconds).
   194→   */
   195→  async start(userId: string, scanIntervalSec: number = 30): Promise<{ success: boolean; message: string }> {
   196→    // Don't start if already running
   197→    if (this.timers.has(userId)) {
   198→      return { success: false, message: "Bot is already running for this user" };
   199→    }
   200→
   201→    // Validate prerequisites
   202→    const validation = await this.validatePrerequisites(userId);
   203→    if (!validation.valid) {
   204→      return { success: false, message: validation.error! };
   205→    }
   206→
   207→    // Initialize stats tracking
   208→    const stats: BotEngineStats = {
   209→      userId,
   210→      status: "running",
   211→      scanIntervalSec,
   212→      totalScans: 0,
   213→      totalBetsPlaced: 0,
   214→      totalStakeUsed: 0,
   215→      totalProfit: 0,
   216→      lastScanAt: null,
   217→      lastBetAt: null,
   218→      startedAt: new Date(),
   219→      errorCount: 0,
   220→      lastError: null,
   221→    };
   222→    this.runningUsers.set(userId, stats);
   223→
   224→    // Run the first scan immediately
   225→    try {
   226→      const result = await this.runScanCycle(userId);
   227→      stats.totalScans++;
   228→      stats.totalBetsPlaced += result.betsPlaced;
   229→      stats.totalStakeUsed += result.totalStake;
   230→      stats.lastScanAt = new Date();
   231→      if (result.betsPlaced > 0 || result.tipsGenerated > 0) {
   232→        stats.lastBetAt = new Date();
   233→      }
   234→      if (result.error) {
   235→        stats.errorCount++;
   236→        stats.lastError = result.error;
   237→      }
   238→    } catch (error) {
   239→      console.error(`[BotEngine] First scan failed for user ${userId}:`, error);
   240→      stats.errorCount++;
   241→      stats.lastError = error instanceof Error ? error.message : "Unknown error";
   242→    }
   243→
   244→    // Set up the recurring timer
   245→    const timer = setInterval(async () => {
   246→      if (this.isShuttingDown) return;
   247→
   248→      try {
   249→        const currentStats = this.runningUsers.get(userId);
   250→        if (!currentStats || currentStats.status !== "running") {
   251→          // Bot was stopped externally, clean up
   252→          this.stop(userId, "auto_stopped");
   253→          return;
   254→        }
   255→
   256→        // Check if the session is still running in the DB
   257→        const session = await prisma.botSession.findUnique({ where: { userId } });
   258→        if (!session || session.status !== "running") {
   259→          this.stop(userId, "session_expired");
   260→          return;
   261→        }
   262→
   263→        const result = await this.runScanCycle(userId);
   264→
   265→        currentStats.totalScans++;
   266→        currentStats.totalBetsPlaced += result.betsPlaced;
   267→        currentStats.totalStakeUsed += result.totalStake;
   268→        currentStats.lastScanAt = new Date();
   269→        currentStats.lastError = null; // Clear error on success
   270→
   271→        if (result.betsPlaced > 0) {
   272→          currentStats.lastBetAt = new Date();
   273→        }
   274→
   275→        if (result.error) {
   276→          currentStats.errorCount++;
   277→          currentStats.lastError = result.error;
   278→        }
   279→
   280→        // Update session stats in DB
   281→        await prisma.botSession.update({
   282→          where: { userId },
   283→          data: {
   284→            totalScans: { increment: 1 },
   285→            totalBetsPlaced: { increment: result.betsPlaced },
   286→            totalStakeUsed: { increment: result.totalStake },
   287→            lastScanAt: new Date(),
   288→            lastBetAt: result.betsPlaced > 0 ? new Date() : undefined,
   289→          },
   290→        });
   291→
   292→        // Log scan activity periodically (every 10 scans)
   293→        if (currentStats.totalScans % 10 === 0) {
   294→          await prisma.botLog.create({
   295→            data: {
   296→              userId,
   297→              action: "bot_scan",
   298→              reasoning: `Auto-scan #${currentStats.totalScans}: ${result.matches} matches found, ${result.betsPlaced} bets placed, ${result.skipped} skipped`,
   299→              details: JSON.stringify({
   300→                scanNumber: currentStats.totalScans,
   301→                betsPlaced: result.betsPlaced,
   302→                totalStake: result.totalStake,
   303→                matches: result.matches,
   304→                skipped: result.skipped,
   305→              }),
   306→            },
   307→          });
   308→        }
   309→
   310→        // Auto-settle finished matches every 3rd scan
   311→        if (currentStats.totalScans % 3 === 0) {
   312→          try {
   313→            await this.autoSettle(userId);
   314→          } catch (settleError) {
   315→            console.error(`[BotEngine] Auto-settle error for user ${userId}:`, settleError);
   316→          }
   317→        }
   318→
   319→        // Auto-cashout evaluation every 5th scan
   320→        if (currentStats.totalScans % 5 === 0) {
   321→          try {
   322→            await this.autoCashoutEvaluation(userId);
   323→          } catch (cashoutError) {
   324→            console.error(`[BotEngine] Auto-cashout error for user ${userId}:`, cashoutError);
   325→          }
   326→        }
   327→      } catch (error) {
   328→        console.error(`[BotEngine] Scan cycle error for user ${userId}:`, error);
   329→        const currentStats = this.runningUsers.get(userId);
   330→        if (currentStats) {
   331→          currentStats.errorCount++;
   332→          currentStats.lastError = error instanceof Error ? error.message : "Unknown error";
   333→
   334→          // Auto-stop after too many consecutive errors
   335→          if (currentStats.errorCount >= 10) {
   336→            console.error(`[BotEngine] Too many errors for user ${userId}, stopping bot`);
   337→            await this.stop(userId, "too_many_errors");
   338→          }
   339→        }
   340→      }
   341→    }, scanIntervalSec * 1000);
   342→
   343→    // NOTE: We intentionally do NOT call timer.unref() here.
   344→    // The scan timer IS the core business logic — it should keep the
   345→    // Node.js process alive. With .unref(), if the HTTP server goes idle,
   346→    // the process can exit and all Telegram alerts die silently.
   347→    // The heartbeat timer is also kept alive for the same reason.
   348→
   349→    this.timers.set(userId, timer);
   350→
   351→    console.log(`[BotEngine] Bot started for user ${userId} with interval ${scanIntervalSec}s`);
   352→
   353→    // Log bot start
   354→    await prisma.botLog.create({
   355→      data: {
   356→        userId,
   357→        action: "bot_started",
   358→        reasoning: `Bot engine started in background mode. Scanning every ${scanIntervalSec}s.`,
   359→        details: JSON.stringify({
   360→          mode: "background",
   361→          scanIntervalSec,
   362→          broker: validation.connectedAccount?.platform,
   363→          allocation: validation.connectedAccount?.allocatedAmount,
   364→        }),
   365→      },
   366→    });
   367→
   368→    return {
   369→      success: true,
   370→      message: stats.tipsGenerated > 0
   371→        ? `Advisor started! Generated ${stats.tipsGenerated} tip(s) in first scan. Running in background.`
   372→        : stats.betsPlaced > 0
   373→        ? `Bot started! Placed ${stats.betsPlaced} bet(s) in first scan. Running in background.`
   374→        : "Bot started! Scanning for matches in background. Will generate tips automatically.",
   375→    };
   376→  }

   378→  /**
   379→   * Stop the bot for a specific user. Clears the timer and updates the session.
   380→   */
   381→  async stop(userId: string, reason: string = "user_stopped"): Promise<void> {
   382→    const timer = this.timers.get(userId);
   383→    if (timer) {
   384→      clearInterval(timer);
   385→      this.timers.delete(userId);
   386→    }
   387→
   388→    const stats = this.runningUsers.get(userId);
   389→    if (stats) {
   390→      stats.status = "stopped";
   391→      this.runningUsers.delete(userId);
   392→    }
   393→
   394→    // Update the database session
   395→    try {
   396→      await prisma.botSession.upsert({
   397→        where: { userId },
   398→        update: {
   399→          status: "stopped",
   400→          stoppedAt: new Date(),
   401→          stopReason: reason,
   402→        },
   403→        create: {
   404→          userId,
   405→          status: "stopped",
   406→          stoppedAt: new Date(),
   407→          stopReason: reason,
   408→        },
   409→      });
   410→
   411→      // Log bot stop
   412→      await prisma.botLog.create({
   413→        data: {
   414→          userId,
   415→          action: "bot_stopped",
   416→          reasoning: `Bot engine stopped. Reason: ${reason}`,
   417→          details: JSON.stringify({
   418→            reason,
   419→            totalScans: stats?.totalScans || 0,
   420→            totalBetsPlaced: stats?.totalBetsPlaced || 0,
   421→            totalStakeUsed: stats?.totalStakeUsed || 0,
   422→            totalProfit: stats?.totalProfit || 0,
   423→            runDuration: stats?.startedAt
   424→              ? Math.round((Date.now() - stats.startedAt.getTime()) / 60000)
   425→              : 0,
   426→            errorCount: stats?.errorCount || 0,
   427→          }),
   428→        },
   429→      });
   430→    } catch (error) {
   431→      console.error(`[BotEngine] Error stopping bot for user ${userId}:`, error);
   432→    }
   433→
   434→    console.log(`[BotEngine] Bot stopped for user ${userId}. Reason: ${reason}`);
   435→  }

   437→  /**
   438→   * Gracefully shut down all running bots.
   439→   */
   440→  async shutdown(): Promise<void> {
   441→    if (this.isShuttingDown) return;
   442→    this.isShuttingDown = true;
   443→
   444→    // Stop heartbeat
   445→    if (this.heartbeatTimer) {
   446→      clearInterval(this.heartbeatTimer);
   447→      this.heartbeatTimer = null;
   448→    }
   449→
   450→    console.log(`[BotEngine] Shutting down ${this.timers.size} running bots...`);
   451→
   452→    for (const [userId] of this.timers) {
   453→      await this.stop(userId, "server_shutdown");
   454→    }
   455→
   456→    this.timers.clear();
   457→    this.runningUsers.clear();
   458→  }

   524→  // ==================== SCAN CYCLE ====================
   525→
   526→  /**
   527→   * Run a single scan cycle - analyze matches and generate tips or place bets.
   528→   * In advisor mode: AI finds value bets → creates Tip → sends Telegram alert.
   529→   * In auto mode: AI finds value bets → places bet via broker → creates Tip.
   530→   */
   531→  private async runScanCycle(userId: string): Promise<ScanResult> {
   532→    const result: ScanResult = { betsPlaced: 0, tipsGenerated: 0, totalStake: 0, matches: 0, skipped: 0 };
   533→
   534→    try {
   535→      // Refresh match data from external APIs (throttled to avoid rate limits)
   536→      try {
   537→        await syncMatchData(false);
   538→      } catch (syncErr) {
   539→        console.warn("[BotEngine] Match sync failed, continuing with existing data:", syncErr);
   540→      }
   541→
   542→      // Get user settings
   543→      const settings = await prisma.userSettings.findUnique({ where: { userId } });
   544→      if (!settings) {
   545→        result.error = "Settings not found";
   546→        return result;
   547→      }
   548→
   549→      const isAdvisorMode = settings.botMode === "advisor";
   550→
   551→      if (!settings.autoBettingEnabled && !isAdvisorMode) {
   552→        result.error = "Auto-betting is disabled";
   553→        await this.stop(userId, "auto_betting_disabled");
   554→        return result;
   555→      }
   556→
   557→      // Check schedule (using user's timezone setting)
   558→      if (!isWithinBetSchedule(settings.betScheduleStart, settings.betScheduleEnd, settings.timezone)) {
   559→        // Log once every ~10 scans to avoid spam
   560→        const stats = this.runningUsers.get(userId);
   561→        if (stats && stats.totalScans % 10 === 0) {
   562→          console.log(`[BotEngine] Outside bet schedule (${settings.betScheduleStart}-${settings.betScheduleEnd} ${settings.timezone}), skipping scan for user ${userId}`);
   563→        }
   564→        return result;
   565→      }
   566→
   567→      // Check risk limits
   568→      const user = await prisma.user.findUnique({ where: { id: userId } });
   569→      const riskCheck = checkRiskLimits(user?.dailyPnl || 0, user?.weeklyPnl || 0, {
   570→        stopLossDaily: settings.stopLossDaily,
   571→        stopLossWeekly: settings.stopLossWeekly,
   572→        profitTargetDaily: settings.profitTargetDaily,
   573→        profitTargetWeekly: settings.profitTargetWeekly,
   574→      });
   575→
   576→      if (!riskCheck.canBet) {
   577→        const reason = (user?.dailyPnl ?? 0) <= -settings.stopLossDaily ? "stop_loss" : "profit_target";
   578→        await this.stop(userId, reason);
   579→        await prisma.botLog.create({
   580→          data: {
   581→            userId,
   582→            action: reason === "stop_loss" ? "stop_loss_hit" : "profit_target_hit",
   583→            reasoning: riskCheck.reason,
   584→          },
   585→        });
   586→        return result;
   587→      }
   588→
   589→      // ---- Advisor mode: no broker required ----
   590→      // ---- Auto mode: need connected broker ----
   591→      let bettingAccount: Awaited<ReturnType<typeof prisma.bettingAccount.findFirst>> = null;
   592→      let activeAllocation: Awaited<ReturnType<typeof prisma.allocation.findFirst>> = null;
   593→      let remainingDailyLimit = settings.dailyBetLimit;
   594→      let existingBetMatchIds: string[] = [];
   595→
   596→      if (!isAdvisorMode) {
   597→        bettingAccount = await prisma.bettingAccount.findFirst({
   598→          where: { userId, isConnected: true },
   599→          orderBy: { allocatedAmount: "desc" },
   600→        });
   601→
   602→        if (!bettingAccount || bettingAccount.allocatedAmount <= 0) {
   603→          await this.stop(userId, "no_allocation");
   604→          await prisma.botLog.create({
   605→            data: {
   606→              userId,
   607→              action: "bot_scan",
   608→              reasoning: "No connected broker or allocation available. Stopping bot.",
   609→            },
   610→          });
   611→          return result;
   612→        }
   613→
   614→        // Check daily limit
   615→        const todayStart = new Date();
   616→        todayStart.setHours(0, 0, 0, 0);
   617→        const todayBets = await prisma.bet.findMany({
   618→          where: {
   619→            userId,
   620→            placedAt: { gte: todayStart },
   621→            status: { in: ["pending", "won", "lost", "cashed_out", "partial_cashout"] },
   622→          },
   623→        });
   624→        const dailyStake = todayBets.reduce((sum, b) => sum + b.stake, 0);
   625→        existingBetMatchIds = todayBets.map((b) => b.matchId);
   626→
   627→        if (dailyStake >= settings.dailyBetLimit) {
   628→          return result;
   629→        }
   630→
   631→        remainingDailyLimit = Math.min(
   632→          settings.dailyBetLimit - dailyStake,
   633→          bettingAccount.allocatedAmount
   634→        );
   635→
   636→        if (remainingDailyLimit < 5) {
   637→          return result;
   638→        }
   639→
   640→        activeAllocation = await prisma.allocation.findFirst({
   641→          where: { userId, bettingAccountId: bettingAccount.id, status: "active" },
   642→        });
   643→      } else {
   644→        // In advisor mode, check for existing tips to avoid duplicates
   645→        const todayStart = new Date();
   646→        todayStart.setHours(0, 0, 0, 0);
   647→        const todayTips = await prisma.tip.findMany({
   648→          where: { userId, createdAt: { gte: todayStart } },
   649→          select: { matchId: true },
   650→        });
   651→        existingBetMatchIds = todayTips.map((t) => t.matchId);
   652→      }
   653→
   654→      // Get upcoming matches
   655→      // tipSports may contain category names (e.g. "football") or sport keys (e.g. "soccer_epl")
   656→      // We need to match both — expand category names to include all sub-keys
   657→      const rawSports = isAdvisorMode
   658→        ? (settings.tipSports?.split(",") || ["football"])
   659→        : (settings.preferredSports?.split(",") || ["football"]);
   660→
   661→      const sportFilter = expandSportKeys(rawSports);

   697→              awayTeam: match.awayTeam,

   782→                riskLevel: prediction.riskLevel || settings.riskLevel,
   783→                matchTime: new Date(match.commenceTime).toLocaleString("en-GB", {
   784→                  timeZone: settings.timezone || "Africa/Accra",
   785→                  dateStyle: "medium",

   806→              where: { id: match.id },

   808→                aiHomeWinProb: prediction.homeWinProb,

   811→                aiConfidence: prediction.confidence,
   812→                aiRecommended: prediction.recommended,

   918→                betId: bet.id,

   942→                action: "bet_placed",

   944→                betId: bet.id,
   945→                details: JSON.stringify({
   946→                  stake,
   947→                  odds: recOdds,
   948→                  selection,

   954→                confidence: prediction.confidence,
   955→                profitImpact: -stake,
   956→              },
   957→            });
   958→
   959→            // Also create a Tip for the auto-placed bet
   960→            try {
   961→              const tip = await prisma.tip.create({
   962→                data: {
   963→                  userId,
   964→                  matchId: match.id,
   965→                  sport: match.sport,
   966→                  league: match.league,
   967→                  homeTeam: match.homeTeam,

   980→
   981→              // Send Telegram + in-app notification

   983→                sport: match.sport,
   984→                league: match.league,
   985→                homeTeam: match.homeTeam,
   986→                awayTeam: match.awayTeam,
   987→                selection,
   988→                odds: recOdds,
   989→                confidence: prediction.confidence,
   990→                valueEdge: prediction.valueEdge,
   991→                kellyStake: prediction.kellyStake,
   992→                riskLevel: settings.riskLevel,
   993→                matchTime: new Date(match.commenceTime).toLocaleString("en-GB", {

   998→                matchId: match.id,

  1001→
  1002→              // Update tip with delivery status

  1009→              }).catch(() => {});
  1010→            } catch (tipError) {
  1011→              console.error("[BotEngine] Failed to create tip/notification for auto-bet:", tipError);
  1012→            }
  1013→
  1014→            result.betsPlaced++;

  1020→        }
  1021→      }
  1022→
  1023→      // Log scan summary
  1024→      if (result.betsPlaced === 0 && result.tipsGenerated === 0 && result.skipped > 0) {

  1026→          data: {
  1027→            userId,
  1028→            action: "bot_scan",

  1030→          },
  1031→        });
  1032→      }
  1033→
  1034→    } catch (error) {
  1035→      console.error(`[BotEngine] Scan cycle error for user ${userId}:`, error);
  1036→      result.error = error instanceof Error ? error.message : "Unknown error";
  1037→    }
  1038→
  1039→    return result;
  1040→  }

  1044→  private async validatePrerequisites(userId: string): Promise<{
  1045→    valid: boolean;
  1046→    error?: string;
  1047→    connectedAccount?: { platform: string; allocatedAmount: number };
  1048→  }> {
  1049→    const settings = await prisma.userSettings.findUnique({ where: { userId } });
  1050→    if (!settings) {
  1051→      return { valid: false, error: "Settings not found. Please configure your settings first." };
  1052→    }
  1053→
  1054→    const isAdvisorMode = settings.botMode === "advisor";
  1055→
  1056→    if (!settings.autoBettingEnabled && !isAdvisorMode) {
  1057→      return { valid: false, error: "Auto-betting is disabled. Enable it first." };
  1058→    }
  1059→
  1060→    // In advisor mode, no broker is required — just need settings
  1061→    if (isAdvisorMode) {
  1062→      return { valid: true };
  1063→    }
  1064→
  1065→    // Auto mode requires a connected broker
  1066→    const connectedAccount = await prisma.bettingAccount.findFirst({
  1067→      where: { userId, isConnected: true },
  1068→    });
  1069→
  1070→    if (!connectedAccount) {
  1071→      return { valid: false, error: "No connected broker account. Connect a broker and set allocation first." };
  1072→    }
  1073→
  1074→    if (connectedAccount.allocatedAmount <= 0) {
  1075→      return { valid: false, error: "No allocation set. Allocate funds from your broker account first." };
  1076→    }
  1077→
  1078→    return {
  1079→      valid: true,
  1080→      connectedAccount: {
  1081→        platform: connectedAccount.platform,
  1082→        allocatedAmount: connectedAccount.allocatedAmount,
  1083→      },
  1084→    };
  1085→  }

  1087→  // ==================== AUTO SETTLEMENT ====================
  1088→
  1089→  /**
  1090→   * Automatically settle bets for finished matches.
  1091→   * Called periodically by the engine (every 3rd scan cycle).
  1092→   */
  1093→  private async autoSettle(userId: string): Promise<number> {
  1094→    const user = await prisma.user.findUnique({
  1095→      where: { id: userId },
  1096→      include: { settings: true },
  1097→    });
  1098→
  1099→    if (!user) return 0;
  1100→
  1101→    const commissionRate = user.settings?.commissionRate ?? config.commission.defaultRate;
  1102→
  1103→    // Find pending bets for finished matches
  1104→    const unsettledBets = await prisma.bet.findMany({
  1105→      where: {
  1106→        userId,
  1107→        status: "pending",
  1108→        match: { status: "finished" },
  1109→      },
  1110→      include: { match: true, bettingAccount: true },
  1111→    });
  1112→
  1113→    if (unsettledBets.length === 0) return 0;
  1114→
  1115→    let settledCount = 0;
  1116→
  1117→    for (const bet of unsettledBets) {
  1118→      if (!bet.match) continue;
  1119→
  1120→      const match = bet.match;
  1121→      const homeScore = match.homeScore ?? 0;
  1122→      const awayScore = match.awayScore ?? 0;
  1123→
  1124→      let betWon = false;
  1125→      let resultReason = "";
  1126→
  1127→      if (bet.betType === "single" || bet.betType === "accumulator_leg") {
  1128→        if (bet.selection === match.homeTeam) {
  1129→          betWon = homeScore > awayScore;
  1130→          resultReason = betWon
  1131→            ? `${match.homeTeam} won ${homeScore}-${awayScore}`
  1132→            : `${match.homeTeam} did not win (${homeScore}-${awayScore})`;
  1133→        } else if (bet.selection === match.awayTeam) {
  1134→          betWon = awayScore > homeScore;
  1135→          resultReason = betWon
  1136→            ? `${match.awayTeam} won ${awayScore}-${homeScore}`
  1137→            : `${match.awayTeam} did not win (${awayScore}-${homeScore})`;
  1138→        } else if (bet.selection === "Draw") {
  1139→          betWon = homeScore === awayScore;
  1140→          resultReason = betWon
  1141→            ? `Match drawn ${homeScore}-${awayScore}`
  1142→            : `Match not drawn (${homeScore}-${awayScore})`;
  1143→        } else if (bet.selection === "Over 2.5") {
  1144→          const totalGoals = homeScore + awayScore;
  1145→          betWon = totalGoals > 2.5;
  1146→          resultReason = betWon
  1147→            ? `Total goals ${totalGoals} > 2.5`
  1148→            : `Total goals ${totalGoals} <= 2.5`;
  1149→        } else if (bet.selection === "Under 2.5") {

  1153→            ? `Total goals ${totalGoals} < 2.5`

  1156→      }
  1157→
  1158→      const now = new Date();

  1160→      if (betWon) {

  1274→  // ==================== AUTO CASHOUT EVALUATION ====================
  1275→
  1276→  /**
  1277→   * Evaluate live bets for cashout opportunities.
  1278→   * Called periodically by the engine (every 5th scan cycle).
  1279→   * Uses the AI engine's cashout recommendation system.
  1280→   */
  1281→  private async autoCashoutEvaluation(userId: string): Promise<number> {
  1282→    const settings = await prisma.userSettings.findUnique({ where: { userId } });
  1283→    if (!settings?.autoCashoutEnabled) return 0;
  1284→
  1285→    // Find live bets that could be cashed out
  1286→    const liveBets = await prisma.bet.findMany({
  1287→      where: {
  1288→        userId,
  1289→        status: "pending",
  1290→        isAutoPlaced: true,
  1291→        match: { status: "live" },
  1292→      },
  1293→      include: { match: true },
  1294→    });
  1295→
  1296→    if (liveBets.length === 0) return 0;
  1297→
  1298→    let cashoutCount = 0;
  1299→
  1300→    for (const bet of liveBets) {
  1301→      if (!bet.match) continue;
  1302→
  1303→      // Simple cashout evaluation based on match state
  1304→      const match = bet.match;
  1305→      const homeScore = match.homeScore ?? 0;
  1306→      const awayScore = match.awayScore ?? 0;
  1307→      const minute = match.minute ?? 0;
  1308→
  1309→      // Determine if the bet is winning
  1310→      let isWinning = false;
  1311→      if (bet.selection === match.homeTeam) {
  1312→        isWinning = homeScore > awayScore;
  1313→      } else if (bet.selection === match.awayTeam) {
  1314→        isWinning = awayScore > homeScore;
  1315→      } else if (bet.selection === "Draw") {
  1316→        isWinning = homeScore === awayScore;
  1317→      } else if (bet.selection === "Over 2.5") {
  1318→        isWinning = (homeScore + awayScore) > 2.5;
  1319→      } else if (bet.selection === "Under 2.5") {
  1320→        isWinning = (homeScore + awayScore) < 2.5;
  1321→      }
  1322→
  1323→      // Cashout logic: if the bet is winning and we're past 70 minutes,
  1324→      // and the cashout threshold is met, execute cashout
  1325→      if (isWinning && minute >= 70) {
  1326→        const cashoutOdds = 1 + (1 / bet.odds) * (minute / 90);
  1327→        const cashoutAmount = Math.round(bet.stake * cashoutOdds * 100) / 100;
  1328→
  1329→        if (cashoutAmount > bet.stake * (1 + settings.cashoutThreshold)) {
  1330→          // Execute partial or full cashout based on settings
  1331→          if (settings.partialCashoutEnabled) {
  1332→            // Partial cashout: take 50% profit, keep the rest riding
  1333→            const partialAmount = Math.round(cashoutAmount * settings.partialCashoutPercent * 100) / 100;
  1334→            const partialGrossProfit = partialAmount - bet.stake * settings.partialCashoutPercent;
  1335→            const commissionRate = settings.commissionRate ?? config.commission.defaultRate;
  1336→            const commission = partialGrossProfit > 0 ? partialGrossProfit * commissionRate : 0;
  1337→            const netProfit = partialGrossProfit - commission;
  1338→
  1339→            await prisma.bet.update({
  1340→              where: { id: bet.id },
  1341→              data: {
  1342→                status: "partial_cashout",
  1343→                partialCashoutAmount,
  1344→                partialCashoutPercent: settings.partialCashoutPercent,
  1345→                cashoutOdds,
  1346→                cashedOutAt: new Date(),
  1347→              },
  1348→            });
  1349→
  1350→            await prisma.user.update({
  1351→              where: { id: userId },
  1352→              data: {
  1353→                balance: { increment: partialAmount },
  1354→                dailyPnl: { increment: netProfit },
  1355→                weeklyPnl: { increment: netProfit },
  1356→              },
  1357→            });
  1358→
  1359→            await prisma.transaction.create({
  1360→              data: {

  1364→                currency: "USD",

  1366→                description: `Auto partial cashout: ${match.homeTeam} vs ${match.awayTeam} - ${bet.selection} ($${partialAmount.toFixed(2)}, commission $${commission.toFixed(2)})`,
  1367→                betId: bet.id,
  1368→              },
  1369→            });
  1370→
  1371→            // Create commission ledger entry if commission was deducted
  1372→            if (commission > 0) {
  1373→              await prisma.commissionLedger.create({
  1374→                data: {
  1375→                  userId,
  1376→                  bettingAccountId: bet.bettingAccountId,
  1377→                  betId: bet.id,
  1378→                  grossProfit: partialGrossProfit,
  1379→                  commissionRate,
  1380→                  commissionAmount: commission,
  1381→                  netProfit,
  1382→                  status: "pending",
  1383→                },
  1384→              }).catch(() => {});

  1387→            await prisma.botLog.create({

  1400→                reasoning: `Auto partial cashout at ${minute}' - bet winning. ${Math.round(settings.partialCashoutPercent * 100)}% cashed out for $${partialAmount.toFixed(2)} (commission $${commission.toFixed(2)})`,
  1401→                profitImpact: netProfit,
  1402→              },
  1403→            });
  1404→          } else {

  1406→            const grossProfit = cashoutAmount - bet.stake;
  1407→            const commissionRate = settings.commissionRate ?? config.commission.defaultRate;
  1408→            const commission = grossProfit > 0 ? grossProfit * commissionRate : 0;
  1409→            const netProfit = grossProfit - commission;

  1417→                profit: netProfit,

  1422→            });

  1424→            await prisma.user.update({
  1425→              where: { id: userId },

  1427→                balance: { increment: cashoutAmount },
  1428→                totalProfit: { increment: netProfit },
  1429→                dailyPnl: { increment: netProfit },
  1430→                weeklyPnl: { increment: netProfit },

  1434→            await prisma.transaction.create({

  1489→    return cashoutCount;
  1490→  }
